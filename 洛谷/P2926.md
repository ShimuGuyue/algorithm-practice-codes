void solve()
{
    int m = 1'000'000;

    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    vector<int> counts(m + 1);
    for (auto a : as)
    {
        ++counts[a];
    }

    for (int i = m; i >= 1; --i)
    {
        for (int j = i * 2; j <= m; j += i)
        {
            counts[j] += counts[i];
        }
    }

    for (auto a : as)
    {
        cout << counts[a] - 1 << '\n';
    }
}
