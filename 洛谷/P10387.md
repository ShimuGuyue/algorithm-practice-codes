void solve()
{
    int64_t n, s;
    std::cin >> n >> s;
    std::vector<std::pair<int64_t, int>> soldiers(n);
    for (auto& [p, c] : soldiers)
    {
        std::cin >> p >> c;
    }

    std::sort(soldiers.begin(), soldiers.end(),
        [](std::pair<int64_t, int> a, std::pair<int64_t, int> b)
        {
            return a.second > b.second;
        }
    );

    std::vector<int64_t> presums(n);
    presums[0] = soldiers[0].first;
    for (int i{ 1 }; i < n; ++i)
    {
        presums[i] = presums[i - 1] + soldiers[i].first;
    }

    int max{ 0 };
    for (auto [p, c] : soldiers)
    {
        max = std::max(max, c);
    }

    std::vector<int> indexs(max + 1, -1);
    for (int i{ 0 }; i < n; ++i)
    {
        if (i == n - 1 || soldiers[i].second != soldiers[i + 1].second)
            indexs[soldiers[i].second] = i;
    }
    for (int i{ max - 1 }; i > 0; --i)
    {
        if (indexs[i] == -1)
            indexs[i] = indexs[i + 1];
    }

    int64_t ans{ 0 };
    for (int i{ max }; i > 0; --i)
    {
        if (indexs[i] == -1)
            continue;
        int index{ indexs[i] };
        ans += std::min(s, presums[index]);
    }
    std::cout << ans << '\n';
}
