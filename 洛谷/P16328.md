void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    std::vector<int> bs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> bs[i];
    }

    std::vector<int> difs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        difs[i] = as[i] - bs[i];
    }

    std::vector<int> precounts(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        precounts[i] = precounts[i - 1] + (difs[i] == 0);
    }

    std::map<int, std::vector<int>> indexs;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (difs[i] == 0)
            continue;
        indexs[difs[i]].push_back(i);
    }

    int sum{ std::count(difs.begin() + 1, difs.end(), 0) };
    int ans{ sum };
    for (auto& [_, v] : indexs)
    {
        int l{ 0 }, r{ 0 };
        while (r < v.size())
        {
            int ll{ v[l] };
            int rr{ v[r] };
            int count{ r - l + 1 };
            int dif{ precounts[rr] - precounts[ll - 1] };
            if (count - dif <= 0)
            {
                count = 1;
                dif = 0;
                l = r;
            }
            ans = std::max(ans, sum + count - dif);
            ++r;
        }
    }
    std::cout << ans << '\n';
}
