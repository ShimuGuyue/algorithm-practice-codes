void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<double>> grid(n, std::vector<double>(n + 1));
    for (int i{ 0 }; i < n; ++i)
    {
        for (int j{ 0 }; j <= n; ++j)
        {
            std::cin >> grid[i][j];
        }
    }

    for (int i{ 0 }; i < n; ++i)
    {
        int max_index{ i };
        for (int ii{ 0 }; ii < n; ++ii)
        {
            if (ii < i && fabs(grid[ii][ii]) > 1e-9)
                continue;
            if (fabs(grid[ii][i]) > fabs(grid[max_index][i]))
                max_index = ii;
        }
        std::swap(grid[i], grid[max_index]);

        if (fabs(grid[i][i]) < 1e-9)
            continue;

        double temp{ grid[i][i] };
        for (int j{ i }; j <= n; ++j)
        {
            grid[i][j] /= temp;
        }
        for (int ii{ 0 }; ii < n; ++ii)
        {
            if (ii == i)
                continue;
            double x{ grid[ii][i] / grid[i][i] };
            for (int j{ i }; j <= n; ++j)
            {
                grid[ii][j] -= grid[i][j] * x;
            }
        }
    }

    int mod{ 1 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (fabs(grid[i][i]) < 1e-9)
        {
            if (fabs(grid[i][n]) >= 1e-9)
            {
                mod = -1;
                break;
            }
            else
            {
                mod = 0;
            }
        }
    }
    if (mod == -1)
    {
        std::cout << -1 << '\n';
    }
    if (mod == 0)
    {
        std::cout << 0 << '\n';
    }
    if (mod == 1)
    {
        for (int i{ 0 }; i < n; ++i)
        {
            std::cout << 'x' << i + 1 << '=';
            std::cout << std::fixed << std::setprecision(2) << grid[i][n] << '\n';
        }
    }
