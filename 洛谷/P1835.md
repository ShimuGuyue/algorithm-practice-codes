void solve()
{
    int64_t l, r;
    cin >> l >> r;

    int n = 1000000;

    vector<int> is_prime(n, true);
    is_prime[1] = false;
    vector<int> primes;
    for (int i = 2; i < n; ++i)
    {
        if (!is_prime[i])
            continue;
        primes.push_back(i);
        for (int j = i * 2; j < n; j += i)
        {
            is_prime[j] = false;
        }
    }

    int ans = 0;
    if (r < n)
    {
        ans = accumulate(is_prime.begin() + l, is_prime.begin() + r + 1, 0);
    }
    else
    {
        is_prime.assign(r - l + 1, true);
        for (int p : primes)
        {
            for (int64_t i = (l + p - 1) / p * p; i <= r; i += p)
            {
                is_prime[i - l] = false;
            }
        }
        ans = accumulate(is_prime.begin(), is_prime.end(), 0);
    }
    cout << ans << '\n';
}
