void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    as[1] /= std::gcd(as[0], as[1]);
    for (int i = 2; i < n; ++i)
    {
        as[1] /= std::gcd(as[i], as[1]);
    }

    cout << (as[1] == 1 ? "Yes" : "No") << '\n';
}
