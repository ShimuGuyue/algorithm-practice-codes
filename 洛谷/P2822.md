void solve()
{
    int t, k;
    cin >> t >> k;

    vector<vector<int>> c(2000 + 1, vector<int>(2000 + 1, -1));
    for (int i = 0; i <= 2000; ++i)
    {
        c[i][0] = 1;
        c[i][i] = 1;
    }
    for (int i = 1; i <= 2000; ++i)
    {
        for (int j = 1; j < i; ++j)
        {
            c[i][j] = (c[i - 1][j - 1] + c[i - 1][j]) % k;
        }
    }

    vector<vector<int>> pres(2000 + 1, vector<int>(2000 + 1));
    for (int i = 1; i <= 2000; ++i)
    {
        for (int j = 1; j <= 2000; ++j)
        {
            pres[i][j] = pres[i][j - 1] + pres[i - 1][j] - pres[i - 1][j - 1] + (c[i][j] == 0);
        }
    }

    while (t--)
    {
        int n, m;
        cin >> n >> m;

        m = std::min(n, m);
        cout << pres[n][m] << '\n';
    }
}
