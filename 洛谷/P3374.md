class BinaryIndexedTree
{
private:
    std::vector<int> tree_;

public:
    BinaryIndexedTree(const std::vector<int>& v)
    {
        init(v);
    }

public:
    int query(const int l, const int r) const
    {
        // TODO：规定区间值做差方式
        return get_pre(r) - get_pre(l - 1);
    }

    void update(int loc, int data)
    {
        while (loc < tree_.size())
        {
            // TODO：规定单点更新方式
            tree_[loc] = tree_[loc] + data;
            loc += lowbit(loc);
        }
    }

private:
    static int lowbit(const int x)
    {
        return x & -x;
    }

private:
    void init(const std::vector<int>& v)
    {
        // TODO：规定运算单位元
        tree_.assign(v.size(), 0);
        for (int i{ 1 }; i < v.size(); ++i)
        {
            update(i, v[i]);
        }
    }

    int get_pre(int n) const
    {
        int ans{ 0 };
        while (n)
        {
            // TODO：规定区间值归并方式
            ans = ans + tree_[n];
            n -= lowbit(n);
        }
        return ans;
    }
};

void solve()
{
    int n, m;
    std::cin >> n >> m;

    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    BinaryIndexedTree bit(as);

    while (m--)
    {
        int op, x, y;
        std::cin >> op >> x >> y;
        if (op == 1)
            bit.update(x, y);
        else
            std::cout << bit.query(x, y) << '\n';
    }
}
