void solve()
{
    int x, y;
    cin >> x >> y;

    if (y % x)
    {
        cout << 0 << '\n';
        return;
    }

    int ans = 0;
    int flag = y / x;
    for (int i = 1; i * i <= flag; ++i)
    {
        if (flag % i)
            continue;
        int j = flag / i;
        if (std::gcd(i, j) != 1)
            continue;
        ans += i == j ? 1 : 2;
    }
    cout << ans << '\n';
}
