```cpp
struct DisjointSet
{
    int64_t inf{ std::numeric_limits<int64_t>::max() };

    std::vector<int> fathers;
    std::vector<int64_t> dists;

    DisjointSet(int n)
    {
        fathers.resize(n);
        dists.resize(n);
        std::iota(fathers.begin(), fathers.end(), 0);
    }

    int find(int x)
    {
        if (fathers[x] == x)
            return x;
        int father{ fathers[x] };
        fathers[x] = find(fathers[x]);
        dists[x] += dists[father];
        return fathers[x];
    }

    bool merge(int x, int y, int64_t dist)
    {
        int setx{ find(x) };
        int sety{ find(y) };
        if (setx == sety)
        {
            if (query(x, y) == dist)
                return true;
            else
                return false;
        }
        fathers[sety] = setx;
        dists[sety] = dists[x] + dist - dists[y];
        return true;
    }

    int64_t query(int x, int y)
    {
        if (find(x) != find(y))
            return inf;
        return dists[y] - dists[x];
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;

    DisjointSet ds(n + 2);
    bool ans{ true };
    while (m--)
    {
        int64_t l, r, s;
        std::cin >> l >> r >> s;
        ++r;
        ans &= ds.merge(l, r, s);
    }
    std::cout << std::boolalpha << ans << '\n';
}
```
