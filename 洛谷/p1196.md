```cpp
struct DisjointSet
{
    int64_t inf{ std::numeric_limits<int64_t>::max() };

    std::vector<int> fathers;
    std::vector<int64_t> dists;
    std::vector<int> sizes;

    DisjointSet(int n)
    {
        fathers.resize(n);
        std::iota(fathers.begin(), fathers.end(), 0);

        dists.resize(n);

        sizes.assign(n, 1);
    }

    int find(int x)
    {
        if (fathers[x] == x)
            return x;
        int father{ fathers[x] };
        fathers[x] = find(fathers[x]);
        dists[x] += dists[father];
        return fathers[x];
    }

    bool merge(int x, int y, int64_t dist)
    {
        int setx{ find(x) };
        int sety{ find(y) };
        if (setx == sety)
        {
            return true;
        }
        fathers[sety] = setx;
        sizes[setx] += sizes[sety];
        dists[sety] = dists[x] + dist - dists[y];
        return true;
    }

    int64_t query(int x, int y)
    {
        if (find(x) != find(y))
            return inf;
        return dists[y] - dists[x];
    }
};
```

```cpp
void solve()
{
    constexpr int n{ 30000 };

    DisjointSet ds(n + 1);

    int t;
    std::cin >> t;
    while (t--)
    {
        char op;
        int i, j;
        std::cin >> op >> i >> j;
        if (op == 'M')
        {
            int x{ ds.find(i) };
            int y{ ds.find(j) };
            int dist{ ds.sizes[y] };
            ds.merge(y, x, dist);
        }
        else
        {
            int64_t ans{ ds.query(i, j) };
            if (ans == ds.inf)
                ans = -1;
            else
                ans = abs(ans) - 1;
            std::cout << ans << '\n';
        }
    }
}
```
