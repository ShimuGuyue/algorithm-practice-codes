void solve()
{
    int n;
    cin >> n;
    vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        cin >> as[i];
    }

    sort(as.begin(), as.end());

    for (int i = 1; i <= n; ++i)
    {
        if (as[i] < i)
        {
            cout << 0 << '\n';
            return;
        }
    }

    int64_t ans = 1;
    int64_t mod = 1000000007;
    for (int i = 1; i <= n; ++i)
    {
        ans = ans * (as[i] - i + 1) % mod;
    }
    cout << ans << '\n';
}
