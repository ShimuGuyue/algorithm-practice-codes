struct BinaryIndexedTree
{
private:
    std::vector<int> tree_dif_;

public:
    BinaryIndexedTree(const std::vector<int>& v)
    {
        init(v);
    }

public:
    int get(int n) const
    {
        int ans{ 0 };
        while (n)
        {
            ans += tree_dif_[n];
            n -= lowbit(n);
        }
        return ans;
    }

    void update(const int l, const int r, const int data)
    {
        update(l, data);
        if (r + 1 < tree_dif_.size())
            update(r + 1, -data);
    }

private:
    static int lowbit(const int n)
    {
        return n & -n;
    }

private:
    void init(const std::vector<int>& v)
    {
        tree_dif_.assign(v.size(), 0);
        for (int i{ 1 }; i < v.size(); ++i)
        {
            update(i, v[i] - v[i - 1]);
        }
    }

    void update(int loc, const int data)
    {
        while (loc < tree_dif_.size())
        {
            tree_dif_[loc] += data;
            loc += lowbit(loc);
        }
    }
};

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    BinaryIndexedTree bit(as);

    while (m--)
    {
        int op;
        std::cin >> op;
        if (op == 1)
        {
            int x, y, k;
            std::cin >> x >> y >> k;
            bit.update(x, y, k);
        }
        else
        {
            int x;
            std::cin >> x;
            std::cout << bit.get(x) << '\n';
        }
    }
}
