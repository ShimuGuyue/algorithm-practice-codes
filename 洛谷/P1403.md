void solve()
{
    int n;
    cin >> n;

    vector<int> counts(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        for (int j = i; j <= n; j += i)
        {
            ++counts[j];
        }
    }
    int ans = accumulate(counts.begin(), counts.end(), 0);
    cout << ans << '\n';
}
