```cpp
/**
 *  BASE：字符集大小
 */
template<const int BASE = 52>   // TODO：定义字符集大小
class KMP
{
private:
    std::vector<int> fails_;
    std::vector<std::array<int, BASE>> nexts_;

    const std::string& pattern_;

public:
    KMP(const std::string& pattern) : pattern_(pattern)
    {
        fails_ = get_borders(pattern_);

        const int n{ static_cast<int>(pattern_.size()) };
        nexts_.assign(n + 1, { });
        for (int i{ 0 }; i <= n; ++i)
        {
            for (int index{ 0 }; index < BASE; ++index)
            {
                if (deturn(index) == pattern_[i])
                    if (i == n)
                        nexts_[i][index] = nexts_[fails_[i - 1]][index];
                    else
                        nexts_[i][index] = i + 1;
                else
                    if (i == 0)
                        nexts_[i][index] = 0;
                    else
                        nexts_[i][index] = nexts_[fails_[i - 1]][index];
            }
        }
    }

public:
    static std::vector<int> get_borders(const std::string& s)
    {
        const int n{ static_cast<int>(s.size()) };
        std::vector<int> borders(n, 0);
        for (int i{ 1 }; i < n; ++i)
        {
            int border{ borders[i - 1] };
            while (border != 0 && s[border] != s[i])
            {
                border = borders[border - 1];
            }
            if (s[border] == s[i])
                ++border;
            borders[i] = border;
        }
        return borders;
    }

public:
    int match(const std::string& text) const
    {
        const int n{ static_cast<int>(text.size()) };
        const int m{ static_cast<int>(pattern_.size()) };

        int index{ 0 };
        for (int i{ 0 }; i < n; ++i)
        {
            index = nexts_[index][enturn(text[i])];
            if (index == m)
                return i - m + 1;
        }
        return -1;
    }

    std::vector<int> match_all(const std::string& text) const
    {
        const int n{ static_cast<int>(text.size()) };
        const int m{ static_cast<int>(pattern_.size()) };

        std::vector<int> ans;

        int index{ 0 };
        for (int i{ 0 }; i < n; ++i)
        {
            index = nexts_[index][enturn(text[i])];
            if (index == m)
                ans.push_back(i - m + 1);
        }
        return ans;
    }

private:
    static int enturn(const char c)
    {
        // TODO：定义字符到整数下标转换逻辑
        // return ;
        if (std::islower(c))
            return c - 'a';
        if (std::isupper(c))
            return c - 'A' + 26;
        return c - '0' + 52;
    }

    static char deturn(const int i)
    {
        // TODO：定义整数下标到字符转换逻辑
        // return ;
        if (i < 26)
            return i + 'a';
        if (i < 52)
            return i + 'A' - 26;
        return i + '0' - 52;
    }
};
```

```cpp
void solve()
{
    std::string s;
    std::string t;
    std::cin >> s;
    std::cin >> t;

    KMP kmp(t);
    auto indexs{ kmp.match_all(s) };
    for (auto i : indexs)
    {
        std::cout << i + 1 << '\n';
    }
    auto borders{ kmp.get_borders(t) };
    for (auto b : borders)
    {
        std::cout << b << ' ';
    }
    std::cout << '\n';
}
```
