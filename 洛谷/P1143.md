void solve()
{
    int n;
    cin >> n;
    string s;
    cin >> s;
    int m;
    cin >> m;

    int digit = 0;
    for (char c : s)
    {
        int a = isdigit(c) ? c - '0' : c - 'A' + 10;
        digit = digit * n + a;
    }

    string ans;
    while (digit)
    {
        int a = digit % m;
        digit /= m;
        char c = a < 10 ? a + '0' : a + 'A' - 10;
        ans += c;
    }
    reverse(ans.begin(), ans.end());
    cout << ans << '\n';
}
