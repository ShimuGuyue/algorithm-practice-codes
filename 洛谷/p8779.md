```cpp
struct DisjointSet
{
    int64_t inf{ std::numeric_limits<int64_t>::max() };

    std::vector<int> fathers;
    std::vector<int64_t> dists;

    DisjointSet(int n)
    {
        fathers.resize(n);
        dists.resize(n);
        std::iota(fathers.begin(), fathers.end(), 0);
    }

    int find(int x)
    {
        if (fathers[x] == x)
            return x;
        int father{ fathers[x] };
        fathers[x] = find(fathers[x]);
        dists[x] += dists[father];
        return fathers[x];
    }

    void merge(int x, int y, int64_t dist)
    {
        int setx{ find(x) };
        int sety{ find(y) };
        fathers[sety] = setx;
        dists[sety] = dists[x] + dist - dists[y];
    }

    int64_t query(int x, int y)
    {
        if (find(x) != find(y))
            return inf;
        return dists[y] - dists[x];
    }
};
```

```cpp
void solve()
{
    int n, m, q;
    std::cin >> n >> m >> q;

    DisjointSet ds(n + 2);
    while (m--)
    {
        int64_t l, r, s;
        std::cin >> l >> r >> s;
        ++r;
        ds.merge(l, r, s);
    }

    while (q--)
    {
        int l, r;
        std::cin >> l >> r;
        ++r;
        int64_t ans{ ds.query(l, r) };
        if (ans == ds.inf)
            std::cout << "UNKNOWN\n";
        else
            std::cout << ans << '\n';
    }
}
```
