template <const int N, const int64_t MOD = 0>
class Combination
{
private:
    inline static std::vector<int64_t> factorials;
    inline static std::vector<int64_t> inv_factorials;

public:
    Combination()
    {
        factorials.assign(N + 1, 1);
        inv_factorials.assign(N + 1, 1);
        for (int i{ 1 }; i <= N; ++i)
        {
            factorials[i] = factorials[i - 1] * i;
            if (MOD)
                factorials[i] %= MOD;
        }
        if (MOD)
        {
            inv_factorials[N] = power(factorials[N], MOD - 2);
            for (int i{ N - 1 }; i > 0; --i)
            {
                inv_factorials[i] = inv_factorials[i + 1] * (i + 1) % MOD;
            }
        }
    }

public:
    int64_t calc(const int64_t n, const int64_t k)
    {
        if (k == 0 || k == n)
            return 1;
        if (k < 0 || k > n)
            return 0;
        return MOD
            ? factorials[n] * inv_factorials[k] % MOD * inv_factorials[n - k] % MOD
            : factorials[n] / factorials[k] / factorials[n - k];
    }

private:
    int64_t power(int64_t base, int64_t p)
    {
        int64_t ans{ 1 };
        while (p)
        {
            if (p & 1)
                ans = ans * base % MOD;
            base = base * base % MOD;
            p /= 2;
        }
        return ans;
    }
};

void solve()
{
    Combination<5000, 1000000007> c;

    int n, m;
    std::cin >> n >> m;
    std::cout << c.calc(n, m) << '\n';
}
