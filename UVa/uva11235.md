class SparseTable
{
private:
    std::vector<std::pair<int, int>> v_;
    std::vector<std::vector<int>> table_;

public:
    SparseTable(const std::vector<std::pair<int, int>>& v)
    {
        build(v);
    }

public:
    int query(const int l, const int r) const
    {
        const int power{ static_cast<int>(log2(r - l + 1)) };
        const int len{ 1 << power };
        return merge(l, r - len + 1, power);
    }

private:
    void build(const std::vector<std::pair<int, int>>& v)
    {
        v_ = v;

        const int n{ static_cast<int>(v.size()) };
        const int m{ static_cast<int>(std::log2(n)) + 1 };
        table_.assign(n, std::vector<int>(m));

        for (int i{ 0 }; i < n; ++i)
        {
            table_[i][0] = contribute(i);
        }
        for (int j{ 1 }; j < m; ++j)
        {
            int len{ 1 << (j - 1) };
            for (int i{ 0 }; i + len * 2 - 1 < n; ++i)
            {
                table_[i][j] = merge(i, i + len, j - 1);
            }
        }
    }

    int contribute(const int index) const
    {
        // TODO: 规定初始贡献值
        return v_[index].second - v_[index].first + 1;
    }

    int merge(const int i1, const int i2, const int j) const
    {
        // TODO: 规定贡献值合并结果
        return std::max(table_[i1][j], table_[i2][j]);
    }
};

void solve()
{
    int n, q;
    std::cin >> n;
    if (n == 0)
        exit(0);
    std::cin >> q;
    std::vector<int> as(n + 2, std::numeric_limits<int>::max());
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::map<int, int> m;
    std::vector<std::pair<int, int>> ranges;
    for (int l{ 1 }, r{ 2 }; r <= n + 1; ++r)
    {
        if (as[r] != as[l])
        {
            m[as[l]] = ranges.size();
            ranges.push_back({l, r - 1});
            l = r;
        }
    }

    SparseTable st(ranges);

    while (q--)
    {
        int l, r;
        std::cin >> l >> r;
        int ll{ m[as[l]] }, rr{ m[as[r]] };
        if (ll == rr)
        {
            std::cout << r - l + 1 << '\n';
            continue;
        }
        int ans{ std::max(ranges[ll].second - l + 1, r - ranges[rr].first + 1) };
        if (ll + 1 == rr)
        {
            std::cout << ans << '\n';
            continue;
        }
        ans = std::max(ans, st.query(ll + 1, rr - 1));
        std::cout << ans << '\n';
    }
}
