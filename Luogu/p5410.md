```cpp
class Z
{
public:
    static std::vector<int> calc(const std::string& s)
    {
        int n{ static_cast<int>(s.length()) };
        std::vector<int> zs(n);

        // TODO：设置Z[0]默认值
        zs[0] = n;
        int l, r{ -1 };
        for (int i{ 1 }; i < n; ++i)
        {
            if (i <= r)
                zs[i] = std::min(zs[i - l], r - i);
            while (i + zs[i] < n && s[zs[i]] == s[i + zs[i]])
            {
                ++zs[i];
            }
            if (i + zs[i] - 1 > r)
            {
                l = i;
                r = i + zs[i] - 1;
            }
        }
        return zs;
    }
};
```

```cpp
void solve()
{
    std::string a;
    std::string b;
    std::cin >> a;
    std::cin >> b;

    auto z1{ Z::calc(b) };
    auto z2{ Z::calc(b + a) };

    int64_t ans1{ 0 }, ans2{ 0 };
    for (int i{ 0 }; i < b.length(); ++i)
    {
        ans1 ^= int64_t(i + 1) * (z1[i] + 1);
    }
    for (int i{ int(b.length()) }; i < a.length() + b.length(); ++i)
    {
        ans2 ^= int64_t(i - b.length() + 1) * (std::min<int>(z2[i], b.length()) + 1);
    }
    std::cout << ans1 << '\n';
    std::cout << ans2 << '\n';
}
```
