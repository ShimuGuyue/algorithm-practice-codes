void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::vector<int> sufs_p_ext(n + 1);
    for (int i{ n - 1 }; i >= 0; --i)
    {
        sufs_p_ext[i] = sufs_p_ext[i + 1] + (s[i] != 'L');
    }

    int64_t sum{ 0 };
    for (int i{ 0 }, count{ 0 }; i < n; ++i)
    {
        if (s[i] == 'L')
            ++count;
        else
            sum += count;
    }

    int64_t ans{ sum };
    for (int i{ 0 }, count{ 0 }; i < n; ++i)
    {
        if (s[i] == 'Q')
            continue;
        if (s[i] == '?')
        {
            sum = sum - count + sufs_p_ext[i + 1];
        }
        ++count;
        ans = std::max(ans, sum);
    }
    std::cout << ans << '\n';
}
