void solve()
{
    int n, m, s;
    std::cin >> n >> m >> s;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    int size{ int(std::ceil(std::log2(n))) + 1 };

    std::vector<int> depths(n + 1);
    std::vector<std::vector<int>> ups(n + 1, std::vector<int>(size));

    auto dfs = [&tree, &depths, &ups, size](this auto&& dfs, int u, int last, int deepth) -> void
    {
        depths[u] = deepth;
        ups[u][0] = last;
        for (int i{ 1 }; i < size; ++i)
        {
            ups[u][i] = ups[ups[u][i - 1]][i - 1];
        }
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u, deepth + 1);
        }
    };
    dfs(s, 0, 1);

    auto lca = [&depths, &ups, size](int a, int b) -> int
    {
        if (depths[a] < depths[b])
            std::swap(a, b);
        int dif{ depths[a] - depths[b] };
        for (int i{ 0 }; i < size; ++i)
        {
            if (dif >> i & 1)
                a = ups[a][i];
        }

        if (a == b)
            return a;

        for (int i{ size - 1 }; i >= 0; --i)
        {
            if (ups[a][i] != ups[b][i])
            {
                a = ups[a][i];
                b = ups[b][i];
            }
        }
        return ups[a][0];
    };

    while (m--)
    {
        int a, b;
        std::cin >> a >> b;
        std::cout << lca(a, b) << '\n';
    }
}
