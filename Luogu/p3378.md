class Heap
{
private:
    std::vector<int> heap_;

public:
    void push(int x)
    {
        int index{ static_cast<int>(heap_.size()) };
        heap_.push_back(x);
        while (slip_up(index))
        { }
    }

    void pop()
    {
        heap_.front() = heap_.back();
        heap_.pop_back();
        int index{ 0 };
        while (slip_down(index))
        { }
    }

    int top()
    {
        return heap_.front();
    }

    bool empty()
    {
        return heap_.empty();
    }

    int size()
    {
        return  heap_.size();
    }

private:
    bool operate(int a, int b)
    {
        return a < b;
    }

    bool slip_up(int& node)
    {
        if (node == 0)
            return false;
        int father{ (node - 1) / 2 };
        if (!operate(heap_[node], heap_[father]))
            return false;
        std::swap(heap_[node], heap_[father]);
        node = father;
        return true;
    }

    bool slip_down(int& node)
    {
        int l{ node * 2 + 1 }, r{ node * 2 + 2 };
        if (l >= heap_.size())
            return false;
        int child{ r < heap_.size() && heap_[r] < heap_[l] ? r : l };
        if (!operate(heap_[child], heap_[node]))
            return false;
        std::swap(heap_[node], heap_[child]);
        node = child;
        return true;
    }
};
