

```cpp
template <const int SIZE>
class Eratosthenes
{
private:
    inline static bool is_init{ false };
    inline static std::vector<int> primes_{ };
    inline static std::vector<bool> is_primes_{ };

public:
    static bool judge(const int n)
    {
        if (!is_init)
            init();
        return (n & 1) ? is_primes_[n] : (n == 2);
    }

    static const std::vector<int>& primes()
    {
        if (!is_init)
            init();
        return primes_;
    }

private:
    static void init()
    {
        primes_.reserve(SIZE / std::log(SIZE));
        is_primes_.assign(SIZE + 1, true);

        if (SIZE >= 2)
            primes_.push_back(2);

        is_primes_[1] = false;
        for (int i{ 3 }; i <= SIZE; i += 2)
        {
            if (!is_primes_[i])
                continue;
            primes_.push_back(i);

            if (int64_t(i) * i > SIZE)
                continue;
            for (int j{ i * i }; j <= SIZE; j += i * 2)
            {
                is_primes_[j] = false;
            }
        }
    }
};
```

```cpp
void solve()
{
    int n, q;
    std::cin >> n >> q;

    auto primes{ Eratosthenes<100000000>::primes() };
    while (q--)
    {
        int k;
        std::cin >> k;
        std::cout << primes[k - 1] << '\n';
    }
}
```
