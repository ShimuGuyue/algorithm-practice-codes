```cpp
class DisjointSet
{
private:
    struct Data
    {
        int father;
    };
    std::vector<Data> nodes_;

public:
    DisjointSet(const int n)
    {
        build(n);
    }

public:
    int find(const int x)
    {
        return nodes_[x].father == x ? x : nodes_[x].father = find(nodes_[x].father);
    }

    bool merge(const int x, const int y)
    {
        const int set_x{ find(x) };
        const int set_y{ find(y) };
        if (set_x == set_y)
            return false;
        nodes_[set_x].father = set_y;
        return true;
    }

    int count()
    {
        std::set<int> t;
        for (int i{ 1 }; i <= nodes_.size() / 2; ++i)
        {
            t.insert(find(i));
        }
        return t.size();
    }

private:
    void build(const int n)
    {
        nodes_.assign(n, { });
        for (int i{ 0 }; i < n; ++i)
        {
            nodes_[i].father = i;
        }
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;

    DisjointSet ds(n * 2 + 1);

    for (int i = 0; i < m; ++i) {
        char op;
        int x, y;
        std::cin >> op >> x >> y;
        if (op == 'F')
        {
            ds.merge(x, y);
        }
        else
        {
            ds.merge(x, y + n);
            ds.merge(y, x + n);
        }
    }
    std::cout << ds.count() << std::endl;
}
```
