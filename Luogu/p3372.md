class BinaryIndexedTree
{
private:
    std::vector<int64_t> tree1; // dif(i)
    std::vector<int64_t> tree2; // dif(i) * (i - 1)

public:
    BinaryIndexedTree(const std::vector<int64_t>& v) : tree1(v.size()), tree2(v.size())
    {
        init(v);
    }

public:
    int64_t query(const int l, const int r) const
    {
        return get(r) - get(l - 1);
    }

    void update(const int l, const int r, const int64_t data)
    {
        update(l, data);
        if (r + 1 <= tree1.size())
            update(r + 1, -data);
    }

private:
    static int lowbit(const int x)
    {
        return x & -x;
    }

private:
    void init(const std::vector<int64_t>& v)
    {
        for (int i{ 1 }; i < v.size(); ++i)
        {
            update(i, v[i] - v[i - 1]);
        }
    }

    int64_t get(const int index) const
    {
        int64_t ans{ 0 };
        int i{ index };
        while (i)
        {
            ans += tree1[i] * index - tree2[i];
            i -= lowbit(i);
        }
        return ans;
    }

    void update(const int index, const int64_t data)
    {
        int i{ index };
        while (i < tree1.size())
        {
            tree1[i] += data;
            tree2[i] += data * (index - 1);
            i += lowbit(i);
        }
    }
};

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    BinaryIndexedTree bit(as);

    while (m--)
    {
        int op;
        std::cin >> op;
        if (op == 1)
        {
            int64_t x, y, k;
            std::cin >> x >> y >> k;
            bit.update(x, y, k);
        }
        else
        {
            int64_t x, y;
            std::cin >> x >> y;
            std::cout << bit.query(x, y) << '\n';
        }
    }
}
