class Solution
{
public:
    int findTheLongestSubstring(std::string& s)
    {
        int ans{ 0 };
        std::map<int, int> m;
        m[0] = -1;
        int pre{ 0 };
        for (int i{ 0 }; i < s.length(); ++i)
        {
            char c{ s[i] };
            int flag{ turn(c) };
            if (flag != -1)
                pre ^= 1 << flag;
            if (m.count(pre))
                ans = std::max(ans, i - m[pre]);
            else
                m[pre] = i;
        }
        return ans;
    }

    int turn(char c)
    {
        int ans;
        switch (c)
        {
        case 'a':
            ans = 0;
            break;
        case 'e':
            ans = 1;
            break;
        case 'i':
            ans = 2;
            break;
        case 'o':
            ans = 3;
            break;
        case 'u':
            ans = 4;
            break;
        default:
            ans = -1; 
        }
        return ans;
    }
};
