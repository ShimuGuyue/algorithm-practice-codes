class Solution
{
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target)
    {
        int n = static_cast<int>(matrix.size());
        int m = static_cast<int>(matrix[0].size());

        int x = 0, y = m - 1;
        while (x < n && y >= 0)
        {
            if (matrix[x][y] < target)
                ++x;
            else if (matrix[x][y] > target)
                --y;
            else
                return true;
        }
        return false;
    }
};
