class Solution
{
public:
    int maxSubArray(vector<int>& as)
    {
        int ans;
        vector<int> dp(as.size());
        ans = dp[0] = as[0];
        for (int i{ 1 }; i < as.size(); ++i)
        {
            ans = std::max(
                ans,
                dp[i] = std::max(dp[i - 1] + as[i], as[i])
            );
        }
        return ans;
    }
};
