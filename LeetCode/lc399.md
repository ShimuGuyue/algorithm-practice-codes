```cpp
struct DisjointSet
{
    std::map<std::string, std::string> fathers;
    std::map<std::string, double> dists;

    std::string find(std::string& x)
    {
        if (fathers[x] == x)
            return x;
        std::string father{ fathers[x] };
        fathers[x] = find(fathers[x]);
        dists[x] *= dists[father];
        return fathers[x];
    }

    void merge(std::string& x, std::string& y, double dist)
    {
        if (!fathers.count(x))
        {
            fathers[x] = x;
            dists[x] = 1;
        }
        if (!fathers.count(y))
        {
            fathers[y] = y;
            dists[y] = 1;
        }
        std::string setx{ find(x) };
        std::string sety{ find(y) };

        fathers[sety] = setx;
        dists[sety] = dists[x] / (dist * dists[y]);
    }

    double query(std::string& x, std::string& y)
    {
        if (!fathers.count(x) || !fathers.count(y))
            return -1;
        if (find(x) != find(y))
            return -1;
        return dists[x] / dists[y];
    }
};
```

```cpp
class Solution
{
public:
    std::vector<double> calcEquation(
        std::vector<std::vector<std::string>>& equations,
        std::vector<double>& values,
        std::vector<std::vector<std::string>>& queries)
    {
        std::vector<double> ans(queries.size());

        DisjointSet ds;
        for (int i{ 0 }; i < equations.size(); ++i)
        {
            ds.merge(equations[i][0], equations[i][1], values[i]);
        }

        for (int i{ 0 }; i < queries.size(); ++i)
        {
            std::string x{ queries[i][0] };
            std::string y{ queries[i][1] };
            ans[i] = ds.query(x, y);
        }
        return ans;
    }
};
```
