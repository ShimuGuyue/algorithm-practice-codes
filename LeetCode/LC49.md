class Solution
{
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs)
    {
        map<int64_t, vector<string>> m;
        for (string& s : strs)
        {
            m[hash(s)].push_back(std::move(s));
        }
        vector<vector<string>> ans;
        for (auto& [x, v] : m)
        {
            ans.push_back(std::move(v));
        }
        return ans;
    }

    int64_t hash(string& s)
    {
        array<int, 27> counts{ };
        for (char c : s)
        {
            ++counts[c - 'a' + 1];
        } 
        int64_t ans{ 0 };
        for (int i{ 1 }; i <= 26; ++i)
        {
            ans = (ans * 27 + counts[i]) % (1 << 20 + 7);
        }
        return ans;
    }
};
