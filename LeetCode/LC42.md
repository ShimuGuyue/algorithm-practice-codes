class Solution
{
public:
    int trap(vector<int>& hs)
    {
        int n = static_cast<int>(hs.size());
        hs.insert(hs.begin(), 0);
        hs.insert(hs.end(), 0);

        vector<int> dpl(n + 2);
        vector<int> dpr(n + 2);
        for (int i{ 1 }; i <= n; ++i)
        {
            dpl[i] = std::max(dpl[i - 1], hs[i]);
        }
        for (int i{ n }; i >= 1; --i)
        {
            dpr[i] = std::max(dpr[i + 1], hs[i]);
        }

        int ans{ 0 };
        for (int i{ 1 }; i <= n; ++i)
        {
            ans += std::min(dpl[i], dpr[i]) - hs[i]; 
        }
        return ans;
    }
};
