class Solution
{
public:
    vector<int> maxSlidingWindow(vector<int>& as, int k)
    {
        vector<int> ans;

        map<int, int> m;
        for (int i{ 0 }; i < k; ++i)
        {
            ++m[as[i]];
        }
        ans.push_back(m.rbegin()->first);

        for (int i{ k }; i < as.size(); ++i)
        {
            ++m[as[i]];
            if (--m[as[i - k]] == 0)
                m.erase(as[i - k]);
            ans.push_back(m.rbegin()->first);
        }

        return ans;
    }
};
