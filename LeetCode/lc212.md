```cpp
class Trie
{
private:
    static constexpr int base{ 26 };    // TODO：设置字符集大小
    struct Node
    {
        std::array<int, base> children{ };
        int count_pass{ 0 };
        int string_id{ -1 };
    };
    std::vector<Node> tree{ Node() };

public:
    Trie() = default;

public:
    void insert(const std::string& s, const int id)
    {
        int node{ 0 };
        ++tree[0].count_pass;
        for (const char c : s)
        {
            const int index{ turn(c) };
            if (tree[node].children[index] == 0)
            {
                tree[node].children[index] = tree.size();
                tree.emplace_back();
            }
            node = tree[node].children[index];
            ++tree[node].count_pass;
        }
        tree[node].string_id = id;
    }

    void erase(const std::string& s)
    {
        int node{ 0 };
        --tree[0].count_pass;
        for (const char c : s)
        {
            const int index{ turn(c) };
            node = tree[node].children[index];
            --tree[node].count_pass;
        }
        tree[node].string_id = -1;
    }

    void search(
        int node,
        int x,
        int y,
        std::vector<std::vector<char>>& board,
        std::vector<std::string>& words,
        std::vector<bool>& flags)
    {
        static constexpr std::array<std::pair<int, int>, 4> dxy{
            {{-1, 0}, {0, -1}, {0, 1}, {1, 0}}
        };

        if (node == 0)
        {
            int index{ turn(board[x][y]) };
            int next = tree[0].children[index];
            if (next == 0 || tree[next].count_pass == 0)
                return;
            char cur{ board[x][y] };
            board[x][y] = 0;
            search(next, x, y, board, words, flags);
            board[x][y] = cur;
        }

        if (tree[node].string_id != -1)
        {
            flags[tree[node].string_id] = true;
            erase(words[tree[node].string_id]);
        }

        for (auto [dx, dy] : dxy)
        {
            int xx{ x + dx };
            int yy{ y + dy };
            if (xx < 0 || xx == board.size())
                continue;
            if (yy < 0 || yy == board[0].size())
                continue;
            if (board[xx][yy] == 0)
                continue;
            int index{ turn(board[xx][yy]) };
            int next{ tree[node].children[index] };
            if (next == 0 || tree[next].count_pass == 0)
                continue;
            char cur{ board[xx][yy] };
            board[xx][yy] = 0;
            search(next, xx, yy, board, words, flags);
            board[xx][yy] = cur;
        }
    }

private:
    static int turn(const char c)
    {
        return 0 + c - 'a';
    }
};
```

```cpp
class Solution
{
public:
    std::vector<std::string> findWords(
        std::vector<std::vector<char>>& board,
        std::vector<std::string>& words)
    {
        Trie trie;
        for (int i{ 0 }; i < words.size(); ++i)
        {
            trie.insert(words[i], i);
        }
        std::vector<bool> flags(words.size());
        for (int i{ 0 }; i < board.size(); ++i)
        {
            for (int j{ 0 }; j < board[0].size(); ++j)
            {
                trie.search(0, i, j, board, words, flags);
            }
        }
        std::vector<std::string> ans;
        for (int i{ 0 }; i < flags.size(); ++i)
        {
            if (flags[i])
                ans.push_back(words[i]);
        }
        return ans;
    }
};
```
