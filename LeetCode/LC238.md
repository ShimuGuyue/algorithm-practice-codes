class Solution
{
public:
    vector<int> productExceptSelf(vector<int>& as)
    {
        vector<int> pres(as.size());
        vector<int> sufs(as.size());
        partial_sum(as.begin(), as.end(), pres.begin(),
            [](int pre, int cur)
            {
                return pre * cur;
            }
        );
        partial_sum(as.rbegin(), as.rend(), sufs.rbegin(),
            [](int pre, int cur)
            {
                return pre * cur;
            }
        );
        vector<int> ans(as.size());
        for (int i{ 0 }; i < as.size(); ++i)
        {
            int pre{ i == 0 ? 1 : pres[i - 1] };
            int suf{ i == as.size() - 1 ? 1 : sufs[i + 1] };
            ans[i] = pre * suf;
        }
        return ans;
    }
};
