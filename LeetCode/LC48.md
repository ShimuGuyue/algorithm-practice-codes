class Solution
{
public:
    void rotate(vector<vector<int>>& matrix)
    {
        int n = static_cast<int>(matrix.size());
        for (int i = 0; i < n / 2; ++i)
        {
            for (int j = i; j < n - 1 - i; ++j)
            {
                std::tie(
                    matrix[i][j],
                    matrix[j][n - 1 - i],
                    matrix[n - 1 - i][n - 1 - j],
                    matrix[n - 1 - j][i]
                ) = std::make_tuple(
                    matrix[n - 1 - j][i],
                    matrix[i][j],
                    matrix[j][n - 1 - i],
                    matrix[n - 1 - i][n - 1 - j]
                );
            }
        }
    }
};
