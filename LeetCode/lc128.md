class Solution
{
public:
    int longestConsecutive(vector<int>& as)
    {
        unordered_set<int> t(as.begin(), as.end());
        int ans{ 0 };
        for (int a : t)
        {
            if (t.contains(a - 1))
                continue;
            int len{ 0 };
            while (t.contains(a))
            {
                ++len;
                ++a;
            }
            ans = std::max(ans, len);
        }
        return ans;
    }
};
