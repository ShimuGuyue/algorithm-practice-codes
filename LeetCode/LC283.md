class Solution
{
public:
    void moveZeroes(vector<int>& as)
    {
        int count{ 0 };
        for (int i{ 0 }; i < as.size(); ++i)
        {
            if (as[i] == 0)
                ++count;
            else
                as[i - count] = as[i];
        }
        for (int i{ 1 }; i <= count; ++i)
        {
            as[as.size() - i] = 0;
        }
    }
};
