class Solution
{
public:
    int firstMissingPositive(vector<int>& as)
    {
        vector<int> flags(as.size() + 2);
        for (int a : as)
        {
            if (a > as.size() || a < 0)
                continue;
            flags[a] = 1;
        }
        for (int i{ 1 }; i <= as.size() + 1; ++i)
        {
            if (flags[i] == 0)
                return i;
        }
        return 0;
    }
};
