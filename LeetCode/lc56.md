class Solution
{
public:
    vector<vector<int>> merge(vector<vector<int>>& ranges)
    {
        sort(ranges.begin(), ranges.end(),
            [](vector<int>& a, vector<int>& b)
            {
                return a[0] < b[0];
            }
        );
        ranges.push_back({ 100000, 100000 });

        vector<vector<int>> ans;
        int l{ ranges[0][0] }, r{ ranges[0][1] };
        for (int i{ 1 }; i < ranges.size(); ++i)
        {
            int ll{ ranges[i][0] };
            int rr{ ranges[i][1] };
            if (ll > r)
            {
                ans.push_back({ l, r });
                l = ll; r = rr;
            }
            else
            {
                r = std::max(r, rr);
            }
        }
        return ans;
    }
};
