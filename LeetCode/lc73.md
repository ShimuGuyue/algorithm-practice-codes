class Solution
{
public:
    void setZeroes(vector<vector<int>>& matrix)
    {
        int n = static_cast<int>(matrix.size());
        int m = static_cast<int>(matrix[0].size());

        bool c{ false }, r{ false };
        for (int i{ 0 }; i < n; ++i)
        {
            if (matrix[i][0] == 0)
                c = true;
        }
        for (int j{ 0 }; j < m; ++j)
        {
            if (matrix[0][j] == 0)
                r = true;
        }
        for (int i{ 1 }; i < n; ++i)
        {
            for (int j{ 1 }; j < m; ++j)
            {
                if (matrix[i][j] == 0)
                    matrix[i][0] = matrix[0][j] = 0;
            }
        }

        if (matrix[0][0] == 0)
            c = r = true;
        for (int i{ 1 }; i < n; ++i)
        {
            if (matrix[i][0] != 0)
                continue;
            for (int j{ 1 }; j < m; ++j)
            {
                matrix[i][j] = 0;
            }
        }
        for (int j{ 1 }; j < m; ++j)
        {
            if (matrix[0][j] != 0)
                continue;
            for (int i{ 1 }; i < n; ++i)
            {
                matrix[i][j] = 0;
            }
        }
        if (c)
        {
            for (int i{ 0 }; i < n; ++i)
            {
                matrix[i][0] = 0;
            }
        }
        if (r)
        {
            for (int j{ 0 }; j < m; ++j)
            {
                matrix[0][j] = 0;
            }
        }
        cout << c << ' ' << r << '\n';
    }
};
