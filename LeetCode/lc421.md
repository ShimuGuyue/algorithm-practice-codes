```cpp
class Trie
{
private:
    static constexpr int base{ 2 };    // TODO：设置字符集大小
    struct Node
    {
        std::array<int, base> children{ };
    };
    std::vector<Node> tree{ Node() };

public:
    Trie() = default;

public:
    void insert(int x)
    {
        int node{ 0 };
        for (int i{ 30 }; i >= 0; --i)
        {
            int index{ x >> i & 1 };
            if (tree[node].children[index] == 0)
            {
                tree[node].children[index] = tree.size();
                tree.emplace_back();   
            }
            node = tree[node].children[index];
        }
    }

    int max_xor(int x)
    {
        int ans{ 0 };
        int node{ 0 };
        for (int i{ 30 }; i >= 0; --i)
        {
            int index{ x >> i & 1 };
            if (tree[node].children[index ^ 1] != 0)
            {
                ans |= 1 << i;
                node = tree[node].children[index ^ 1];
            }
            else
            {
                node = tree[node].children[index];
            }
        }
        return ans;
    }
};
```

```cpp
class Solution
{
public:
    int findMaximumXOR(std::vector<int>& nums)
    {
        Trie trie;
        for (int a : nums)
        {
            trie.insert(a);
        }
        int ans{ 0 };
        for (int a : nums)
        {
            ans = std::max(ans, trie.max_xor(a));
        }
        return ans;
    }
};
```
