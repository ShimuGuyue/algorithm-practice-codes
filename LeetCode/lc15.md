class Solution
{
public:
    vector<vector<int>> threeSum(vector<int>& as)
    {
        sort(as.begin(), as.end());
        vector<int> bs;
        for (int a : as)
        {
            if (bs.size() >= 3
            &&  bs[bs.size() - 1] == a
            &&  bs[bs.size() - 2] == a
            &&  bs[bs.size() - 3] == a)
                continue;
            bs.push_back(a);
        }
        as = std::move(bs);

        unordered_map<int, int> m;
        for (int a : as)
        {
            ++m[a];
        }

        vector<vector<int>> ans;
        for (int i{ 0 }; i < as.size(); ++i)
        {
            int a{ as[i] };
            --m[a];
            for (int j{ i + 1 }; j < as.size(); ++j)
            {
                int b{ as[j] };
                --m[b];
                int c{ 0 - a - b };
                if (m[c])
                {
                    ans.push_back({ a, b, c });
                }
                ++m[b];
            }
        }
        for (auto& v : ans)
        {
            sort(v.begin(), v.end());
        }
        sort(ans.begin(), ans.end());
        ans.erase(unique(ans.begin(), ans.end()), ans.end());
        return ans;
    }
};
