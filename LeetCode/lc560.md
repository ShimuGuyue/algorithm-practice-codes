class Solution
{
public:
    int subarraySum(std::vector<int>& nums, int k)
    {
        int64_t presum{ 0 };
        int64_t ans{ 0 };
        std::map<int64_t, int> m;
        for (int a : nums)
        {
            presum += a;
            if (presum == k)
                ++ans;
            ans += m[presum - k];
            ++m[presum];
        }
        return ans;
    }
};
