class Solution
{
public:
    int subarraySum(vector<int>& as, int k)
    {
        vector<int> pres(as.size());
        partial_sum(as.begin(), as.end(), pres.begin());

        int ans{ 0 };
        map<int, int> m;
        ++m[0];
        for (int a : pres)
        {
            ans += m[a - k];
            ++m[a];
        }
        return ans;
    }
};
