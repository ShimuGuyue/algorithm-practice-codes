class NumArray
{
public:
    std::vector<int> presums;

    NumArray(std::vector<int>& nums)
    {
        presums.resize(nums.size());
        std::partial_sum(nums.begin(), nums.end(), presums.begin());
    }
    
    int sumRange(int left, int right)
    {
        return presums[right] - (left ? presums[left - 1] : 0);    
    }
};
