class Solution
{
public:
    int minSubarray(std::vector<int>& nums, int p)
    {
        int sum{
            std::accumulate(nums.begin(), nums.end(), 0,
                [p](int pre, int cur)
                {
                    return (pre + cur) % p;
                }
            )
        };

        if (sum == 0)
            return 0;

        int ans{ int(nums.size()) };
        std::map<int, int> last_indexs;
        last_indexs[0] = -1;
        int pre{ 0 };
        for (int i{ 0 }; i < nums.size(); ++i)
        {
            pre = (pre + nums[i]) % p;
            int target{ (pre - sum + p) % p };
            if (last_indexs.count(target))
                ans = std::min(ans, i - last_indexs[target]);
            last_indexs[pre] = i;
        }
        return ans == nums.size() ? -1 : ans;
    }
};
