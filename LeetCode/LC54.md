class Solution
{
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix)
    {
        int n = static_cast<int>(matrix.size());
        int m = static_cast<int>(matrix[0].size());

        vector<int> ans;
        int x{ 0 }, y{ 0 };
        int mode{ 0 };
        for (int i{ 0 }; i < n * m; ++i)
        {
            ans.push_back(matrix[x][y]);
            matrix[x][y] = 114514;
            switch (mode)
            {
            case 0:
                if (y + 1 == m || matrix[x][y + 1] == 114514)
                {
                    mode = 1;
                    ++x;
                }
                else
                {
                    ++y;
                }
                break;
            case 1:
                if (x + 1 == n || matrix[x + 1][y] == 114514)
                {
                    mode = 2;
                    --y;
                }
                else
                {
                    ++x;
                }
                break;
            case 2:
                if (y - 1 == -1 || matrix[x][y - 1] == 114514)
                {
                    mode = 3;
                    --x;
                }
                else
                {
                    --y;
                }
                break;
            case 3:
                if (x - 1 == -1 || matrix[x - 1][y] == 114514)
                {
                    mode = 0;
                    ++y;
                }
                else
                {
                    --x;
                }
                break;
            }
        }
        return ans;
    }
};
