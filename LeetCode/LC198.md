class Solution
{
public:
    int rob(std::vector<int>& as)
    {
        int n{ int(as.size()) };

        std::vector<int> bs(n + 1);
        for (int i{ 1 }; i <= n; ++i)
        {
            bs[i] = as[i - 1];
        }

        std::vector<std::vector<int>> dp(n + 1, std::vector<int>(2));
        for (int i{ 1 }; i <= n; ++i)
        {
            dp[i][0] = std::max(dp[i - 1][0], dp[i - 1][1]);
            dp[i][1] = dp[i - 1][0] + bs[i];
        }
        return std::max(dp[n][0], dp[n][1]);
    }
};
