class Solution
{
public:
    string minWindow(string , string t)
    {
        array<int, 128> count0{ };
        array<int, 128> count1{ };
        for (char c : t)
        {
            ++count0[c];
        }

        int l{ 0 }, r{ 0 };
        while (!match(count0, count1) && r < .length())
        {
            ++count1[[r]];
            ++r;
        }
        if (!match(count0, count1))
            return "";

        int ll{ l }, rr{ r - 1 };
        while (count1[[l]] > count0[[l]])
        {
            --count1[[l]];
            ++l;
            ++ll;
        }
        while (r < .length())
        {
            ++count1[[r]];
            while (count1[[l]] > count0[[l]])
            {
                --count1[[l]];
                ++l;
            }
            if (r - l < rr - ll)
            {
                ll = l; rr = r;
            }
            ++r;
        }
        return string(.begin() + ll, .begin() + rr + 1);
    }

    bool match(array<int, 128>& count0, array<int, 128>& count1)
    {
        for (char i{ 'a' }; i <= 'z'; ++i)
        {
            if (count1[i] < count0[i])
                return false;
        }
        for (char i{ 'A' }; i <= 'Z'; ++i)
        {
            if (count1[i] < count0[i])
                return false;
        }
        return true;
    }
};
