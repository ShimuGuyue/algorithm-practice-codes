class Solution
{
public:
    int maxArea(vector<int>& as)
    {
        int n = static_cast<int>(as.size());
        int ans{ 0 };
        int l{ 0 }, r{ n - 1 };
        while (l < r)
        {
            int len{ r - l };
            ans = std::max(
                ans,
                len * std::min(as[l], as[r])
            );
            if (as[l] < as[r])
                ++l;
            else
                --r;
        }
        return ans;
    }
};
