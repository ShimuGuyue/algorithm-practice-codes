class Solution
{
public:
    int longestWPI(std::vector<int>& hours)
    {
        std::map<int, int> m;
        m[0] = -1;

        int ans{ 0 };
        int sum{ 0 };
        for (int i{ 0 }; i < int(hours.size()); ++i)
        {
            sum += hours[i] > 8 ? 1 : -1;
            if (sum > 0)
                ans = i + 1;
            else
                if (m.count(sum - 1))
                    ans = std::max(ans, i - m[sum - 1]);

            if (!m.count(sum))
                m[sum] = i;
        }
        return ans;
    }
};
