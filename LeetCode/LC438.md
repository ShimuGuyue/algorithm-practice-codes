class Solution
{
public:
    vector<int> findAnagrams(string , string p)
    {
        if (.length() < p.length())
            return { };

        array<int, 26> counts0{ };
        array<int, 26> counts1{ };
        for (char c : p)
        {
            ++counts0[turn(c)];
        }

        vector<int> ans;

        int n = static_cast<int>(p.length());
        for (int i{ 0 }; i < n; ++i)
        {
            ++counts1[turn([i])];
        }
        if (match(counts0, counts1))
            ans.push_back(0);
        for (int i{ n }; i < .length(); ++i)
        {
            ++counts1[turn([i])];
            --counts1[turn([i - n])];
            if (match(counts0, counts1))
                ans.push_back(i - n + 1);
        }
        return ans;
    }

    int turn(char c)
    {
        return c - 'a';
    }

    bool match(array<int, 26>& a, array<int, 26>& b)
    {
        for (int i{ 0 }; i < 26; ++i)
        {
            if (a[i] != b[i])
                return false;
        }
        return true;
    }
};
