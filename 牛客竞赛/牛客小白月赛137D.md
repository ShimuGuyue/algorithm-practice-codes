void solve()
{
    int n;
    cin >> n;
    if (n < 6)
    {
        cout << "No\n";
        return;
    }

    cout << "Yes\n";

    vector<int> ans;
    ans.reserve(n);
    ans.push_back(3);
    ans.push_back(6);
    ans.push_back(2);
    ans.push_back(4);
    for (int i = 8; i <= n; i += 2)
    {
        ans.push_back(i);
    }
    for (int i = 1; i <= n; i += 2)
    {
        if (i != 3)
            ans.push_back(i);
    }

    for (auto a : ans)
    {
        cout << a << ' ';
    }
    cout << '\n';
    reverse(ans.begin(), ans.end());
    for (auto a : ans)
    {
        cout << a << ' ';
    }
    cout << '\n';
}
