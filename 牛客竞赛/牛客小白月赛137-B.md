void solve()
{
    int n, k;
    cin >> n >> k;
    vector<int> as(k);
    for (auto& a : as)
    {
        cin >> a;
    }

    int ans = std::max(as.front() - 1, n - as.back());
    for (int i = 1; i < k; ++i)
    {
        ans = std::max(ans, (as[i] - as[i - 1]) / 2);
    }
    cout << ans << '\n';
}
