void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    vector<int> bs(n);
    vector<int> cs(n);
    vector<int> ds(n);
    for (auto& a : as)
    {
        cin >> a;
    }
    for (auto& b : bs)
    {
        cin >> b;
    }
    for (auto& c : cs)
    {
        cin >> c;
    }
    for (auto& d : ds)
    {
        cin >> d;
    }

    vector<pair<int, int>> abs(n);
    vector<pair<int, int>> cds(n);
    for (int i = 0; i < n; ++i)
    {
        abs[i] = std::minmax(as[i], bs[i]);
        cds[i] = std::minmax(cs[i], ds[i]);
    }
    sort(abs.begin(), abs.end());
    sort(cds.begin(), cds.end());
    if (abs != cds)
    {
        cout << "No\n";
        return;
    }

    int count = 0;
    set<pair<int, int>> t;
    for (int i = 0; i < n; ++i)
    {
        t.insert({ as[i], bs[i] });
    }
    for (int i = 0; i < n; ++i)
    {
        if (!t.contains({ cs[i], ds[i] }))
            ++count;
    }
    cout << (count & 1 ? "No" : "Yes") << '\n';
}
