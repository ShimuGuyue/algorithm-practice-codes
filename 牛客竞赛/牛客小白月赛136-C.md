void solve()
{
    int64_t n, a, b, c, d;
    cin >> n >> a >> b >> c >> d;
    string s;
    cin >> s;

    int64_t ans1;
    string ans2;

    for (char ch : s)
    {
        if (ch == 'L')
        {
            if (a < c)
            {
                --a;
                ans2 += 'A';
            }
            else
            {
                --c;
                ans2 += 'B';
            }
        }
        if (ch == 'R')
        {
            if (a > c)
            {
                ++a;
                ans2 += 'A';
            }
            else
            {
                ++c;
                ans2 += 'B';
            }
        }
        if (ch == 'U')
        {
            if (b > d)
            {
                ++b;
                ans2 += 'A';
            }
            else
            {
                ++d;
                ans2 += 'B';
            }
        }
        if (ch == 'D')
        {
            if (b < d)
            {
                --b;
                ans2 += 'A';
            }
            else
            {
                --d;
                ans2 += 'B';
            }
        }
    }
    ans1 = llabs(a - c) + llabs(b - d);

    cout << ans1 << '\n';
    cout << ans2 << '\n';
}
