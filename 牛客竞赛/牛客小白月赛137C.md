void solve()
{
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<pair<int, int>> xys(k);
    for (auto& [x, y] : xys)
    {
        cin >> x >> y;
    }

    s = " " + s;

    vector<int> pres(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        pres[i] = pres[i - 1] + (s[i] == 'N');
    }

    int xx = 0, yy = 0;
    for (int i = 1; i <= n; ++i)
    {
        if (s[i] == 'L')
            --yy;
        else if (s[i] == 'R')
            ++yy;
        else if (s[i] == 'U')
            --xx;
        else if (s[i] == 'D')
            ++xx;

        int min = 1 << 30;
        for (auto [x, y] : xys)
        {
            min = std::min(min, abs(x - xx) + abs(y - yy));
        }
        if (min <= pres[i])
        {
            cout << "Yes\n";
            return;
        }
    }
    cout << "No\n";
}
