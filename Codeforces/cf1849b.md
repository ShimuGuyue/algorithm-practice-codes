void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<std::pair<int, int>> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i].first;
        as[i].first %= k;
        if (as[i].first == 0)
            as[i].first = k;
        as[i].second = i;
    }

    std::sort(as.begin() + 1, as.end(),
        [](std::pair<int, int> a, std::pair<int, int> b)
        {
            if (a.first != b.first)
                return a.first > b.first;
            return a.second < b.second;
        }
    );

    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << as[i].second << ' ';
    }
    std::cout << '\n';
}
