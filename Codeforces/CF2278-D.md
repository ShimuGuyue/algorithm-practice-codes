void solve()
{
    int n;
    cin >> n;
    map<int64_t, int> m;
    for (int i = 0; i < n; ++i)
    {
        int a;
        cin >> a;
        ++m[a];
    }

    int64_t sum = 0;
    for (auto [k, v] : m)
    {
        sum += k * v;
    }

    int index;
    int64_t ans1 = int64_t{1} << 60;

    int64_t lower_count = 0;
    int64_t lower_sum = 0;
    for (auto [k, v] : m)
    {
        int64_t greater_count = n - lower_count - v;
        int64_t greater_sum = sum - lower_sum - k * v;
        int64_t cur = (k * lower_count - lower_sum)
                    + (greater_sum - k * greater_count);
        if (cur < ans1)
        {
            ans1 = cur;
            index = k;
        }
        lower_count += v;
        lower_sum += k * v;
    }

    int ans2 = (index - m.begin()->first)
                + (m.rbegin()->first - index);
    cout << ans1 << ' ' << ans2 << '\n';
}
