void solve()
{
    int n, m, x, y;
    std::cin >> n >> m >> x >> y;
    std::vector<std::string> grid(n);
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    std::vector<int> as(m + 1);
    for (int i{ 1 }; i <= m; ++i)
    {
        for (int j{ 0 }; j < n; ++j)
        {
            as[i] += grid[j][i - 1] == '#';
        }
    }

    std::vector<int> presums(m + 1);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    std::vector<std::vector<int>> dp(m + 1, std::vector<int>(2, 1000000));
    dp[0][0] = 0;
    dp[0][1] = 0;
    dp[x][0] = presums[x];
    dp[x][1] = n * x - presums[x];
    for (int i{ x + 1 }; i <= m; ++i)
    {
        bool ok{ true };
        for (int j{ i - x }; j >= std::max(0, i - y); --j)
        {
            dp[i][0] = std::min(dp[i][0], dp[j][1] + presums[i] - presums[j]);
            dp[i][1] = std::min(dp[i][1], dp[j][0] + (n * (i - j) - (presums[i] - presums[j])));
        }
    }

    int ans{ std::min(dp[m][0], dp[m][1]) };
    std::cout << ans << '\n';
}
