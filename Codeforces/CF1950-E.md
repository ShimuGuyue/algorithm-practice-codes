void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int ans{ n };
    auto check = [n, &s](int k)
    {
        int dif{ 0 };
        for (int i{ 0 }; i < k; ++i)
        {
            std::map<char, int> m;
            for (int j{ i }; j < n; j += k)
            {
                ++m[s[j]];
            }
            int max{ 0 };
            for (auto [a, b] : m)
            {
                max = std::max(b, max);
            }
            dif += n / k - max;
        }
        return dif <= 1;
    };
    for (int i{ 1 }; i * i <= n; ++i)
    {
        if (n % i)
            continue;
        if (check(i))
            ans = std::min(ans, i);
        if (check(n / i))
            ans = std::min(ans, n / i);
    }
    std::cout << ans << '\n';
}
