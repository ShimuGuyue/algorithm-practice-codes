void solve()
{
    int n;
    std::cin >> n;
    std::multiset<int> t;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        if (a != 0)
            t.insert(a);
    }
    while (--n)
    {
        auto i = *t.rbegin();
        t.erase(t.find(i));
        if (i - 1 != 0)
            t.insert(i - 1);
    }
    int ans = t.size();
    std::cout << ans << '\n';
}
