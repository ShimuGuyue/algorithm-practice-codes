void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    for (int k{ n / 2 }; k > 0; --k)
    {
        std::vector<std::pair<int, int>> indexs(n + 1, { -1, -1 });
        std::vector<int> counts(n + 1);
        int sum{ 0 };
        std::deque<int> q_max;
        std::deque<int> q_min;
        for (int i{ 0 }; i < k; ++i)
        {
            ++counts[as[i]];
            if (counts[as[i]] == 1)
                ++sum;
            while (!q_max.empty() && as[i] >= as[q_max.back()])
            {
                q_max.pop_back();
            }
            q_max.push_back(i);
            while (!q_min.empty() && as[i] <= as[q_min.back()])
            {
                q_min.pop_back();
            }
            q_min.push_back(i);
        }
        if (sum == k && as[q_max.front()] - as[q_min.front()] + 1 == k)
            indexs[as[q_min.front()]] = { 0, 0 };
        for (int i{ k }; i < n; ++i)
        {
            if (counts[as[i - k]] == 1)
                --sum;
            --counts[as[i - k]];
            while (!q_max.empty() && q_max.front() < i - k + 1)
            {
                q_max.pop_front();
            }
            while (!q_min.empty() && q_min.front() < i - k + 1)
            {
                q_min.pop_front();
            }
            ++counts[as[i]];
            if (counts[as[i]] == 1)
                ++sum;
            while (!q_max.empty() && as[i] >= as[q_max.back()])
            {
                q_max.pop_back();
            }
            q_max.push_back(i);
            while (!q_min.empty() && as[i] <= as[q_min.back()])
            {
                q_min.pop_back();
            }
            q_min.push_back(i);
            if (sum == k && as[q_max.front()] - as[q_min.front()] + 1 == k)
            {
                if (indexs[as[q_min.front()]].first == -1)
                    indexs[as[q_min.front()]].first = i - k + 1;
                indexs[as[q_min.front()]].second = i - k + 1;
            }
        }
        for (int i{ 1 }; i + k <= n; ++i)
        {
            if (indexs[i].first == -1 || indexs[i + k].first == -1)
                continue;
            if (indexs[i].first + k <= indexs[i + k].second || indexs[i + k].first + k <= indexs[i].second)
            {
                std::cout << k << '\n';
                return;
            }
        }
    }
    std::cout << 0 << '\n';
}
