void solve()
{
    int n, x1, y1, x2, y2;
    std::cin >> n >> x1 >> y1 >> x2 >> y2;
    std::vector<std::pair<int, int>> locs(n);
    for (auto& [a, b] : locs)
    {
        std::cin >> a;
    }
    for (auto& [a, b] : locs)
    {
        std::cin >> b;
    }

    std::map<int, std::pair<int, int>> m;   // min - max
    for (auto [a, b] : locs)
    {
        if (m[a] == std::make_pair(0, 0))
            m[a] = {b, b};
        else
        {
            m[a].first = std::min(m[a].first, b);
            m[a].second = std::max(m[a].second, b);
        }
    }
    m[x1] = {y1, y1};
    m[x2] = {y2, y2};

    std::vector<std::pair<int, std::pair<int, int>>> v(m.begin(), m.end());
    int64_t ans{ x2 - x1 };
    std::vector<std::pair<int64_t, int64_t>> dp(v.size());
    for (int i = 1; i < v.size(); ++i)
    {
        auto [min1, max1]{ v[i - 1].second };
        auto [min2, max2]{ v[i].second };
        int len{ max2 - min2 };
        dp[i].first = std::min(
            std::abs(max2 - max1) + dp[i - 1].second,
            std::abs(max2 - min1) + dp[i - 1].first
        ) + len;
        dp[i].second = std::min(
            std::abs(min2 - max1) + dp[i - 1].second,
            std::abs(min2 - min1) + dp[i - 1].first
        ) + len;
    }
    ans += dp.back().first;
    std::cout << ans << '\n';
}
