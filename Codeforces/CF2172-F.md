template <typename T>
class ST
{
private:
    std::vector<std::vector<T>> st{};

public:
    ST()
    {}
    ST(std::vector<T> &v)
    {
        build(v);
    }

private:
    // 此函数规定贡献方式
    T merge(T a, T b)
    {
        return std::gcd(a, b);
    }

public:
    void build(std::vector<T> &v)
    {
        size_t n{ v.size() };
        size_t m { int(log2(n)) + 1 };
        st.assign(n, std::vector<T>(m));

        for (int i{ 0 }; i < n; ++i)
        {
            st[i][0] = v[i];
        }
        for (int j{ 1 }; j < m; ++j)
        {
            int len{ 1 << (j - 1) };
            for (int i{ 0 }; i < n - (1 << j) + 1; ++i)
            {
                st[i][j] = merge(st[i][j - 1], st[i + len][j - 1]);
            }
        }
    }

    T query(int l, int r)
    {
        int power{ int(log2(r - l + 1)) };
        int len{ 1 << power };
        return merge(st[l][power], st[r - len + 1][power]);
    }
};

void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }

    ST st(as);

    int64_t ans{ st.query(0, n - 1) };
    int l{ 0 }, r{ n - 1 };
    while (l + 1 != r)
    {
        if (st.query(0, r - 1) < st.query(l + 1, n - 1))
        {
            ans += st.query(0, r - 1);
            --r;
        }
        else
        {
            ans += st.query(l + 1, n - 1);
            ++l;
        }
        debug_(l, r, ans);
    }
    std::cout << ans << '\n';
}
