void solve()
{
    int n, c;
    std::cin >> n >> c;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::pair<int, int>> costs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        costs[i] = {as[i] + std::min(i - 0, n + 1 - i), i};
    }
    std::sort(costs.begin() + 1, costs.end());

    std::vector<int64_t> presums(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        presums[i] = presums[i - 1] + costs[i].first;
    }

    std::vector<int> indexs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        indexs[costs[i].second] = i;
    }

    int ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] + i > c)
            continue;
        auto it{ std::prev(std::upper_bound(presums.begin(), presums.end(), c - (as[i] + i))) };
        int index{ it - presums.begin() };
        if (indexs[i] > index)
        {
            ans = std::max(ans, index + 1);
        }
        else
        {
            int64_t remain{ c - *it + costs[indexs[i]].first - (as[i] + i) };
            if (index == n)
                ans = std::max(ans, index);
            else
                if (remain >= costs[index + 1].first)
                    ans = std::max(ans, index + 1);
                else
                    ans = std::max(ans, index);
        }
    }
    std::cout << ans << '\n';
}
