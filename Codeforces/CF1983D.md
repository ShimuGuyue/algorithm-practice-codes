void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    if (!judge(as, bs))
    {
        std::cout << "No\n";
        return;
    }

    if (calc(as, 0, n - 1) % 2 == calc(bs, 0, n - 1) % 2)
        std::cout << "Yes\n";
    else
        std::cout << "No\n";
}

bool judge(std::vector<int> as, std::vector<int> bs)
{
    std::sort(as.begin(), as.end());
    std::sort(bs.begin(), bs.end());
    return as == bs;
}

int64_t calc(std::vector<int>& v, int l, int r)
{
    if (l == r)
        return 0;
    int64_t ans{ 0 };
    int mid{ (l + r) / 2 };
    ans += calc(v, l, mid) + calc(v, mid + 1, r);

    static std::vector<int> temp(100000 + 1);
    int i{ l }, j{ mid + 1 }, k{ l };
    while (i <= mid && j <= r)
    {
        if (v[j] < v[i])
        {
            ans += mid - i + 1;
            temp[k++] = v[j++];
        }
        else
        {
            temp[k++] = v[i++];
        }
    }
    while (i <= mid)
    {
        temp[k++] = v[i++];
    }
    while (j <= r)
    {
        temp[k++] = v[j++];
    }
    k = l;
    while (k <= r)
    {
        v[k] = temp[k];
        ++k;
    }
    return ans;
}
