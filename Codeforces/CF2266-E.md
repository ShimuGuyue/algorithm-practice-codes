```cpp
vector<vector<int>> factors;
void init()
{
    int n = 200000;
    factors.resize(n + 1);
    for (int i = 2; i <= n; ++i)
    {
        if (factors[i].size())
            continue;
        for (int j = i; j <= n; j += i)
        {
            factors[j].push_back(i);
        }
    }
}
```

```cpp
void solve()
{
    int n, k;
    cin >> n >> k;
    vector<int> counts(n + 1);
    for (int i = 0; i < n; ++i)
    {
        int a;
        cin >> a;
        ++counts[a];
    }

    int64_t ans = 0;
    vector<int64_t> dp(n + 1, INT64_C(1) << 60);
    for (int i = 0; i <= k; ++i)
    {
        dp[i] = 0;
    }
    for (int i = k + 1; i <= n; ++i)
    {
        for (int f : factors[i])
        {
            dp[i] = std::min(dp[i], f * dp[i / f] + 1);
        }
        ans += dp[i] * counts[i];
    }
    cout << ans << '\n';
}
```
