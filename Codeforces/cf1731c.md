```cpp
std::vector<int> flags;
void init()
{
    for (int i{ 0 }; i * i <= (1 << 19); ++i)
    {
        flags.push_back(i * i);
    }
}
```

```cpp
void solve(int time)
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t ans{ 0 };
    static std::vector<std::pair<int, int>> m((1 << 19));
    m[0] = {1, time};
    int pre{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        ans += i + 1;
        pre ^= as[i];
        for (auto f : flags)
        {
            if (m[pre ^ f].second == time)
                ans -= m[pre ^ f].first;
        }
        if (m[pre].second != time)
            m[pre] = {0, time};
        ++m[pre].first;
    }
    std::cout << ans << '\n';
}
```
