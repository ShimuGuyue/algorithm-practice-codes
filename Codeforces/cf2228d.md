void solve()
{
    int n;
    std::cin >> n;

    std::vector<std::vector<int>> nodes(n + 2);
    std::vector<int> cnt_r(n + 2, 0);
    std::vector<int> flags(n + 2, 0);

    for (int i = 0; i < n; ++i)
    {
        int x, y;
        std::cin >> x >> y;
        nodes[x].push_back(y);
        cnt_r[y]++;
        flags[y] = 1;
    }

    std::vector<int> pres(n + 2, 0);
    std::partial_sum(flags.begin(), flags.end(), pres.begin());

    std::vector<int> list_l(n + 2, 0), list_r(n + 2, 0);
    int head = 0, tail = n + 1;
    int pre = head;
    for (int y = 1; y <= n; ++y)
    {
        if (cnt_r[y] == 0)
            continue;
        list_r[pre] = y;
        list_l[y] = pre;
        pre = y;
    }
    list_r[pre] = tail;
    list_l[tail] = pre;

    int min_l = n + 2;
    int max_l = -1;

    int64_t ans{ 0 };

    for (int i = 1; i <= n; ++i)
    {
        if (nodes[i].empty())
            continue;

        for (int j : nodes[i])
        {
            min_l = std::min(min_l, j);
            max_l = std::max(max_l, j);

            cnt_r[j]--;
            if (cnt_r[j] == 0)
            {
                int prv = list_l[j];
                int nxt = list_r[j];
                list_r[prv] = nxt;
                list_l[nxt] = prv;
            }
        }

        int min_r = list_r[head];
        int max_r = list_l[tail];

        if (max_l == -1 || min_r == tail)
            continue;

        int l = std::max(min_l, min_r);
        int r = std::min(max_l, max_r);

        if (l <= r)
        {
            int cnt = pres[r] - pres[l - 1] - 1;
            if (cnt > 0)
            {
                ans += cnt;
            }
        }
    }
    std::cout << ans << '\n';
}
