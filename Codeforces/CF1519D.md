void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    std::vector<int64_t> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    int64_t sum{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        sum += as[i] * bs[i];
    }

    int64_t ans{ sum };
    for (int i{ 0 }; i < n; ++i)
    {
        int64_t cur{ sum };
        int l{ i }, r{ i };
        while (l >= 0 && r < n)
        {
            cur -= as[l] * bs[l] + as[r] * bs[r];
            cur += as[r] * bs[l] + as[l] * bs[r];
            ans = std::max(ans, cur);
            --l; ++r;
        }
    }
    for (int i{ 1 }; i < n; ++i)
    {
        int64_t cur{ sum };
        int l{ i - 1 }, r{ i };
        while (l >= 0 && r < n)
        {
            cur -= as[l] * bs[l] + as[r] * bs[r];
            cur += as[r] * bs[l] + as[l] * bs[r];
            ans = std::max(ans, cur);
            --l; ++r;
        }
    }
    std::cout << ans << '\n';
}
