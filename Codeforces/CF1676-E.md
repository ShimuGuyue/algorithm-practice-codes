void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end(), std::greater<int>());

    std::vector<int> presums(n);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    while (q--)
    {
        int x;
        std::cin >> x;
        auto it{ std::lower_bound(presums.begin(), presums.end(), x) };
        if (it == presums.end())
            std::cout << -1 << '\n';
        else
            std::cout << it - presums.begin() + 1 << '\n';
    }
}
