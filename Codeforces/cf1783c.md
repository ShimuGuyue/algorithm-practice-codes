void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> init{ as };

    std::sort(as.begin(), as.end());

    std::vector<int> pres(n);
    std::partial_sum(as.begin(), as.end(), pres.begin());

    auto check = [&init, &as, &pres, m](int k) -> bool
    {
        if (pres[k - 1] <= m)
            return true;

        if (k >= 2 && pres[k - 2] <= m)
        {
            if (init[k - 1] <= as[k - 2])
                return true;
            if (pres[k - 2] - as[k - 2] + init[k - 1] <= m)
                return true;
        }

        return false;
    };
    int l{ 0 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    int ans{ n + 1 - l };
    std::cout << ans << '\n';
}
