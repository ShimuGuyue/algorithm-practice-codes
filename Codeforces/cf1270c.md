void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t sum_add{ std::accumulate(as.begin(), as.end(), INT64_C(0)) };
    int64_t sum_xor{ std::accumulate(as.begin(), as.end(), INT64_C(0), std::bit_xor<>()) };

    if (sum_xor == 0)
    {
        std::cout << 1 << '\n';
        std::cout << sum_add << '\n';
    }
    else
    {
        std::cout << 2 << '\n';
        std::cout << sum_xor << ' ' << sum_add + sum_xor << '\n';
    }
}
