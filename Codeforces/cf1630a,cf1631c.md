void solve()
{
    int n, k;
    std::cin >> n >> k;

    if (k == 0)
    {
        for (int i = 0; i < n - 1 - i; ++i)
        {
            std::cout << i << ' ' << n - 1 - i << '\n';
        }
        return;
    }

    if (k == n - 1)
    {
        if (n == 4)
        {
            std::cout << -1 << '\n';
            return;
        }
        std::cout << n - 2 << ' ' << n - 1 << '\n';
        std::cout << 1 << ' ' << n - 3 << '\n';
        std::cout << 0 << ' ' << 2 << '\n';
        for (int i = 3; i < n - 1 - i; ++i)
        {
            std::cout << i << ' ' << n - 1 - i << '\n';
        }
        return;
    }

    std::cout << k << ' ' << n - 1 - 0 << '\n';
    std::cout << 0 << ' ' << n - 1 - k << '\n';
    for (int i = 1; i < n - 1 - i; ++i)
    {
        if (i == k || i == n - 1 - k)
            continue;
        std::cout << i << ' ' << n - 1 - i << '\n';
    }
}
