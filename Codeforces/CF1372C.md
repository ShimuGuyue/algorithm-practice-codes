void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    if (std::is_sorted(as.begin(), as.end()))
    {
        std::cout << 0 << '\n';
        return;
    }

    int l{ 1 }, r{ n };
    while (l <= n && as[l] == l)
    {
        ++l;
    }
    while (r > 0 && as[r] == r)
    {
        --r;
    }

    int ans{ 1 };
    for (int i{ l }; i <= r; ++i)
    {
        if (as[i] == i)
            ans = 2;
    }
    std::cout << ans << '\n';
}
