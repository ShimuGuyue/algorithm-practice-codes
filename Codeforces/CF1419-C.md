void solve()
{
    int n, x;
    std::cin >> n >> x;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (std::count(as.begin(), as.end(), x) == n)
    {
        std::cout << 0 << '\n';
        return;
    }

    int sum{ std::accumulate(as.begin(), as.end(), 0) };
    bool found{ std::find(as.begin(), as.end(), x) != as.end() };
    if (sum == n * x || found)
        std::cout << 1 << '\n';
    else
        std::cout << 2 << '\n';
}
