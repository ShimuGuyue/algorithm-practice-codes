void solve()
{
    int p, q;
    std::cin >> p >> q;

    /*
        n(m + 1) + m(n + 1) = p + 2q;
        2nm + n + m = p + 2q;
        4nm + 2n + 2m + 1 = 2p + 4q + 1;
        (2n + 1)(2m + 1) = 2p + 4q + 1;

        x = 2n + 1, y = 2m + 1;
        x * y = 2p + 4q + 1;(x, y >= 3, odd)
    */
    int s{ 2 * p + 4 * q + 1 };
    for (int x{ 3 }; x * x <= s; x += 2)
    {
        if (s % x)
            continue;
        int y{ s / x };
        if (y % 2 == 0)
            continue;
        if (y < 3)
            continue;
        int n{ x / 2 };
        int m{ y / 2 };
        if (n > m)
            std::swap(n, m);
        if (q > n * (m + 1))
            continue;
        std::cout << n << ' ' << m << '\n';
        return;
    }

    std::cout << -1 << '\n';
}
