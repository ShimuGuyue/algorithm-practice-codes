void solve()
{
    int n;
    std::cin >> n;

    std::vector<std::pair<int, int>> ans;
    std::vector<bool> visited(n + 1);
    auto dfs = [&ans, &visited](this auto&& dfs, int u, int v) -> void
    {
        visited[v] = true;
        std::cout << "? " << u << ' ' << v << std::endl;
        int x;
        std::cin >> x;
        if (x == u)
        {
            ans.push_back({ u, v });
            return;
        }
        if (!visited[x])
            dfs(u, x);
        dfs(x, v);
    };
    for (int i{ 2 }; i <= n; ++i)
    {
        if (!visited[i])
            dfs(1, i);
    }

    std::cout << "! ";
    for (auto [u, v] : ans)
    {
        std::cout << u << ' ' << v << ' ';
    }
    std::cout << std::endl;
}
