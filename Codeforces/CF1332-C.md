void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    std::vector<std::map<char, int>> flags((k + 1) / 2);
    for (int i{ 0 }; i < n; ++i)
    {
        int loc{ i % k };
        if (loc >= (k + 1) / 2)
            loc = k - loc - 1;
        ++flags[loc][s[i]];
    }

    int ans{ 0 };
    for (int i{ 0 }; i < (k + 1) / 2; ++i)
    {
        int max{ 0 };
        for (auto [a, b] : flags[i])
        {
            max = std::max(max, b);
        }
        if (i + i + 1 == k)
            ans += n / k - max;
        else
            ans += n / k * 2 - max;
    }
    std::cout << ans << '\n';
}
