void solve()
{
    std::string s;
    std::cin >> s;
    int x;
    std::cin >> x;

    int n{ int(s.length()) };

    std::vector<int> ans(n, 1);
    for (int i{ 0 }; i < n; ++i)
    {
        if (s[i] == '1')
            continue;
        if (i - x >= 0)
            ans[i - x] = 0;
        if (i + x < n)
            ans[i + x] = 0;
    }
    for (int i = 0; i < n; ++i)
    {
        if (s[i] == '0')
            continue;
        bool b{ false };
        if (i - x >= 0 && ans[i - x] == 1)
            b = true;
        if (i + x < n && ans[i + x] == 1)
            b = true;
        if (!b)
        {
            std::cout << -1 << '\n';
            return;
        }
    }
    for (auto a : ans)
    {
        std::cout << a;
    }
    std::cout << '\n';
}
