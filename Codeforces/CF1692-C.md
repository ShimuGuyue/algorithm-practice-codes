void solve()
{
    std::array<std::string, 8> grid;
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    for (int i = 1; i < 7; ++i)
    {
        for (int j = 1; j < 7; ++j)
        {
            if (grid[i][j] == '#' &&
                grid[i - 1][j - 1] == '#' &&
                grid[i - 1][j + 1] == '#' &&
                grid[i + 1][j - 1] == '#' &&
                grid[i + 1][j + 1] == '#')
            {
                std::cout << i + 1 << ' ' << j + 1 << '\n';
                return;
            }
        }
    }
}
