void solve()
{
    std::string s;
    std::cin >> s;

    int n{ int(s.length()) } ;

    int count1{ 0 }, count2{ 0 };
    for (char c : s)
    {
        if (c == '(')
            ++count1;
        else if (c == ')')
            ++count2;
    }
    if (count1 == n / 2 || count2 == n / 2)
    {
        std::cout << "YES\n";
        return;
    }

    int index1{ -1 }, index2{ -1 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (s[i] != '?')
            continue;
        if (count1 < n / 2)
        {
            s[i] = '(';
            ++count1;
            if (count1 == n / 2)
                index1 = i;
        }
        else
        {
            s[i] = ')';
            if (index2 == -1)
                index2 = i;
            ++count2;
        }
    }

    std::swap(s[index1], s[index2]);
    count1 = 0; count2 = 0;
    for (char c : s)
    {
        if (c == '(')
            ++count1;
        else
            ++count2;
        if (count1 < count2)
        {
            std::cout << "YES\n";
            return;
        }
    }
    std::cout << "NO\n";
}
