void solve()
{
    std::array<int, 7> as;
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());
    for (int i{ 0 }; i < 6; ++i)
    {
        as[i] = -as[i];
    }

    int ans{ std::accumulate(as.begin(), as.end(), 0) };
    std::cout << ans << '\n';
}
