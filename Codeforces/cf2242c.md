void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> counts;
    int count{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (i == 0 || as[i] == as[i - 1])
            ++count;
        if (i != 0 && as[i] != as[i - 1])
        {
            counts.push_back(count);
            count = 1;
        }
    }
    counts.push_back(count);

    std::set<int> t;
    int sum{ n };
    int size{ static_cast<int>(counts.size()) };
    while (size)
    {
        if (k >= sum && (k - sum) % size == 0)
            t.insert(size);

        int min{ 500000 };
        for (auto& a : counts)
        {
            if (a != 0 && a < min)
                min = a;
        }
        if (min != 1)
            --min;
        sum -= size * min;
        for (auto& a : counts)
        {
            if (a == 0)
                continue;
            a -= min;
            if (a == 0)
                --size;
        }
    }
    int ans{ static_cast<int>(t.size()) };
    std::cout << ans << '\n';
}
