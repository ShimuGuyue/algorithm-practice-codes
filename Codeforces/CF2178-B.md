void solve()
{
    std::string s;
    std::cin >> s;

    int ans{ 0 };
    if (s.front() == 'u')
    {
        s.front() = 's';
        ++ans;
    }
    if (s.back() == 'u')
    {
        s.back() = 's';
        ++ans;
    }

    for (int i = 1; i < s.length() - 1; ++i)
    {
        if (s[i] == 's')
            continue;
        if (s[i - 1] != 's')
        {
            s[i - 1] = 's';
            ++ans;
        }
        if (s[i + 1] != 's')
        {
            s[i + 1] = 's';
            ++ans;
        }
    }
    std::cout << ans << '\n';
}
