void solve()
{
    int q;
    std::cin >> q;

    int64_t len1{ 0 }, len2{ 0 };
    bool alla1{ true }, alla2{ true };
    while (q--)
    {
        int64_t d, k;
        std::string x;
        std::cin >> d >> k >> x;
        if (d == 1)
            len1 += k * x.length();
        else
            len2 += k * x.length();
        for (char c : x)
        {
            if (c != 'a')
                if (d == 1)
                    alla1 = false;
                else
                    alla2 = false;
        }

        if (alla2 == false)
            std::cout << "YES\n";
        else
            if (alla1 == false)
                std::cout << "NO\n";
            else
                if (len1 < len2)
                    std::cout << "YES\n";
                else
                    std::cout << "NO\n";
    }
}
