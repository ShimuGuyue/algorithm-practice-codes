void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> grid(n, std::vector<int>(m));
    for (auto& v : grid)
    {
        for (auto& a : v)
        {
            std::cin >> a;
        }
    }

    std::vector<std::vector<int>> ans(n, std::vector<int>(m));
    ans[0][0] = grid[0][0];
    for (int i{ 0 }; i < n; ++i)
    {
        for (int j{ 0 }; j < m; ++j)
        {
            if (i + 1 != n && ans[i + 1][j] == 0)
            {
                if ((grid[i + 1][j] & 1) == (ans[i][j] & 1))
                    ans[i + 1][j] = grid[i + 1][j] + 1;
                else
                    ans[i + 1][j] = grid[i + 1][j];
            }
            if (j + 1 != m && ans[i][j + 1] == 0)
            {
                if ((grid[i][j + 1] & 1) == (ans[i][j] & 1))
                    ans[i][j + 1] = grid[i][j + 1] + 1;
                else
                    ans[i][j + 1] = grid[i][j + 1];
            }
        }
    }

    for (auto& v : ans)
    {
        for (auto& a : v)
        {
            std::cout << a << ' ';
        }
        std::cout << '\n';
    }
}
