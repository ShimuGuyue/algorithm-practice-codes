```cpp
class DisjointSet
{
private:
    struct Data
    {
        int set;
        int rank;
        int size;
    };
    std::vector<Data> nodes_;

    int count_;

public:
    DisjointSet(const int n)
    {
        build(n);
    }

public:
    int find(const int x)
    {
        return nodes_[x].set == x ? x : nodes_[x].set = find(nodes_[x].set);
    }

    bool merge(const int x, const int y)
    {
        const int set_x{ find(x) };
        const int set_y{ find(y) };
        if (set_x == set_y)
            return false;
        if (nodes_[set_x].rank < nodes_[set_y].rank)
        {
            nodes_[set_x].set = set_y;
            nodes_[set_y].size += nodes_[set_x].size;
        }
        else
        {
            nodes_[set_y].set = set_x;
            nodes_[set_x].size += nodes_[set_y].size;
            if (nodes_[set_x].rank == nodes_[set_y].rank)
                ++nodes_[set_x].rank;
        }
        --count_;
        return true;
    }

    int count() const
    {
        return count_;
    }

    int get_size(const int x)
    {
        return nodes_[find(x)].size;
    }

private:
    void build(const int n)
    {
        count_ = n;

        nodes_.assign(n, { });
        for (int i{ 0 }; i < n; ++i)
        {
            nodes_[i].set = i;
            nodes_[i].rank = 1;
            nodes_[i].size = 1;
        }
    }
};
```

```cpp
void solve()
{
    int n;
    std::cin >> n;
    DisjointSet ds(n + 1);
    std::vector<int> degrees(n + 1);
    std::set<std::pair<int, int>> visited;
    for (int i{ 1 }; i <= n; ++i)
    {
        int j;
        std::cin >> j;
        ds.merge(i, j);
        if (visited.find({i, j}) == visited.end())
        {
            visited.insert({i, j});
            visited.insert({j, i});
            ++degrees[i];
            ++degrees[j];
        }
    }

    int max{ ds.count() - 1 };
    std::vector<bool> is_circles(n + 1, true);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (degrees[i] == 1)
            is_circles[ds.find(i)] = false;
    }
    bool find{ false };
    int circle{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (ds.find(i) != i)
            continue;
        circle += is_circles[i];
        if (!is_circles[i])
            find = true;
    }
    int min{ circle + find };

    std::cout << min << ' ' << max << '\n';
}
```
