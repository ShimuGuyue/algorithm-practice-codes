void solve()
{
    constexpr int64_t inf{ INT64_C(1) << 60 };

    int n;
    std::cin >> n;
    std::vector<std::vector<int64_t>> grid(n + 1, std::vector<int64_t>(n + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= n; ++j)
        {
            std::cin >> grid[i][j];
        }
    }
    std::vector<int> xs(n);
    for (auto& x : xs)
    {
        std::cin >> x;
    }

    std::reverse(xs.begin(), xs.end());
    std::vector<int64_t> ans;
    std::vector<bool> visited(n + 1);
    for (int x : xs)
    {
        visited[x] = true;

        int64_t sum{ 0 };
        for (int i{ 1 }; i <= n; ++i)
        {
            for (int j{ 1 }; j <= n; ++j)
            {
                grid[i][j] = std::min(grid[i][j], grid[i][x] + grid[x][j]);
                if (visited[i] && visited[j])
                    sum += grid[i][j];
            }
        }
        ans.push_back(sum);
    }
    std::reverse(ans.begin(), ans.end());
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
