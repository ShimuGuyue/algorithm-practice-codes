```c++
std::vector<std::vector<int64_t>> presums;

void init()
{
    constexpr int n{ 2023 };
    presums.assign(2023 + 1, std::vector<int64_t>(2023 + 1));
    int64_t flag{ 1 };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= i; ++j)
        {
            presums[i][j] = flag * flag;
            ++flag;
        }
    }

    for (int j{ 1 }; j <= n; ++j)
    {
        for (int i{ j }; i <= n; ++i)
        {
            presums[i][j] += presums[i - 1][j];
        }
    }
}
```

```c++
void solve()
{
    int n;
    std::cin >> n;

    int x{ 1 }, y;
    while (x <= 2023)
    {
        if (n <= x * (x + 1) / 2)
            break;
        ++x;
    }
    y = n - (x - 1) * x / 2;

    int64_t ans{ 0 };
    for (int i{ x }, j{ y }; j > 0; --i, --j)
    {
        ans += presums[i][j];
    }
    std::cout << ans << '\n';
}
```
