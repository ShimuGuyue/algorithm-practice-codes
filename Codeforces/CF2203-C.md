void solve()
{
    int64_t s, m;
    std::cin >> s >> m;

    int64_t min{
        [m]
        {
            int64_t ans;
            for (int64_t i{ 0 }; i <= 60; ++i)
            {
                if (1 << i & m)
                {
                    ans = 1 << i;
                    break;
                }
            }
            return ans;
        }()
    };

    if (s % min)
    {
        std::cout << -1 << '\n';
        return;
    }

    auto check = [s, m](int64_t k) -> bool
    {
        int64_t sum{ s };
        for (int i{ 60 }; i >= 0; --i)
        {
            if (m >> i & 1)
                sum -= std::min(sum >> i, k) << i;
        }
        return sum == 0;
    };
    int64_t l{ 1 }, r{ s };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
