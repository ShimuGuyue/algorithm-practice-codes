void solve()
{
    std::string s;
    std::cin >> s;

    std::queue<int> odds;
    std::queue<int> evens;
    for (char c : s)
    {
        int a{ c - '0' };
        if (a & 1)
            odds.push(a);
        else
            evens.push(a);
    }

    while (!odds.empty() && !evens.empty())
    {
        if (odds.front() < evens.front())
        {
            std::cout << odds.front();
            odds.pop();
        }
        else
        {
            std::cout << evens.front();
            evens.pop();
        }
    }
    while (!odds.empty())
    {
        std::cout << odds.front();
        odds.pop();
    }
    while (!evens.empty())
    {
        std::cout << evens.front();
        evens.pop();
    }
    std::cout << '\n';
}
