void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> grid(n + 2, std::vector<int>(m + 2));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            std::cin >> grid[i][j];
        }
    }

    std::vector<std::vector<int>> visited(n + 2, std::vector<int>(m + 2));
    auto dfs = [&grid, &visited](this auto& dfs, int x, int y) -> int
    {
        static constexpr std::array<std::array<int, 2>, 4> dxy{
            {{-1, 0}, {0, -1}, {0, 1}, {1, 0}}
        };

        int sum{ grid[x][y] };
        for (auto [dx, dy] : dxy)
        {
            int xx{ x + dx };
            int yy{ y + dy };
            if (grid[xx][yy] == 0)
                continue;
            if (visited[xx][yy])
                continue;
            visited[xx][yy] = true;
            sum += dfs(xx, yy);
        }
        return sum;
    };
    int ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            if (grid[i][j] == 0)
                continue;
            if (visited[i][j])
                continue;
            visited[i][j] = true;
            ans = std::max(ans, dfs(i, j));
        }
    }
    std::cout << ans << '\n';
}
