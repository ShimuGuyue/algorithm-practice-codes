void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    std::vector<int> bs(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> bs[i];
    }

    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(2));
    for (int i = 1; i <= n; ++i)
    {
        int64_t min{ dp[i - 1][0] };
        int64_t max{ dp[i - 1][1] };
        dp[i][0] = std::min(min - as[i], bs[i] - max);
        dp[i][1] = std::max(max - as[i], bs[i] - min);
    }
    int64_t ans{ dp[n][1] };
    std::cout << ans << '\n';
}
