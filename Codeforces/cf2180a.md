void solve()
{
    int l, a, b;
    std::cin >> l >> a >> b;
    std::vector<bool> visited(l);
    int ans{ a };
    while (!visited[(a + b) % l])
    {
        visited[(a + b) % l] = true;
        ans = std::max(ans, (a + b) % l);
        a = (a + b) % l;
    }
    std::cout << ans << '\n';
}
