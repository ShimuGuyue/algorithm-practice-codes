void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> indegrees(n + 1);
    std::vector<std::vector<int>> graph(n + 1);
    std::vector<int> as(n);
    while (k--)
    {
        for (auto& a : as)
        {
            std::cin >> a;
        }
        for (int i{ 2 }; i < n; ++i)
        {
            ++indegrees[as[i]];
            graph[as[i - 1]].push_back(as[i]);
        }
    }

    std::queue<int> q;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (indegrees[i] == 0)
            q.push(i);
    }
    int count{ 0 };
    while (!q.empty())
    {
        ++count;
        int u{ q.front() };
        q.pop();
        for (int v : graph[u])
        {
            --indegrees[v];
            if (indegrees[v] == 0)
                q.push(v);
        }
    }
    std::cout << (count == n ? "Yes\n" : "No\n");
}
