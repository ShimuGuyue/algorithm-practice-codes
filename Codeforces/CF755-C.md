```c++
class DisjointSet
{
private:
	struct Node
	{
		int parent;
		int rank;
	};

	std::vector<Node> nodes;

public:
	DisjointSet(int n) : nodes(n)
	{
		for (int i = 0; i < n; ++i)
		{
			nodes[i] = {i, 1};
		}
	}

public:
	int find(int x)
	{
		return (nodes[x].parent == x) ? x : (nodes[x].parent = find(nodes[x].parent));
	}

	void merge(int x, int y)
	{
		int set_x = find(x);
		int set_y = find(y);
		if (nodes[set_x].rank < nodes[set_y].rank)
		{
			nodes[set_x].parent = set_y;
		}
		else
		{
			nodes[set_y].parent = set_x;
			if (nodes[set_x].rank == nodes[set_y].rank)
				++nodes[set_x].rank;
		}
	}
};
```

```c++
void solve()
{
    int n;
    std::cin >> n;
    DisjointSet ds(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        int a;
        std::cin >> a;
        ds.merge(i, a);
    }
    std::set<int> t;
    for (int i = 1; i <= n; ++i)
    {
        t.insert(ds.find(i));
    }
    std::cout << t.size() << '\n';
}
```
