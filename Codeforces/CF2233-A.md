void solve()
{
    int n, a, b, c;
    std::cin >> n >> a >> b >> c;

    int ans{ 1 << 30 };
    ans = std::min(ans, div(n, a + b));
    ans = std::min(ans, c + div(std::max(0, n - c * a), a + 10 * b));
    std::cout << ans << '\n';
}

int div(int n, int k)
{
    return (n + k - 1) / k;
}
