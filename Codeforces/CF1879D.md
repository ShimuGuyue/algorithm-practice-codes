void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t mod{ 998244353 };

    int64_t ans{ 0 };
    for (int b{ 0 }; b < 30; ++b)
    {
        std::vector<int64_t> dp0_len(n + 1);
        std::vector<int64_t> dp0_count(n + 1);
        std::vector<int64_t> dp1_len(n + 1);
        std::vector<int64_t> dp1_count(n + 1);
        for (int i{ 1 }; i <= n; ++i)
        {

            if (as[i] >> b & 1)
            {
                dp0_count[i] = (dp1_count[i - 1]) % mod;
                dp1_count[i] = (dp0_count[i - 1] + 1) % mod;
                dp0_len[i] = (dp1_len[i - 1] + dp1_count[i - 1]) % mod;
                dp1_len[i] = (dp0_len[i - 1] + dp0_count[i - 1] + 1) % mod;
            }
            else
            {
                dp0_count[i] = (dp0_count[i - 1] + 1) % mod;
                dp1_count[i] = (dp1_count[i - 1]) % mod;
                dp0_len[i] = (dp0_len[i - 1] + dp0_count[i - 1] + 1) % mod;
                dp1_len[i] = (dp1_len[i - 1] + dp1_count[i - 1]) % mod;
            }
            ans = (ans + (INT64_C(1) << b) % mod * dp1_len[i]) % mod;
        }
    }
    std::cout << ans << '\n';
}
