void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::map<int, std::vector<int>> indexs;
    for (int i = 1; i <= n; ++i)
    {
        indexs[as[i]].push_back(i);
    }

    std::map<int, int> maxlens;
    for (int i = 1; i <= n; ++i)
    {
        int a{ as[i] };
        maxlens[a] = std::max(maxlens[a], 1);
        if (maxlens.count(a - 1))
            maxlens[a] = std::max(maxlens[a], maxlens[a - 1] + 1);
    }

    int x, max{ 0 };
    for (auto [k, v] : maxlens)
    {
        if (v > max)
        {
            max = v;
            x = k;
        }
    }

    std::vector<int> ans;
    int index{ n + 1 };
    while (max--)
    {
        index = *std::prev(std::upper_bound(indexs[x].begin(), indexs[x].end(), index));
        ans.push_back(index);
        --x;
    }
    std::reverse(ans.begin(), ans.end());
    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
