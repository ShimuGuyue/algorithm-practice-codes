void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    bool up{ false };
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] > as[i - 1])
        {
            up = true;
        }
        if (as[i] < as[i - 1])
        {
            if (up)
            {
                std::cout << "NO\n";
                return;
            }
        }
    }
    std::cout << "YES\n";
}
