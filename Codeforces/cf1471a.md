void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t min{ 0 }, max{ 0 };
    for (auto a : as)
    {
        max += (a + m - 1) / m;
        min += a;
    }
    min = (min + m - 1) / m;

    std::cout << min << ' ' << max << '\n';
}
