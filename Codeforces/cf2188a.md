void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> ans(n);
    int mod{ 0 };
    int a{ n }, b{ 1 };
    for (int i = n - 1; i >= 0; --i)
    {
        if (mod == 0)
            ans[i] = a--;
        else
            ans[i] = b++;
        mod = (mod + 1) % 2;
    }
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
