void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 0 };
    for (int k{ 1 }; k <= n; ++k)
    {
        if (n % k)
            continue;
        int gcd{ 0 };
        for (int i{ k }; i < n; ++i)
        {
            gcd = std::gcd(gcd, abs(as[i] - as[i - k]));
        }
        if (gcd != 1)
            ++ans;
    }
    std::cout << ans << '\n';
}
