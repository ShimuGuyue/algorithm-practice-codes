void solve()
{
    std::string op;
    std::cin >> op;
    if (op[0] == 'f')
    {
        int n;
        std::cin >> n;
        while (n--)
        {
            int a;
            std::cin >> a;
            std::cout << char(a + 'a' - 1);
        }
    }
    else
    {
        std::string s;
        std::cin >> s;
        std::cout << s.length() << '\n';
        for (char c : s)
        {
            std::cout << int(c - 'a' + 1) << ' ';
        }
    }
    std::cout << '\n';
}
