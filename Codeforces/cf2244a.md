void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int max{ 0 };
    int count{ 0 };
    for (char c : s)
    {
        if (c == '#')
            ++count;
        else
            count = 0;
        max = std::max(max, count);
    }
    int ans{ (max + 1) / 2 };
    std::cout << ans << '\n';
}
