void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::string temp = s;
    std::sort(temp.begin(), temp.end());
    if (temp == s)
    {
        std::cout << "Bob\n";
        return;
    }

    std::vector<int> indexs_1;
    for (int i = 0; i < n; ++i)
    {
        if (s[i] == '1')
            indexs_1.push_back(i);
    }

    std::vector<int> ans;
    int index{ n - 1 };
    while (s[index] == '1')
    {
        --index;
    }
    for (int i : indexs_1)
    {
        if (i > index)
            break;

        ans.push_back(i);
        ans.push_back(index--);
        while (s[index] == '1')
        {
            --index;
        }
    }

    std::cout << "Alice\n";
    std::cout << ans.size() << '\n';
    std::sort(ans.begin(), ans.end());
    for (auto a : ans)
    {
        std::cout << a + 1 << ' ';
    }
    std::cout << '\n';
}
