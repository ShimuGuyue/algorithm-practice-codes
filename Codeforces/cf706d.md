```cpp
class Trie
{
private:
    static constexpr int base{ 2 };    // TODO：设置字符集大小
    struct Node
    {
        std::array<int, base> children{ };
        int count_pass{ 0 };
        int count_end{ 0 };
    };
    std::vector<Node> tree{ Node() };

public:
    Trie() = default;

public:
    void insert(const int n)
    {
        int node{ 0 };
        ++tree[0].count_pass;
        for (int b{ 30 }; b >= 0; --b)
        {
            const int index{ n >> b & 1 };
            if (tree[node].children[index] == 0)
            {
                tree[node].children[index] = tree.size();
                tree.emplace_back();
            }
            node = tree[node].children[index];
            ++tree[node].count_pass;
        }
        ++tree[node].count_end;
    }

    void erase(const int n)
    {
        int node{ search(n) };
        if (node == -1 || tree[node].count_end == 0)
            return;
        node = 0;
        --tree[0].count_pass;
        for (int b{ 30 }; b >= 0; --b)
        {
            int index{ n >> b & 1 };
            node = tree[node].children[index];
            --tree[node].count_pass;
        }
        --tree[node].count_end;
    }

    int query(int n)
    {
        int ans{ 0 };
        int node{ 0 };
        for (int b{ 30 }; b >= 0; --b)
        {
            int index{ n >> b & 1 };
            if (tree[node].children[index ^ 1] == 0 || tree[tree[node].children[index ^ 1]].count_pass == 0)
            {
                node = tree[node].children[index];
            }
            else
            {
                ans |= 1 << b;
                node = tree[node].children[index ^ 1];
            }
        }
        return ans;
    }

private:
    int search(const int n) const
    {
        int node{ 0 };
        for (int b{ 30 }; b >= 0; --b)
        {
            const int index{ n >> b & 1 };
            if (tree[node].children[index] == 0 || tree[tree[node].children[index]].count_pass == 0)
                return -1;
            node = tree[node].children[index];
        }
        return node;
    }
};
```

```cpp
void solve()
{
    Trie trie;
    trie.insert(0);

    int q;
    std::cin >> q;
    while (q--)
    {
        char op;
        int x;
        std::cin >> op >> x;
        if (op == '+')
            trie.insert(x);
        else if (op == '-')
            trie.erase(x);
        else if (op == '?')
            std::cout << trie.query(x) << '\n';
    }
}
```
