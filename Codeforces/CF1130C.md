void solve()
{
    int n;
    std::cin >> n;
    int x1, y1;
    int x2, y2;
    std::cin >> x1 >> y1;
    std::cin >> x2 >> y2;
    std::vector<std::string> grid(n);
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    --x1; --y1;
    --x2; --y2;

    std::vector<std::vector<bool>> visited(n, std::vector<bool>(n));
    auto dfs = [&visited, n, &grid](this auto&& dfs, int x, int y, std::vector<std::pair<int, int>>& nodes) -> void
    {
        visited[x][y] = true;
        nodes.push_back({ x, y });
        static std::array<std::array<int, 2>, 4> dxy{
            { { -1, 0 }, { 0, -1 }, { 0, 1 }, { 1, 0 } }
        };
        for (auto [dx, dy] : dxy)
        {
            int xx{ x + dx };
            int yy{ y + dy };
            if (xx < 0 || xx == n)
                continue;
            if (yy < 0 || yy == n)
                continue;
            if (visited[xx][yy])
                continue;
            if (grid[xx][yy] == '1')
                continue;
            dfs(xx, yy, nodes);
        }
    };

    std::vector<std::pair<int, int>> nodes_1;
    std::vector<std::pair<int, int>> nodes_2;
    dfs(x1, y1, nodes_1);
    dfs(x2, y2, nodes_2);

    int ans{ 1 << 30 };
    for (auto [xx1, yy1] : nodes_1)
    {
        for (auto [xx2, yy2] : nodes_2)
        {
            ans = std::min(ans, (xx1 - xx2) * (xx1 - xx2) + (yy1 - yy2) * (yy1 - yy2));
        }
    }
    std::cout << ans << '\n';
}
