void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::set<int> t(as.begin(), as.end());
    if (t.size() != n)
    {
        std::cout << -1 << '\n';
        return;
    }

    std::sort(as.rbegin(), as.rend());
    for (auto a : as)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
