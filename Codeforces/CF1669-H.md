void solve()
{
    int n, k;
    std::cin >> n >> k;

    std::array<int, 31> counts{ };
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        for (int b = 0; b <= 30; ++b)
        {
            if (a >> b & 1)
                ++counts[b];
        }
    }

    int ans{ 0 };
    for (int i = 30; i >= 0; --i)
    {
        if (n - counts[i] <= k)
        {
            k -= n - counts[i];
            ans |= (1 << i);
        }
    }
    std::cout << ans << '\n';
}
