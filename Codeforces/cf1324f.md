void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        if (as[i] == 0)
            as[i] = -1;
    }
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    std::vector<int> dp(n + 1);
    auto dfs1 = [&as, &tree, &dp](this auto& dfs, int u, int last) -> void
    {
        dp[u] = as[u];
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            if (dp[v] > 0)
                dp[u] += dp[v];
        }
    };
    dfs1(1, 0);

    std::vector<int> ans(n + 1);
    ans[1] = dp[1];
    auto dfs2 = [&as, &tree, &dp, &ans](this auto& dfs, int u, int last) -> void
    {
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            ans[v] = dp[v] + std::max(0, ans[u] - std::max(0, dp[v]));
            dfs(v, u);
        }
    };
    dfs2(1, 0);

    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
