void solve()
{
    int64_t u, v;
    std::cin >> u >> v;

    if (v < u)
    {
        std::cout << -1 << '\n';
        return;
    }

    std::vector<int64_t> ans(100000);

    ans[0] = u;
    v -= u;

    for (int b{ 60 }; b >= 0; --b)
    {
        int index{ ans[0] >> b & 1 ? 1 : 0 };
        int64_t flag{ INT64_C(1) << b };
        while (v >= flag * 2)
        {
            ans[index] |= flag;
            ans[index + 1] |= flag;
            v -= flag * 2;
        }
    }
    if (v)
    {
        std::cout << -1 << '\n';
        return;
    }

    int size{ std::find(ans.begin(), ans.end(), 0) - ans.begin() };
    std::cout << size << '\n';
    for (int i{ 0 }; i < size; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
