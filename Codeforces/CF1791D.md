void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::map<char, int> m1;
    std::map<char, int> m2;
    for (char c : s)
    {
        ++m1[c];
    }

    int ans{ 0 };
    for (int i{ 0 }; i < s.length() - 1; ++i)
    {
        ++m2[s[i]];
        --m1[s[i]];
        if (m1[s[i]] == 0)
            m1.erase(s[i]);
        ans = std::max(ans, int(m1.size() + m2.size()));
    }
    std::cout << ans << '\n';
}
