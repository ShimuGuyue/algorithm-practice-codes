void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<bool>> dp(n + 1, std::vector<bool>(1 << 8));
    dp[0][0] = 1;
    int ans{ 0 };
    for (int i = 1; i <= n; ++i)
    {
        dp[i][as[i]] = true;
        ans = std::max(ans, as[i]);
        for (int j = 0; j < (1 << 8); ++j)
        {
            if (!dp[i - 1][j])
                continue;
            dp[i][j ^ as[i]] = true;
            ans = std::max(ans, j ^ as[i]);
        }
    }
    std::cout << ans << '\n';
}
