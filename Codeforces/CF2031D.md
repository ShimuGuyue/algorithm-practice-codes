void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> premaxs(n);
    std::vector<int> sufmins(n);
    std::partial_sum(as.begin(), as.end(), premaxs.begin(),
        [](int pre, int cur)
        {
            return std::max(pre, cur);
        }
    );
    std::partial_sum(as.rbegin(), as.rend(), sufmins.rbegin(),
        [](int pre, int cur)
        {
            return std::min(pre, cur);
        }
    );

    std::vector<int> ans(n);
    ans[n - 1] = *std::max_element(as.begin(), as.end());
    for (int i{ n - 2 }; i >= 0; --i)
    {
        ans[i] = premaxs[i];
        if (premaxs[i] > sufmins[i + 1])
            ans[i] = std::max(ans[i], ans[i + 1]);
    }

    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
