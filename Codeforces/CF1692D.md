void solve()
{
    char c;
    int h, m, x;
    std::cin >> h >> c >> m >> x;

    int count{ std::lcm(x, 1440) / x };
    int ans{ 0 };
    while (count--)
    {
        m += x;
        int k{ m / 60 };
        m %= 60;
        h += k;
        h %= 24;
        std::string hh{ std::to_string(h) };
        std::string mm{ std::to_string(m) };
        if (hh.size() == 1)
            hh = "0" + hh;
        if (mm.size() == 1)
            mm = "0" + mm;
        if (hh[0] == mm[1] && hh[1] == mm[0])
            ++ans;
    }
    std::cout << ans << '\n';
}
