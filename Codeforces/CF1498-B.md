void solve()
{
    int n, m;
    std::cin >> n >> m;

    std::map<int, int> counts;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    int ans{ };
    while (n)
    {
        int flag{ m };
        int temp{ 1 << 30 };
        while (temp)
        {
            if (counts[temp])
            {
                int count{ std::min(counts[temp], flag / temp) };
                counts[temp] -= count;
                n -= count;
                flag -= count * temp;
            }
            temp /= 2;
        }
        ++ans;
    }
    std::cout << ans << '\n';
}
