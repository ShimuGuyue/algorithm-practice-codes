void solve()
{
    int64_t n, x, y;
    std::cin >> n >> x >> y;
    int64_t lcm{ std::lcm(x, y) };

    int64_t count_lcm{ n / lcm };
    int64_t count_x{ n / x - count_lcm };
    int64_t count_y{ n / y - count_lcm };

    int64_t ans{ 0 };
    ans += count_x * (n + n - count_x + 1) / 2;
    ans -= count_y * (1 + count_y) / 2;
    std::cout << ans << '\n';
}
