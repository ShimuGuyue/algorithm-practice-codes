void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<std::pair<int, int64_t>>> graph(n + 1);
    while (m--)
    {
        int64_t u, v, w;
        std::cin >> u >> v >> w;
        graph[u].push_back({ v, w * 2 });
        graph[v].push_back({ u, w * 2 });
    }
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        graph[0].push_back({ i, as[i] });
    }

    std::vector<int64_t> lens(n + 1, INT64_C(1) << 60);
    std::priority_queue<std::pair<int64_t, int>, std::vector<std::pair<int64_t, int>>, std::greater<std::pair<int64_t, int>>> h;
    lens[0] = 0;
    h.push({ 0, 0 });
    while (!h.empty())
    {
        auto [len, u]{ h.top() };
        h.pop();
        if (len > lens[u])
            continue;
        for (auto [v, w] : graph[u])
        {
            if (len + w >= lens[v])
                continue;
            lens[v] = len + w;
            h.push({ lens[v], v });
        }
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << lens[i] << ' ';
    }
    std::cout << '\n';
}
