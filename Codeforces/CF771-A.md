```cpp
class DisjointSet
{
private:
    struct Data
    {
        int set;
        int rank;
        int size;
    };
    std::vector<Data> nodes_;

    int count_;

public:
    DisjointSet(const int n)
    {
        build(n);
    }

public:
    int find(const int x)
    {
        return nodes_[x].set == x ? x : nodes_[x].set = find(nodes_[x].set);
    }

    bool merge(const int x, const int y)
    {
        const int set_x{ find(x) };
        const int set_y{ find(y) };
        if (set_x == set_y)
            return false;
        if (nodes_[set_x].rank < nodes_[set_y].rank)
        {
            nodes_[set_x].set = set_y;
            nodes_[set_y].size += nodes_[set_x].size;
        }
        else
        {
            nodes_[set_y].set = set_x;
            nodes_[set_x].size += nodes_[set_y].size;
            if (nodes_[set_x].rank == nodes_[set_y].rank)
                ++nodes_[set_x].rank;
        }
        --count_;
        return true;
    }

    int count() const
    {
        return count_;
    }

    int get_size(const int x)
    {
        return nodes_[find(x)].size;
    }

    bool ok(int m)
    {
        std::vector<bool> visited(nodes_.size());
        int64_t sum{ 0 };
        for (int i{ 1 }; i < visited.size(); ++i)
        {
            int n{ find(i) };
            if (visited[n])
                continue;
            visited[n] = true;
            int64_t size{ get_size(n) };
            sum += size * (size - 1) / 2;
        }
        return m == sum;
    }

private:
    void build(const int n)
    {
        count_ = n;

        nodes_.assign(n, { });
        for (int i{ 0 }; i < n; ++i)
        {
            nodes_[i].set = i;
            nodes_[i].rank = 1;
            nodes_[i].size = 1;
        }
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;
    DisjointSet ds(n + 1);
    for (int i{ 0 }; i < m; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        ds.merge(u, v);
    }
    std::cout << (ds.ok(m) ? "YES\n" : "NO\n");
}
```
