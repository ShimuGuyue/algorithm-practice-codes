void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::priority_queue<int, std::vector<int>, std::greater<int>> h;
    int64_t sum{ 0 };
    int count{ 0 };
    for (auto a : as)
    {
        if (a >= 0)
        {
            sum += a;
            ++count;
        }
        else
        {
            sum += a;
            ++count;
            h.push(a);
            while (sum < 0)
            {
                sum -= h.top();
                --count;
                h.pop();
            }
        }
    }
    std::cout << count << '\n';
}
