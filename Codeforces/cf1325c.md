void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    std::map<std::pair<int, int>, int> m;
    for (int i{ 0 }; i < n - 1; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        if (u > v)
            std::swap(u, v);
        m[{u, v}] = i;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    bool make{ std::find_if(tree.begin(), tree.end(),
        [](std::vector<int>& a)
        {
            return a.size() > 2;
        }) != tree.end()
    };
    if (!make)
    {
        for (int i{ 0 }; i < n - 1; ++i)
        {
            std::cout << i << '\n';
        }
        return;
    }

    std::vector<int> ans(n - 1, -1);
    for (int u{ 1 }; u <= n; ++u)
    {
        if (tree[u].size() <= 2)
            continue;
        for (int i{ 0 }; i <= 2; ++i)
        {
            int v{ tree[u][i] };
            auto [p, q]{ std::minmax(u, v) };
            ans[m[{p, q}]] = i;
        }
        break;
    }
    int flag{ 3 };
    for (auto& a : ans)
    {
        if (a == -1)
            a = flag++;
        std::cout << a << '\n';
    }
}
