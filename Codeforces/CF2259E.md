void solve()
{
    int n;
    cin >> n;
    vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        cin >> as[i];
    }

    vector<int> difs(n + 2);
    vector<int> ans(n + 2, 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] == -1 || as[i] == 0)
            continue;
        int l = std::max<int>(1, i - as[i] + 1);
        int r = std::min<int>(n, i + as[i] - 1);
        ++difs[l];
        --difs[r + 1];
    }

    std::partial_sum(difs.begin(), difs.end(), ans.begin());
    for (auto& a : ans)
    {
        a = a > 0 ? 0 : 1;
    }

    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] == -1)
            continue;
        if ((i - as[i] >  0  && ans[i - as[i]] == 1)
        ||  (i + as[i] <= n  && ans[i + as[i]] == 1))
            continue;
        cout << -1 << '\n';
        return;
    }

    if (std::find(ans.begin() + 1, ans.end(), 1) == ans.end())
    {
        cout << -1 << '\n';
        return;
    }

    for (int i{ 1 }; i <= n; ++i)
    {
        cout << ans[i];
    }
    cout << '\n';
}
