void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::string s;
    std::cin >> s;
    std::string t;
    std::cin >> t;

    std::vector<int> ls(m);
    std::vector<int> rs(m);
    for (int i{ 0 }, j{ 0 }; i < n && j < m; ++i)
    {
        if (s[i] == t[j])
            ls[j++] = i;
    }
    for (int i{ n - 1 }, j{ m - 1 }; i >= 0 && j >= 0; --i)
    {
        if (s[i] == t[j])
            rs[j--] = i;
    }

    int ans{ 0 };
    for (int i{ 1 }; i < m; ++i)
    {
        ans = std::max(ls[i] - ls[i - 1], ans);
        ans = std::max(rs[i] - rs[i - 1], ans);
        ans = std::max(rs[i] - ls[i - 1], ans);
    }
    std::cout << ans << '\n';
}
