void solve()
{
    int n;
    char c;
    std::cin >> n >> c;
    std::string s;
    std::cin >> s;

    s += s;

    std::vector<int> nears(n * 2);
    for (int i{ n * 2 - 1 }; i >= 0; --i)
    {
        if (s[i] == 'g')
            nears[i] = i;
        else
            nears[i] = nears[i + 1];
    }

    int ans{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (s[i] == c)
            ans = std::max(ans, nears[i] - i);
    }
    std::cout << ans << '\n';
}
