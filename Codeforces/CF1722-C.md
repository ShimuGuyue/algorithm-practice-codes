void solve()
{
    int n;
    std::cin >> n;

    std::map<std::string, int> m;
    std::vector<std::vector<std::string>> ss(3, std::vector<std::string>(n));
    for (int i = 0; i < 3; ++i)
    {
        for (auto& s : ss[i])
        {
            std::cin >> s;
            ++m[s];
        }
    }

    for (int i = 0; i < 3; ++i)
    {
        int ans{ 0 };
        for (auto& s : ss[i])
        {
            if (m[s] == 1)
                ans += 3;
            if (m[s] == 2)
                ans += 1;
        }
        std::cout << ans << ' ';
    }
    std::cout << '\n';
}
