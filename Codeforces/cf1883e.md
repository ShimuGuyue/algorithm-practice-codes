void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t ans{ 0 };
    std::vector<int> flags(n);
    for (int i{ 1 }; i < n; ++i)
    {
        int64_t l{ as[i - 1] }, r{ as[i] };
        while  (l * 2 <= r)
        {
            --flags[i];
            l *= 2;
        }
        while (r < l)
        {
            r *= 2;
            ++flags[i];
        }
        flags[i] += flags[i - 1];
        flags[i] = std::max(flags[i], 0);
        ans += flags[i];
    }
    std::cout << ans << '\n';
}
