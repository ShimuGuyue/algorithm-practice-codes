void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int max{ *std::max_element(as.begin(), as.end()) };
    int64_t sum{ std::accumulate(as.begin(), as.end(), INT64_C(0)) };
    if (sum % 2 == 0 && max * 2 <= sum)
        std::cout << "YES\n";
    else
        std::cout << "NO\n";
}
