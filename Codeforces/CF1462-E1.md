void solve()
{
    int n;
    std::cin >> n;

    std::map<int, int64_t> m;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++m[a];
    }

    std::vector<std::pair<int, int64_t>> v(m.begin(), m.end());
    int64_t ans{ 0 };
    for (int i = 0; i < v.size(); ++i)
    {
        int64_t a{ v[i].second };
        if (a >= 3)
            ans += a * (a - 1) * (a - 2) / 6;

        if (i + 1 >= v.size() || v[i + 1].first > v[i].first + 2)
            continue;
        int64_t b{ v[i + 1].second };

        if (a >= 2 && b >= 1)
            ans += a * (a - 1) / 2 * b;
        if (a >= 1 && b >= 2)
            ans += a * b * (b - 1) / 2;

        if (i + 2 >= v.size() || v[i + 2].first > v[i].first + 2)
            continue;
        int64_t c{ v[i + 2].second };

        if (a >= 2 && c >= 1)
            ans += a * (a - 1) / 2 * c;
        if (a >= 1 && c >= 2)
            ans += a * c * (c - 1) / 2;

        if (a >= 1 && b >= 1 && c >= 1)
            ans += a * b * c;
    }
    std::cout << ans << '\n';
}
