void solve()
{
    int n, m;
    cin >> n >> m;

    if (n == 1)
    {
        cout << 0 << '\n';
        return;
    }

    int len = n / 2;
    cout << n - (1 + len) + 1 << '\n';
    for (int i = 1; i <= n - (1 + len) + 1; ++i)
    {
        cout << i + 1 << ' ' << i + len << '\n';
    }
}
