class Manacher
{
public:
    static std::string turn(const std::string& s)
    {
        std::string ans;
        ans += '#';
        for (const char c : s)
        {
            ans += c;
            ans += '#';
        }
        return ans;
    }

    static std::vector<int> calc(const std::string& s)
    {
        const int n{ static_cast<int>(s.size()) };
        std::vector<int> lens(n);

        int p{ -1 }, r{ -1 };
        for (int i{ 0 }; i < n; ++i)
        {
            const int j{ p - (i - p) };
            int len{ i <= r ? std::min(lens[j], r - i + 1) : 1 };
            while (i - len >= 0 && i + len < n)
            {
                if (s[i - len] != s[i + len])
                    break;
                ++len;
            }
            lens[i] = len;

            if (i + len - 1 > r)
            {
                p = i;
                r = i + len - 1;
            }
        }
        return lens;
    }
};

void solve()
{
    std::string s;
    std::cin >> s;

    if (s.length() == 1)
    {
        std::cout << s << '\n';
        return;
    }

    auto t{ Manacher::turn(s) };
    auto flags{ Manacher::calc(t) };

    int n{ int(t.length()) };

    int l{ 0 }, r{ n - 1 };
    while (l < r)
    {
        if (t[l] != t[r])
        {
            --l;
            ++r;
            break;
        }
        ++l;
        --r;
    }
    if (l >= r)
    {
        --l;
        ++r;
    }

    int loc;
    int max{ 0 };
    for (int i{ l + 1 }; i - l <= r - i; ++i)
    {
        int len{ flags[i] };
        if (i - len + 1 > l + 1)
            continue;
        if (i - len + 1 < l + 1)
            len = i - l;
        if (len > max)
        {
            max = len;
            loc = i;
        }
    }
    for (int i{ r - 1 }; r - i <= i - l; --i)
    {
        int len{ flags[i] };
        if (i + len - 1 < r - 1)
            continue;
        if (i + len - 1 > r - 1)
            len = r - i;
        if (len > max)
        {
            max = len;
            loc = i;
        }
    }

    for (int i{ 0 }; i <= l; ++i)
    {
        if (t[i] != '#')
            std::cout << t[i];
    }
    if (max)
    {
        int ll{ loc - max + 1 };
        int rr{ loc + max - 1 };
        for (int i{ ll }; i <= rr; ++i)
        {
            if (t[i] != '#')
                std::cout << t[i];
        }
    }
    for (int i{ r }; i < n; ++i)
    {
        if (t[i] != '#')
            std::cout << t[i];
    }
    std::cout << '\n';
}
