void solve()
{
    int n;
    std::cin >> n;

    if (n == 1)
    {
        std::cout << char(rand() % 26 + 'a') << '\n';
        return;
    }

    if (n & 1)
    {
        std::cout << std::string(n / 2, 'a');
        std::cout << 'b';
        std::cout << 'c';
        std::cout << std::string(n / 2 - 1, 'a');
    }
    else
    {
        std::cout << std::string(n / 2, 'a');
        std::cout << 'b';
        std::cout << std::string(n / 2 - 1, 'a');
    }
    std::cout << '\n';
}
