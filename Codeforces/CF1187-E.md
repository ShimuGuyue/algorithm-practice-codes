void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    std::vector<int> sizes(n + 1);
    std::vector<int64_t> dp(n + 1);
    auto dfs1 = [&tree, &sizes, &dp](this auto&& dfs, int u, int last) -> void
    {
        sizes[u] = 1;
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            sizes[u] += sizes[v];
        }
        dp[u] = sizes[u];
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dp[u] += dp[v];
        }
    };
    dfs1(1, 0);

    std::vector<int64_t> ans(n + 1);
    ans[1] = dp[1];
    auto dfs2 = [&tree, &sizes, &dp, &ans, n](this auto&& dfs, int u, int last) -> void
    {
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            ans[v] = ans[u] + n - 2 * sizes[v];
            dfs(v, u);
        }
    };
    dfs2(1, 0);

    std::cout << *std::max_element(ans.begin(), ans.end()) << '\n';
}
