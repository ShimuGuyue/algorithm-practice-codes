```cpp
class DisjointSet
{
private:
    struct Data
    {
        int set;
        int rank;
        int size;
    };
    std::vector<Data> nodes_;

    int count_;

public:
    DisjointSet(const int n)
    {
        build(n);
    }

public:
    int find(const int x)
    {
        return nodes_[x].set == x ? x : nodes_[x].set = find(nodes_[x].set);
    }

    bool merge(const int x, const int y)
    {
        const int set_x{ find(x) };
        const int set_y{ find(y) };
        nodes_[set_x].set = set_y;
        return true;
    }

private:
    void build(const int n)
    {
        count_ = n;

        nodes_.assign(n, { });
        for (int i{ 0 }; i < n; ++i)
        {
            nodes_[i].set = i;
            nodes_[i].rank = 1;
            nodes_[i].size = 1;
        }
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;

    DisjointSet ds(n + 2);

    std::vector<int> ans(n + 1);
    while (m--)
    {
        int l, r, x;
        std::cin >> l >> r >> x;
        int i{ ds.find(l) };
        while (i <= r)
        {
            if (i == x)
            {
                i = ds.find(i + 1);
                continue;
            }
            ans[i] = x;
            ds.merge(i, i + 1);
            i = ds.find(i);
        }
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
```
