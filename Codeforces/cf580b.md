void solve()
{
    int n, d;
    std::cin >> n >> d;
    std::map<int, int64_t> flags;
    while (n--)
    {
        int m, s;
        std::cin >> m >> s;
        flags[m] += s;
    }

    int64_t ans{ 0 };
    int64_t sum{ 0 };
    auto lt{ flags.begin() };
    auto rt{ flags.begin() };
    while (rt != flags.end())
    {
        sum += rt->second;
        while (rt->first - lt->first >= d)
        {
            sum -= lt->second;
            ++lt;
        }
        ans = std::max(ans, sum);
        ++rt;
    }
    std::cout << ans << '\n';
}
