void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int64_t sum{ 0 };
    std::priority_queue<int, std::vector<int>, std::less<int>> h;
    for (int i = 0; i < n; ++i)
    {
        int l{ i }, r{ n - i - 1 };
        if (s[i] == 'L')
        {
            sum += l;
            h.push(std::max(r - l, 0));
        }
        else
        {
            sum += r;
            h.push(std::max(l - r, 0));
        }
    }

    while (!h.empty())
    {
        sum += h.top();
        h.pop();
        std::cout << sum << ' ';
    }
    std::cout << '\n';
}
