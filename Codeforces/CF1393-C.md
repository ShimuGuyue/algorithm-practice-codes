void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> counts(n + 1);
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    int max{ -1 };
    int count{ 0 };
    for (auto c : counts)
    {
        if (c > max)
        {
            max = c;
            count = 1;
        }
        else if (c == max)
        {
            ++count;
        }
    }

    int ans{ (n - max * count) / (max - 1) + count - 1 };
    std::cout << ans << '\n';
}
