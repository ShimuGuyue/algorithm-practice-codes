void solve()
{
    int n;
    std::cin >> n;

    std::vector<std::vector<int>> ans;
    for (int i{ 1 }; i <= n; ++i)
    {
        ans.emplace_back();
        auto& v{ ans.back() };
        v.push_back(1);
        v.push_back(i);
        for (int j{ 1 }; j <= n; ++j)
        {
            v.push_back(j);
        }
    }

    int sum{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        sum += i * (i * 2 - 1);
    }

    std::cout << sum << ' ' << n * 2 << '\n';
    for (int i{ n }; i > 0; --i)
    {
        std::cout << 1 << ' ' << i << ' ';
        for (int j{ 1 }; j <= n; ++j)
        {
            std::cout << j << ' ';
        }
        std::cout << '\n';
        std::cout << 2 << ' ' << i << ' ';
        for (int j{ 1 }; j <= n; ++j)
        {
            std::cout << j << ' ';
        }
        std::cout << '\n';
    }
}
