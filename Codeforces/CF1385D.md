void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    int ans{ merge(s, 0, n - 1, 'a') };
    std::cout << ans << '\n';
}

int merge(std::string &s, int l, int r, char c)
{
    if (l == r)
        return s[l] != c;

    int mid{ (l + r) / 2 };
    int ans1{ }, ans2{ };
    for (int i = l; i <= mid; ++i)
    {
        ans1 += s[i] != c;
    }
    ans1 += merge(s, mid + 1, r, c + 1);
    for (int i = mid + 1; i <= r; ++i)
    {
        ans2 += s[i] != c;
    }
    ans2 += merge(s, l, mid, c + 1);

    return std::min(ans1, ans2);
}
