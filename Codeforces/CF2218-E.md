void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        for (int j{ i + 1 }; j < n; ++j)
        {
            ans = std::max(ans, as[i] ^ as[j]);
        }
    }
    std::cout << ans << '\n';
}
