void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> ps(n);
    std::vector<int> as(n);
    for (auto& p : ps)
    {
        std::cin >> p;
    }
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> indexs(n + 1);
    for (int i{ 0 }; i < n; ++i)
    {
        indexs[ps[i]] = i;
    }

    std::vector<int> flags(n);
    for (int i{ 0 }; i < n; ++i)
    {
        flags[i] = indexs[as[i]];
    }
    if (std::is_sorted(flags.begin(), flags.end()))
        std::cout << "yes\n";
    else
        std::cout << "no\n";
}
