void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    std::vector<int> flags(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        flags[as[i]] = i;
    }

    std::vector<bool> visited(n + 1);
    int ans{ 0 };
    for (int i{ n }; i > 0; --i)
    {
        if (visited[i])
            continue;
        ++ans;
        int x{ i };
        while (flags[x] < x)
        {
            visited[flags[x]] = true;
            x = flags[x];
        }
    }
    std::cout << ans << '\n';
}
