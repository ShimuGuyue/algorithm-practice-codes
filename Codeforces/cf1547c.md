void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> ans(n + 1, std::numeric_limits<int>::max());
    std::priority_queue<std::pair<int, int>, std::vector<std::pair<int, int>>, std::greater<std::pair<int, int>>> h;
    std::vector<int> as(k);
    for (int i{ 0 }; i < k; ++i)
    {
        std::cin >> as[i];
    }
    for (int i{ 0 }; i < k; ++i)
    {
        int t;
        std::cin >> t;
        ans[as[i]] = t;
        h.push({t, as[i]});
    }

    while (!h.empty())
    {
        auto [t, a]{ h.top() };
        h.pop();

        if (t > ans[a])
            continue;
        if (a < n && t + 1 < ans[a + 1])
        {
            ans[a + 1] = t + 1;
            h.push({t + 1, a + 1});
        }
        if (a > 1 && t + 1 < ans[a - 1])
        {
            ans[a - 1] = t + 1;
            h.push({t + 1, a - 1});
        }
    }

    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
