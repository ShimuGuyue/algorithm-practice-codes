void solve()
{
    int n, a, b;
    std::cin >> n >> a >> b;
    std::vector<std::vector<std::pair<int, int>>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v, w;
        std::cin >> u >> v >> w;
        tree[u].push_back({v, w});
        tree[v].push_back({u, w});
    }

    std::set<int> t;
    auto dfs1 = [&tree, &t, b](this auto& dfs, int u, int last, int flag) -> void
    {
        if (u != b)
            t.insert(flag);
        for (auto [v, w] : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u, flag ^ w);
        }
    };
    dfs1(b, 0, 0);

    std::string ans{ "NO" };
    auto dfs2 = [&tree, &t, &ans, b](this auto& dfs, int u, int last, int flag) -> void
    {
        if (t.find(flag) != t.end())
        {
            ans = "YES";
            return;
        }
        for (auto [v, w] : tree[u])
        {
            if (v == last || v == b)
                continue;
            dfs(v, u, flag ^ w);
        }
    };
    dfs2(a, 0, 0);
    std::cout << ans << '\n';
}
