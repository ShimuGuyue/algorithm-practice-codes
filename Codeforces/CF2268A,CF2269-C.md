void solve()
{
    int n, k;
    cin >> n >> k;
    vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        cin >> as[i];
    }

    int64_t ans = 0;
    for (int i = k; i <= n - k + 1; ++i)
    {
        ans += as[i];
    }
    for (int i = 1; i <= std::min(k - 1, n - k + 1); ++i)
    {
        ans += std::max(as[i], as[n - i + 1]);
    }
    cout << ans << '\n';
}
