void solve()
{
    int n;
    std::cin >> n;
    if (n & 1)
        std::cout << "Bob\n";
    else
        if (is_power2(n))
            if (calc_power2(n) & 1)
                std::cout << "Bob\n";
            else
                std::cout << "Alice\n";
        else
            std::cout << "Alice\n";
}

bool is_power2(uint32_t n)
{
    return std::popcount(n) == 1;
}

int calc_power2(int n)
{
    int ans{ 0 };
    while (n != 1)
    {
        n /= 2;
        ++ans;
    }
    return ans;
}
