static constexpr int64_t mod{ 1000000007 };

void solve()
{
    struct Node
    {
        int l{ 0 }, r{ 0 }, u{ 0 };
    };

    int n;
    std::cin >> n;
    std::vector<Node> tree(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        int l, r;
        std::cin >> l >> r;
        if (l == 0)
            continue;
        tree[i].l = l;
        tree[i].r = r;
        tree[l].u = tree[r].u = i;
    }

    std::vector<int64_t> values(n + 1);
    auto dfs1 = [&tree, &values](this auto& dfs, int n)
    {
        auto [l, r, u]{ tree[n] };
        if (l == 0)
        {
            values[n] = 1;
            return;
        }
        dfs(l);
        dfs(r);
        values[n] = (values[l] + values[r] + 3) % mod;
    };
    dfs1(1);

    std::vector<int64_t> presums(n + 1);
    auto dfs2 = [&tree, &values, &presums](this auto& dfs, int n)
    {
        auto [l, r, u]{ tree[n] };

        presums[n] = (presums[u] + values[n]) % mod;

        if (l == 0)
            return;
        dfs(l);
        dfs(r);
    };
    dfs2(1);

    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << presums[i] << ' ';
    }
    std::cout << std::endl;;
}
