void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<std::vector<int64_t>> grid(n + 1, std::vector<int64_t>(m + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            std::cin >> grid[i][j];
        }
    }
    std::vector<std::vector<bool>> modes(n + 1, std::vector<bool>(m + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        std::string s;
        std::cin >> s;
        for (int j{ 1 }; j <= m; ++j)
        {
            modes[i][j] = s[j - 1] == '1';
        }
    }

    std::vector<std::vector<int64_t>> precounts(n + 1, std::vector<int64_t>(m + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            precounts[i][j] += precounts[i - 1][j];
            precounts[i][j] += precounts[i][j - 1];
            precounts[i][j] -= precounts[i - 1][j - 1];
            precounts[i][j] += modes[i][j] ? 1 : -1;
        }
    }

    int64_t sum{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            sum += (modes[i][j] ? 1 : -1) * grid[i][j];
        }
    }

    std::vector<int64_t> flags;
    for (int i{ k }; i <= n; ++i)
    {
        for (int j{ k }; j <= m; ++j)
        {
            int64_t flag{ precounts[i][j] };
            flag -= precounts[i][j - k];
            flag -= precounts[i - k][j];
            flag += precounts[i - k][j - k];
            flags.push_back(flag);
        }
    }

    int64_t gcd{
        std::accumulate(flags.begin(), flags.end(), INT64_C(0),
            [](int64_t pre, int64_t cur)
            {
                return std::gcd(pre, std::llabs(cur));
            }
        )
    };

    if (gcd == 0)
        std::cout << (sum == 0 ? "Yes" : "No") << '\n';
    else
        std::cout << (sum % gcd == 0 ? "Yes" : "No") << '\n';
}
