```cpp
struct DisjointSet
{
    std::vector<int> fathers;

    DisjointSet(int n) : fathers(n)
    {
        for (int i{ 1 }; i < n; ++i)
        {
            fathers[i] = i;
        }
    };

    int find(int x)
    {
        return fathers[x] == x ? x : fathers[x] = find(fathers[x]);
    }

    void merge(int x, int y)
    {
        fathers[find(x)] = find(y);
    }
};
```

```cpp
void solve()
{
    std::string s;
    std::cin >> s;

    int n{ static_cast<int>(s.length()) };

    DisjointSet ds('z' + 1);

    std::vector<std::vector<int>> graph('z' + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        if (std::find(graph[s[i]].begin(), graph[s[i]].end(), s[i - 1]) != graph[s[i]].end())
            continue;
        if (graph[s[i]].size() == 2 || graph[s[i - 1]].size() == 2)
        {
            std::cout << "NO\n";
            return;
        }
        if (ds.find(s[i]) == ds.find(s[i - 1]))
        {
            std::cout << "NO\n";
            return;
        }
        graph[s[i]].push_back(s[i - 1]);
        graph[s[i - 1]].push_back(s[i]);
        ds.merge(s[i], s[i - 1]);
    }
    std::cout << "YES\n";


    std::vector<bool> visited('z' + 1);
    auto output = [&visited, &graph](this auto&& dfs, char u) -> void
    {
        visited[u] = true;
        std::cout << u;
        for (char v : graph[u])
        {
            if (visited[v])
                continue;
            dfs(v);
        }
    };
    for (char c{ 'a' }; c <= 'z'; ++c)
    {
        if (visited[c])
            continue;
        if (graph[c].size() == 2)
            continue;
        output(c);
    }
    std::cout << '\n';
}
```
