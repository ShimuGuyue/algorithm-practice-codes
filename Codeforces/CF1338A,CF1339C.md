void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (std::is_sorted(as.begin(), as.end()))
    {
        std::cout << 0 << '\n';
        return;
    }

    std::vector<int> difs(n);
    std::adjacent_difference(as.begin(), as.end(), difs.begin());

    int flag{ 0 };
    int max{ as[0] };
    for (int i = 1; i < n; ++i)
    {
        flag = std::max(flag, max - as[i]);
        max = std::max(max, as[i]);
    }
    debug_(flag);

    int ans{ 0 };
    while (flag)
    {
        ++ans;
        flag /= 2;
    }
    std::cout << ans << '\n';
}
