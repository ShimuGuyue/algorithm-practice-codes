void solve()
{
    int n;
    std::cin >> n;

    std::multiset<int> ans;
    for (int i{ 1 }; i < n; ++i)
    {
        if (std::gcd(i, n) == 1)
            ans.insert(i);
    }

    int multi{ std::accumulate(ans.begin(), ans.end(), 1,
        [n](int64_t pre, int cur)
        {
            return pre * cur % n;
        })
    };
    if (multi != 1)
        ans.erase(ans.find(multi));

    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
