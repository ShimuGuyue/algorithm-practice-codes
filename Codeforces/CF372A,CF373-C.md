void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    std::multiset<int> t1(as.begin(), as.begin() + n / 2);
    std::multiset<int> t2(as.begin() + n / 2, as.end());
    int ans{ n };
    for (auto a : t1)
    {
        auto it{ t2.lower_bound(a * 2) };
        if (it == t2.end())
            continue;
        t2.erase(it);
        --ans;
    }
    std::cout << ans << '\n';
}
