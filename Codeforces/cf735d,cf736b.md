void solve()
{
    int n;
    std::cin >> n;
    if (is_prime(n))
        std::cout << 1 << '\n';
    else if (is_prime(n - 2) || n % 2 == 0)
        std::cout << 2 << '\n';
    else
        std::cout << 3 << '\n';

}

bool is_prime(int n)
{
    for (int i{ 2 }; i * i <= n; ++i)
    {
        if (n % i == 0)
            return false;
    }
    return true;
}
