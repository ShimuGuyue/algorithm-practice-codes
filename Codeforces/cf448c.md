void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    auto dfs = [&as](this auto&& dfs, int l, int r, int low) -> int64_t
    {
        if (l > r)
            return 0;
        if (l == r)
            return as[l] > low ? 1 : 0;
        int64_t ans{ r - l + 1 };
        int64_t min{ *std::min_element(as.begin() + l, as.begin() + r + 1) };
        int64_t sum{ min - low };

        int left{ l };
        for (int i{ l }; i <= r; ++i)
        {
            if (as[i] == min)
            {
                sum += dfs(left, i - 1, min);
                left = i + 1;
            }
        }
        sum += dfs(left, r, min);
        return std::min(ans, sum);
    };
    int64_t ans{ dfs(1, n, 0) };
    std::cout << ans << '\n';
}
