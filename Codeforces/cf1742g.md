void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<bool> visited(n);
    int x{ 0 };
    for (int i{ 30 }; i >= 0; --i)
    {
        int index{ -1 };
        for (int j{ 0 }; j < n; ++j)
        {
            if (visited[j])
                continue;
            if ((x | as[j]) == x)
                continue;
            if (index == -1 || (x | as[j]) > (x | as[index]))
                index = j;
        }
        if (index == -1)
            break;
        visited[index] = true;
        std::cout << as[index] << ' ';
        x |= as[index];
    }
    for (int i { 0 }; i < n; ++i)
    {
        if (!visited[i])
            std::cout << as[i] << ' ';
    }
    std::cout << '\n';
}
