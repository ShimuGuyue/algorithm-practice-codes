void solve()
{
    int n, x;
    std::cin >> n >> x;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    auto check = [&as, x](int64_t k)
    {
        int64_t sum{ 0 };
        for (auto a : as)
        {
            sum += std::max<int64_t>(0, k - a);
            if (sum > x)
                return false;
        }
        return true;
    };
    int64_t l{ 0 }, r{ 1000000000000000000 };
    while (l < r)
    {
        int64_t mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    std::cout << l << '\n';
}
