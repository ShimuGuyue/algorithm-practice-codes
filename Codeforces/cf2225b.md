void solve()
{
    std::string s;
    std::cin >> s;

    auto judge = [&s](char c)
    {
        std::vector<int> indexs;
        for (int i{ 0 }; i < s.length(); ++i)
        {
            if (i & 1)
                if (s[i] == c)
                    indexs.push_back(i);
                else ;
            else
                if (s[i] != c)
                    indexs.push_back(i);
                else ;
        }
        if (indexs.empty())
            return true;
        int len{ indexs.back() - indexs.front() + 1 };
        return len == indexs.size();
    };

    if (judge('a') || judge('b'))
        std::cout << "Yes\n";
    else
        std::cout << "No\n";
}
