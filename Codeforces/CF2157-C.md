void solve()
{
    int n, k, q;
    std::cin >> n >> k >> q;

    std::vector<int> flags(n + 1);

    while (q--)
    {
        int c, l, r;
        std::cin >> c >> l >> r;
        if (c == 1)
        {
            for (int i = l; i <= r; ++i)
            {
                if (flags[i] == 3)
                    continue;
                flags[i] = flags[i] == 2 ? 3 : 1;
            }
        }
        else
        {
            for (int i = l; i <= r; ++i)
            {
                if (flags[i] == 3)
                    continue;
                flags[i] = flags[i] == 1 ? 3 : 2;
            }
        }
    }

    for (int i = 1; i <= n; ++i)
    {
        if (flags[i] == 0)
            std::cout << 0 << ' ';
        if (flags[i] == 1)
            std::cout << k << ' ';
        if (flags[i] == 2)
            std::cout << i % k << ' ';
        if (flags[i] == 3)
            std::cout << k + 1 << ' ';
    }
    std::cout << '\n';
}
