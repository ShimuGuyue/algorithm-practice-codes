void solve()
{
    int n, x;
    std::cin >> n >> x;
    std::string s;
    std::cin >> s;

    std::vector<int> pres(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        pres[i] = pres[i - 1] + (s[i - 1] == '0' ? 1 : -1);
    }

    if (pres[n] == 0)
    {
        if (std::find(pres.begin(), pres.end(), x) != pres.end())
            std::cout << -1 << '\n';
        else
            std::cout << 0 << '\n';
        return;
    }

    int ans{ 0 };
    int per{ pres[n] };
    for (int i{ 0 }; i < n; ++i)
    {
        if ((x - pres[i]) % per)
            continue;
        if (int64_t(x - pres[i]) * per >= 0)
            ++ans;
    }
    std::cout << ans << '\n';
}
