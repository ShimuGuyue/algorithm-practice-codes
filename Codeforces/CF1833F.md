static constexpr int64_t mod{ 1000000007 };

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::map<int, int64_t> counts;
    std::map<int, int> flags;
    counts[0] = 1;
    flags[0] = 1;
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
        flags[a] = 1;
    }

    std::map<int, int64_t> precounts{ counts };
    std::map<int, int> preflags{ flags };
    for (auto it{ std::next(precounts.begin()) }; it != precounts.end(); ++it)
    {
        it->second *= std::prev(it)->second;
        it->second %= mod;
    }
    for (auto it{ std::next(preflags.begin()) }; it != preflags.end(); ++it)
    {
        it->second += std::prev(it)->second;
    }

    int64_t ans{ 0 };
    for (auto it{ std::next(preflags.begin()) }; it != preflags.end(); ++it)
    {
        int r{ it->first + m - 1 };
        int l{ std::prev(it)->first };
        if (!preflags.count(r))
            continue;
        if (preflags[r] - preflags[l] != m)
            continue;
        ans += precounts[r] * inv(precounts[l]);
        ans %= mod;
    }
    std::cout << ans << '\n';
}

int64_t inv(int64_t n)
{
    return quickpow(n, mod - 2);
}

int64_t quickpow(int64_t base, int64_t power)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}
