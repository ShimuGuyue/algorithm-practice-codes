void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int odd{ 0 }, even{ 0 };
    bool mid{ false };

    for (int i{ 0 }, j{ n - 1 }; i <= j; ++i, --j)
    {
        if (i == j)
            if (s[i] == '0')
                mid = true;
            else
                ;
        else
            if (s[i] == '0' && s[j] == '0')
                ++even;
            else if (s[i] == '0' || s[j] == '0')
                ++odd;
    }

    if (odd == 0)
        if (mid && even)
            std::cout << "ALICE\n";
        else
            std::cout << "BOB\n";
    else if (odd >= 2)
        std::cout << "ALICE\n";
    else
        if (mid && even == 0)
            std::cout << "DRAW\n";
        else
            std::cout << "ALICE\n";
}
