void solve()
{
    int mult12, mult23, mult34, mult45;
    std::cout << "? 1 2" << std::endl;
    std::cin >> mult12;
    std::cout << "? 2 3" << std::endl;
    std::cin >> mult23;
    std::cout << "? 3 4" << std::endl;
    std::cin >> mult34;
    std::cout << "? 4 5" << std::endl;
    std::cin >> mult45;

    std::array<int, 6> flags{4, 8, 15, 16, 23, 42};
    do
    {
        if (flags[0] * flags[1] == mult12 &&
            flags[1] * flags[2] == mult23 &&
            flags[2] * flags[3] == mult34 &&
            flags[3] * flags[4] == mult45)
            break;
    } while (std::next_permutation(flags.begin(), flags.end()));
    std::cout << "! ";
    for (auto a : flags)
    {
        std::cout << a << ' ';
    }
    std::cout << std::endl;
}
