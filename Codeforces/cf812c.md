void solve()
{
    int n, s;
    std::cin >> n >> s;
    std::vector<std::pair<int64_t, int64_t>> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i].first;
        as[i].second = i;
    }

    auto check = [&as, n, s](int k) -> std::pair<bool, int>
    {
        auto v{ as };
        std::sort(as.begin() + 1, as.end(), 
            [k](std::pair<int64_t, int64_t> a, std::pair<int64_t, int64_t> b)
            {
                int64_t val1{ a.first + a.second * k };
                int64_t val2{ b.first + b.second * k };
                return val1 < val2;
            }
        );
        int64_t sum{ 0 };
        for (int i{ 1 }; i <= k; ++i)
        {
            int64_t val{ as[i].first + as[i].second * k };
            sum += val;
        }
        return {sum <= s, sum};
    };
    int l{ 0 }, r{ n };
    int ans{ 0 };
    while (l < r)
    {
        int mid{ (l + r + 1) / 2 };
        auto [b, a]{ check(mid) };
        if (b)
            l = mid, ans = a;
        else
            r = mid - 1;
    }
    std::cout << l << ' ' << ans << '\n';
}
