void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t ans{ 0 };
    for (int l{ 0 }; l < n; ++l)
    {
        for (int r{ l }; r < std::min(l + 4, n); ++r)
        {
            if (r - l + 1 == 3)
            {
                if (as[l] <= as[l + 1] && as[l + 1] <= as[l + 2])
                    continue;
                if (as[l] >= as[l + 1] && as[l + 1] >= as[l + 2])
                    continue;
            }
            if (r - l + 1 == 4)
            {
                if (as[l] <= as[l + 1] && as[l + 1] <= as[l + 2])
                    continue;
                if (as[l] <= as[l + 1] && as[l + 1] <= as[l + 3])
                    continue;
                if (as[l] <= as[l + 2] && as[l + 2] <= as[l + 3])
                    continue;
                if (as[l + 1] <= as[l + 2] && as[l + 2] <= as[l + 3])
                    continue;
                if (as[l] >= as[l + 1] && as[l + 1] >= as[l + 2])
                    continue;
                if (as[l] >= as[l + 1] && as[l + 1] >= as[l + 3])
                    continue;
                if (as[l] >= as[l + 2] && as[l + 2] >= as[l + 3])
                    continue;
                if (as[l + 1] >= as[l + 2] && as[l + 2] >= as[l + 3])
                    continue;
            }
            ++ans;
        }
    }
    std::cout << ans << '\n';
}
