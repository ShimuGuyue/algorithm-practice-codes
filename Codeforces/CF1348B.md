void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> digits(as);
    std::sort(digits.begin(), digits.end());
    digits.erase(std::unique(digits.begin(), digits.end()), digits.end());

    if (digits.size() > k)
    {
        std::cout << -1 << '\n';
        return;
    }

    std::vector<bool> visited(n + 1);
    for (auto a : digits)
    {
        visited[a] = true;
    }
    for (int i = 1; i <= n && digits.size() < k; ++i)
    {
        if (!visited[i])
            digits.push_back(i);
    }

    std::vector<int> ans;
    int i{ 0 }, j{ 0 };
    while (i != n)
    {
        ans.push_back(digits[j]);
        if (as[i] == digits[j])
            ++i;
        j = (j + 1) % digits.size();
    }

    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
