void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int count{ 1 };
    for (int i{ 2 }; i <= n; ++i)
    {
        if (as[i] == as[i - 1])
            ++count;
        else
            count = 1;
        if (count == m)
        {
            std::cout << "No\n";
            return;
        }
    }
    std::cout << "Yes\n";
}
