void solve()
{
    std::string s;
    std::cin >> s;

    int ans{ 20 };
    std::sort(s.begin() + 1, s.end());
    {
        int sum{ s[0] - '0' };
        int count{ 0 };
        for (int i{ 1 }; i < int(s.length()); ++i)
        {
            int a{ s[i] - '0' };
            if (sum + a >= 10)
                ++count;
            else
                sum += a;
        }
        ans = std::min(ans, count);
    }
    {
        int sum{ 1 };
        int count{ 1 };
        for (int i{ 1 }; i < int(s.length()); ++i)
        {
            int a{ s[i] - '0' };
            if (sum + a >= 10)
                ++count;
            else
                sum += a;
        }
        ans = std::min(ans, count);
    }
    std::cout << ans << '\n';
}
