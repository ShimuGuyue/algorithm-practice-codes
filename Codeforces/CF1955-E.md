void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int ans{ 0 };
    for (int k{ 1 }; k <= n; ++k)
    {
        bool ok{ true };
        int change{ 0 };
        std::vector<int> flags(n + 1);
        for (int i{ 0 }; i < n; ++i)
        {
            change ^= flags[i];
            int cur{ (s[i] - '0') ^ change };
            if (cur == 1)
                continue;
            if (i + k <= n)
            {
                change ^= 1;
                flags[i + k] ^= 1;
            }
            else
            {
                ok = false;
            }
        }
        if (ok)
            ans = k;
    }
    std::cout << ans << '\n';
}
