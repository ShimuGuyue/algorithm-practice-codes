void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::stack<char> k;
    for (char c : s)
    {
        if (k.empty() || k.top() != c)
            k.push(c);
        else
            k.pop();
    }
    std::cout << (k.empty() ? "YES\n" : "NO\n");
}
