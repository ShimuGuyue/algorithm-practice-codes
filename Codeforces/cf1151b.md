void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> grid(n + 1, std::vector<int>(m + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            std::cin >> grid[i][j];
        }
    }

    int max{ 1024 };
    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(max, -1));
    std::vector<std::vector<int>> last(n + 1, std::vector<int>(max, -1));
    dp[0][0] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int k{ 0 }; k < max; ++k)
        {
            for (int j{ 1 }; j <= m; ++j)
            {
                int a{ grid[i][j] };
                if (dp[i - 1][k ^ a] == -1)
                    continue;
                dp[i][k] = j;
                last[i][k] = k ^ a;
            }
        }
    }

    for (int k{ 1 }; k < max; ++k)
    {
        if (dp[n][k] != -1)
        {
            std::cout << "TAK\n";
            std::vector<int> ans;
            for (int i{ n }; i > 0; --i)
            {
                ans.push_back(dp[i][k]);
                k = last[i][k];
            }
            std::reverse(ans.begin(), ans.end());
            for (auto a : ans)
            {
                std::cout << a << ' ';
            }
            std::cout << '\n';
            return;
        }
    }
    std::cout << "NIE\n";
}
