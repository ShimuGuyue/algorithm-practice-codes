void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t sum{ as[0] };
    int64_t max{ as[0] };
    std::cout << as[0] << ' ';
    for (int i{ 1 }; i < n; ++i)
    {
        sum += as[i];
        int64_t dif{ max - as[i] };
        if (dif)
        {
            if (sum < max * (i + 1))
            {
                max = sum / (i + 1);
            }
        }
        std::cout << max << ' ';
    }
    std::cout << '\n';
}
