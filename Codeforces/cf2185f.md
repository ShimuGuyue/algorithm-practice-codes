void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as_init(1 << n);
    for (auto& a : as_init)
    {
        std::cin >> a;
    }

    std::vector<int> as{ as_init };

    std::map<std::pair<int, int>, int> flags;
    auto dfs1 = [&as, &flags](auto&& dfs, int l, int r) -> int
    {
        if (l == r)
        {
            flags[{l, r}] = as[l];
            return as[l];
        }
        int mid{ (l + r) / 2 };
        int a{ dfs(dfs, l , mid) };
        int b{ dfs(dfs, mid + 1, r) };
        flags[{l, r}] = a ^ b;
        return a ^ b;
    };
    dfs1(dfs1, 0, (1 << n) - 1);

    while (q--)
    {
        int b, c;
        std::cin >> b >> c;
        --b;
        as[b] = c;

        int ans{ 0 };
        auto dfs2 = [&as, &flags, &ans, b](auto&& dfs, int l, int r, int level) -> int
        {
            if (l == r)
                return as[l];
            int mid{ (l + r) / 2 };
            int ll, rr;
            if (b <= mid)
            {
                ll = dfs(dfs, l, mid, level - 1);
                rr = flags[{mid + 1, r}];
                if (rr > ll)
                    ans += 1 << (level - 1);
            }
            else
            {
                ll = flags[{l, mid}];
                rr = dfs(dfs, mid + 1, r, level - 1);
                if (ll >= rr)
                    ans += 1 << (level - 1);
            }
            return ll ^ rr;
        };
        dfs2(dfs2, 0, (1 << n) - 1, n);
        std::cout << ans << '\n';

        as[b] = as_init[b];
    }
}
