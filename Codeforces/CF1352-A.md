void solve()
{
    std::string s;
    std::cin >> s;

    int n{ int(s.length()) };

    std::vector<std::string> ans;
    for (int i = 0; i < n; ++i)
    {
        if (s[i] == '0')
            continue;
        std::string temp;
        temp += s[i];
        for (int j = i + 1; j < n; ++j)
        {
            temp += '0';
        }
        ans.push_back(temp);
    }

    std::cout << ans.size() << '\n';
    for (auto& str : ans)
    {
        std::cout << str << ' ';
    }
    std::cout << '\n';
}
