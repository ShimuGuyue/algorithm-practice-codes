void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::array<int, 3>> edges(m);
    std::vector<std::vector<std::pair<int, int>>> graph(n + 1);
    for (auto& [u, v, w] : edges)
    {
        std::cin >> u >> v >> w;
        graph[u].push_back({ v, w });
        graph[v].push_back({ u, w });
        auto [min, max]{ std::minmax(u, v) };
    }

    int64_t sum{ 0 };
    std::vector<std::vector<std::pair<int, int>>> tree(n + 1);
    std::priority_queue<std::array<int, 3>, std::vector<std::array<int, 3>>, std::greater<std::array<int, 3>>> h;
    std::vector<bool> visited(n + 1);
    visited[1] = true;
    for (auto [v, w] : graph[1])
    {
        h.push({ w, 1, v });
    }
    while (!h.empty())
    {
        auto [w, u, v]{ h.top() };
        h.pop();
        if (visited[u] && visited[v])
            continue;
        if (visited[v])
            std::swap(u, v);
        tree[u].push_back({ v, w });
        tree[v].push_back({ u, w });
        sum += w;
        visited[v] = true;
        for (auto [vv, ww] : graph[v])
        {
            if (visited[vv])
                continue;
            h.push({ ww, vv, v });
        }
    }

    int k{ int(std::log2(n) + 1) };
    std::vector<std::vector<int>> ups(n + 1, std::vector<int>(k, 1));
    std::vector<std::vector<int>> maxs(n + 1, std::vector<int>(k));
    std::vector<int> depths(n + 1);
    auto build = [&tree, &ups, &maxs, &depths, k](this auto&& build, int u, int last, int w) -> void
    {
        ups[u][0] = last;
        maxs[u][0] = w;
        depths[u] = depths[last] + 1;
        for (int i{ 1 }; i < k; ++i)
        {
            ups[u][i] = ups[ups[u][i - 1]][i - 1];
            maxs[u][i] = std::max(maxs[u][i - 1], maxs[ups[u][i - 1]][i - 1]);
        }
        for (auto [v, w] : tree[u])
        {
            if (v == last)
                continue;
            build(v, u, w);
        }
    };
    build(1, 1, 0);

    for (auto [u, v, w] : edges)
    {
        int max{ 0 };

        if (depths[u] < depths[v])
            std::swap(u, v);
        int dif{ depths[u] - depths[v] };
        for (int i{ 0 }; i < k; ++i)
        {
            if ((dif >> i & 1) == 0)
                continue;
            max = std::max(max, maxs[u][i]);
            u = ups[u][i];
        }
        if (u != v)
        {
            for (int i{ k - 1 }; i >= 0; --i)
            {
                if (ups[u][i] == ups[v][i])
                    continue;
                max = std::max({ max, maxs[u][i], maxs[v][i] });
                u = ups[u][i];
                v = ups[v][i];
            }
            max = std::max({ max, maxs[u][0], maxs[v][0] });
        }

        int64_t ans{ sum - max + w };
        std::cout << ans << '\n';
    }
}
