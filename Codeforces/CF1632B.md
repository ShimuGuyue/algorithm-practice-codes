void solve()
{
    int n;
    std::cin >> n;
    int flag{ 1 };
    while (flag <= n - 1)
    {
        flag *= 2;
    }
    flag /= 2;

    for (int i = n - 1; i >= flag; --i)
    {
        std::cout << i << ' ';
    }
    for (int i = 0; i < flag; ++i)
    {
        std::cout << i << ' ';
    }
    std::cout << '\n';
}
