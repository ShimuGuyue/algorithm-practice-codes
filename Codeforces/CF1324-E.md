void solve()
{
    int n, h, l, r;
    std::cin >> n >> h >> l >> r;
    std::vector<int> as(n);
    for (auto&a : as)
    {
        std::cin >> a;
    }

    std::map<int, int> m;
    m[0] = 0;
    for (auto a : as)
    {
        std::map<int, int> temp;
        for (auto [k, v] : m)
        {
            int t1{ (k + a) % h };
            if (t1 >= l && t1 <= r)
                temp[t1] = std::max(temp[t1], v + 1);
            else
                temp[t1] = std::max(temp[t1], v);
            int t2{ (k + a - 1) % h };
            if (t2 >= l && t2 <= r)
                temp[t2] = std::max(temp[t2], v + 1);
            else
                temp[t2] = std::max(temp[t2], v);
        }
        m = temp;
    }
    int ans{ 0 };
    for (auto [k, v] : m)
    {
        ans = std::max(ans, v);
    }
    std::cout << ans << '\n';
}
