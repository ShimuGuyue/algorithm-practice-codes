

```c++
static constexpr int mod{ 1000000007 };
std::array<int64_t, 200001> factorials;
void init()
{
    factorials[0] = 1;
    factorials[1] = 1;
    for (int i = 2; i <= 200000; ++i)
    {
        factorials[i] = factorials[i - 1] * i % mod;
    }
}
```

```c++
void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int flag = std::accumulate(as.begin(), as.end(), INT_MAX,
        [](int sum, int next)
        {
            return sum & next;
        }
    );

    int64_t count = std::count(as.begin(), as.end(), flag);

    int64_t ans{ count * (count - 1) % mod * factorials[n - 2] % mod };
    std::cout << ans << '\n';
}
```
