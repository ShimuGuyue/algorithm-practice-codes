void solve()
{
    int n, m, h;
    std::cin >> n >> m >> h;
    std::vector<int> as_init(n);
    for (auto& a : as_init)
    {
        std::cin >> a;
    }

    std::vector<std::pair<int, int>> as(n);
    for (int i = 0; i < n; ++i)
    {
        as[i].first = as_init[i];
    }

    int count{ 0 };
    while (m--)
    {
        int b, c;
        std::cin >> b >> c;
        --b;
        if (as[b].second < count )
        {
            as[b] = { as_init[b] + c, count };
        }
        else
        {
            as[b].first += c;
        }
        if (as[b].first > h)
            ++count;
    }

    for (int i = 0; i < n; ++i)
    {
        std::cout << (as[i].second == count ? as[i].first : as_init[i]) << ' ';
    }
    std::cout << '\n';
}
