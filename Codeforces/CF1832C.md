void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    as.erase(std::unique(as.begin(), as.end()), as.end());

    int ans{ static_cast<int>(as.size()) };
    for (int i{ 1 }; i < static_cast<int>(as.size()) - 1; ++i)
    {
        if (as[i] > as[i - 1] && as[i] > as[i + 1])
            continue;
        if (as[i] < as[i - 1] && as[i] < as[i + 1])
            continue;
        --ans;
    }
    std::cout << ans << '\n';
}
