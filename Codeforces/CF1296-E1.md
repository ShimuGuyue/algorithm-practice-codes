void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::vector<std::vector<int>> edges(n);
    for (int i{ 0 }; i < n; ++i)
    {
        for (int j{ 0 }; j < i; ++j)
        {
            if (s[j] > s[i])
            {
                edges[i].push_back(j);
                edges[j].push_back(i);
            }
        }
    }

    std::vector<int> colors(n, -1);
    std::queue<int> q;
    for (int i{ 0 }; i < n; ++i)
    {
        if (colors[i] != -1)
            continue;
        colors[i] = 0;
        q.push(i);
        while (!q.empty())
        {
            int u{ q.front() };
            q.pop();
            for (int v : edges[u])
            {
                if (colors[v] == colors[u])
                {
                    std::cout << "NO\n";
                    return;
                }
                if (colors[v] == -1)
                {
                    colors[v] = colors[u] == 0 ? 1 : 0;
                    q.push(v);
                }
            }
        }
    }
    std::cout << "Yes\n";
    for (int a : colors)
    {
        std::cout << a;
    }
    std::cout << '\n';
}
