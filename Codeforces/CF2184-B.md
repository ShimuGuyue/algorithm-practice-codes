void solve()
{
    int s, k, m;
    std::cin >> s >> k >> m;

    int ans{ -1 };
    if (k >= s)
    {
        int leave{ m % k };
        ans = s - leave < 0 ? 0 : s - leave;
    }
    else
    {
        int count{ m / k };
        int sum{ (count & 1) ? k : s };
        int leave{ m % k };
        ans = sum - leave < 0 ? 0 : sum - leave;
    }
    std::cout << ans << '\n';
}
