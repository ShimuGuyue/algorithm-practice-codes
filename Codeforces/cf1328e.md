```cpp
class LCA
{
private:
    int n_, m_;
    std::vector<int> depths_;
    std::vector<std::vector<int>> ups_;
    const std::vector<std::vector<int>>& tree_;

    std::set<std::pair<int, int>> edges;

public:
    LCA(const std::vector<std::vector<int>>& tree, const int root) : tree_(tree)
    {
        n_ = static_cast<int>(tree.size());
        m_ = static_cast<int>(std::ceil(std::log2(tree.size()))) + 1;

        depths_.assign(n_, 0);
        ups_.assign(n_, std::vector<int>(m_, 0));

        for (int u{ 1 }; u < n_; ++u)
        {
            for (int v : tree[u])
            {
                edges.insert({ u, v });
                edges.insert({ v, u });
            }
        }

        build(root, 0, 1);
    }

    int calc(int u, int v) const
    {
        if (depths_[u] < depths_[v])
            std::swap(u, v);
        const int dif{ depths_[u] - depths_[v] };
        for (int i{ 0 }; i < m_; ++i)
        {
            if (dif >> i & 1)
                u = ups_[u][i];
        }

        if (u == v)
            return u;

        for (int i{ m_ - 1 }; i >= 0; --i)
        {
            if (ups_[u][i] != ups_[v][i])
            {
                u = ups_[u][i];
                v = ups_[v][i];
            }
        }
        return ups_[u][0];
    }

    bool judge(std::vector<int>& as)
    {
        std::vector<int>& depths(depths_);
        std::sort(as.begin(), as.end(),
            [&depths](int a, int b)
            {
                return depths[a] > depths[b];
            }
        );

        for (int i{ 1 }; i < as.size(); ++i)
        {
            int l{ calc(as[0], as[i]) };
            if (l == as[0] || l == as[i])
                continue;
            if (edges.count({ l, as[i] }))
                continue;
            return false;
        }
        return true;
    }

private:
    void build(const int u, const int last, const int deepth)
    {
        depths_[u] = deepth;
        ups_[u][0] = last;
        for (int i{ 1 }; i < m_; ++i)
        {
            ups_[u][i] = ups_[ups_[u][i - 1]][i - 1];
        }
        for (const int v : tree_[u])
        {
            if (v == last)
                continue;
            build(v, u, deepth + 1);
        }
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    LCA lca(tree, 1);

    while (m--)
    {
        int k;
        std::cin >> k;
        std::vector<int> as(k);
        for (auto& a : as)
        {
            std::cin >> a;
        }
        if (lca.judge(as))
            std::cout << "YES\n";
        else
            std::cout << "NO\n";
    }
}
```
