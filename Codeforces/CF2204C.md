void solve()
{
    int64_t a, b, c, m;
    std::cin >> a >> b >> c >> m;

    std::array<int64_t, 3> ans{ };

    int64_t l123{ std::lcm(std::lcm(a, b), c) };
    int64_t c123{ m / l123 };
    ans[0] += 2 * c123;
    ans[1] += 2 * c123;
    ans[2] += 2 * c123;

    int64_t l12{ std::lcm(a, b) };
    int64_t c12{ m / l12 - c123 };
    ans[0] += 3 * c12;
    ans[1] += 3 * c12;

    int64_t l13{ std::lcm(a, c) };
    int64_t c13{ m / l13 - c123 };
    ans[0] += 3 * c13;
    ans[2] += 3 * c13;

    int64_t l23{ std::lcm(b, c) };
    int64_t c23{ m / l23 - c123 };
    ans[1] += 3 * c23;
    ans[2] += 3 * c23;

    int64_t c1{ m / a - c12 - c13 - c123 };
    ans[0] += 6 * c1;

    int64_t c2{ m / b - c12 - c23 - c123 };
    ans[1] += 6 * c2;

    int64_t c3{ m / c - c13 - c23 - c123 };
    ans[2] += 6 * c3;

    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
