void solve()
{
    int a, b, c;
    std::cin >> a >> b >> c;

    std::map<int, int> m;   // deep, count
    m[0] = 1;
    while (a)
    {
        auto [k, v]{ *m.begin() };
        m.erase(m.begin());

        if (a >= v)
        {
            m[k + 1] += v * 2;
            a -= v;
        }
        else
        {
            m[k] += v - a;
            m[k + 1] += a * 2;
            a = 0;
        }
    }
    while (b)
    {
        auto [k, v]{ *m.begin() };
        m.erase(m.begin());

        if (b >= v)
        {
            m[k + 1] += v;
            b -= v;
        }
        else
        {
            m[k] += v - b;
            m[k + 1] += b;
            b = 0;
        }
    }

    int sum{ 0 };
    for (auto [k, v] : m)
    {
        sum += v;
    }
    int ans{ -1 };
    if (sum == c)
        ans = m.rbegin()->first;
    std::cout << ans << '\n';
}
