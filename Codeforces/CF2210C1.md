void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    int ans{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        int g1{ i == 0 ? 1 : std::gcd(as[i], as[i - 1]) };
        int g2{ i == n - 1 ? 1 : std::gcd(as[i], as[i + 1]) };
        int l{ std::lcm(g1, g2) };
        if (as[i] != l)
            ++ans;
    }
    std::cout << ans << '\n';
}
