void solve()
{
    int n;
    std::cin >> n;

    std::map<int, int64_t> flags_1; // 水平
    std::map<int, int64_t> flags_2; // 竖直
    std::map<int, int64_t> flags_3; // 右上
    std::map<int, int64_t> flags_4; // 右下
    while (n--)
    {
        int x, y;
        std::cin >> x >> y;
        ++flags_1[y];
        ++flags_2[x];
        ++flags_3[y - x];
        ++flags_4[y + x];
    }

    int64_t ans{ 0 };
    for (auto [k, v] : flags_1)
    {
        ans += v * (v - 1);
    }
    for (auto [k, v] : flags_2)
    {
        ans += v * (v - 1);
    }
    for (auto [k, v] : flags_3)
    {
        ans += v * (v - 1);
    }
    for (auto [k, v] : flags_4)
    {
        ans += v * (v - 1);
    }
    std::cout << ans << '\n';
}
