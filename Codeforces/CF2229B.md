void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    for (int i{ 0 }; i < n; ++i)
    {
        if (as[i] > bs[i])
            std::swap(as[i], bs[i]);
    }

    int64_t ans{ std::accumulate(bs.begin(), bs.end(), INT64_C(0)) + *std::max_element(as.begin(), as.end()) };
    std::cout << ans << '\n';
}
