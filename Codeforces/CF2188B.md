void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    if (std::count(s.begin(), s.end(), '1') == 0)
    {
        std::cout << (n + 2) / 3 << '\n';
        return;
    }

    int front{ -1 }, back{ -1 };
    for (int i = 0; i < n; ++i)
    {
        if (s[i] == '1')
        {
            if (front == -1)
                front = i;
            back = i;
        }
    }

    int ans{ std::count(s.begin(), s.end(), '1') };
    ans += (front + 1) / 3;
    ans += (n - back - 1 + 1) / 3;
    int count{ 0 };
    for (int i = front; i <= back; ++i)
    {
        if (s[i] == '1')
        {
            ans += count / 3;
            count = 0;
        }
        else
        {
            ++count;
        }
    }
    std::cout << ans << '\n';
}
