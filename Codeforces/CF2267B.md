void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    map<int, int> m;
    for (auto a : as)
    {
        ++m[a];
    }

    while (1)
    {
        bool end = true;
        int flag = 0;
        for (auto it = m.rbegin(); it != m.rend(); ++it)
        {
            if (it->second != 0)
            {
                flag = it->second;
                end = false;
                break;
            }
        }
        for (auto it = m.rbegin(); it != m.rend(); ++it)
        {
            auto& [k, v] = *it;
            int count = std::min(v, flag);
            v -= count;
            for (int i = 0; i < count; ++i)
            {
                cout << k << ' ';
            }
        }
        if (end)
            break;
    }

    cout << '\n';
}
