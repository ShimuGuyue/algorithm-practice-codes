void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::map<int, int> precounts;
    while (n--)
    {
        int a;
        std::cin >> a;
        ++precounts[a];
    }

    for (auto it{ precounts.begin() }; std::next(it) != precounts.end(); ++it)
    {
        std::next(it)->second += it->second;
    }
    precounts[-2000000000] = 0;

    while (m--)
    {
        int b;
        std::cin >> b;
        std::cout << std::prev(precounts.upper_bound(b))->second << ' ';
    }
    std::cout << '\n';
}
