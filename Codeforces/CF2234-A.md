void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.rbegin(), as.rend());

    for (int i{ 0 }; i + 2 < n; ++i)
    {
        if (as[i] % as[i + 1] != as[i + 2])
        {
            std::cout << -1 << '\n';
            return;
        }
    }
    std::cout << as[0] << ' ' << as[1] << '\n';
}
