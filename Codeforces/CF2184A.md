void solve()
{
    int n;
    std::cin >> n;

    for (int i = 0; i <= n; ++i)
    {
        int temp{ n + i };
        if (temp & 1)
            continue;
        int x{ temp / 2 };
        int y{ x - i };
        if (judge(x) && judge(y))
        {
            std::cout << i << '\n';
            return;
        }
    }
}

bool judge(int x)
{
    return x != 1;
}
