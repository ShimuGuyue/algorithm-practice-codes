void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i = 2; i <= n; ++i)
    {
        int a;
        std::cin >> a;
        tree[i].push_back(a);
        tree[a].push_back(i);
    }
    std::string s;
    std::cin >> s;
    s = " " + s;

    int ans{ 0 };
    auto dfs = [&ans, &tree, &s](auto&& dfs, int u, int last) -> std::pair<int, int>
    {
        int a{ 0 }, b{ 0 };
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            auto [aa, bb]{ dfs(dfs, v, u) };
            a += aa;
            b += bb;
        }
        if (s[u] == 'W')
            ++a;
        else
            ++b;
        if (a == b)
            ++ans;
        return {a, b};
    };
    dfs(dfs, 1, 0);

    std::cout << ans << '\n';
}
