void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int64_t> suf_gcds(n + 1);
    std::partial_sum(as.rbegin(), as.rend(), suf_gcds.rbegin() + 1,
        [](int64_t pre, int64_t next)
        {
            return std::gcd(pre, next);
        }
    );

    int64_t ans{ 0 };
    for (int i = 0; i < n; ++i)
    {
        ans = std::gcd(ans, as[i] * suf_gcds[i + 1]);
    }
    ans /= suf_gcds[0];
    std::cout << ans << '\n';
}
