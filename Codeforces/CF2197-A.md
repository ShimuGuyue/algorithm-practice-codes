void solve()
{
    int x;
    std::cin >> x;
    int ans{ 0 };
    for (int i = 0; i <= 81; ++i)
    {
        int y{ x + i };
        int d{ 0 };
        while (y)
        {
            d += y % 10;
            y /= 10;
        }
        if (d == i)
            ++ans;
    }
    std::cout << ans << '\n';
}
