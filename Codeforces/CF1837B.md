void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int ans{ 1 };
    int sum{ 1 };
    for (int i{ 1 }; i < n; ++i)
    {
        if (s[i] == s[i - 1])
            ++sum;
        else
        {
            ans = std::max(ans, sum);
            sum = 1;
        }
    }
    ans = std::max(ans, sum);
    std::cout << ans + 1 << '\n';
}
