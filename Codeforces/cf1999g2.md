void solve()
{
    int l{ 2 }, r{ 999 };
    while (l < r)
    {
        int ll{ (l + l + r) / 3 };
        int rr{ (l + r + r) / 3 };
        std::cout << "? " << ll << ' ' << rr << std::endl;
        int x;
        std::cin >> x;
        if (x == ll * rr)
            l = rr + 1;
        else if (x == ll * (rr + 1))
            l = ll + 1,
            r = rr;
        else if (x == (ll + 1) * (rr + 1))
            r = ll;
    }
    std::cout << "! " << r << std::endl;
}
