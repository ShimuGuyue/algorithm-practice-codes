void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    if (as.front() != 1)
    {
        std::cout << "no\n";
        return;
    }

    int presum{ 1 };
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] > presum)
        {
            std::cout << "no\n";
            return;
        }
        presum += as[i];
    }
    std::cout << "yes\n";
}
