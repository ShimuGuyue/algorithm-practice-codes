void solve()
{
    int n, x, y;
    std::cin >> n >> x >> y;

    int64_t ans{ 0 };
    std::map<std::pair<int, int>, int> counts;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        int target_x{ x - (a % x) };
        if (target_x == x)
            target_x = 0;
        int target_y{ a % y };
        ans += counts[{target_x, target_y}];
        ++counts[std::make_pair(a % x, a % y)];
    }
    std::cout << ans << '\n';
}
