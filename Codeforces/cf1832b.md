void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::sort(as.begin(), as.end());

    std::vector<int64_t> pres(n + 1);
    std::partial_sum(as.begin(), as.end(), pres.begin());

    int64_t ans{ 0 };
    for (int i{ 0 }; i <= k; ++i)
    {
        ans = std::max(ans, pres[n - k + i] - pres[i * 2]);
    }
    std::cout << ans << '\n';
}
