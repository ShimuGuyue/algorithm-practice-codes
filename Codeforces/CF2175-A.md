void solve()
{
    int n;
    std::cin >> n;
    std::set<int> t;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        t.insert(a);
    }
    while (t.find(t.size()) == t.end())
    {
        t.insert(t.size());
    }
    std::cout << t.size() << '\n';
}
