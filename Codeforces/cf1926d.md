void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::map<int, int> m;
    for (auto& a : as)
    {
        ++m[a];
    }

    int flag{ std::numeric_limits<int>::max() };

    int ans{ n };
    for (auto [k, v] : m)
    {
        if (k > (flag ^ k))
            break;
        if (m.count(flag ^ k))
            ans -= std::min(v, m[flag ^ k]);
    }
    std::cout << ans << '\n';
}
