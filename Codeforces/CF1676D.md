void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> grid(n, std::vector<int>(m));
    for (auto& v : grid)
    {
        for (auto& a : v)
        {
            std::cin >> a;
        }
    }

    std::map<int, int> flags1;  // y =  x + b
    std::map<int, int> flags2;  // y = -x + b
    for (int x = 0; x < n; ++x)
    {
        for (int y = 0; y < m; ++y)
        {
            int b1{ y - x };
            int b2{ y + x };
            flags1[b1] += grid[x][y];
            flags2[b2] += grid[x][y];
        }
    }

    int ans{ 0 };
    for (int x = 0; x < n; ++x)
    {
        for (int y = 0; y < m; ++y)
        {
            int b1{ y - x };
            int b2{ y + x };
            ans = std::max(ans, flags1[b1] + flags2[b2] - grid[x][y]);
        }
    }
    std::cout << ans << '\n';
}
