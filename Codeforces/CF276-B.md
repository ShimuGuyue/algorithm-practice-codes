void solve()
{
    std::string s;
    std::cin >> s;

    std::array<int, 'z' + 1> counts{ };
    for (char c : s)
    {
        ++counts[c];
    }

    int count_odd{
        std::count_if(counts.begin(), counts.end(),
            [](int a)
            {
                return a % 2;
            }
        )
    };
    if (count_odd <= 1)
    {
        std::cout << "First\n";
        return;
    };

    std::cout << ((s.length() & 1) ? "First" : "Second") << '\n';
}
