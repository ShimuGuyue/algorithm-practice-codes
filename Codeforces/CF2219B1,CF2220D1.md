void solve()
{
    int n;
    std::cin >> n;
    n = 2 * n + 1;

    int a, b, c;

    int l{ 1 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        std::cout << "? " << mid << ' ';
        for (int i{ 1 }; i <= mid; ++i)
        {
            std::cout << i << ' ';
        }
        std::cout << std::endl;

        int x;
        std::cin >> x;
        if ((mid - x) & 1)
            r = mid;
        else
            l = mid + 1;
    }
    c = r;

    l = 1; r = c - 1;
    while (l < r)
    {
        int mid{ (l + r + 1) / 2 };
        std::cout << "? " << c - mid + 1 << ' ';
        for (int i{ mid }; i <= c; ++i)
        {
            std::cout << i << ' ';
        }
        std::cout << std::endl;

        int x;
        std::cin >> x;
        if ((c - mid + 1 - x) & 1)
            l = mid;
        else
            r = mid - 1;
    }
    a = l;

    l = a + 1, r = c - 1;
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        std::cout << "? " << mid - a + 2 << ' ';
        for (int i{ a }; i <= mid; ++i)
        {
            std::cout << i << ' ';
        }
        std::cout << c << std::endl;

        int x;
        std::cin >> x;
        if ((mid - a + 2 - x) & 1)
            r = mid;
        else
            l = mid + 1;
    }
    b = r;

    std::cout << "! " << a << ' ' << b << ' ' << c << std::endl;
}
