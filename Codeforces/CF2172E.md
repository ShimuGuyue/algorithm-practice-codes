void solve()
{
    std::string s;
    int j, k;
    std::cin >> s >> j >> k;
    if (j > k)
        std::swap(j, k);
    for (int i{ 1 }; i < j; ++i)
    {
        std::next_permutation(s.begin(), s.end());
    }
    std::string s1{ s };
    for (int i{ j }; i < k; ++i)
    {
        std::next_permutation(s.begin(), s.end());
    }
    std::string s2{ s };
    int a{ 0 }, b{ 0 };
    for (int i{ 0 }; i < s.length(); ++i)
    {
        if (s1[i] == s2[i])
            ++a;
        else
            ++b;
    }
    std::cout << a << 'A' << b << 'B' << '\n';
}
