void solve()
{
    constexpr int m{ 200 };

    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<int>> indexs(m + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        indexs[as[i]].push_back(i);
    }

    std::vector<std::vector<int>> presums(m + 1, std::vector<int>(n + 1));
    for (int i{ 1 }; i <= m; ++i)
    {
        for (int j{ 1 }; j <= n; ++j)
        {
            presums[i][j] = presums[i][j - 1] + (as[j] == i);
        }
    }

    int ans{ 1 };
    for (int i{ 1 }; i <= m; ++i)
    {
        if (indexs[i].empty())
            continue;
        int l{ 0 }, r{ int(indexs[i].size()) - 1 };
        while (l < r)
        {
            int ll{ indexs[i][l] };
            int rr{ indexs[i][r] };
            int max{ 0 };
            for (int j{ 1 }; j <= m; ++j)
            {
                max = std::max(max, presums[j][rr - 1] - presums[j][ll]);
            }
            ans = std::max(ans, (l + 1) * 2 + max);
            ++l;
            --r;
        }
    }
    std::cout << ans << '\n';
}
