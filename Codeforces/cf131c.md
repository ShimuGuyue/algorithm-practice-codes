void solve()
{
    int n, m, t;
    std::cin >> n >> m >> t;

    int64_t ans{ 0 };
    for (int i = 4; i <= n; ++i)
    {
        int j{ t - i };
        if (j < 1 || j > m)
            continue;
        ans += combinate(n, i) * combinate(m, j);
    }
    std::cout << ans << '\n';
}

int64_t combinate(int n, int k)
{
    int64_t ans{ 1 };
    for (int i = 1; i <= k; ++i)
    {
        ans = ans * (n - i + 1) / i;
    }
    return ans;
}
