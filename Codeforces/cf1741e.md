void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<bool> dp(n + 1);
    dp[0] = true;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (i - as[i] - 1 >= 0)
            dp[i] = (dp[i] || dp[i - as[i] - 1]);
        if (i + as[i] <= n)
            dp[i + as[i]] = (dp[i + as[i]] || dp[i - 1]);
    }
    std::cout << (dp[n] ? "yes" : "no") << '\n';
}
