void solve()
{
    int64_t n, k;
    std::cin >> n >> k;

    int64_t sum{ 0 }, ans{ 0 };
    for (int b{ 1 }; ; b *= 2)
    {
        int64_t flag{ b * k };
        if (sum + flag <= n)
        {
            ans += k;
            sum += flag;
        }
        else
        {
            ans += (n - sum) / b;
            break;
        }
    }
    std::cout << ans << '\n';
}
