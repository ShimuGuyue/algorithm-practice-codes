void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::string> grid(2);
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    std::vector<std::vector<bool>> visited(2, std::vector<bool>(n));
    auto dfs = [&grid, n, &visited](this auto&& dfs, int x, int y) -> int
    {
        visited[x][y] = true;
        int ans{ 1 };
        if (x == 0 && grid[x + 1][y] == '.' && !visited[x + 1][y])
            ans += dfs(x + 1, y);
        if (x == 1 && grid[x - 1][y] == '.' && !visited[x - 1][y])
            ans += dfs(x - 1, y);
        if (y + 1 != n && grid[x][y + 1] == '.' && !visited[x][y + 1])
            ans += dfs(x, y + 1);
        return ans;
    };
    for (int j{ 0 }; j < n; ++j)
    {
        for (int i : { 0, 1 })
        {
            if (grid[i][j] == '#')
                continue;
            if (visited[i][j])
                continue;
            int sum{ dfs(i, j) };
            if (sum & 1)
            {
                std::cout << "None\n";
                return;
            }
        }
    }

    bool more{ false };
    for (int j{ 0 }; j < n - 1; ++j)
    {
        if (grid[0][j] == '.' && grid[1][j] == '.')
        {
            if (grid[0][j + 1] == '.' && grid[1][j + 1] == '.')
                more = true;
            grid[0][j] = grid[1][j] = '#';
        }
        else
        {
            if (grid[0][j] == '.')
            {
                if (grid[0][j + 1] != '.')
                {
                    std::cout << "None\n";
                    return;
                }
                grid[0][j] = grid[0][j + 1] = '#';
            }
            if (grid[1][j] == '.')
            {
                if (grid[1][j + 1] != '.')
                {
                    std::cout << "None\n";
                    return;
                }
                grid[1][j] = grid[1][j + 1] = '#';
            }
        }
    }
    std::cout << (more ? "Multiple" : "Unique") << '\n';
}
