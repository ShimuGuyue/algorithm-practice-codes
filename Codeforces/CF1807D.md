void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        as[i] %= 2;
    }

    std::vector<int> presums(n + 1);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    while (q--)
    {
        int l, r, k;
        std::cin >> l >> r >> k;
        int sum { presums[n] - presums[r] + presums[l - 1] + (k % 2) * (r - l + 1) };
        std::cout << ((sum & 1) ? "YES\n" : "NO\n");
    }
}
