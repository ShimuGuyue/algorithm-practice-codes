void solve()
{
    int64_t n, y;
    std::cin >> n >> y;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    static int m{ 200000 };
    std::vector<int> counts(m + 1);
    for (auto a : as)
    {
        ++counts[a];
    }
    std::vector<int> pres(m + 1);
    std::partial_sum(counts.begin(), counts.end(), pres.begin());

    int64_t ans{ -(INT64_C(1) << 60) };
    for (int x{ 2 }; x <= m; ++x)
    {
        int64_t sum{ 0 };
        int64_t count{ 0 };
        for (int k{ 1 }; k <= (m + x - 1) / x; ++k)
        {
            int l{ (k - 1) * x + 1 };
            int r{ std::min(m, k * x) };
            int cur{ pres[r] - pres[l - 1] };
            sum += int64_t(k) * cur;
            int last{ counts[k] };
            count += std::max(0, cur - last);
        }
        ans = std::max(ans, sum - y * count);
    }
    std::cout << ans << '\n';
}
