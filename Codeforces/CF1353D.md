```c++
struct Pair
{
    int l, r;
    int len;

    bool operator<(const Pair& o) const
    {
        return len != o.len ?
                len < o.len :
                l > o.l;
    }
};
```

```c++
void solve()
{
    int n;
    std::cin >> n;

    std::vector<int> ans(n);
    std::priority_queue<Pair, std::vector<Pair>, std::less<Pair>> h;
    h.push({0, n - 1, n});
    int flag{ 1 };
    while (!h.empty())
    {
        auto [l, r, len]{ h.top() };
        h.pop();
        int mid{ (l + r) / 2 };
        ans[mid] = flag++;
        if (mid != l)
            h.push({l, mid - 1, mid - l});
        if (mid != r)
            h.push({mid + 1, r, r - mid});
    }
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
```
