void solve()
{
    static constexpr int64_t mod{ 1000000007 };

    int64_t n;
    std::cin >> n;

    int64_t ans{ n };
    int64_t lcm{ 1 };
    for (int i{ 1 }; lcm <= n; ++i)
    {
        lcm = std::lcm(lcm, i);
        ans += n / lcm;
        ans %= mod;
    }
    std::cout << ans << '\n';
}
