void solve()
{
    std::string s;
    std::cin >> s;

    if (s.find('1') == std::string::npos)
    {
        std::cout << 0 << '\n';
        return;
    }
    if (s.find('2') == std::string::npos)
    {
        std::cout << 0 << '\n';
        return;
    }
    if (s.find('3') == std::string::npos)
    {
        std::cout << 0 << '\n';
        return;
    }

    int ans{ static_cast<int>(s.length()) };
    std::map<char, int> m;
    int l{ 0 }, r{ 0 };
    while (r < s.length())
    {
        ++m[s[r]];
        while (m[s[l]] > 1)
        {
            --m[s[l]];
            ++l;
        }
        if (m['1'] && m['2'] && m['3'])
            ans = std::min(ans, r - l + 1);
        ++r;
    }
    std::cout << ans << '\n';
}
