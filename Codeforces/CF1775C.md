void solve()
{
    int64_t n, m;
    std::cin >> n >> m;

    auto highbit = [](int64_t x)
    {
        for (int b{ 60 }; b >= 0; --b)
        {
            if (x >> b & 1)
                return b;
        }
        return -1;
    };
    if (highbit(n) != highbit(m) && m != 0)
    {
        // debug_("highbit不同");
        std::cout << -1 << '\n';
        return;
    }

    for (int b{ 0 }; b < 60; ++b)
    {
        int64_t x{ n >> b & 1 };
        int64_t y{ m >> b & 1 };
        if (x == 0 && y != 0)
        {
            // debug_("n为零m不为零");
            std::cout << -1 << '\n';
            return;
        }
    }

    for (int b{ 1 }; b < 60; ++b)
    {
        int64_t x{ n >> b & 1 };
        int64_t y{ n >> (b - 1) & 1 };
        if (x && y)
        {
            int64_t p{ m >> b & 1 };
            int64_t q{ m >> (b - 1) & 1 };
            if ((p == 1 && q == 1) || (p == 0 && q == 0))
                continue;
            // debug_("n相邻两个一且m只有一个");
            std::cout << -1 << '\n';
            return;
        }
    }

    int64_t ans{ 0 };
    bool find{ false };
    for (int64_t b{ 62 }; b >= 0; --b)
    {
        int64_t x{ n >> b & 1 };
        int64_t y{ m >> b & 1 };
        if (find)
        {
            if (x == 1 && y == 1)
            {
                // debug_("最高不同位之后仍有相同位");
                std::cout << -1 << '\n';
                return;
            }
            continue;
        }
        if (x == 1 && y == 0)
        {
            ans |= INT64_C(1) << (b + 1);
            find = true;
            continue;
        }
        if (x)
            ans |= INT64_C(1) << b;
    }
    std::cout << ans << '\n';
}
