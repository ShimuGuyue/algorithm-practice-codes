void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::vector<int> ans(n);
    std::vector<int> dp(26, 0);
    for (int i{ n - 1 }; i >= 0; --i)
    {
        dp[s[i] - 'a'] = std::max(dp[s[i] - 'a'], 1);
        for (int j{ 0 }; j < s[i] - 'a'; ++j)
        {
            dp[s[i] - 'a'] = std::max(dp[s[i] - 'a'], dp[j] + 1);
        }
        ans[i] = dp[s[i] - 'a'];
    }
    std::cout << *std::max_element(ans.begin(), ans.end()) << '\n';
    for (int a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
