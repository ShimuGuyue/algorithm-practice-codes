void solve()
{
    int a, b;
    std::cin >> a >> b;
    int ans{ std::max(calc(a, b), calc(b, a)) };
    std::cout << ans << '\n';
}

int calc(int a, int b)
{
    int ans{ 0 };
    while (1)
    {
        if (ans & 1)
        {
            if (b < (1 << ans))
                break;
            b -= 1 << ans;
        }
        else
        {
            if (a < (1 << ans))
                break;
            a -= 1 << ans;
        }
        ++ans;
    }
    return ans;
}
