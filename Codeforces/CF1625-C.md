void solve()
{
    int n, l, k;
    std::cin >> n >> l >> k;
    std::vector<int64_t> ds(n);
    std::vector<int64_t> as(n);
    for (auto& d : ds)
    {
        std::cin >> d;
    }
    ds.push_back(l);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(k + 1, 2000000000));
    dp[0][0] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 0 }; j <= std::min(i - 1, k); ++j)
        {
            for (int x{ 0 }; x <= j; ++x)
            {
                if (x == i)
                    continue;
                dp[i][j] = std::min(dp[i][j], dp[i - x - 1][j - x] + as[i - x - 1] * (ds[i] - ds[i - x - 1]));
            }
        }
    }

    int64_t ans{ *std::min_element(dp[n].begin(), dp[n].end()) };
    std::cout << ans << '\n';
}
