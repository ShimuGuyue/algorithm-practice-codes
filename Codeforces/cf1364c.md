    std::vector<int> ans(n + 1, -1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] != as[i - 1])
            ans[i] = as[i - 1];
    }
    int flag{ 0 };
    for (int l{ 1 }, r{ 1 }; r <= n; ++r)
    {
        if (ans[r] == -1)
            continue;
        while (flag < as[r])
        {
            if (flag == ans[r])
            {
                ++flag;
                continue;
            }
            while (ans[l] != -1)
            {
                ++l;
            }
            ans[l] = flag;
            ++l;
            ++flag;
        }
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << (ans[i] == -1 ? 114514 : ans[i]) << ' ';
    }
    std::cout << '\n';
}
