void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<char>> grid(n + 2, std::vector<char>(m + 2, '.'));
    int sum{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            std::cin >> grid[i][j];
            if (grid[i][j] == '*')
                ++sum;
        }
    }

    int count{ 0 };
    for (int i{ 1 }; i + 1 <= n; ++i)
    {
        for (int j{ 1 }; j + 1 <= m; ++j)
        {
            /*
                . . . .
                . * * .
                . * . .
                . . . -
            */
            auto check1 = [&grid, i, j]
            {
                static std::string flag1{".....**..*......"};
                static std::string flag2{".....**..*.....*"};
                std::string s;
                for (int x{ i - 1 }; x <= i + 2; ++x)
                {
                    for (int y{ j - 1 }; y <= j + 2; ++y)
                    {
                        s += grid[x][y];
                    }
                }
                return s == flag1 || s == flag2;
            };
            /*
                . . . .
                . * * .
                . . * .
                - . . .
            */
            auto check2 = [&grid, i, j]
            {
                static std::string flag1{".....**...*....."};
                static std::string flag2{".....**...*.*..."};
                std::string s;
                for (int x{ i - 1 }; x <= i + 2; ++x)
                {
                    for (int y{ j - 1 }; y <= j + 2; ++y)
                    {
                        s += grid[x][y];
                    }
                }
                return s == flag1 || s == flag2;
            };
            /*
                . . . -
                . * . .
                . * * .
                . . . .
            */
            auto check3 = [&grid, i, j]
            {
                static std::string flag1{".....*...**....."};
                static std::string flag2{"...*.*...**....."};
                std::string s;
                for (int x{ i - 1 }; x <= i + 2; ++x)
                {
                    for (int y{ j - 1 }; y <= j + 2; ++y)
                    {
                        s += grid[x][y];
                    }
                }
                return s == flag1 || s == flag2;
            };
            /*
                - . . .
                . . * .
                . * * .
                . . . .
            */
            auto check4 = [&grid, i, j]
            {
                static std::string flag1{"......*..**....."};
                static std::string flag2{"*.....*..**....."};
                std::string s;
                for (int x{ i - 1 }; x <= i + 2; ++x)
                {
                    for (int y{ j - 1 }; y <= j + 2; ++y)
                    {
                        s += grid[x][y];
                    }
                }
                return s == flag1 || s == flag2;
            };
            if (check1() || check2() || check3() || check4())
                ++count;
        }
    }

    if (count * 3 == sum)
        std::cout << "yes\n";
    else
        std::cout << "no\n";
}
