void solve()
{
    int n, a, b, c, d;
    std::cin >> n >> a >> b >> c >> d;
    int ans{ 1 };
    if (a > c)
        ans = std::max(ans, n - c);
    if (a < c)
        ans = std::max(ans, c - 0);
    if (b > d)
        ans = std::max(ans, n - d);
    if (b < d)
        ans = std::max(ans, d - 0);
    std::cout << ans << '\n';
}
