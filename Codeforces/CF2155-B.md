void solve()
{
    int n, k;
    std::cin >> n >> k;

    std::vector<std::vector<bool>> willout(n + 2, std::vector<bool>(n + 2));
    for (int i = 0; i <= n + 1; ++i)
    {
        willout[0][i] = true;
        willout[i][0] = true;
        willout[n + 1][i] = true;
        willout[i][n + 1] = true;
    }

    std::vector<std::vector<char>> grid(n + 1, std::vector<char>(n + 1));
    for (int i = 1; i <= n; ++i)
    {
        for (int j = 1; j <= n; ++j)
        {
            if (k)
            {
                --k;
                willout[i][j] = true;
                if (willout[i - 1][j])
                    grid[i][j] = 'U';
                else if (willout[i + 1][j])
                    grid[i][j] = 'D';
                else if (willout[i][j - 1])
                    grid[i][j] = 'L';
                else if (willout[i][j + 1])
                    grid[i][j] = 'R';
                else
                {
                    std::cout << "NO\n";
                    return;
                }
            }
            else
            {
                if (!willout[i - 1][j])
                    grid[i][j] = 'U';
                else if (!willout[i + 1][j])
                    grid[i][j] = 'D';
                else if (!willout[i][j - 1])
                    grid[i][j] = 'L';
                else if (!willout[i][j + 1])
                    grid[i][j] = 'R';
                else
                {
                    std::cout << "NO\n";
                    return;
                }
            }
        }
    }

    std::cout << "YES\n";
    for (int i = 1; i <= n; ++i)
    {
        for (int j = 1; j <= n; ++j)
        {
            std::cout << grid[i][j];
        }
        std::cout << '\n';
    }
}
