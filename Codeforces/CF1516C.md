void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int sum{ std::accumulate(as.begin(), as.end(), 0) };
    if (sum & 1)
    {
        std::cout << 0 << '\n';
        return;
    }

    int m{ sum / 2 };
    std::vector<bool> dp(m + 1);
    dp[0] = true;
    for (auto a : as)
    {
        for (int i{ m - a }; i >= 0; --i)
        {
            if (dp[i])
                dp[i + a] = true;
        }
    }
    if (!dp[m])
    {
        std::cout << 0 << '\n';
        return;
    }

    std::cout << 1 << '\n';

    auto it{ std::find_if(as.begin(), as.end(), [](int x){ return x & 1; }) };
    if (it != as.end())
    {
        std::cout << std::distance(as.begin(), it) << '\n';
        return;
    }

    int min{ 64 }, index;
    for (int i{ 1 }; i <= n; ++i)
    {
        int a{ as[i] };
        int count{ 0 };
        while (a % 2 == 0)
        {
            ++count;
            a /= 2;
        }
        if (count < min)
        {
            min = count;
            index = i;
        }
    }
    std::cout << index << '\n';
}
