void solve()
{
    int n;
    std::cin >> n;

    if (n == 2 || n == 3)
    {
        std::cout << -1 << '\n';
        return;
    }

    if (n == 4)
    {
        std::cout << "2 4 1 3\n";
        return;
    }

    std::vector<bool> visited(n + 1);
    int index{ 1 };
    std::cout << "1 ";
    visited[1] = true;
    while (index + 2 <= n)
    {
        index += 2;
        std::cout << index << ' ';
        visited[index] = true;
    }
    index -= 3;
    std::cout << index << ' ';
    visited[index] = true;
    if (index + 4 == n)
        index += 4;
    else
        index += 2;
    std::cout << index << ' ';
    visited[index] = true;
    while (index - 2 > 0)
    {
        index -= 2;
        if (visited[index])
            continue;
        std::cout << index << ' ';
        visited[index] = true;
    }
    std::cout << '\n';
}
