void solve()
{
    std::string s;
    std::cin >> s;

    int ans{ static_cast<int>(s.length()) };

    auto calc1 = [&s](char a)
    {
        int ans{ 0 };
        for (char c : s)
        {
            if (c != a)
                ++ans;
        }
        return ans;
    };
    auto calc2 = [&s](char a, char b)
    {
        int ans{ 0 };
        int counta{ 0 }, countb{ 0 };
        for (char c : s)
        {
            if (counta == countb)
            {
                if (c == a)
                    ++counta;
                else
                    ++ans;
            }
            else
            {
                if (c == b)
                    ++countb;
                else
                    ++ans;
            }
        }
        if (counta != countb)
            ++ans;
        return ans;
    };
    for (char a{ '0' }; a <= '9'; ++a)
    {
        ans = std::min(ans, calc1(a));
        for (char b{ '0' }; b <= '9'; ++b)
        {
            if (a == b)
                continue;
            ans = std::min(ans, calc2(a, b));
        }
    }
    std::cout << ans << '\n';
}
