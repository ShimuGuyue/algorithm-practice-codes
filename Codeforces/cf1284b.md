void solve()
{
    int64_t n;
    std::cin >> n;

    std::map<int, int64_t> mins;
    std::vector<int> counts_max(1000000 + 1);

    for (int i = 0; i < n; ++i)
    {
        int m;
        std::cin >> m;
        std::vector<int> as(m);
        for (auto& a : as)
        {
            std::cin >> a;
        }
        if (!is_down(as))
            continue;
        auto [min, max]{ std::minmax_element(as.begin(), as.end()) };
        ++mins[*min];
        ++counts_max[*max];
    }

    std::vector<int> presums_max(1000000 + 1);
    std::partial_sum(counts_max.begin(), counts_max.end(), presums_max.begin());

    int64_t ans{ n * n };
    for (auto [k, v] : mins)
    {
        ans -= v * presums_max[k];
    }
    std::cout << ans << '\n';
}

bool is_down(std::vector<int>& v)
{
    int min{ v.front() };
    for (int i = 1; i < v.size(); ++i)
    {
        if (v[i] > min)
            return false;
        min = std::min(min, v[i]);
    }
    return true;
}
