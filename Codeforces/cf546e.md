void solve()
{
    int n;
    std::cin >> n;
    std::array<int, 10> as;
    int min{ 200000 }, flag;
    for (int i{ 1 }; i < 10; ++i)
    {
        std::cin >> as[i];
        if (as[i] < min)
        {
            min = as[i];
            flag = i;
        }
    }

    if (min > n)
    {
        std::cout << -1 << '\n';
        return;
    }

    int count{ n / min };
    n -= count * min;
    std::vector<int> ans(count, flag);
    for (auto& a : ans)
    {
        for (int i{ 9 }; i > 0; --i)
        {
            int dif{ as[i] - min };
            if (dif <= n)
            {
                a = i;
                n -= dif;
                break;
            }
        }
        std::cout << a;
    }
    std::cout << '\n';
}
