void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::priority_queue<int, std::vector<int>, std::less<int>> h;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        h.push(a);
    }

    int ans{ 0 };
    int64_t min{ std::numeric_limits<int>::max() };
    int count{ 0 };
    while (!h.empty())
    {
        min = h.top();
        h.pop();
        ++count;
        if (min * count >= k)
        {
            ++ans;
            count = 0;
        }
    }
    std::cout << ans << '\n';
}
