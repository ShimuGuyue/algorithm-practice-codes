void solve()
{
    int n;
    cin >> n;
    string s;
    cin >> s;

    s = " " + s;

    vector<int> ans;

    stack<int> k;
    for (int i = 1; i <= n; ++i)
    {
        if (s[i] == '1')
        {
            k.push(i);
        }
        else if (s[i] == '2')
        {
            if (k.empty())
            {
                continue;
            }
            else
            {
                k.pop();
                ans.push_back(i);
            }
        }
        else
        {
            continue;
        }
    }

    while (!k.empty())
    {
        ans.push_back(k.top());
        k.pop();
    }

    sort(ans.begin(), ans.end());
    cout << ans.size() << '\n';
    for (auto a : ans)
    {
        cout << a << ' ';
    }
    cout << '\n';
}
