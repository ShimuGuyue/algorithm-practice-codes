void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 0 };
    int min{ 1 << 30 };
    for (auto a : as)
    {
        ans += std::min(min, a);
        min = std::min(min, a);
    }
    std::cout << ans << '\n';
}
