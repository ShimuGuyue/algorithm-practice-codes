void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 2 }; i <= n; ++i)
    {
        int p;
        std::cin >> p;
        tree[p].push_back(i);
    }

    int ans{ n };
    std::vector<int> deepth(n + 1);
    std::vector<int> maxs(n + 1);
    auto dfs = [&tree, &deepth, &maxs, &ans](this auto&& dfs, int u) -> void
    {
        maxs[u] = deepth[u];
        int max1{ -1 }, max2{ -1 };
        for (int v : tree[u])
        {
            deepth[v] = deepth[u] + 1;
            dfs(v);

            maxs[u] = std::max(maxs[u], maxs[v]);
            if (maxs[v] > max1)
            {
                max2 = max1;
                max1 = maxs[v];
            }
            else if (maxs[v] > max2)
            {
                max2 = maxs[v];
            }
        }
        if (max2 != -1)
            ans += max2 - deepth[u];
    };
    dfs(1);

    std::cout << ans << '\n';
}
