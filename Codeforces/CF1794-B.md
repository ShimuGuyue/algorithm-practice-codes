void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
        if (a == 1)
            ++a;
    }

    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] % as[i - 1] == 0)
            ++as[i];
    }

    for (auto a : as)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
