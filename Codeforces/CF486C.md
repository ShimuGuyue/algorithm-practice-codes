void solve()
{
    int n, p;
    std::cin >> n >> p;
    std::string s;
    std::cin >> s;

    s = " " + s;

    if (p > n - p + 1)
        p = n - p + 1;
    int l{ p }, r{ p };
    for (int i{ 1 }; i < l; ++i)
    {
        if (s[i] == s[n - i + 1])
            continue;
        l = i;
        break;
    }
    for (int i{ n / 2 }; i > r; --i)
    {
        if (s[i] == s[n - i + 1])
            continue;
        r = i;
        break;
    }

    int ans{ r - l + std::min(r - p, p - l) };
    for (int i{ 1 }; i <= n / 2; ++i)
    {
        int x{ (s[i] - s[n - i + 1] + 26) % 26 };
        ans += std::min(x, 26 - x);
    }
    std::cout << ans << '\n';
}
