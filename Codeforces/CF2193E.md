void solve()
{
    int n;
    std::cin >> n;
    std::set<int> digits;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        digits.insert(a);
    }

    std::vector<int> dp(n + 1, INT_MAX);
    for (auto a : digits)
    {
        dp[a] = 1;
        if (a == 1)
            continue;
        for (int i = 1; i * a <= n; ++i)
        {
            if (dp[i] == INT_MAX)
                continue;
            dp[i * a] = std::min(dp[i * a], dp[i] + 1);
        }
    }
    for (int i = 1; i <= n; ++i)
    {
        std::cout << (dp[i] == INT_MAX ? -1 : dp[i]) << ' ';
    }
    std::cout << '\n';
}
