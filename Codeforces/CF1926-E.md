void solve()
{
    int n, k;
    std::cin >> n >> k;

    int flag{ 1 };
    while (k)
    {
        int count{ (n + flag) / (2 * flag) };
        if (count < k)
        {
            k -= count;
        }
        else
        {
            int ans{ flag };
            while (--k)
            {
                ans += flag * 2;
            }
            std::cout << ans << '\n';
        }
        flag *= 2;
    }
}
