void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int max{ *std::max_element(as.begin(), as.end()) };

    auto calc = [&as](int k) -> int64_t
    {
        int64_t sum1{ 0 }, sum2{ 0 };
        for (auto a : as)
        {
            sum2 += (k - a) / 2;
            sum1 += (a & 1) != (k & 1);
        }

        if (sum1 >= sum2)
            return std::max(sum1 * 2 - 1, sum2 * 2);

        int64_t dif{ sum2 - sum1 };
        dif /= 3;
        sum1 += dif * 2;
        sum2 -= dif;
        int64_t ans{ std::max(sum1 * 2 - 1, sum2 * 2) };
        if (sum2 > sum1)
        {
            sum1 += 2;
            sum2 -= 1;
            ans = std::min(ans, std::max(sum1 * 2 - 1, sum2 * 2));
        }
        return ans;
    };

    int64_t ans{ std::min(calc(max), calc(max + 1)) };
    std::cout << ans << '\n';
}
