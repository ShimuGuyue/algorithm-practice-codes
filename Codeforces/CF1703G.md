void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<std::vector<int>> as(n + 1, std::vector<int>(32));
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i][0];
        for (int j = 1; j < 32; ++j)
        {
            if (j > i)
                break;
            as[i][j] = as[i][j - 1] / 2;
        }
    }

    int64_t ans{ 0 };
    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(32, -1e18));
    dp[0][0] = 0;
    for (int i = 1; i <= n; ++i)
    {
        for (int j = 0; j < 32; ++j)
        {
            dp[i][j] = dp[i - 1][j] + as[i][j] - k;
            if (j != 0)
                dp[i][j] = std::max(dp[i][j], dp[i - 1][j - 1] + as[i][j]);
            ans = std::max(ans, dp[i][j]);
        }
    }
    std::cout << ans << '\n';
}
