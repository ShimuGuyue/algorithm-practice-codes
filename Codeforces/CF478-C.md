void solve()
{
    std::array<int64_t, 3> as;
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.rbegin(), as.rend());
    if (as[0] > (as[1] + as[2]) * 2)
        std::cout << (as[1] + as[2]) << '\n';
    else
        std::cout << (as[0] + as[1] + as[2]) / 3 << '\n';
}
