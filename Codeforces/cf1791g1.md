void solve()
{
    int n, c;
    std::cin >> n >> c;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        as[i] += i;
    }

    std::sort(as.begin() + 1, as.end());
    int ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (c >= as[i])
        {
            ++ans;
            c -= as[i];
        }
    }
    std::cout << ans << '\n';
}
