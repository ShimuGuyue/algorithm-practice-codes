void solve()
{
    int64_t n, s;
    std::cin >> n >> s;

    auto calc = [](int64_t n)
    {
        int ans{ 0 };
        while (n)
        {
            ans += n % 10;
            n /= 10;
        }
        return ans;
    };

    int sum{ calc(n) };
    if (sum <= s)
    {
        std::cout << 0 << '\n';
        return;
    }

    int64_t ans{ 0 };
    int64_t per{ 1 };
    while (sum > s)
    {
        if (n % 10 == 0)
        {
            n /= 10;
            per *= 10;
            continue;
        }

        ans += (10 - n % 10) * per;
        sum -= calc(n) - calc(n + 10 - n % 10);

        n += 10 - n % 10;
        n /= 10;
        per *= 10;
    }
    std::cout << ans << '\n';
}
