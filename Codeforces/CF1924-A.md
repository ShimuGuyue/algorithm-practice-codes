void solve()
{
    int n, k, m;
    std::cin >> n >> k >> m;
    std::string s;
    std::cin >> s;

    int flag{ 0 };
    int dif{ k };
    std::string ans;
    std::vector<int> counts(k);
    for (char c : s)
    {
        int a{ c - 'a' };
        if (counts[a] == flag)
        {
            ++counts[a];
            --dif;
        }
        if (dif == 0)
        {
            dif = k;
            ++flag;
            ans += c;
        }
    }
    if (flag >= n)
    {
        std::cout << "YES\n";
    }
    else
    {
        std::cout << "NO\n";
        for (int i{ 0 }; i < k; ++i)
        {
            if (counts[i] == flag)
            {
                while (ans.size() != n)
                {
                    ans += i + 'a';
                }
                break;
            }
        }
        std::cout << ans << '\n';
    }
}
