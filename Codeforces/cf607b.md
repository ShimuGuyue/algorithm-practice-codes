void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(n + 1, n));
    for (int i{ 1 }; i <= n; ++i)
    {
        dp[0][i] = 1;
        dp[1][i] = 1;
    }
    for (int i{ 2 }; i <= n; ++i)
    {
        for (int j{ 1 }; j + i - 1 <= n; ++j)
        {
            if (as[j] == as[j + i - 1])
                dp[i][j] = std::min(dp[i][j], dp[i - 2][j + 1]);
            for (int k1{ 1 }; k1 < i; ++k1)
            {
                int k2{ i - k1 };
                dp[i][j] = std::min(dp[i][j], dp[k1][j] + dp[k2][j + k1]);
            }
        }
    }
    std::cout << dp[n][1] << '\n';
}
