void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int i{ 0 };
    while (i + 1 < n)
    {
        if (as[i] > as[i + 1])
        {
            std::swap(as[i], as[i + 1]);
            as[i + 1] += as[i];
        }
        ++i;
    }
    std::cout << as.back() << '\n';
}
