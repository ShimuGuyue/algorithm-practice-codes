void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    ++as.front();

    int ans{
        std::accumulate(as.begin(), as.end(), 1,
            [](int pre, int next)
            {
                return pre * next;
            }
        )
    };
    std::cout << ans << '\n';
}
