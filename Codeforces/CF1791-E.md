void solve()
{
    int n;
    std::cin >> n;

    std::vector<int> posi_zeros;
    std::vector<int> negas;
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        if (a >= 0)
            posi_zeros.push_back(a);
        else
            negas.push_back(a);
    }

    std::sort(posi_zeros.begin(), posi_zeros.end());
    std::sort(negas.begin(), negas.end());

    for (int i{ 0 }; i < int(negas.size()); i += 2)
    {
        if (i + 1 == int(negas.size()))
        {
            if (!posi_zeros.empty() && -negas[i] > posi_zeros[0])
            {
                negas[i] *= -1;
                posi_zeros[0] *= -1;
            }
        }
        else
        {
            negas[i] *= -1;
            negas[i + 1] *= -1;
        }
    }

    int64_t ans{ 0 };
    ans += std::accumulate(negas.begin(), negas.end(), INT64_C(0));
    ans += std::accumulate(posi_zeros.begin(), posi_zeros.end(), INT64_C(0));
    std::cout << ans << '\n';
}
