void solve()
{
    std::vector<int> factors(10000000 + 1);
    std::vector<std::pair<int, int>> ans(10000000 + 1, {-1, -1});
    for (int i{ 2 }; i <= 10000000; ++i)
    {
        if (factors[i] == 0)
            factors[i] = i;
        int x{ i };
        while (x <= 10000000)
        {
            factors[x] = factors[i];
            x += i;
        }
    }
    for (int i{ 2 }; i <= 10000000; ++i)
    {
        int x{ i };
        while (x % factors[i] == 0)
        {
            x /= factors[i];
        }
        if (x != 1)
            ans[i] = {factors[i], x};
    }

    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto a : as)
    {
        std::cout << ans[a].first << ' ';
    }
    std::cout << '\n';
    for (auto a : as)
    {
        std::cout << ans[a].second << ' ';
    }
    std::cout << '\n';
}
