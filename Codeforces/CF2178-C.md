void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int64_t> presums(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        presums[i] = presums[i - 1] + as[i];
    }
    std::vector<int64_t> presums_abs(n + 1);
    presums_abs[1] = as[1];
    for (int i = 2; i <= n; ++i)
    {
        presums_abs[i] = presums_abs[i - 1] + abs(as[i]);
    }

    int64_t ans{ INT64_MIN };
    for (int i = 1; i <= n; ++i)
    {
        int64_t front{ presums_abs[i - 1] };
        int64_t back{ presums[n] - presums[i] };
        ans = std::max(ans, front - back);
    }
    std::cout << ans << '\n';
}
