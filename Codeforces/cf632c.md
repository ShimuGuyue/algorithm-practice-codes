void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::string> ss(n);
    for (auto& s : ss)
    {
        std::cin >> s;
    }

    std::sort(ss.begin(), ss.end(),
        [](std::string& a, std::string& b)
        {
            return a + b < b + a;
        }
    );

    for (auto& s : ss)
    {
        std::cout << s;
    }
    std::cout << '\n';
}
