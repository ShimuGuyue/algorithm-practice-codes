void solve()
{
    int n, l, r;
    std::cin >> n >> l >> r;
    std::vector<int> as(n);

    int64_t ans{ 0 };
    for (auto &a : as)
    {
        std::cin >> a;
        if (a < l)
        {
            ans += l - a;
            a = l;
        }
        if (a > r)
        {
            ans += a - r;
            a = r;
        }
    }
    std::sort(as.begin(), as.end());
    for (int i = 0, j = n - 1; i < j; ++i, --j)
    {
        ans += as[j] - as[i];
    }
    std::cout << ans << '\n';
}
