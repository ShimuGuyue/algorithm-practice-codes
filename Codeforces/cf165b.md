void solve()
{
    int64_t n, k;
    std::cin >> n >> k;

    auto check = [n, k](int64_t v)
    {
        int64_t sum{ 0 };
        int64_t flag{ 1 };
        while (flag <= v)
        {
            sum += v / flag;
            flag *= k;
        }
        return sum >= n;
    };

    int64_t l{ 0 }, r{ n };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
