void solve()
{
    int n, m;
    std::cin >> n >> m;

    std::queue<int> q;
    q.push(n);
    while (!q.empty())
    {
        int a{ q.front() };
        q.pop();
        if (a == m)
        {
            std::cout << "YES\n";
            return;
        }
        if (a % 3)
            continue;
        q.push(a / 3);
        q.push(a / 3 * 2);
    }
    std::cout << "NO\n";
}
