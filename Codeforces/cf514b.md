void solve()
{
    int n, x0, y0;
    std::cin >> n >> x0 >> y0;
    std::set<double> ks;
    for (int i = 0; i < n; ++i)
    {
        int x, y;
        std::cin >> x >> y;
        int dx{ x - x0 };
        int dy{ y - y0 };
        double k{ dx == 0 ? std::numeric_limits<int>::max() : double(dy) / dx };
        ks.insert(k);
    }
    std::cout << ks.size() << '\n';
}
