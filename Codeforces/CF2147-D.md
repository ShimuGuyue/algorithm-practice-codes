void solve()
{
    int n;
    std::cin >> n;

    int64_t ans1{ 0 }, ans2{ 0 };

    std::map<int64_t, int> odds;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        if (a % 2 == 0)
        {
            ans1 += a / 2;
            ans2 += a / 2;
        }
        else
        {
            ++odds[a];
        }
    }
    std::vector<std::pair<int64_t, int>> v(odds.begin(), odds.end());
    std::sort(v.begin(), v.end(),
        [](std::pair<int64_t, int> &a, std::pair<int64_t, int> &b)
        {
            return a.second > b.second;
        });
    for (int i = 0; i < v.size(); ++i)
    {
        auto &[a, b]{ v[i] };
        if (i % 2 == 0)
        {
            ans1 += (a / 2 + 1) * b;
            ans2 += (a / 2) * b;
        }
        else
        {
            ans1 += (a / 2) * b;
            ans2 += (a / 2 + 1) * b;
        }
    }
    std::cout << ans1 << ' ' << ans2 << '\n';
}
