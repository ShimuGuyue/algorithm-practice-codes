    }

        int m;

        std::cin >> s;
        for (char c : s)
        {
            if (c == 'U')
                --as[i];
            else
                ++as[i];
        }
        as[i] = (as[i] % 10 + 10) % 10;
    }
    for (auto a : as)
    {
        std::cout << a << ' ';
    }

}
