void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> bs(as);
    std::sort(bs.begin(), bs.end());

    for (int i{ 1 }; i < n * 2; i *= 2)
    {
        std::vector<int> cs(as);
        for (int l{ 0 }; l < n; l += i)
        {
            int r{ std::min(l + i, n) };
            std::sort(cs.begin() + l, cs.begin() + r);
        }
        if (cs == bs)
        {
            std::cout << i / 2 << '\n';
            return;
        }
    }
}
