void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    std::set<int> t(as.begin(), as.end());

    std::vector<int> ans;
    ans.reserve(n);
    std::set<int> visited;
    for (int a : as)
    {
        if (visited.find(a) != visited.end())
            continue;
        if (k / a > n)
        {
            std::cout << -1 << '\n';
            return;
        }
        int temp{ a };
        while (temp <= k)
        {
            if (t.find(temp) == t.end())
            {
                std::cout << -1 << '\n';
                return;
            }
            visited.insert(temp);
            temp += a;
        }
        ans.push_back(a);
    }
    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
