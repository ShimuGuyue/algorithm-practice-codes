void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> dp(n + 1);
    std::vector<int> flags(n + 1, -1000000000);
    for (int i{ 1 }; i <= n; ++i)
    {
        dp[i] = std::max(dp[i - 1], i + 1 + flags[as[i]]);
        flags[as[i]] = std::max(flags[as[i]], dp[i - 1] - i);
    }
    std::cout << dp[n] << '\n';
}
