int64_t mod{ 998244353 };

void solve()
{
    int64_t n, m;
    std::cin >> n >> m;

    if (n == 2)
    {
        std::cout << 0 << '\n';
        return;
    }

    int64_t ans{ 1 };
    // 1: max
    // 2: double
    // n - 3: other
    ans = ans * combinate(m, n - 1) % mod;
    ans = ans * quickpow(2, n - 3) % mod;
    ans = ans * (n - 2) % mod;
    std::cout << ans << '\n';
}

int64_t combinate(int64_t n, int64_t m)
{
    int64_t ans{ 1 };
    for (int64_t i{ 1 }; i <= m; ++i)
    {
        ans = ans * (n - i + 1) % mod * inv(i) % mod;
    }
    return ans;
}

int64_t inv(int64_t n)
{
    return quickpow(n, mod - 2);
}

int64_t quickpow(int64_t base, int64_t power)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}
