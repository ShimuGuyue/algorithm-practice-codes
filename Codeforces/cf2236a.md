void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int max{ *std::max_element(as.begin(), as.end()) };
    int min{ *std::min_element(as.begin(), as.end()) };
    int ans{ max - min + 1 };
    std::cout << ans << '\n';
}
