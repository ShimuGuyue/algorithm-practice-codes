void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> grid(n);
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    bool find{ false };
    std::vector<std::vector<bool>> visited(n, std::vector<bool>(m)); 
    auto dfs = [n, m, &grid, &visited, &find](this auto& dfs, std::pair<int, int> loc, std::pair<int, int> last, std::pair<int, int> begin, char c) -> void
    {
        static constexpr std::array<std::pair<int, int>, 4> dxy{
            {{-1, 0}, {0, -1}, {0, 1}, {1, 0}}
        };
        auto [x, y]{ loc };
        for (auto [dx, dy] : dxy)
        {
            int xx{ x + dx };
            int yy{ y + dy };
            if (xx < 0 || xx == n)
                continue;
            if (yy < 0 || yy == m)
                continue;
            if (std::make_pair(xx, yy) == last)
                continue;
            if (grid[xx][yy] != c)
                continue;
            if (visited[xx][yy])
                continue;
            visited[xx][yy] = true;
            if (std::make_pair(xx, yy) == begin)
            {
                find = true;
                return;
            }
            dfs({xx, yy}, {x, y}, begin, c);
        }
    };
    for (int i{ 0 }; i < n; ++i)
    {
        for (int j{ 0 }; j < m; ++j)
        {
            visited.assign(n, std::vector<bool>(m, false));
            dfs({i, j}, {-1, -1}, {i, j}, grid[i][j]);
            if (find)
            {
                std::cout << "Yes\n";
                return;
            }
        }
    }
    std::cout << "No\n";
}
