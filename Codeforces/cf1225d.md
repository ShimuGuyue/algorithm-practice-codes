void solve()
{
    int n, k;
    std::cin >> n >> k;

    int64_t ans{ 0 };
    int base{ 100003 };
    std::map<uint64_t, int> hashs;
    while (n--)
    {
        int a;
        std::cin >> a;
        uint64_t hash{ 0 };
        uint64_t hash_inv{ 0 };
        for (int i{ 2 }; i * i <= a; ++i)
        {
            if (a % i)
                continue;
            int count{ 0 };
            while (a % i == 0)
            {
                a /= i;
                ++count;
            }
            count %= k;
            if (count == 0)
                continue;
            hash = hash * base + i;
            hash = hash * 200 + count;
            hash_inv = hash_inv * base + i;
            hash_inv = hash_inv * 200 + k - count;
        }
        if (a != 1)
        {
            hash = hash * base + a;
            hash = hash * 200 + 1;
            hash_inv = hash_inv * base + a;
            hash_inv = hash_inv * 200 + k - 1;
        }
        ans += hashs[hash_inv];
        ++hashs[hash];
    }
    std::cout << ans << '\n';
}
