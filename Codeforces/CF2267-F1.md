void solve()
{
    int n, q;
    cin >> n >> q;
    vector<vector<int>> as(1);
    as[0].resize(n);
    for (auto& a : as[0])
    {
        cin >> a;
    }
    sort(as[0].begin(), as[0].end());

    while (1)
    {
        if (as.back().back() == 0)
            break;
        vector<int> bs;
        for (int j = 0; j < n; ++j)
        {
            for (int k = j + 1; k < n; ++k)
            {
                bs.push_back(as.back()[j] ^ as.back()[k]);
            }
        }
        sort(bs.begin(), bs.end());
        as.push_back(vector<int>(bs.begin(), bs.begin() + n));
    }

    while (q--)
    {
        int x;
        cin >> x;
        if (x >= as.size())
            cout << 0 << '\n';
        else
            cout << as[x].back() - as[x].front() << '\n';
    }
}
