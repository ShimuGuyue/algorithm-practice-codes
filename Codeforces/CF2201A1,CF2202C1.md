void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 1 };
    int l{ as[0] + 1 }, r{ l };
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] == r)
        {
            r = as[i] + 1;
        }
        else if (as[i] > r)
        {
            ++ans;
            l = as[i] + 1;
            r = l;
        }
        else if (as[i] < r)
        {
            if (as[i] >= l)
            {
                r = as[i] + 1;
            }
            else
            {
                ++ans;
                l = as[i] + 1;
                r = l;
            }
        }
    }
    std::cout << ans << '\n';
}
