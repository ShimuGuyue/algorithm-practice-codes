void solve()
{
    int n;
    std::cin >> n;

    std::string mode{ "back" };
    std::string s;
    for (int k = n * 2; k >= 0; )
    {
        if (s.size() == n)
        {
            std::cout << "! " << s << std::endl;
            return;
        }

        if (mode == "back")
        {
            --k;
            std::cout << "? " << s << "0" << std::endl;
            bool judge0;
            std::cin >> judge0;
            if (judge0)
            {
                s = s + "0";
                continue;
            }

            --k;
            std::cout << "? " << s << "1" << std::endl;
            bool judge1;
            std::cin >> judge1;
            if (judge1)
            {
                s = s + "1";
                continue;
            }

            mode = "front";
        }
        else
        {
            --k;
            std::cout << "? " << "0" << s << std::endl;
            bool judge0;
            std::cin >> judge0;
            if (judge0)
            {
                s = "0" + s;
                continue;
            }

            s = "1" + s;
            continue;
        }
    }
}
