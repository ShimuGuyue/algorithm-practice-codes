void solve()
{
    std::string s;
    std::cin >> s;

    std::vector<int> counts('z' + 1);
    for (char c : s)
    {
        ++counts[c];
    }

    std::string ans;
    std::stack<char> k;
    for (char c : s)
    {
        k.push(c);
        --counts[c];
        while (!k.empty())
        {
            bool end{ false };
            char t{ k.top() };
            for (char ch{ 'a' }; ch < t; ++ch)
            {
                if (counts[ch])
                    end = true;
            }
            if (end)
                break;
            ans.push_back(t);
            k.pop();
        }
    }
    while (!k.empty())
    {
        ans.push_back(k.top());
        k.pop();
    }

    std::cout << ans << '\n';
}
