void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    vector<int> v0; v0.reserve(n);
    vector<int> v1; v1.reserve(n);
    for (int i = 0; i < n; ++i)
    {
        if (i & 1)
            v1.push_back(as[i]);
        else
            v0.push_back(as[i]);
    }

    sort(v0.begin(), v0.end());
    sort(v1.begin(), v1.end());

    vector<int> v00, v11;

    int a = 0, b = 0;
    int i = 0, j = 0;
    int mode;
    while (1)
    {
        while (i < v0.size() && v0[i] <= b)
        {
            v00.push_back(v0[i]);
            ++i;
        }
        if (i != v0.size())
        {
            a = v0[i];
            ++i;
            mode = 1;
        }
        else
        {
            while (j < v1.size())
            {
                v11.push_back(v1[j]);
                ++j;
            }
            break;
        }

        while (j < v1.size() && v1[j] <= a)
        {
            v11.push_back(v1[j]);
            ++j;
        }
        if (j != v1.size())
        {
            b = v1[j];
            ++j;
            mode = 0;
        }
        else
        {
            while (i < v0.size())
            {
                v00.push_back(v0[i]);
                ++i;
            }
            break;
        }
    }

    while (!v00.empty() || !v11.empty())
    {
        if (mode == 1)
        {
            if (v11.empty() && !v00.empty())
            {
                cout << "No\n";
                return;
            }
            if (v11.back() > a)
            {
                cout << "No\n";
                return;
            }
            b = v11.back();
            v11.pop_back();
        }
        else
        {
            if (v00.empty() && !v11.empty())
            {
                cout << "No\n";
                return;
            }
            if (v00.back() > b)
            {
                cout << "No\n";
                return;
            }
            a = v00.back();
            v00.pop_back();
        }
        mode = mode ^ 1;
    }
    cout << "Yes\n";
}
