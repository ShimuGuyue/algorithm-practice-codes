void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    if (n == 1)
    {
        std::cout << "1 1\n";
        std::cout << -as[1] << '\n';
        std::cout << "1 1\n";
        std::cout << 0 << '\n';
        std::cout << "1 1\n";
        std::cout << 0 << '\n';
        return;
    }

    std::cout << "1 1\n";
    std::cout << n - as[1] << '\n';
    as[1] = n;

    std::cout << 2 << ' ' << n << '\n';
    for (int i{ 2 }; i <= n; ++i)
    {
        std::cout << (n - 1) * as[i] << ' ';
        as[i] += as[i] * (n - 1);
    }
    std::cout << '\n';

    std::cout << 1 << ' ' << n << '\n';
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << -as[i] << ' ';
    }
    std::cout << '\n';
}
