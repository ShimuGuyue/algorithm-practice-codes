void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    std::vector<std::vector<int>> grid(n + 1);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        grid[u].push_back(v);
        grid[v].push_back(u);
    }

    std::vector<int> ans(k + 1);
    std::vector<int> lens(n + 1);
    std::vector<bool> visited(n + 1);
    std::queue<int> q;
    q.push(1);
    visited[1] = true;
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        for (int v : grid[u])
        {
            if (visited[v])
                continue;
            visited[v] = true;
            lens[v] = lens[u] + 1;
            ans[as[v]] = lens[v];
            q.push(v);
        }
    }
    for (int i{ 1 }; i <= k; ++i)
    {
        std::cout << ans[i] << " \n"[i == k];
    }
}
