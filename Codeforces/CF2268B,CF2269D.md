void solve()
{
    int n, q;
    cin >> n >> q;
    vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        cin >> as[i];
    }

    int count_even = 0;
    for (int i = 1; i <= n; ++i)
    {
        if (popcount(uint32_t(as[i])) % 2 == 0)
            ++count_even;
    }

    cout << count_even;

    while (q--)
    {
        int p, x;
        cin >> p >> x;
        if (popcount(uint32_t(as[p])) % 2 == 0)
            --count_even;
        as[p] = x;
        if (popcount(uint32_t(as[p])) % 2 == 0)
            ++count_even;

        cout << ' ' << count_even;
    }
    cout << '\n';
}
