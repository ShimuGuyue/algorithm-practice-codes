void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::map<int, int> m;
    for (auto a : as)
    {
        ++m[a];
    }

    if (m[0] < 2)
    {
        std::cout << -1 << '\n';
        return;
    }

    int ans{ 0 };
    ans += as.front() != 0;
    ans += as.back()  != 0;
    std::cout << ans << '\n';
}
