void solve()
{
    int n;
    cin >> n;
    string s1, s2;
    cin >> s1 >> s2;

    int odd1 = 0, even1 = 0;
    int odd2 = 0, even2 = 0;
    for (int i = 0, count = 0; i < n; ++i)
    {
        if (s1[i] == 'f')
        {
            ++count;
        }
        if (s1[i] == 'o' || i == n - 1)
        {
            odd1 += count & 1;
            even1 += count - (count & 1);
            count = 0;
        }
    }
    for (int i = 0, count = 0; i < n; ++i)
    {
        if (s2[i] == 'f')
        {
            ++count;
        }
        if (s2[i] == 'o' || i == n - 1)
        {
            odd2 += count & 1;
            even2 += count - (count & 1);
            count = 0;
        }
    }

    int dif = abs(odd1 - odd2);
    if (odd1 < odd2)
        odd1 += std::min(even1, (dif & 1) ? dif - 1 : dif);
    else
        odd2 += std::min(even2, (dif & 1) ? dif - 1 : dif);
    dif = abs(odd1 - odd2);
    cout << dif << '\n';
}
