class SparseTable
{
private:
    std::vector<int> v_;
    std::vector<std::vector<int>> table_;

public:
    SparseTable(const std::vector<int>& v)
    {
        build(v);
    }

public:
    int query(const int l, const int r) const
    {
        const int power{ static_cast<int>(log2(r - l + 1)) };
        const int len{ 1 << power };
        return merge(l, r - len + 1, power);
    }

private:
    void build(const std::vector<int>& v)
    {
        v_ = v;

        const int n{ static_cast<int>(v.size()) };
        const int m{ static_cast<int>(std::log2(n)) + 1 };
        table_.assign(n, std::vector<int>(m));

        for (int i{ 0 }; i < n; ++i)
        {
            table_[i][0] = contribute(i);
        }
        for (int j{ 1 }; j < m; ++j)
        {
            int len{ 1 << (j - 1) };
            for (int i{ 0 }; i + len * 2 - 1 < n; ++i)
            {
                table_[i][j] = merge(i, i + len, j - 1);
            }
        }
    }

    int contribute(const int index) const
    {
        // TODO: 规定初始贡献值
        return v_[index];
    }

    int merge(const int i1, const int i2, const int j) const
    {
        // TODO: 规定贡献值合并结果
        return std::max(table_[i1][j], table_[i2][j]);
    }
};

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(m + 1);
    for (int i{ 1 }; i <= m; ++i)
    {
        std::cin >> as[i];
    }

    SparseTable st(as);

    int q;
    std::cin >> q;
    while (q--)
    {
        int x1, y1, x2, y2, k;
        std::cin >> x1 >> y1 >> x2 >> y2 >> k;
        if (abs(x1 - x2) % k || abs(y1 - y2) % k)
        {
            std::cout << "NO\n";
            continue;
        }
        auto [l, r]{ std::minmax(y1, y2) };
        int max{ st.query(l, r) };
        if (x1 > max && x2 > max)
        {
            std::cout << "YES\n";
            continue;
        }
        int flag{ x1 + ((n - x1) / k) * k };
        if (flag > max)
            std::cout << "YES\n";
        else
            std::cout << "NO\n";
    }
}
