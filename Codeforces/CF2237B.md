void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    int ans{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        bool find{ false };
        for (int j{ i }; j < n; ++j)
        {
            if (as[j] > bs[i])
                continue;
            find = true;
            for (int k{ j }; k - 1 >= i; --k)
            {
                std::swap(as[k], as[k - 1]);
                ++ans;
            }
            break;
        }
        if (!find)
        {
            std::cout << -1 << '\n';
            return;
        }
    }
    std::cout << ans << '\n';
}
