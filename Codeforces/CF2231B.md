void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int max{ 0 };
    for (int i{ 1 }; i < n; ++i)
    {
        max = std::max(as[i - 1] - as[i], max);
    }

    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] < as[i - 1])
            as[i] += max;
    }

    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] < as[i - 1])
        {
            std::cout << "No\n";
            return;
        }
    }
    std::cout << "Yes\n";
}
