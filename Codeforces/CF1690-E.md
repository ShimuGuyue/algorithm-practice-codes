void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t ans{ 0 };
    for (auto& a : as)
    {
        ans += a / k;
        a %= k;
    }

    std::sort(as.begin(), as.end());

    int l{ 0 }, r{ n - 1 };
    while (l < r)
    {
        if (as[l] + as[r] < k)
        {
            ++l;
        }
        else
        {
            ++ans;
            ++l;
            --r;
        }
    }
    std::cout << ans << '\n';
}
