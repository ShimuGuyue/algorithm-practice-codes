

```cpp
static constexpr int64_t mod{ 1000000007 };

static constexpr int n{ 40000 };
std::array<int64_t, n + 1> dp{ };

void init()
{
    std::vector<int> flags;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (i % 10 == 0)
            continue;
        auto s{ std::to_string(i) };
        auto t{ std::string(s.rbegin(), s.rend()) };
        if (s == t)
            flags.push_back(i);
    }

    dp[0] = 1;
    for (auto a : flags)
    {
        for (int i{ 0 }; i + a <= n; ++i)
        {
            dp[i + a] = (dp[i + a] + dp[i]) % mod;
        }
    }
}
```

```cpp
void solve()
{
    int n;
    std::cin >> n;
    std::cout << dp[n] << '\n';
}
```
