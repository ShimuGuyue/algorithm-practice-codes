```c++
void solve()
{
struct Data
{
    int count_bit{ 0 };
    int count_pow_2_k{ 0 };
    int count_pow_2_k_add_1{ 0 };
};

int n, q;
std::cin >> n >> q;
std::vector<Data> presums(n + 1);
for (int i = 1; i <= n; ++i)
{
    int a;
    std::cin >> a;
    presums[i].count_bit = presums[i - 1].count_bit + count_bit(a);
    presums[i].count_pow_2_k = presums[i - 1].count_pow_2_k + is_pow_2_k(a);
    presums[i].count_pow_2_k_add_1 = presums[i - 1].count_pow_2_k_add_1 + is_pow_2_k_add_1(a);
}

while (q--)
{
    int l, r;
    std::cin >> l >> r;
    int count_bit{ presums[r].count_bit - presums[l - 1].count_bit };
    int count_pow_2_k{ presums[r].count_pow_2_k - presums[l - 1].count_pow_2_k };
    int count_pow_2_k_add_1{ presums[r].count_pow_2_k_add_1 - presums[l - 1].count_pow_2_k_add_1 };

    int ans{ count_bit + (r - l + 1 - count_pow_2_k - count_pow_2_k_add_1) + count_pow_2_k_add_1 / 2 };
    std::cout << ans << '\n';
}
}
```

```c++
int count_bit(int n)
{
    int ans{ 0 };
    while (n != 1)
    {
        ++ans;
        n /= 2;
    }
    return ans;
}

bool is_pow_2_k(int n)
{
    for (int i = 2; i <= n; i *= 2)
    {
        if (i == n)
            return true;
    }
    return false;
}

bool is_pow_2_k_add_1(int n)
{
    return is_pow_2_k(n - 1);
}
```
