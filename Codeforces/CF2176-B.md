void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    s += s;

    int ans{};
    int count{};
    for (char c : s)
    {
        if (c == '0')
            ++count;
        else
            count = 0;
        ans = std::max(ans, count);
    }
    std::cout << ans << '\n';
}
