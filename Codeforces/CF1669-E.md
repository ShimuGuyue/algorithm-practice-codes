void solve()
{
    int n;
    std::cin >> n;

    std::vector<int> counts_1_1('k' - 'a' + 1);
    std::vector<int> counts_1_2('k' - 'a' + 1);
    std::vector<std::vector<int64_t>> counts_2('k' - 'a' + 1, std::vector<int64_t>('k' - 'a' + 1));
    for (int i = 0; i < n; ++i)
    {
        std::string s;
        std::cin >> s;
        int a{ s.front() - 'a' };
        int b{ s.back() - 'a' };
        ++counts_1_1[a];
        ++counts_1_2[b];
        ++counts_2[a][b];
    }

    int64_t ans{ 0 };
    for (int i = 0; i <= 'k' - 'a'; ++i)
    {
        for (int j = 0; j <= 'k' - 'a'; ++j)
        {
            if (counts_2[i][j] == 0)
                continue;
            ans += counts_2[i][j] * (counts_1_1[i] - counts_2[i][j]);
            ans += counts_2[i][j] * (counts_1_2[j] - counts_2[i][j]);
        }
    }
    ans /= 2;
    std::cout << ans << '\n';
}
