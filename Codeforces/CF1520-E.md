void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::vector<int> indexs;
    for (int i = 0; i < n; ++i)
    {
        if (s[i] == '*')
            indexs.push_back(i);
    }

    int flag{ int(indexs.size() / 2) };
    int64_t ans{ 0 };
    for (int i = 0; i < int(indexs.size()); ++i)
    {
        ans += abs(indexs[i] - (indexs[flag] - (flag - i)));
    }
    std::cout << ans << '\n';
}
