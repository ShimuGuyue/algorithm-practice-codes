void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    bool only{ true };
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] != as[i - 1])
            only = false;
    }
    if (only)
    {
        std::cout << 1 << '\n';
        for (int i{ 0 }; i < n; ++i)
        {
            std::cout << 1 << ' ';
        }
        std::cout << '\n';
        return;
    }

    if (n % 2 == 0)
    {
        std::cout << 2 << '\n';
        for (int i{ 0 }; i < n; ++i)
        {
            std::cout << (i & 1 ? 1 : 2) << ' ';
        }
        std::cout << '\n';
        return;
    }

    if (as.front() == as.back())
    {
        std::cout << 2 << '\n';
        for (int i{ 0 }; i < n - 1; ++i)
        {
            std::cout << (i & 1 ? 1 : 2) << ' ';
        }
        std::cout << 2 << '\n';
        return;
    }
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] != as[i - 1])
            continue;
        std::vector<int> ans(n);
        ans[i - 1] = ans[i] = 1;
        for (int l{ i - 2 }; l >= 0; --l)
        {
            ans[l] = ans[l + 1] == 1 ? 2 : 1;
        }
        for (int r{ i + 1 }; r < n; ++r)
        {
            ans[r] = ans[r - 1] == 1 ? 2 : 1;
        }
        std::cout << 2 << '\n';
        for (auto a : ans)
        {
            std::cout << a << ' ';
        }
        std::cout << '\n';
        return;
    }

    std::cout << 3 << '\n';
    for (int i{ 0 }; i < n - 1; ++i)
    {
        std::cout << (i & 1 ? 1 : 2) << ' ';
    }
    std::cout << 3 << '\n';
}
