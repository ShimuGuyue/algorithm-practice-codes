void solve()
{
    int n;
    std::cin >> n;
    while (n--)
    {
        int a;
        std::cin >> a;
        int ans{ 1 };
        while (a)
        {
            a /= 2;
            ans *= 2;
        }
        ans /= 2;
        std::cout << ans << ' ';
    }
    std::cout << '\n';
}
