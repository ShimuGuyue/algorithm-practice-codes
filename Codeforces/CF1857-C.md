void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> bs(n * (n - 1) / 2);
    for (auto& b : bs)
    {
        std::cin >> b;
    }
    std::sort(bs.begin(), bs.end());

    std::vector<int> ans(n);
    int index{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        ans[i] = bs[index];
        index += n - i - 1;
    }
    ans.back() = 1000000000;
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
