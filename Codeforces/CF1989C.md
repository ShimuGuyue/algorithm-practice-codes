void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    int a{ 0 }, b{ 0 };
    int posi{ 0 }, nega{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (as[i] == bs[i])
        {
            if (as[i] == 1)
                ++posi;
            if (as[i] == -1)
                ++nega;
        }
        else
        {
            int max{ std::max(as[i], bs[i]) };
            if (max == 0)
                continue;
            if (as[i] == max)
                ++a;
            else
                ++b;
        }
    }

    int min{ std::min(a, b) };
    int max{ std::max(a, b) };
    int dif{ max - min };
    if (posi < dif)
    {
        min += posi;
    }
    else
    {
        min = max;
        posi -= dif;
        int half{ posi / 2 };
        min += half;
        max += posi - half;
    }

    dif = max - min;
    if (dif > nega)
    {
        max -= nega;
    }
    else
    {
        max = min;
        nega -= dif;
        int half{ nega / 2 };
        max -= half;
        min -= nega - half;
    }

    std::cout << min << '\n';
}
