void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<std::string> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<int>> pres(n + 1, std::vector<int>(6));
    std::vector<std::vector<int>> sufs(n + 1, std::vector<int>(6));
    for (int i{ 1 }; i <= n; ++i)
    {
        pres[i] = pres[i - 1];
        pres[i][turn(as[i])] = i;
    }
    for (int i{ n }; i >= 1; --i)
    {
        if (i != n)
            sufs[i] = sufs[i + 1];
        sufs[i][turn(as[i])] = i;
    }

    while (q--)
    {
        int ans{ 1 << 30 };
        int x, y;
        std::cin >> x >> y;
        if (x > y)
            std::swap(x, y);
        for (std::string s : { "BG", "BR", "BY", "GR", "GY", "RY" })
        {
            bool b1{ false }, b2{ false };
            if (s[0] == as[x][0] || s[0] == as[x][1] || s[1] == as[x][0] || s[1] == as[x][1])
                b1 = true;
            if (s[0] == as[y][0] || s[0] == as[y][1] || s[1] == as[y][0] || s[1] == as[y][1])
                b2 = true;
            if (!(b1 && b2))
                continue;
            int k{ turn(s) };
            if (pres[y][k])
                ans = std::min(ans, std::abs(pres[y][k] - x) + std::abs(pres[y][k] - y));
            if (sufs[x][k])
                ans = std::min(ans, std::abs(sufs[x][k] - x) + std::abs(sufs[x][k] - y));
        }
        if (ans == (1 << 30))
            ans = -1;
        std::cout << ans << '\n';
    }
}

int turn(std::string s)
{
    int ans;
    if (s == "BG")
        ans = 0;
    if (s == "BR")
        ans = 1;
    if (s == "BY")
        ans = 2;
    if (s == "GR")
        ans = 3;
    if (s == "GY")
        ans = 4;
    if (s == "RY")
        ans = 5;
    return ans;
}
