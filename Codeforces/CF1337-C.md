void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i = 1; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    std::vector<int> values(n + 1);
    auto dfs = [&tree, &values](auto&& dfs, int u, int last, int count) -> int
    {
        int value{ count };
        int count_son{ 0 };
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            int cur{ dfs(dfs, v, u, count + 1) };
            count_son += cur;
            value -= cur;
        }
        values[u] = value;
        return count_son + 1;
    };
    dfs(dfs, 1, 0, 0);

    int64_t ans{ 0 };
    std::sort(values.begin() + 1, values.end(), std::greater<int>());
    for (int i = 1; i <= k; ++i)
    {
        ans += values[i];
    }
    std::cout << ans << '\n';
}
