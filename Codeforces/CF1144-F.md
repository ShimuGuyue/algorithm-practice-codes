void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> graph(n + 1);
    std::vector<std::pair<int, int>> edges(m);
    for (auto &[u, v] : edges)
    {
        std::cin >> u >> v;
        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    std::vector<int> colors(n + 1);
    colors[1] = 1;
    std::queue<int> q;
    q.push(1);
    while (!q.empty())
    {
        int u{ q.front() };
        q.pop();
        for (int v : graph[u])
        {
            if (colors[v])
            {
                if (colors[v] != colors[u])
                    continue;
                std::cout << "NO\n";
                return;
            }
            colors[v] = colors[u] == 1 ? 2 : 1;
            q.push(v);
        }
    }

    std::cout << "YES\n";
    for (auto [u, v] : edges)
    {
        if (colors[u] == 1)
            std::cout << 1;
        else
            std::cout << 0;
    }
    std::cout << '\n';
}
