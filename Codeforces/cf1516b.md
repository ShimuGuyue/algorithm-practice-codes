void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int xorsum = std::accumulate(as.begin(), as.end(), 0,
        [](int a, int b)
        {
            return a ^ b;
        }
    );

    int flag1{ 0 };
    for (int i = 0; i < n - 1; ++i)
    {
        flag1 ^= as[i];
        if (flag1 == (xorsum ^ flag1))
        {
            std::cout << "yes\n";
            return;
        }
        int flag2{ 0 };
        for (int j = i + 1; j < n - 1; ++j)
        {
            flag2 ^= as[j];
            if (flag2 == flag1 && (flag2 == (xorsum ^ flag1 ^ flag2)))
            {
                std::cout << "yes\n";
                return;
            }
        }
    }
    std::cout << "no\n";
}
