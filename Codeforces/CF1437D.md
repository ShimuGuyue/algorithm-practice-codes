void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> deepths(n + 1);
    std::queue<int> q;
    q.push(1);
    int last{ 0 };
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] < last)
            q.pop();
        deepths[as[i]] = deepths[q.front()] + 1;
        last = as[i];
        q.push(as[i]);
    }
    int ans{ *std::max_element(deepths.begin(), deepths.end()) };
    std::cout << ans << '\n';
}
