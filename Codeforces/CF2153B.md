void solve()
{
    int x, y, z;
    std::cin >> x >> y >> z;
    bool ok{ true };
    for (int i = 0; i < 30; ++i)
    {
        int count{ 0 };
        count += (x >> i) & 1;
        count += (y >> i) & 1;
        count += (z >> i) & 1;
        if (count % 2 == 0 && count != 0)
            ok = false;
    }
    std::cout << (ok ? "yes" : "no") << '\n';
}
