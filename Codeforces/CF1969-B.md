void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    bool zero{ false }, one{ false };
    for (auto a : as)
    {
        if (a)
        {
            one = true;
            if (zero)
            {
                std::cout << 2 << '\n';
                return;
            }
        }
        else
        {
            if (one)
                zero = true;
        }
    }

    std::cout << (one ? 1 : 0) << '\n';
}
