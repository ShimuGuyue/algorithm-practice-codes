void solve()
{
    int n;
    std::cin >> n;

    static constexpr int inf{ 500000 };

    int m11{ inf };
    int m10{ inf };
    int m01{ inf };

    while (n--)
    {
        int m;
        std::string s;
        std::cin >> m >> s;
        if (s == "11")
            m11 = std::min(m11, m);
        if (s == "01")
            m01 = std::min(m01, m);
        if (s == "10")
            m10 = std::min(m10, m);
    }
    int ans1{ m11 }, ans2{ std::min(m10 + m01, inf) };
    int ans{ std::min(ans1, ans2) };
    std::cout << (ans == inf ? -1 : ans) << '\n';
}
