void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::string s;
    std::cin >> s;

    s = " " + s;

    std::vector<int> l0s(n + 2, 0);
    std::vector<int> r1s(n + 2, n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        l0s[i] = l0s[i - 1];
        if (s[i] == '0')
            l0s[i] = i;
    }
    for (int i{ n }; i >= 1; --i)
    {
        r1s[i] = r1s[i + 1];
        if (s[i] == '1')
            r1s[i] = i;
    }

    std::set<std::pair<int, int>> t;
    while (m--)
    {
        int l, r;
        std::cin >> l >> r;
        l = r1s[l];
        r = l0s[r];
        if (l <= r)
            t.insert({ l, r });
        else
            t.insert({ 0, 0 });
    }
    std::cout << t.size() << '\n';
}
