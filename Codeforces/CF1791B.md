void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int x{ 0 }, y{ 0 };
    for (char c : s)
    {
        if (c == 'U')
            ++y;
        if (c == 'D')
            --y;
        if (c == 'L')
            --x;
        if (c == 'R')
            ++x;
        if (x == 1 && y == 1)
        {
            std::cout << "yes\n";
            return;
        }
    }
    std::cout << "no\n";
}
