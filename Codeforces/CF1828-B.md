void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        as[i] = std::abs(as[i] - i);
    }

    int ans{
        std::accumulate(as.begin() + 1, as.end(), as[1],
            [](int a, int b)
            {
                return std::gcd(a, b);
            }
        )
    };

    std::cout << ans << '\n';
}
