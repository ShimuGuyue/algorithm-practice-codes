void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<bool> visited(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (visited[i])
            continue;
        std::vector<int> v;
        for (int j{ i }; j <= n; j *= 2)
        {
            visited[j] = true;
            v.push_back(as[j]);
        }
        std::sort(v.rbegin(), v.rend());
        for (int j{ i }; j <= n; j *= 2)
        {
            as[j] = v.back();
            v.pop_back();
        }
    }
    if (std::is_sorted(as.begin() + 1, as.end()))
        std::cout << "YES\n";
    else
        std::cout << "NO\n";
}
