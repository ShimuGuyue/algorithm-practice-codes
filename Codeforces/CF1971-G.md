```cpp
struct DisjointSet
{
    std::map<int, int> fathers;

    int find(int x)
    {
        return fathers[x] == x ? x : fathers[x] = find(fathers[x]);
    }

    void merge(int x, int y)
    {
        if (!fathers.count(x))
            fathers[x] = x;
        if (!fathers.count(y))
            fathers[y] = y;
        fathers[find(x)] = find(y);
    }
};
```

```cpp
void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    DisjointSet ds;
    for (auto a : as)
    {
        ds.merge(a, a ^ 1);
        ds.merge(a, a ^ 2);
        ds.merge(a, a ^ 3);
    }

    std::map<int, std::vector<int>> flags;
    for (auto a : as)
    {
        flags[ds.find(a)].push_back(a);
    }
    for (auto& [k, v] : flags)
    {
        std::sort(v.rbegin(), v.rend());
    }

    for (auto a : as)
    {
        auto& v{ flags[ds.find(a)] };
        std::cout << v.back() << ' ';
        v.pop_back();
    }
    std::cout << '\n';
}
```
