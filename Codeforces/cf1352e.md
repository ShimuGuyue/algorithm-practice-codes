void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<bool> flags(n + 1);
    for (int i = 0; i < n; ++i)
    {
        int sum{ as[i] };
        for (int j = i + 1; j < n && sum <= n; ++j)
        {
            sum += as[j];
            if (sum <= n)
                flags[sum] = true;
        }
    }

    int ans{ 0 };
    for (auto a : as)
    {
        if (flags[a])
            ++ans;
    }
    std::cout << ans << '\n';
}
