void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> mins_l(n + 1);
    std::vector<int> mins_r(n + 1);
    mins_l[1] = as[1];
    mins_r[n] = as[n];
    for (int i{ 2 }; i <= n; ++i)
    {
        mins_l[i] = std::min(as[i], mins_l[i - 1]);
    }
    for (int i{ n - 1 }; i >= 1; --i)
    {
        mins_r[i] = std::min(as[i], mins_r[i + 1]);
    }

    for (int i{ 2 }; i < n; ++i)
    {
        if (mins_l[i - 1] > as[i] || mins_r[i + 1] > as[i])
            continue;
        std::cout << "Yes\n";
        for (int j{ 1 }; j < i; ++j)
        {
            if (as[j] < as[i])
            {
                std::cout << j << ' ';
                break;
            }
        }
        std::cout << i << ' ';
        for (int j{ n }; j > i; --j)
        {
            if (as[j] < as[i])
            {
                std::cout << j << '\n';
                break;
            }
        }
        return;
    }
    std::cout << "No\n";
}
