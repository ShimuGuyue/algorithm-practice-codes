void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::multiset<int> t(as.begin() + 1, as.end());
    int l{ 1 }, r{ n };
    while (!t.empty())
    {
        if (as[l] == *t.begin())
        {
            t.erase(t.begin());
            ++l;
        }
        else if (as[l] == *std::prev(t.end()))
        {
            t.erase(std::prev(t.end()));
            ++l;
        }
        else if (as[r] == *t.begin())
        {
            t.erase(t.begin());
            --r;
        }
        else if (as[r] == *std::prev(t.end()))
        {
            t.erase(std::prev(t.end()));
            --r;
        }
        else
        {
            std::cout << l << ' ' << r << '\n';
            return;
        }
    }

    std::cout << -1 << '\n';
}
