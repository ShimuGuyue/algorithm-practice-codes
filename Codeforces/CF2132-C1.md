void solve()
{
    static constexpr std::array<int64_t, 20> flags{{
        1, 3, 9, 27, 81, 243, 729, 2187, 6561, 19683, 59049, 177147, 531441,
        1594323, 4782969, 14348907, 43046721, 129140163, 387420489, 1162261467
    }};

    int64_t n;
    std::cin >> n;

    int64_t ans{};
    while (n)
    {
        int x = std::prev(std::upper_bound(flags.begin(), flags.end(), n)) - flags.begin();
        int64_t count{ n / flags[x] };
        n %= flags[x];
        int64_t per = flags[x + 1] + (x ? x * flags[x - 1] : 0);
        ans += per * count;
    }
    std::cout << ans << '\n';
}
