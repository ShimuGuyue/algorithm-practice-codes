void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> odds;
    std::vector<int> evens;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (i & 1)
            odds.push_back(as[i]);
        else
            evens.push_back(as[i]);
    }

    std::sort(odds.rbegin(), odds.rend());
    std::sort(evens.rbegin(), evens.rend());

    int count_odd{ 0 }, count_even{ 0 };
    while (m--)
    {
        int x;
        std::cin >> x;
        if (x & 1)
            ++count_odd;
        else
            ++count_even;
    }

    int64_t ans{ std::accumulate(as.begin() + 1, as.end(), INT64_C(0)) };
    for (int i{ 0 }; i < std::min<int>(count_odd, odds.size()); ++i)
    {
        if (odds[i] <= 0)
        {
            if (i == 0)
                ans -= odds[i];
            break;
        }
        ans -= odds[i];
    }
    for (int i{ 0 }; i < std::min<int>(count_even, evens.size()); ++i)
    {
        if (evens[i] <= 0)
        {
            if (i == 0)
                ans -= evens[i];
            break;
        }
        ans -= evens[i];
    }
    std::cout << ans << '\n';
}
