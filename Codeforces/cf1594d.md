```cpp
class DisjointSet
{
private:
    struct Data
    {
        int father;
        int size;
    };
    std::vector<Data> nodes_;

public:
    DisjointSet(const int n)
    {
        build(n);
    }

public:
    int find(const int x)
    {
        return nodes_[x].father == x ? x : nodes_[x].father = find(nodes_[x].father);
    }

    bool merge(const int x, const int y)
    {
        const int set_x{ find(x) };
        const int set_y{ find(y) };
        if (set_x == set_y)
            return false;
        nodes_[set_x].father = set_y;
        nodes_[set_y].size += nodes_[set_x].size;
        return true;
    }

    int get_size(const int x)
    {
        return nodes_[find(x)].size;
    }

private:
    void build(const int n)
    {
        nodes_.assign(n * 2 + 1, { });
        for (int i{ 1 }; i <= n * 2; ++i)
        {
            nodes_[i].father = i;
            nodes_[i].size = i <= n;
        }
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;
    DisjointSet ds(n);
    bool ok{ true };
    while (m--)
    {
        int i, j;
        std::string s;
        std::cin >> i >> j >> s;
        if (s[0] == 'c')
        {
            if (ds.find(i) == ds.find(j + n))
                ok = false;
            ds.merge(i, j);
            ds.merge(i + n, j + n);
        }
        else
        {
            if (ds.find(i) == ds.find(j))
                ok = false;
            ds.merge(i, j + n);
            ds.merge(j, i + n);
        }
    }
    if (!ok)
    {
        std::cout << -1 << '\n';
        return;
    }

    int ans{ 0 };
    std::vector<bool> visited(n * 2 + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (visited[ds.find(i)] || visited[ds.find(i + n)])
            continue;
        visited[ds.find(i)] = visited[ds.find(i + n)] = true;
        ans += std::max(
            ds.get_size(ds.find(i)),
            ds.get_size(ds.find(i + n))
        );
    }
    std::cout << ans << '\n';
}
```
