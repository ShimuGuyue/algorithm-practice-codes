void solve()
{
    constexpr int mod{1000000007};

    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> vs(n);
    for (auto& s : vs)
    {
        std::cin >> s;
    }

    int64_t ans{ 1 };
    for (int i = 0; i < m; ++i)
    {
        std::set<char> t;
        for (int j = 0; j < n; ++j)
        {
            t.insert(vs[j][i]);
        }
        ans = ans * t.size() % mod;
    }
    std::cout << ans << '\n';
}
