void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 0 };
    for (int i{ n - 2 }; i >= 0; --i)
    {
        while (as[i] >= as[i + 1] && as[i] != 0)
        {
            as[i] /= 2;
            ++ans;
        }
        if (as[i] == as[i + 1])
        {
            std::cout << -1 << '\n';
            return;
        }
    }
    std::cout << ans << '\n';
}
