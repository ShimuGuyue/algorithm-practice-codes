void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> indexs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        int a;
        std::cin >> a;
        indexs[a].push_back(i);
    }

    static constexpr int inf{ std::numeric_limits<int>::max() };
    std::vector<int> ans(n + 1, inf);
    for (int a{ 1 }; a <= n; ++a)
    {
        if (indexs[a].empty())
            continue;
        int len{ std::max(indexs[a].front(), n - indexs[a].back() + 1) };
        for (int i{ 1 }; i < indexs[a].size(); ++i)
        {
            len = std::max(len, indexs[a][i] - indexs[a][i - 1]);
        }
        ans[len] = std::min(ans[len], a);
    }
    for (int i{ 2 }; i <= n; ++i)
    {
        ans[i] = std::min(ans[i], ans[i - 1]);
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << (ans[i] == inf ? -1 : ans[i]) << ' ';
    }
    std::cout << '\n';
}
