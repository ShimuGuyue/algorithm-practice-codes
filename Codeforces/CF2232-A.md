void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::map<int, int> counts;
    for (auto a : as)
    {
        ++counts[a];
    }

    int ans{ n };
    int pre{ 0 };
    for (auto [a, c] : counts)
    {
        ans = std::min(ans, std::max(pre, n - pre - c));
        pre += c;
    }
    std::cout << ans << '\n';
}
