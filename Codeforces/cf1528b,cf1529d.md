void solve()
{
    int n;
    std::cin >> n;

    std::vector<int> factors(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ i }; j <= n; j += i)
        {
            ++factors[j];
        }
    }

    int64_t mod{ 998244353 };

    std::vector<int64_t> dp(n + 1);
    std::vector<int64_t> pres(n + 1);
    dp[1] = 1;
    pres[1] = dp[1];
    for (int i{ 2 }; i <= n; ++i)
    {
        dp[i] = (pres[i - 1] + factors[i]) % mod;
        pres[i] = (pres[i - 1] + dp[i]) % mod;
    }
    std::cout << dp[n] << '\n';
}
