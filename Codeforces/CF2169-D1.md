void solve()
{
    int64_t x, y, k;
    std::cin >> x >> y >> k;

    auto check = [x, y, k](int64_t flag) -> bool
    {
        for (int i{ 0 }; i < x; ++i)
        {
            flag -= flag / y;
        }
        return flag >= k;
    };
    int64_t l{ 1 }, r{ 1000000000000 };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    if (check(r))
        std::cout << r << '\n';
    else
        std::cout << -1 << '\n';
}
