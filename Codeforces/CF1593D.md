void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::pair<int64_t, int64_t>> abs(n);
    for (auto& [a, b] : abs)
    {
        std::cin >> a >> b;
    }

    std::sort(abs.begin(), abs.end(),
        [](std::pair<int64_t, int64_t> a, std::pair<int64_t, int64_t> b)
        {
            return a.second < b.second;
        }
    );

    int64_t max{
        std::accumulate(abs.begin(), abs.end(), INT64_C(0),
            [](int64_t pre, std::pair<int64_t, int64_t> cur)
            {
                return pre + cur.first;
            }
        )
    };

    int64_t ans{ 0 };
    int64_t sum{ 0 };
    int l{ 0 }, r{ n - 1 };
    while (sum < max)
    {
        if (sum >= abs[l].second)
        {
            sum += abs[l].first;
            ans += abs[l].first * 1;
            abs[l].first = 0;
            ++l;
        }
        else
        {
            int64_t dif{ abs[l].second - sum };
            if (dif >= abs[r].first)
            {
                sum += abs[r].first;
                ans += abs[r].first * 2;
                abs[r].first = 0;
                --r;
            }
            else
            {
                sum += dif;
                ans += dif * 2;
                abs[r].first -= dif;
            }
        }
    }
    std::cout << ans << '\n';
}
