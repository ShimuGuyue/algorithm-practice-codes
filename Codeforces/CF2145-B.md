void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    std::string ans(n, '+');
    int l{ 0 }, r{ n - 1 };
    int count{ 0 };
    for (char c : s)
    {
        if (c == '0')
            ans[l++] = '-';
        else if (c == '1')
            ans[r--] = '-';
        else
            ++count;
    }
    for (int i{ 0 }; i < count; ++i)
    {
        ans[l + i] = ans[r - i] = count == r - l + 1 ? '-' : '?';
    }
    std::cout << ans << '\n';
}
