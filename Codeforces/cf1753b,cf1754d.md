void solve()
{
    int n, x;
    std::cin >> n >> x;
    std::array<int, 500000 + 1> counts{ };
    while (n--)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }
    for (int i{ 1 }; i < x; ++i)
    {
        counts[i + 1] += counts[i] / (i + 1);
        counts[i] %= i + 1;
        if (counts[i])
        {
            std::cout << "No\n";
            return;
        }
    }
    if (counts[x])
        std::cout << "Yes\n";
    else
        std::cout << "No\n";
}
