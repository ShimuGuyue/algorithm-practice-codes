void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t ans{ 0 };
    std::vector<int> precounts(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        precounts[i] = precounts[i - 1];
        if (as[i] >= i)
            continue;
        int j{ as[i] - 1 };
        if (j >= 0)
            ans += precounts[j];
        ++precounts[i];
    }
    std::cout << ans << '\n';
}
