void solve()
{
    std::string a;
    std::string b;
    std::cin >> a;
    std::cin >> b;

    auto n{ static_cast<int>(a.length()) };

    a = " " + a;
    b = " " + b;

    auto judge = [&a, &b](this auto&& judge, int l1, int r1, int l2, int r2) -> bool
    {
        int len{ r1 - l1 + 1 };
        bool ok{ true };
        for (int i{ 0 }; i < len; ++i)
        {
            if (a[l1 + i] != b[l2 + i])
                ok = false;
        }
        if (ok)
            return true;

        if (len & 1)
            return false;
        int half1{ (l1 + r1) / 2 };
        int half2{ (l2 + r2) / 2 };
        if (judge(l1, half1, half2 + 1, r2) && judge(half1 + 1, r1, l2, half2))
            return true;
        if (judge(l1, half1, l2, half2) && judge(half1 + 1, r1, half2 + 1, r2))
            return true;
        return false;
    };

    if (judge(1, n, 1, n))
        std::cout << "YES\n";
    else
        std::cout << "NO\n";
}
