void solve()
{
    int n, x;
    std::cin >> n >> x;

    int64_t ans{ 0 };
    for (int a = 1; a + 2 <= x && a + 2 <= n; ++a)
    {
        for (int b = 1; a + b + 1 <= x && a * b + 2 <= n; ++b)
        {
            int range_c{ std::min(x - a - b ,(n - a * b) / (a + b)) };
            ans += range_c;
        }
    }
    std::cout << ans << '\n';
}
