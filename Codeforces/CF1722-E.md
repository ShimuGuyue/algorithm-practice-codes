void solve()
{
    constexpr int m{ 1000 };

    int n, q;
    std::cin >> n >> q;

    std::vector<std::vector<int64_t>> pres(m + 1, std::vector<int64_t>(m + 1));
    while (n--)
    {
        int a, b;
        std::cin >> a >> b;
        pres[a][b] += a * b;
    }
    for (int i = 1; i <= m; ++i)
    {
        for (int j = 1; j <= m; ++j)
        {
            pres[i][j] += pres[i - 1][j] + pres[i][j - 1] - pres[i - 1][j - 1];
        }
    }

    while (q--)
    {
        int a1, b1, a2, b2;
        std::cin >> a1 >> b1 >> a2 >> b2;
        --a2; --b2;
        int64_t ans{ pres[a2][b2] - pres[a1][b2] - pres[a2][b1] + pres[a1][b1] };
        std::cout << ans << '\n';
    }
}
