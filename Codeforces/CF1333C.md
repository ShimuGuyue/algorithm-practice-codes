void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t ans{ 0 };
    std::map<int64_t, int> m;
    m[0] = 0;
    int64_t sum{ 0 };
    int l{ 0 }, r{ 1 };
    while (r <= n)
    {
        sum += as[r];
        if (m.count(sum))
            l = std::max(l, m[sum] + 1);
        ans += r - l;
        m[sum] = r;
        ++r;
    }
    std::cout << ans << '\n';
}
