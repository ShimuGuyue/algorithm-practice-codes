void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string a;
    std::string b;
    std::cin >> a;
    std::cin >> b;

    std::string c;
    for (int i{ 0 }; i < n; ++i)
    {
        c.push_back((a[i] != b[i]) + '0');
    }

    int64_t flag{ 1 };
    for (int i{ 1 }; i < k; ++i)
    {
        if (i & 1)
            flag = flag * 2;
        else
            flag = flag * 2 - 1;
    }

    int64_t ans{ 0 };

    int64_t a0{ std::count(a.begin(), a.end(), '0') };
    int64_t b0{ std::count(b.begin(), b.end(), '0') };
    int64_t c0{ std::count(c.begin(), c.end(), '0') };
    int64_t a1{ std::count(a.begin(), a.end(), '1') };
    int64_t b1{ std::count(b.begin(), b.end(), '1') };
    int64_t c1{ std::count(c.begin(), c.end(), '1') };

    ans += a0 * a1 * flag;
    ans += b0 * b1 * flag;
    ans += c0 * c1 * (k & 1 ? flag : flag - 1);

    std::cout << ans << '\n';
}
