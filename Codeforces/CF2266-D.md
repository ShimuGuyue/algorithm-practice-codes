void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    for (int i = 0; i < n; ++i)
    {
        as[i] -= i;
    }
    sort(as.begin(), as.end());
    as.erase(unique(as.begin(), as.end()), as.end());

    int ans = 1;
    int count = 1;
    for (int i = 1; i < n; ++i)
    {
        if (as[i] == as[i - 1] + 1)
            ++count;
        else
            count = 1;
        ans = std::max(ans, count);
    }
    cout << ans << '\n';
}
