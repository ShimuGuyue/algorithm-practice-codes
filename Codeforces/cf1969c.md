void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    SparseTable st(as);

    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(k + 1, INT64_C(1) << 60));
    dp[0][0] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 0 }; j <= k; ++j)
        {
            for (int d{ 0 }; d <= j; ++d)
            {
                if (d == i)
                    break;
                int64_t min{ st.query(i - d, i) };
                dp[i][j] = std::min(dp[i][j], dp[i - d - 1][j - d] + (d + 1) * min);
            }
        }
    }

    int64_t ans{ *std::min_element(dp[n].begin(), dp[n].end()) };
    std::cout << ans << '\n';
}
