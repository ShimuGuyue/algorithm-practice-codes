void solve()
{
    int n, d;
    std::cin >> n >> d;
    std::vector<int64_t> as(n * 3 + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        as[n + i] = as[i];
        as[n + n + i] = as[i];
    }

    std::vector<int64_t> pres(n * 3 + 1);
    std::partial_sum(as.begin(), as.end(), pres.begin());

    int64_t ans{ 0 };
    for (int i{ n + 1 }; i <= n + n; ++i)
    {
        int64_t sum{ as[i] * d * 2 };
        sum -= pres[i - 1] - pres[i - d - 1];
        sum -= pres[i + d] - pres[i];
        if (sum > 0)
            ans += sum;
    }
    std::cout << ans << '\n';
}
