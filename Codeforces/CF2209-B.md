void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    for (int i{ 0 }; i < n; ++i)
    {
        int greater{ 0 }, less{ 0 };
        for (int j{ i + 1 }; j < n; ++j)
        {
            if (as[j] > as[i])
                ++greater;
            if (as[j] < as[i])
                ++less;
        }
        std::cout << std::max(greater, less) << ' ';
    }
    std::cout << '\n';
}
