void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    for (char c{ 'a' }; c <= 'z'; ++c)
    {
        bool b1{ false }, b2{ false };
        for (int i{ 0 }; i < n; ++i)
        {
            if (s[i] != c)
                continue;
            if (i & 1)
                b1 = true;
            else
                b2 = true;
        }
        if (b1 && b2)
        {
            std::cout << "NO\n";
            return;
        }
    }
    std::cout << "YES\n";
}
