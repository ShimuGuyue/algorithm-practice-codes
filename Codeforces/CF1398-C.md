void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    s = " " + s;

    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        as[i] = s[i] ^ '0';
    }
    std::vector<int> presums(n + 1);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    int64_t ans{ };
    std::map<int, int> counts;
    for (int i = 1; i <= n; ++i)
    {
        ++counts[presums[i - 1] - i + 1];
        ans += counts[presums[i] - i];
    }
    std::cout << ans << '\n';
}
