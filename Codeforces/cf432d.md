void solve()
{
    std::string s;
    std::cin >> s;

    std::vector<int> borders{ get_borders(s) };

    std::vector<int> dp(s.length(), 1);
    for (int i{ int(s.length() - 1) }; i >= 0; --i)
    {
        if (borders[i])
            dp[borders[i] - 1] += dp[i];
    }

    std::vector<std::pair<int, int>> ans;
    int index{ int(s.length()) - 1 };
    while (index >= 0)
    {
        ans.push_back({index + 1, dp[index]});
        index = borders[index] - 1;
    }
    std::reverse(ans.begin(), ans.end());
    std::cout << ans.size() << '\n';
    for (auto [a, b] : ans)
    {
        std::cout << a << ' ' << b << '\n';
    }
}

std::vector<int> get_borders(const std::string& s)
{
    const int n{ static_cast<int>(s.size()) };
    std::vector<int> borders(n, 0);
    for (int i{ 1 }; i < n; ++i)
    {
        int border{ borders[i - 1] };
        while (border != 0 && s[border] != s[i])
        {
            border = borders[border - 1];
        }
        if (s[border] == s[i])
            ++border;
        borders[i] = border;
    }
    return borders;
}
