void solve()
{
    int64_t n, m, k;
    std::cin >> n >> m >> k;

    auto check = [n, m, k](int64_t flag) -> bool
    {
        int64_t sum{ 0 };
        for (int i{ 1 }; i <= n; ++i)
        {
            sum += std::min(m, flag / i);
        }
        return sum >= k;
    };

    int64_t l{ 1 }, r{ n * m };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
