void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int ans{ 0 };

    std::priority_queue<int, std::vector<int>, std::greater<int>> h1;
    int64_t sum1{ 0 };
    for (int i{ m + 1 }; i <= n; ++i)
    {
        sum1 += as[i];
        h1.push(as[i]);
        if (sum1 < 0)
        {
            ++ans;
            sum1 -= h1.top() * 2;
            h1.pop();
        }
    }

    std::priority_queue<int, std::vector<int>, std::less<int>> h2;
    int64_t sum2{ 0 };
    for (int i{ m }; i > 1; --i)
    {
        sum2 += as[i];
        h2.push(as[i]);
        if (sum2 > 0)
        {
            ++ans;
            sum2 -= h2.top() * 2;
            h2.pop();
        }
    }

    std::cout << ans << '\n';
}
