void solve()
{
    int64_t n, h, k;
    std::cin >> n >> h >> k;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t sum{ std::accumulate(as.begin(), as.end(), INT64_C(0)) };

    int64_t ans{ h / sum * (n + k) };
    h %= sum;
    if (h == 0)
    {
        std::cout << ans - k << '\n';
        return;
    }

    std::vector<int64_t> premins(n + 1);
    premins[1] = as[1];
    for (int i{ 2 }; i <= n; ++i)
    {
        premins[i] = std::min(premins[i - 1], as[i]);
    }
    std::vector<int64_t> sufmaxs(n + 2);
    sufmaxs[n] = as[n];
    for (int i{ int(n) - 1 }; i > 0; --i)
    {
        sufmaxs[i] = std::max(sufmaxs[i + 1], as[i]);
    }

    std::vector<int64_t> presums(n + 1);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    for (int i{ 1 }; i <= n; ++i)
    {
        if (presums[i] >= h || presums[i] - premins[i] + sufmaxs[i + 1] >= h)
        {
            std::cout << ans + i << '\n';
            return;
        }
    }
}
