void solve()
{
    std::array<int, 5> as;
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int sum{ 0 };
    std::map<int, int> m;
    for (auto a : as)
    {
        sum += a;
        ++m[a];
    }

    int ans{ sum };
    for (auto [k, v] : m)
    {
        if (v >= 3)
            ans = std::min(ans, sum - k * 3);
        else if (v == 2)
            ans = std::min(ans, sum - k * 2);
    }
    std::cout << ans << '\n';
}
