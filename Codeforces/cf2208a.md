void solve()
{
    int n;
    std::cin >> n;
    std::map<int, int> m;
    for (int i{ 0 }; i < n * n; ++i)
    {
        int a;
        std::cin >> a;
        ++m[a];
    }

    for (auto [k, v] : m)
    {
        if (v > n * (n - 1))
        {
            std::cout << "NO\n";
            return;
        }
    }
    std::cout << "YES\n";
}
