void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::string> grid(n);
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    std::set<int> t;
    for (auto& s : grid)
    {
        auto a{ s.find('1') };
        if (a == std::string::npos)
            continue;
        t.insert(a);
    }
    std::cout << (t.size() == 1 ? "SQUARE\n" : "TRIANGLE\n");
}
