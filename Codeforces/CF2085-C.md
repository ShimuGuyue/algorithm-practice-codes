void solve()
{
    int64_t x, y;
    std::cin >> x >> y;

    if (x == y)
    {
        std::cout << -1 << '\n';
        return;
    }

    int64_t flag{ 1 };
    while (flag < x || flag < y)
    {
        flag *= 2;
    }

    int64_t k{ flag - std::max(x, y) };
    std::cout << k << '\n';
}
