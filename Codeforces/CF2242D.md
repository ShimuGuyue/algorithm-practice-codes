void solve()
{
    std::string a;
    std::string b;
    std::cin >> a;
    std::cin >> b;

    int n{ static_cast<int>(a.size()) };
    int m{ static_cast<int>(b.size()) };
    a = " " + a;
    b = " " + b;

    std::vector<int> pres_a(n + 1);
    std::vector<int> pres_b(m + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        pres_a[i] = (pres_a[i - 1] + a[i] - '0') % 10;
    }
    for (int i{ 1 }; i <= m; ++i)
    {
        pres_b[i] = (pres_b[i - 1] + b[i] - '0') % 10;
    }

    if (pres_a[n] != pres_b[m])
    {
        std::cout << -1 << '\n';
        return;
    }

    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(m + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            if (pres_a[i] == pres_b[j])
                dp[i][j] = dp[i - 1][j - 1] + 1;
            else
                dp[i][j] = std::max(dp[i - 1][j], dp[i][j - 1]);
        }
    }
    std::cout << dp[n][m] << '\n';
}
