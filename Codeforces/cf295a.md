void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<int64_t> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::array<int, 3>> orders(m + 1);
    for (int i = 1; i <= m; ++i)
    {
        auto& [l, r, d]{ orders[i] };
        std::cin >> l >> r >> d;
    }

    std::vector<int> difs_order(m + 2);
    while (k--)
    {
        int x, y;
        std::cin >> x >> y;
        ++difs_order[x];
        --difs_order[y + 1];
    }
    std::vector<int64_t> counts_order(m + 2);
    std::partial_sum(difs_order.begin(), difs_order.end(), counts_order.begin());

    std::vector<int64_t> difs(n + 2);
    {
        for (int i = 1; i <= m; ++i)
        {
            auto [l, r, d]{ orders[i] };
            int64_t count{ counts_order[i] };
            int64_t sum{ count * d };
            difs[l] += sum;
            difs[r + 1] -= sum; 
        }
    }

    std::vector<int64_t> increments(n + 2);
    std::partial_sum(difs.begin(), difs.end(), increments.begin());
    for (int i = 1; i <= n; ++i)
    {
        as[i] += increments[i];
    }

    for (int i = 1; i <= n; ++i)
    {
        std::cout << as[i] << ' ';
    }
    std::cout << '\n';
}
