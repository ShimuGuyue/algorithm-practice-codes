void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    if (std::count(as.begin(), as.end(), 1) == n)
    {
        cout << -1 << '\n';
        return;
    }

    if (n == 2)
    {
        cout << as[0] + as[1] - 1 << '\n';
        return;
    }

    int ans;
    if (std::count(as.begin() + 1, as.end() - 1, 1) == n - 2)
        if (as.front() != 1 && as.back() != 1)
            ans = as.front() + as.back() - 1;
        else
            ans = std::max(as.front(), as.back());
    else
        ans = accumulate(as.begin(), as.end(), 0) - 1;
    cout << ans << '\n';
}
