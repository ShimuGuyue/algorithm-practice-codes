void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t ans{ int64_t(n) * (n - 1) - k * n * 2 };
    for (int i = std::max(1, n - 2 * k); i <= n; ++i)
    {
        for (int j = i + 1; j <= n; ++j)
        {
            ans = std::max(ans, int64_t(i) * j - k * (as[i] | as[j]));
        }
    }
    std::cout << ans << '\n';
}
