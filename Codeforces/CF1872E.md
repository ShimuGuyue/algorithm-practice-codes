void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    std::string s;
    std::cin >> s;
    s = " " + s;

    std::vector<int> presums_xor(n + 1);
    std::partial_sum(as.begin(), as.end(), presums_xor.begin(),
        [](int sum, int next)
        {
            return sum ^ next;
        }
    );

    int ans{ 0 };
    for (int i = 1; i <= n; ++i)
    {
        if (s[i] == '1')
            ans ^= as[i];
    }

    int q;
    std::cin >> q;
    while (q--)
    {
        int op;
        std::cin >> op;
        if (op == 1)
        {
            int l, r;
            std::cin >> l >> r;
            ans ^= presums_xor[r] ^ presums_xor[l - 1];
        }
        else
        {
            int g;
            std::cin >> g;
            if (g == 1)
                std::cout << ans << ' ';
            else
                std::cout <<( presums_xor.back() ^ ans) << ' ';
        }
    }
    std::cout << '\n';
}
