void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::vector<int> flags_0;
    std::vector<int> flags_1;
    int count{ 0 };
    std::vector<int> ans(n);
    for (int i = 0; i < n; ++i)
    {
        if (s[i] == '0')
        {
            if (flags_1.empty())
            {
                ++count;
                flags_0.push_back(count);
                ans[i] = count;
            }
            else
            {
                flags_0.push_back(flags_1.back());
                ans[i] = flags_1.back();
                flags_1.pop_back();
            }
        }
        else
        {
            if (flags_0.empty())
            {
                ++count;
                flags_1.push_back(count);
                ans[i] = count;
            }
            else
            {
                flags_1.push_back(flags_0.back());
                ans[i] = flags_0.back();
                flags_0.pop_back();
            }
        }
    }
    std::cout << count << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
