```c++
struct SegementTree
{
    struct Node
    {
        int count_sum{ 0 };
        int count_l{ 0 };
        int count_r{ 0 };
        int sege_l, sege_r;
        Node* l{ nullptr };
        Node* r{ nullptr };
    };
    Node* root{ new Node };

    void build(std::string& s, Node* n, int l, int r)
    {
        n->sege_l = l;
        n->sege_r = r;
        if (l == r)
        {
            if (s[l] == '(')
                n->count_l = 1;
            else
                n->count_r = 1;
            return;
        }
        int mid{ (l + r) / 2 };
        n->l = new Node;
        n->r = new Node;
        build(s, n->l, l, mid);
        build(s, n->r, mid + 1, r);
        int min{ std::min(n->l->count_l, n->r->count_r) };
        n->count_sum = n->l->count_sum + n->r->count_sum + min;
        n->count_l = n->l->count_l + n->r->count_l - min;
        n->count_r = n->r->count_r + n->l->count_r - min;
    }

    Node query(Node* n, int l, int r)
    {
        if (l <= n->sege_l && r >= n->sege_r )
            return *n;

        int mid{ (n->sege_l + n->sege_r) / 2 };

        if (r <= mid)
            return query(n->l, l, r);
        if (l > mid)
            return query(n->r, l, r);

        Node n_l{ query(n->l, l, mid) };
        Node n_r{ query(n->r, mid + 1, r) };

        int matched{ std::min(n_l.count_l, n_r.count_r) };
        Node ans;
        ans.count_sum = n_l.count_sum + n_r.count_sum + matched;
        ans.count_l = n_l.count_l + n_r.count_l - matched;
        ans.count_r = n_l.count_r + n_r.count_r - matched;
        return ans;
    }
};
```

```c++
void solve()
{
    std::string s;
    std::cin >> s;
    s = " " + s;

    SegementTree st;
    st.build(s, st.root, 1, s.length() - 1);

    int q;
    std::cin >> q;
    while (q--)
    {
        int l, r;
        std::cin >> l >> r;
        std::cout << st.query(st.root, l, r).count_sum * 2 << '\n';
    }
}
```
