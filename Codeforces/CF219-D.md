void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::map<int, bool>> graph(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        graph[u][v] = true;
        graph[v][u] = false;
    }

    std::vector<int> dp(n + 1);
    auto dfs1 = [&graph, &dp](this auto&& dfs, int u, int last) -> void
    {
        for (auto [v, w] : graph[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            if (!w)
                ++dp[u];
            dp[u] += dp[v];
        }
    };
    dfs1(1, 0);

    std::vector<int> ans(n + 1);
    ans[1] = dp[1];
    auto dfs2 = [&graph, &ans](this auto&& dfs, int u, int last) -> void
    {
        if (last)
        {
            if (graph[last][u])
                ans[u] = ans[last] + 1;
            else
                ans[u] = ans[last] - 1;
        }
        for (auto [v, w] : graph[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
        }
    };
    dfs2(1, 0);
    int min{ *std::min_element(ans.begin() + 1, ans.end()) };
    std::cout << min << '\n';
    for (int i{ 1 }; i <= n; ++i)
    {
        if (ans[i] == min)
            std::cout << i << ' ';
    }
    std::cout << '\n';
}
