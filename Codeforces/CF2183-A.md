void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    as.push_back(2);

    int count0{ 0 }, count1{ 0 };
    int flag{ as[0] };
    for (int i = 1; i <= n; ++i)
    {
        if (as[i] == flag)
            continue;
        if (flag == 0)
            ++count0;
        else
            ++count1;
        flag = as[i];
    }
    if (count1 >= count0)
        std::cout << "alice\n";
    else
        std::cout << "bob\n";
}
