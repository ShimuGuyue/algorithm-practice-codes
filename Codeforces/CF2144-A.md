void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> pres(n + 1);
    std::partial_sum(as.begin(), as.end(), pres.begin());

    for (int l{ 1 }; l < n; ++l)
    {
        for (int r{ l + 1 }; r <= n; ++r)
        {
            int a{ pres[l] % 3 };
            int b{ (pres[r] - pres[l]) % 3 };
            int c{ (pres[n] - pres[r]) % 3 };
            if (a == b && b == c)
            {
                std::cout << l << ' ' << r << '\n';
                return;
            }
            if (a != b && b != c && a != c)
            {
                std::cout << l << ' ' << r << '\n';
                return;
            }
        }
    }
    std::cout << 0 << ' ' << 0 << '\n';
}
