void solve()
{
    std::string s;
    std::cin >> s;

    int ans{ 0 };
    for (int i{ 1 }; i * 2 <= s.length(); ++i)
    {
        int count{ 0 };
        for (int j{ 0 }; j < i; ++j)
        {
            if (s[j] == s[j + i] || s[j] == '?' || s[j + i] == '?')
                ++count;
        }
        if (count == i)
        {
            ans = i;
            continue;
        }
        for (int j{ i }; j + i < s.length(); ++j)
        {
            if (s[j] == s[j - i] || s[j] == '?' || s[j - i] == '?')
                --count;
            if (s[j] == s[j + i] || s[j] == '?' || s[j + i] == '?')
                ++count;
            if (count == i)
            {
                ans = i;
                break;
            }
        }
    }
    std::cout << ans * 2 << '\n';
}
