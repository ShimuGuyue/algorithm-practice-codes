void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(n + 1);
    std::vector<int> bs(m + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i{ 1 }; i <= m; ++i)
    {
        std::cin >> bs[i];
    }

    std::vector<bool> flags(n + 1);
    for (int i{ 1 }; i <= m; ++i)
    {
        flags[bs[i]] = true;
    }

    std::vector<std::vector<int64_t>> dp(n + 2, std::vector<int64_t>(2));
    dp[n + 1][1] = -(INT64_C(1) << 60);
    for (int i{ n }; i >= 1; --i)
    {
        if (flags[i])
        {
            dp[i][0] = std::max(dp[i + 1][0], dp[i + 1][1]) + as[i];
            dp[i][1] = std::max(dp[i + 1][0], dp[i + 1][1]) - as[i];
        }
        else
        {
            dp[i][0] = dp[i + 1][0] + as[i];
            dp[i][1] = dp[i + 1][1] - as[i];
        }
    }
    int64_t ans{ std::max(dp[1][0], dp[1][1]) };
    std::cout << ans << '\n';
}
