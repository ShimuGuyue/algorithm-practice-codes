void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> flags;
    bool flag1{ false }, flag2{ false };
    for (auto a : as)
    {
        if (a == 0)
        {
            if (flag2)
                flags.push_back(2);
            else if (flag1)
                flags.push_back(1);
            flags.push_back(0);
            flag1 = flag2 = false;
        }
        else if (a == 1)
            flag1 = true;
        else if (a == 2)
            flag2 = true;
    }
    if (as.back())
    {
        if (flag2)
            flags.push_back(2);
        else if (flag1)
            flags.push_back(1);
    }

    int ans{ 0 };
    int m{ int(flags.size()) };
    for (int i{ 0 }; i < m; ++i)
    {
        if (flags[i] == 0 || flags[i] == -1)
            continue;
        if (flags[i] == 1)
        {
            if (i != 0 && flags[i - 1] == 0)
                flags[i - 1] = -1;
            else if (i != m - 1 && flags[i + 1] == 0)
                flags[i + 1] = -1;
        }
        else if (flags[i] == 2)
        {
            if (i != 0 && flags[i - 1] == 0)
                flags[i - 1] = -1;
            if (i != m - 1 && flags[i + 1] == 0)
                flags[i + 1] = -1;
        }
        ++ans;
        flags[i] = -1;
    }
    for (int i{ 0 }; i < m; ++i)
    {
        if (flags[i] == 0)
            ++ans;
    }
    std::cout << ans << '\n';
}
