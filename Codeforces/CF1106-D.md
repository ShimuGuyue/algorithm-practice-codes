void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> map(n + 1);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        map[u].push_back(v);
        map[v].push_back(u);
    }

    std::vector<bool> visited(n + 1);
    std::priority_queue<int, std::vector<int>, std::greater<int>> h;
    h.push(1);
    visited[1] = true;
    while (!h.empty())
    {
        int u{ h.top() };
        h.pop();
        std::cout << u << ' ';
        for (int v : map[u])
        {
            if (visited[v])
                continue;
            visited[v] = true;
            h.push(v);
        }
    }
    std::cout << '\n';
}
