void solve()
{
    int n, c;
    std::cin >> n >> c;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    as[0] += c;

    int64_t max{ -1 };
    int index_max;
    for (int i{ 0 }; i < n; ++i)
    {
        if (as[i] > max)
        {
            max = as[i];
            index_max = i;
        }
    }

    std::vector<int64_t> presums(n);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    for (int i{ 0 }; i < n; ++i)
    {
        if (i == index_max)
        {
            std::cout << 0 << ' ';
            continue;
        }
        int ans{ i };
        int64_t sum{ presums[i] };
        if (sum < max)
            ++ans;
        std::cout << ans << ' ';
    }
    std::cout << '\n';
}
