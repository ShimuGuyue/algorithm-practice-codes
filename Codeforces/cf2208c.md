void solve()
{
    int n;
    std::cin >> n;
    std::vector<double> cs(n + 1);
    std::vector<double> ps(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> cs[i] >> ps[i];
        ps[i] = 1.0 - ps[i] / 100;
    }

    std::vector<double> dp(n + 1);
    dp[n] = cs[n];
    for (int i{ n - 1 }; i >= 1; --i)
    {
        dp[i] = std::max(
            dp[i + 1],
            dp[i + 1] * ps[i] + cs[i]
        );
    }

    double ans{ dp[1] };
    std::cout << std::fixed << std::setprecision(10) << ans << '\n';
}
