void solve()
{
    int64_t n, c;
    std::cin >> n >> c;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t ans{ 0 };

    ans += (c + 1) * (1 + c + 1) / 2;

    /*
        x + y == a  x <= y      y - x == a  y <= c
        x <= a - x              a + x <= c
        x <= a / 2              x <= c - a
    */
    int64_t odd{ 0 };
    int64_t even{ 0 };
    for (auto a : as)
    {
        ans -= a / 2 + 1;
        ans -= c - a + 1;
        if (a & 1)
            ++odd;
        else
            ++even;
    }

    /*
        x + y == a  y - x == b
        x = (a - b) / 2, y = (a + b) / 2
    */
    ans += odd * (odd + 1) / 2 + even * (even + 1) / 2;

    std::cout << ans << '\n';
}
