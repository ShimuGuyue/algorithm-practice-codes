void solve()
{
    std::vector<std::string> grid(8);
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    std::string ans;
    for (int j{ 0 }; j < 8; ++j)
    {
        for (int i{ 0 }; i < 8; ++i)
        {
            if (grid[i][j] != '.')
                ans += grid[i][j];
        }
    }
    std::cout << ans << '\n';
}
