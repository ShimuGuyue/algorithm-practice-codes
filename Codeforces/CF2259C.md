void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    for (int i{ 0 }; i < n; ++i)
    {
        if ( as[i] == 0)
            continue;
        as[i] = 1;
        break;
    }
    for (int i{ n - 1 }; i >= 0; --i)
    {
        if (as[i] == 0)
            continue;
        as[i] = 1;
        break;
    }
    std::for_each(as.begin(), as.end(),
        [](int& a)
        {
            if (a == -1)
                a = 0;
        }
    );

    for (auto a : as)
    {
        cout << a << ' ';
    }
    cout << '\n';
}
