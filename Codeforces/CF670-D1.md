void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    auto check = [n, k, &as, &bs](int64_t count)
    {
        int64_t flag{ 0 };
        for (int i = 0; i < n; ++i)
        {
            int64_t sum{ as[i] * count };
            if (sum > bs[i])
                flag += sum - bs[i];
            if (flag > k)
                return false;
        }
        return true;
    };

    int64_t l{ 0 }, r{ 2000000000 };
    while (l < r)
    {
        int64_t mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    if (!check(l))
        --l;
    std::cout << l << '\n';
}
