void solve()
{
    int n;
    std::cin >> n;
    std::string a;
    std::string b;
    std::cin >> a;
    std::cin >> b;

    a = " " + a;
    b = " " + b;

    std::vector<int> ans;
    int flag{ a[1] };
    for (int i{ 2 }; i <= n; ++i)
    {
        if (a[i] != flag)
        {
            flag = a[i];
            ans.push_back(i - 1);
        }
    }
    for (int i{ n }; i >= 1; --i)
    {
        if (flag != b[i])
        {
            flag = b[i];
            ans.push_back(i);
        }
    }

    std::cout << ans.size();
    for (auto a : ans)
    {
        std::cout << ' ' << a;
    }
    std::cout << '\n';
}
