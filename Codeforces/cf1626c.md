void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> ts(n);
    std::vector<int> hs(n);
    for (auto& t : ts)
    {
        std::cin >> t;
    }
    for (auto& h : hs)
    {
        std::cin >> h;
    }

    std::vector<std::pair<int, int>> segs;
    for (int i{ 0 }; i < n; ++i)
    {
        segs.push_back({ ts[i] - hs[i] + 1, ts[i] });
    }
    std::sort(segs.begin(), segs.end());
    while (1)
    {
        bool ok{ true };
        for (int i{ 1 }; i < segs.size(); ++i)
        {
            auto& [l1, r1]{ segs[i - 1] };
            auto& [l2, r2]{ segs[i] };
            if (l2 > r1)
                continue;
            ok = false;
            r1 = std::max(r1, r2);
            segs.erase(segs.begin() + i);
            break;
        }
        if (ok)
            break;
    }

    int64_t ans{ 0 };
    for (auto [l, r] : segs)
    {
        int64_t len{ r - l + 1 };
        ans += len * (len + 1) / 2;
    }
    std::cout << ans << '\n';
}
