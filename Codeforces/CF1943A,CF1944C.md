void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> counts(n + 1);
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    int ans{ n + 1 };
    bool flag1{ false };
    for (int i{ 0 }; i <= n; ++i)
    {
        if (counts[i] == 0)
            ans = std::min(ans, i);
        if (counts[i] == 1)
            if (!flag1)
                flag1 = true;
            else
                ans = std::min(ans, i);
    }
    std::cout << ans << '\n';
}
