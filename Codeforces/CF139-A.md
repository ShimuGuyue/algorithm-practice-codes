void solve()
{
    int n;
    std::cin >> n;
    std::array<int, 8> as{ };
    for (int i{ 1 }; i <= 7; ++i)
    {
        std::cin >> as[i];
    }

    int sum{ std::accumulate(as.begin(), as.end(), 0) };

    n %= sum;
    if (n == 0)
        n = sum;

    for (int i{ 1 }; i <= 7; ++i)
    {
        n -= as[i];
        if (n <= 0)
        {
            std::cout << i << '\n';
            return;
        }
    }
}
