void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> v;
    for (int i{ 0 }; i < n; ++i)
    {
        v.push_back((i + (as[i] % n + n) % n) % n);
    }
    std::sort(v.begin(), v.end());
    v.erase(std::unique(v.begin(), v.end()), v.end());

    if (v.size() == n)
        std::cout << "YES\n";
    else
        std::cout << "NO\n";
}
