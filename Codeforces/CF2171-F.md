template <typename T>
using min_heap = std::priority_queue<T, std::vector<T>, std::greater<T>>;

void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    std::vector<int> indexs(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
        indexs[as[i]] = i;
    }

    std::vector<int> premins(n + 1);
    std::vector<int> sufmaxs(n + 1);
    std::partial_sum(as.begin() + 1, as.end(), premins.begin() + 1,
        [](int a, int b)
        {
            return std::min(a, b);
        });
    std::partial_sum(as.rbegin(), as.rend() - 1, sufmaxs.rbegin(),
        [](int a, int b)
        {
            return std::max(a, b);
        });

    for (int i = 2; i <= n; ++i)
    {
        if (premins[i - 1] > sufmaxs[i])
        {
            std::cout << "No\n";
            return;
        }
    }

    std::cout << "Yes\n";
    min_heap<int> h;
    h.push(as[1]);
    for (int i = 2; i <= n; ++i)
    {
        int flag{ INT_MAX };
        while (!h.empty() && h.top() < as[i])
        {
            int t{ h.top() };
            h.pop();
            std::cout << t << ' ' << as[i] << '\n';
            flag = std::min(flag, t);
        }
        if (flag != INT_MAX)
            h.push(flag);
        else
            h.push(as[i]);
    }
}
