void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 2);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> pre1s(n + 2);
    std::vector<int> suf0s(n + 2);
    int count1{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] == 1)
            ++count1;
        pre1s[i] = count1;
    }
    int count0{ 0 };
    for (int i{ n }; i >= 1; --i)
    {
        if (as[i] == 0)
            ++count0;
        suf0s[i] = count0;
    }

    int64_t ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] == 1) 
            ans += suf0s[i + 1];
    }

    int64_t sum{ ans };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] == 0)
            ans = std::max(ans, sum - pre1s[i - 1] + suf0s[i + 1]);
        else
            ans = std::max(ans, sum - suf0s[i + 1] + pre1s[i - 1]);
    }
    std::cout << ans << '\n';
}
