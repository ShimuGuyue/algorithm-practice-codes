void solve()
{
    constexpr int n{ 8 };
    std::vector<std::string> grid(n);
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    for (int i{ 0 }; i < n; ++i)
    {
        int count_r{ 0 };
        for (int j{ 0 }; j < n; ++j)
        {
            if (grid[i][j] == 'R')
                ++count_r;
        }
        if (count_r == n)
        {
            std::cout << "R\n";
            return;
        }
    }
    for (int j{ 0 }; j < n; ++j)
    {
        int count_b{ 0 };
        for (int i{ 0 }; i < n; ++i)
        {
            if (grid[i][j] == 'B')
                ++count_b;
        }
        if (count_b == n)
        {
            std::cout << "B\n";
            return;
        }
    }
}
