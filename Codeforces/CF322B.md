void solve()
{
    std::array<int, 3> as;
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    int ans{ 0 };
    int flag{ as[0] };

    for (int i = 0; i < 3; ++i)
    {
        if (flag - i < 0)
            break;
        int sum{ flag - i };
        for (int j = 0; j < 3; ++j)
        {
            sum += (as[j] - (flag - i)) / 3;
        }
        ans = std::max(ans, sum);
    }
    std::cout << ans << '\n';
}
