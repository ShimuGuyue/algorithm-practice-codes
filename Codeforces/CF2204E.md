void solve()
{
    std::string s;
    std::cin >> s;

    if (s.length() == 1)
    {
        std::cout << s << '\n';
        return;
    }

    std::vector<int> counts(10);
    for (char c : s)
    {
        ++counts[c - '0'];
    }

    for (int i{ 1 }; i <= s.length() * 9; ++i)
    {
        bool ok{ true };
        auto cur{ counts };
        for (char c : std::to_string(i))
        {
            --cur[c - '0'];
            if (cur[c - '0'] < 0)
                ok = false;
        }
        if (!ok)
            continue;

        int x{ i };
        while (x > 9)
        {
            int y{ 0 };
            while (x)
            {
                y += x % 10;
                x /= 10;
            }
            x = y;
            for (char c : std::to_string(x))
            {
                --cur[c - '0'];
                if (cur[c - '0'] < 0)
                    ok = false;
            }
        }
        if (!ok)
            continue;

        int sum{ 0 };
        for (int k{ 0 }; k < 10; ++k)
        {
            sum += k * cur[k];
        }
        if (sum != i)
            continue;

        for (int k{ 9 }; k >= 0; --k)
        {
            while (cur[k]--)
            {
                std::cout << k;
            }
        }
        while (sum > 9)
        {
            std::cout << sum;
            int next{ 0 };
            while (sum)
            {
                next += sum % 10;
                sum /= 10;
            }
            sum = next;
        }
        std::cout << sum << '\n';
        return;
    }
}
