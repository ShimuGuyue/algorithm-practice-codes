void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> counts(n + 1);
    for (int i = 0; i < m; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    auto check = [&counts](int flag)
    {
        int64_t remain{ 0 };
        for (int i = 1; i < counts.size(); ++i)
        {
            if (counts[i] <= flag)
                remain += (flag - counts[i]) / 2;
            else
                remain -= (counts[i] - flag);
        }
        return remain >= 0;
    };

    int l{ 1 }, r{ 2 * m };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
