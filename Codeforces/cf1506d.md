void solve()
{
    int n;
    std::cin >> n;
    std::map<int, int> m;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++m[a];
    }

    int max{ 0 };
    for (auto [k, v] : m)
    {
        max = std::max(max, v);
    }

    if (n & 1)
    {
        if (max <= n / 2)
            std::cout << 1 << '\n';
        else
            std::cout << n - (n - max) * 2 << '\n';
    }
    else
    {
        if (max <= n / 2)
            std::cout << 0 << '\n';
        else
            std::cout << n - (n - max) * 2 << '\n';
    }
}
