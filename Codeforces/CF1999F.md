```c++
static constexpr int64_t mod{ 1000000007 };
std::vector<int64_t> factorials;
std::vector<int64_t> inv_factorials;

void init()
{
    constexpr int n{ 200000 };
    factorials.assign(n + 1, 1);
    inv_factorials.assign(n + 1, 1);

    for (int i = 2; i <= n; ++i)
    {
        factorials[i] = factorials[i - 1] * i % mod;
    }

    inv_factorials[n] = quickpow(factorials[n], mod - 2);
    for (int i = n - 1; i > 0; --i)
    {
        inv_factorials[i] = inv_factorials[i + 1] * (i + 1) % mod;
    }
}

int64_t quickpow(int64_t base, int64_t power)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}
```

```c++
void solve()
{
    int n, k;
    std::cin >> n >> k;

    int count0{ 0 }, count1{ 0 };
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        if (a == 0)
            ++count0;
        else
            ++count1;
    }

    int64_t ans{ 0 };
    for (int i = k / 2 + 1; i <= std::min(k, count1); ++i)
    {
        int j{ k - i };
        if (count0 < j)
            continue;
        ans += combinate(count1, i) * combinate(count0, j) % mod;
        ans %= mod;
    }
    std::cout << ans << '\n';
}

int64_t combinate(int n, int k)
{
    if (k < 0 || k > n)
        return 0;
    if (k == 0 || k == n)
        return 1;
    return factorials[n] * inv_factorials[k] % mod * inv_factorials[n - k] % mod;
}
```
