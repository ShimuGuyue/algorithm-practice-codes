void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::set<char> t(s.begin(), s.end());

    int ans{ n };
    std::map<char, int> counts;
    int l{ 0 }, r{ 0 };
    while (r < n)
    {
        ++counts[s[r]];
        if (counts.size() == t.size())
        {
            while (counts[s[l]] > 1)
            {
                --counts[s[l]];
                ++l;
            }
            ans = std::min(ans, r - l + 1);
        }
        ++r;
    }
    std::cout << ans << '\n';
}
