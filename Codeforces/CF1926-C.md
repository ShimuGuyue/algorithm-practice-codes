

```cpp
std::vector<int> ans;
void init()
{
    constexpr int n{ 200000 };
    ans.assign(n + 1, 0);
    for (int i{ 1 }; i <= n; ++i)
    {
        ans[i] = ans[i - 1];
        int x{ i };
        while (x)
        {
            ans[i] += x % 10;
            x /= 10;
        }
    }
}
```

```cpp
void solve()
{
    int n;
    std::cin >> n;
    std::cout << ans[n] << '\n';
}
```
