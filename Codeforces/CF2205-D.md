void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 2);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> maxindex_l(n + 1);
    std::vector<int> maxindex_r(n + 1);
    [n, &as, &maxindex_l]
    {
        std::stack<int> k;
        for (int i{ 1 }; i <= n; ++i)
        {
            while (!k.empty() && as[k.top()] <= as[i])
            {
                k.pop();
            }
            if (k.empty())
                maxindex_l[i] = i;
            else
                maxindex_l[i] = k.top();
            k.push(i);
        }
    }();
    [n, &as, &maxindex_r]
    {
        std::stack<int> k;
        for (int i{ n }; i >= 1; --i)
        {
            while (!k.empty() && as[k.top()] <= as[i])
            {
                k.pop();
            }
            if (k.empty())
                maxindex_r[i] = i;
            else
                maxindex_r[i] = k.top();
            k.push(i);
        }
    }();

    int ans{ 0 };
    std::vector<int> dpl(n + 1);
    std::vector<int> dpr(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        dpl[i] = dpl[maxindex_l[i]] + 1;
    }
    for (int i{ n }; i >= 1; --i)
    {
        dpr[i] = dpr[maxindex_r[i]] + 1;
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        ans = std::max(ans, dpl[i] + dpr[i] - 1);
    }
    ans = n - ans;
    std::cout << ans << '\n';
}
