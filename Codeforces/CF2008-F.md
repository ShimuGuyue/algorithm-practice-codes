static constexpr int64_t mod{ 1000000007 };

void solve()
{
    int64_t n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t sum{ std::accumulate(as.begin(), as.end(), INT64_C(0)) % mod };
    int64_t ans{ 0 };
    for (auto a : as)
    {
        ans += (sum - a + mod) % mod * a % mod;
        ans %= mod;
    }
    ans *= inv(n * (n - 1) % mod) % mod;
    ans %= mod;
    std::cout << ans << '\n';
}

int64_t quickpow(int64_t base, int64_t power)
{
    base %= mod;

    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}

int64_t inv(int64_t n)
{
    return quickpow(n, mod - 2);
}
