void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    std::vector<int64_t> cs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i] >> cs[i];
    }

    int64_t inf{ INT64_C(1) << 60 };
    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(3, inf));
    dp[0] = { 0, 0, 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 0 }; j < 3; ++j)
        {
            for (int k{ 0 }; k < 3; ++k)
            {
                if (dp[i - 1][k] == inf)
                    continue;
                if (as[i] + j == as[i - 1] + k)
                    continue;
                dp[i][j] = std::min(dp[i][j], dp[i - 1][k] + cs[i] * j);
            }
        }
    }
    int64_t ans{ std::min({ dp[n][0], dp[n][1], dp[n][2] }) };
    std::cout << ans << '\n';
}
