void solve()
{
    int64_t n, c;
    std::cin >> n >> c;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    auto check = [&as, c](int64_t k)
    {
        int64_t sum{ 0 };
        for (auto a : as)
        {
            sum += (a + k * 2) * (a + k * 2);
            if (sum >= c)
                return true;
        }
        return false;
    };
    int64_t l{ 1 }, r{ 1000000000 };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
