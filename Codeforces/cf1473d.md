template<typename T, typename K = T>
class SparseTableMax
{
private:
    std::vector<std::vector<K>> table_;

    const std::vector<T>& v_;

public:
    SparseTableMax(const std::vector<T>& v) : v_(v)
    {
        const int n{ static_cast<int>(v.size()) };
        const int m{ static_cast<int>(std::log2(n)) + 1 };
        table_.assign(n, std::vector<K>(m));

        for (int i{ 0 }; i < n; ++i)
        {
            table_[i][0] = contribute(i);
        }
        for (int j{ 1 }; j < m; ++j)
        {
            const int len{ 1 << (j - 1) };
            for (int i{ 0 }; i + len * 2 - 1 < n; ++i)
            {
                table_[i][j] = merge(i, i + len, j - 1);
            }
        }
    }

public:
    K query(const int l, const int r) const
    {
        const int power{ static_cast<int>(log2(r - l + 1)) };
        const int len{ 1 << power };
        return merge(l, r - len + 1, power);
    }

private:
    K contribute(const int index) const

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::string s;
    std::cin >> s;

    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        as[i] = s[i - 1] == '-' ? -1 : +1;
    }

    std::vector<int> pres(n + 1);
    std::partial_sum(as.begin(), as.end(), pres.begin());

    SparseTableMax<int> st_max(pres);
    SparseTableMin<int> st_min(pres);

    while (m--)
    {
        int l, r;
        std::cin >> l >> r;
        int min{ 0 }, max{ 0 };
        if (l != 1)
        {
            min = std::min(min, st_min.query(1, l - 1));
            max = std::max(max, st_max.query(1, l - 1));
        }
        if (r != n)
        {
            int range{ pres[r] - pres[l - 1] };
            min = std::min(min, st_min.query(r + 1, n) - range);
            max = std::max(max, st_max.query(r + 1, n) - range);
        }
        std::cout << max - min + 1 << '\n';
    }
}
