void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int l{ 1 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        int sum{ 0 };

        std::cout << "? ";
        std::cout << mid - l + 1 << ' ';
        for (int i{ l }; i <= mid; ++i)
        {
            std::cout << i << ' ';
            sum += as[i];
        }
        std::cout << std::endl;

        int x;
        std::cin >> x;
        if (sum == x)
            l = mid + 1;
        else
            r = mid;
    }
    std::cout << "! " << r << std::endl;
}
