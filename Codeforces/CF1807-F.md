void solve()
{
    int n, m, i1, j1, i2, j2;
    char d1, d2;
    std::cin >> n >> m >> i1 >> j1 >> i2 >> j2 >> d1 >> d2;

    int ans{ 0 };
    int x{ i1 }, y{ j1 };
    for (int i{ 0 }; i < n * m * 4; ++i)
    {
        if (x == i2 && y == j2)
        {
            std::cout << ans << '\n';
            return;
        }
        if ((x == 1 && d1 == 'U') ||
            (x == n && d1 == 'D') ||
            (y == 1 && d2 == 'L') ||
            (y == m && d2 == 'R'))
            ++ans;
        if (d1 == 'U')
        {
            if (x == 1)
            {
                ++x;
                d1 = 'D';
            }
            else
            {
                --x;
            }
        }
        else
        {
            if (x == n)
            {
                --x;
                d1 = 'U';
            }
            else
            {
                ++x;
            }
        }
        if (d2 == 'L')
        {
            if (y == 1)
            {
                ++y;
                d2 = 'R';
            }
            else
            {
                --y;
            }
        }
        else
        {
            if (y == m)
            {
                --y;
                d2 = 'L';
            }
            else
            {
                ++y;
            }
        }
    }
    std::cout << -1 << '\n';
}
