void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> grid(n, std::vector<int>(m));
    std::map<int, std::pair<int, int>> flags;
    for (int i{ 0 }; i < n; ++i)
    {
        for (int j{ 0 }; j < m; ++j)
        {
            std::cin >> grid[i][j];
        }
    }
    for (int i{ 0 }; i < n; ++i)
    {
        for (int j{ 0 }; j < m; ++j)
        {
            int a;
            std::cin >> a;
            flags[a] = { i, j };
        }
    }

    for (int i{ 0 }; i < n; ++i)
    {
        int flag{ -1 };
        for (int j{ 0 }; j < m; ++j)
        {
            int a{ grid[i][j] };
            if (flags[a].first != flag && flag != -1)
            {
                std::cout << "No\n";
                return;
            }
            flag = flags[a].first;
        }
    }
    for (int j{ 0 }; j < m; ++j)
    {
        int flag{ -1 };
        for (int i{ 0 }; i < n; ++i)
        {
            int a{ grid[i][j] };
            if (flags[a].second != flag && flag != -1)
            {
                std::cout << "No\n";
                return;
            }
            flag = flags[a].second;
        }
    }
    std::cout << "Yes\n";
}
