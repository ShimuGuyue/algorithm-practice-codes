void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> flags(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        flags[i] = (as[i] * 2 > as[i - 1]) ? 1 : 0;
    }

    std::vector<int> presums(n + 1);
    std::partial_sum(flags.begin(), flags.end(), presums.begin());

    int ans{ 0 };
    for (int i = 1; i + k <= n; ++i)
    {
        int sum{ presums[i + k] - presums[i - 1] };
        if (sum == k + 1)
            ++ans;
        else if (sum == k && flags[i] == 0)
            ++ans;
    }
    std::cout << ans << '\n';
}
