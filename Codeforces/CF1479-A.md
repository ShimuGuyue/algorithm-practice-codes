void solve()
{
    int n;
    std::cin >> n;

    std::vector<int> as(n + 2);
    as[0] = as[n + 1] = 200000;

    int l{ 1 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        if (as[mid] == 0)
        {
            std::cout << "? " << mid << std::endl;
            std::cin >> as[mid];
        }
        if (as[mid + 1] == 0)
        {
            std::cout << "? " << mid + 1 << std::endl;
            std::cin >> as[mid + 1];
        }
        if (as[mid] > as[mid + 1])
            l = mid + 1;
        else
            r = mid;
    }
    std::cout << "! " << l << std::endl;
}
