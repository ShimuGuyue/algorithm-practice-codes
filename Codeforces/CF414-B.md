void solve()
{
    constexpr int64_t mod{ 1000000007 };

    int n, k;
    std::cin >> n >> k;

    std::vector<std::vector<int>> factors(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        for (int j = i; j <= n; j += i)
        {
            factors[j].push_back(i);
        }
    }

    std::vector<std::vector<int64_t>> dp(k + 1, std::vector<int64_t>(n + 1));
    for (int i = 1; i <= n; ++i)
    {
        dp[1][i] = 1;
    }
    for (int i = 2; i <= k; ++i)
    {
        for (int j = 1; j <= n; ++j)
        {
            for (int factor : factors[j])
            {
                dp[i][j] += dp[i - 1][factor];
                dp[i][j] %= mod;
            }
        }
    }
    int64_t ans{ std::accumulate(dp[k].begin() + 1, dp[k].end(), INT64_C(0)) % mod };
    std::cout << ans << '\n';
}
