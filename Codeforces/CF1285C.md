void solve()
{
    int64_t x;
    std::cin >> x;

    int64_t a{ 1 }, b{ x };
    for (int64_t i = 1; i * i <= x; ++i)
    {
        if (x % i)
            continue;
        int64_t j{ x / i };
        if (std::lcm(i, j) != x)
            continue;
        a = i;
        b = j;
    }
    std::cout << a << ' ' << b << '\n';
}
