void solve()
{
    int n;
    std::cin >> n;

    std::string sn{ std::to_string(n) };
    int len_n{ int(sn.length()) };

    std::string s;
    while (s.length() < 10)
    {
        s += sn;
    }

    std::vector<std::pair<int, int>> ans;

    for (int a{ 1 }; a <= 10000; ++a)
    {
        for (int len_x{ 1 }; len_x <= 6; ++len_x)
        {
            int b{ a * len_n - len_x };
            if (b >= 1 && b <= 10000 && b <= a * n)
            {
                int ans_int{ n * a - b };
                int ans_str{ std::stoi(s.substr(0, len_x)) };
                if (ans_int == ans_str)
                    ans.push_back({ a, b });
            }
        }
    }

    std::cout << ans.size() << '\n';
    for (auto [a, b] : ans)
    {
        std::cout << a << ' ' << b << '\n';
    }
}
