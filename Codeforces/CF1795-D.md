void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }

    int64_t mod{ 998244353 };
    auto quickpow = [mod](int64_t base, int power)
    {
        int64_t ans{ 1 };
        while (power)
        {
            if (power & 1)
                ans = ans * base % mod;
            base = base * base % mod;
            power /= 2;
        }
        return ans;
    };
    auto inv = [mod, quickpow](int64_t x)
    {
        return quickpow(x, mod - 2);
    };

    int64_t ans{ 1 };
    for (int i{ 1 }; i <= n / 6; ++i)
    {
        ans = ans * (n / 3 + 1 - i) % mod;
        ans = ans * inv(i) % mod;
    }
    for (int i{ 0 }; i < n; i += 3)
    {
        std::array<int, 3> flags{ { as[i], as[i + 1], as[i + 2] } };
        std::sort(flags.begin(), flags.end());
        auto [a, b, c]{ flags };
        if (a == b && b == c)
            ans = ans * 3 % mod;
        if (a == b && a != c)
            ans = ans * 2 % mod;
    }
    std::cout << ans << '\n';
}
