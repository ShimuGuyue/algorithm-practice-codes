void solve()
{
    int n, s, x;
    std::cin >> n >> s >> x;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int sum{ std::accumulate(as.begin(), as.end(), 0) };
    if (sum > s)
    {
        std::cout << "no\n";
        return;
    }
    if ((s - sum) % x == 0)
        std::cout << "yes\n";
    else
        std::cout << "no\n";
}
