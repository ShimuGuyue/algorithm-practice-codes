void solve()
{
    int x;
    std::cin >> x;

    if (x & 1)
    {
        std::cout << -1 << '\n';
        return;
    }

    int a{ x / 2 }; // and

    int ans1{ 0 }, ans2{ 0 };
    for (int bit = 0; bit < 30; ++bit)
    {
        int xx{ x >> bit & 1 };
        int aa{ a >> bit & 1 };

        if (xx == 0 && aa == 0)
        {
            ;
        }
        else if (xx == 0 && aa == 1)
        {
            ans1 |= 1 << bit;
            ans2 |= 1 << bit;
        }
        else if (xx == 1 && aa == 0)
        {
            ans1 |= 1 << bit;
        }
        else if (xx == 1 && aa == 1)
        {
            std::cout << -1 << '\n';
            return;
        }
    }
    if (ans1 + ans2 == x * 2)
        std::cout << ans1 << ' ' << ans2 << '\n';
    else
        std::cout << -1 << '\n';
}
