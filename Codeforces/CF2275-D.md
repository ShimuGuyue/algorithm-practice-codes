void solve()
{
    int64_t n, k;
    cin >> n >> k;
    vector<array<int64_t, 3>> as(n);
    for (auto& [a, b, c] : as)
    {
        cin >> a >> b >> c;
    }

    auto check = [&as, k](int64_t flag) -> bool
    {
        int64_t sum = 0;
        for (auto [a, b, c] : as)
        {
            int64_t cur = int64_t{1} << 60;
            int64_t temp = a + b + c;
            if (temp >= flag)
                continue;
            if (a == b && a == c)
                return false;
            if (a > b || a > c || b > c)
            {
                cur = std::min(cur, flag - temp);
            }
            // a <= b <= c
            else
            {
                if (a < b)
                {
                    // c lower to b - 1
                    int64_t dif = c - b + 1;
                    cur = std::min(cur, flag - temp + dif * 2);
                }
                if (a < c)
                {
                    // b lower to a - 1
                    int64_t dif = b - a + 1;
                    cur = std::min(cur, flag - temp + dif * 2);
                }
            }
            sum += cur;
            if (sum > k)
                return false;
        }
        return true;
    };

    int64_t l = -(int64_t{1} << 60);
    int64_t r = +(int64_t{1} << 60);
    while (l < r)
    {
        int64_t mid = (l + r + 1) / 2;
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    cout << l << '\n';
}
