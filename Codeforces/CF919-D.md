void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::string s;
    std::cin >> s;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        as[i] = s[i - 1] - 'a';
    }
    std::vector<int> indegrees(n + 1);
    std::vector<std::set<int>> graph(n + 1);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        if (!graph[u].count(v))
        {
            ++indegrees[v];
            graph[u].insert(v);
        }
    }

    int ans{ 1 };
    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(26));
    int count{ 0 };
    std::queue<int> q;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (indegrees[i] == 0)
        {
            q.push(i);
            dp[i][as[i]] = 1;
            ++count;
        }
    }
    while (!q.empty())
    {
        int u{ q.front() };
        q.pop();
        for (int v : graph[u])
        {
            --indegrees[v];
            for (int i{ 0 }; i < 26; ++i)
            {
                if (i == as[v])
                    dp[v][i] = std::max(dp[v][i], dp[u][i] + 1);
                else
                    dp[v][i] = std::max(dp[v][i], dp[u][i]);
                ans = std::max(ans, dp[v][i]);
            }
            if (indegrees[v] == 0)
            {
                q.push(v);
                ++count;
            }
        }
    }
    if (count != n)
        ans = -1;
    std::cout << ans << '\n';
}
