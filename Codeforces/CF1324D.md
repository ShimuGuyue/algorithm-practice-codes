void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        as[i] += a;
    }
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        as[i] -= a;
    }

    std::sort(as.begin(), as.end());

    int64_t ans{ 0 };
    for (int i = 0; i < n; ++i)
    {
        auto it{ std::upper_bound(as.begin() + i + 1, as.end(), -as[i]) };
        ans += std::distance(it, as.end());
    }
    std::cout << ans << '\n';
}
