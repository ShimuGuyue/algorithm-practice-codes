void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::vector<bool> flags(n);
    int i{ 0 };
    flags[0] = 1;
    while (n--)
    {
        if (s[i] == 'L')
            --i;
        else
            ++i;
        flags[i] = 1;
    }
    int ans{ std::accumulate(flags.begin(), flags.end(), 0) };
    std::cout << ans << '\n';
}
