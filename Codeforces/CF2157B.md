void solve()
{
    int n, x, y;
    std::cin >> n >> x >> y;
    std::string s;
    std::cin >> s;

    x = abs(x);
    y = abs(y);

    int count{ std::count(s.begin(), s.end(), '4') };

    auto [min, max] = std::minmax(x, y);
    if (max > n)
    {
        std::cout << "no\n";
        return;
    }
    if (max <= n - count)
    {
        std::cout << "yes\n";
        return;
    }
    if ((n - max) + (n - min) >= count)
        std::cout << "yes\n";
    else
        std::cout << "no\n";
}
