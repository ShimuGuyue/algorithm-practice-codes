void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::string ans(n, '0');
    int sum{ 0 };
    for (int i{ n - 1 }; i >= 0; --i)
    {
        if (sum < q && as[i] > sum)
        {
            ++sum;
            ans[i] = '1';
        }
        if (as[i] <= sum)
        {
            ans[i] = '1';
        }
    }
    std::cout << ans << '\n';
}
