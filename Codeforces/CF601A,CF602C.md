void solve()
{
    constexpr int inf{ 200000 };

    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> graph1(n + 1, std::vector<int>(n + 1, inf));
    std::vector<std::vector<int>> graph2(n + 1, std::vector<int>(n + 1, inf));
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        graph1[u][v] = graph1[v][u] = 1;
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= n; ++j)
        {
            if (graph1[i][j] == inf)
                graph2[i][j] = 1;
        }
    }

    for (int k{ 1 }; k <= n; ++k)
    {
        for (int i{ 1 }; i <= n; ++i)
        {
            for (int j{ 1 }; j <= n; ++j)
            {
                graph1[i][j] = std::min(graph1[i][j], graph1[i][k] + graph1[k][j]);
                graph2[i][j] = std::min(graph2[i][j], graph2[i][k] + graph2[k][j]);
            }
        }
    }

    int ans{ std::max(graph1[1][n], graph2[1][n]) };
    if (ans == inf)
        ans = -1;
    std::cout << ans << '\n';
}
