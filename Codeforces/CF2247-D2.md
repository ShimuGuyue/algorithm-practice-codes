struct SegmentTree
{
    struct Data
    {
        int l, r;
        int max, min;
        int k;
    };

    std::vector<Data> tree;

    void build(int n, int l, int r, std::vector<int>& v)
    {
        if (tree.empty())
            tree.resize(v.size() * 4);
        tree[n].l = l;
        tree[n].r = r;
        if (l == r)
        {
            tree[n].max = v[l];
            tree[n].min = v[l];
            tree[n].k = 0;
            return;
        }
        int mid{ (l + r) / 2 };
        build(n * 2 + 1, l, mid, v);
        build(n * 2 + 2, mid + 1, r, v);
        push_up(n);
    }

    void update(int index, int x, int n, int l, int r)
    {
        if (l == r)
        {
            tree[n].max = x;
            tree[n].min = x;
            return;
        }
        int mid{ (l + r) / 2 };
        if (index <= mid)
            update(index, x, n * 2 + 1, l, mid);
        else
            update(index, x, n * 2 + 2, mid + 1, r);
        push_up(n);
    }

    void push_up(int n)
    {
        int l{ n * 2 + 1 };
        int r{ n * 2 + 2 };
        tree[n].max = std::max(tree[l].max, tree[r].max);
        tree[n].min = std::min(tree[l].min, tree[r].min);
        tree[n].k = std::max(tree[l].k, tree[r].k);
        if (tree[l].max > tree[r].min)
            tree[n].k = std::max(tree[n].k, (tree[n].r - tree[n].l + 1) / 2);
    }

    auto k() -> int
    {
        return tree[0].k;
    }
};

void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    while (std::popcount(as.size()) != 1)
    {
        as.push_back(1 << 30);
        ++n;
    }

    SegmentTree tree;
    tree.build(0, 0, n - 1, as);
    std::cout << tree.k() << '\n';

    while (q--)
    {
        int p, x;
        std::cin >> p >> x;
        tree.update(p, x, 0, 0, n - 1);
        std::cout << tree.k() << '\n';
    }
}
