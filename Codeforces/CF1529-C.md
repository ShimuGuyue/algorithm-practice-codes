void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::pair<int64_t, int64_t>> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i].first >> as[i].second;
    }

    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(2));
    auto dfs = [&as, &tree, &dp](this auto& dfs, int u, int last) -> void
    {
        auto [l, r]{ as[u] };
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            dp[u][0] += std::max(abs(l - as[v].first) + dp[v][0], abs(l - as[v].second) + dp[v][1]);
            dp[u][1] += std::max(abs(r - as[v].first) + dp[v][0], abs(r - as[v].second) + dp[v][1]);
        }
    };
    dfs(1, 0);
    int64_t ans{ std::max(dp[1][0], dp[1][1]) };
    std::cout << ans << '\n';
}
