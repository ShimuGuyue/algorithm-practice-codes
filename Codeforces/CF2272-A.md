void solve()
{
    int n, k;
    cin >> n >> k;
    vector<int> as(n + 1);
    vector<int> indexs(n);;
    for (int i = 1; i <= n; ++i)
    {
        cin >> as[i];
        indexs[as[i]] = i;
    }

    k = n - k + 1;

    bool ok = true;
    int index1;
    vector<int> visited(n);
    for (int i = 1; i <= k; ++i)
    {
        visited[as[i]] = true;
    }
    for (int i = 0; i < k; ++i)
    {
        if (!visited[i])
        {
            ok = false;
            index1 = i;
            break;
        }
    }
    if (ok)
    {
        cout << -1 << '\n';
        return;
    }

    int index2;
    for (int i = 1; i <= k; ++i)
    {
        if (as[i] >= k)
        {
            index2 = i;
            break;
        }
    }

    cout << indexs[index1] << ' ' << index2 << '\n';
}
