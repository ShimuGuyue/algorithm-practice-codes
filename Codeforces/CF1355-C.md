void solve()
{
    int a, b, c, d;
    std::cin >> a >> b >> c >> d;

    std::vector<int> difs(1000000 + 2);
    for (int x{ a }; x <= b; ++x)
    {
        ++difs[x + b];
        --difs[x + c + 1];
    }

    std::vector<int64_t> counts(1000000 + 2);
    std::partial_sum(difs.begin(), difs.end(), counts.begin());

    std::vector<int64_t> sufcounts(1000000 + 2);
    std::partial_sum(counts.rbegin(), counts.rend(), sufcounts.rbegin());

    int64_t ans{ 0 };
    for (int i{ c }; i <= d; ++i)
    {
        ans += sufcounts[i + 1];
    }
    std::cout << ans << '\n';
}
