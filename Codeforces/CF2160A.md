void solve()
{
    int n;
    std::cin >> n;
    std::array<bool, 101> counts{};
    while (n--)
    {
        int a;
        std::cin >> a;
        counts[a] = true;
    }

    int ans{ 0 };
    for (int i = 0; i < 100; ++i)
    {
        if (counts[i])
        {
            ++ans;
        }
        else
        {
            break;
        }
    }
    std::cout << ans << '\n';
}
