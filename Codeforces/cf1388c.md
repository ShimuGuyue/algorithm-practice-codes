void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    std::vector<int> hs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> hs[i];
    }
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    bool ok{ true };
    std::vector<int> counts(n + 1);
    std::vector<std::pair<int, int>> flags(n + 1);
    auto dfs = [&as, &hs, &tree, &counts, &ok, &flags](this auto&& dfs, int u, int last) -> void
    {
        if (!ok)
            return;

        counts[u] = as[u];
        int sumx{ 0 }, sumy{ 0 };
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            counts[u] += counts[v];
            auto [x, y]{ flags[v] };
            sumx += x;
            sumy += y;
        }

        if ((counts[u] + hs[u]) & 1)
            ok = false;

        int x{ (counts[u] + hs[u]) / 2 };
        int y{ counts[u] - x };
        if (x < 0 || y < 0)
            ok = false;

        if (x < sumx)
            ok = false;

        flags[u] = { x, y };
    };
    dfs(1, 0);
    std::cout << (ok ? "Yes" : "No") << '\n';
}
