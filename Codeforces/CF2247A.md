void solve()
{
    int n;
    std::cin >> n;
    int count_posi{ 0 }, count_nega{ 0 };
    while (n--)
    {
        int a;
        std::cin >> a;
        if (a == 1)
            ++count_posi;
        else
            ++count_nega;
    }

    int dif{ (count_posi - count_nega) % 4 };
    if (dif == 0)
        std::cout << "Yes\n";
    else
        std::cout << "No\n";
}
