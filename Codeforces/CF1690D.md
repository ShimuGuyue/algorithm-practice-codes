void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    s = " " + s;

    std::vector<int> presums(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        presums[i] = presums[i - 1] + (s[i] == 'W');
    }

    int ans{ n };
    for (int i{ k }; i <= n; ++i)
    {
        ans = std::min(ans, presums[i] - presums[i - k]);
    }
    std::cout << ans << '\n';
}
