void solve()
{
    int n, x;
    std::cin >> n >> x;
    std::vector<int> degrees(n + 1);
    for (int i = 1; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        ++degrees[u];
        ++degrees[v];
    }

    if (degrees[x] <= 1)
    {
        std::cout << "Ayush\n";
        return;
    }

    if (n & 1)
        std::cout << "Ashish\n";
    else
        std::cout << "Ayush\n";
}
