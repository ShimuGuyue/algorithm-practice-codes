void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> ans(n);

    std::vector<bool> visited(n + 1);

    int flag;
    int l{ 1 }, r{ n - 2 };

    flag = 2;
    while (l + 1 < n)
    {
        if (flag > n)
            break;
        if (flag + 2 <= n)
        {
            ans[l] = flag;
            visited[flag] = true;
            flag += 2;
            ans[l + 1] = flag;
            visited[flag] = true;
            flag += 2;
            l += 3;
        }
        else
        {
            ans[l - 1] = flag;
            visited[flag] = true;
            break;
        }
    }
    while (ans[l] == 0)
    {
        --l;
    }
    flag = 3;
    while (r - 1 > l)
    {
        if (flag > n)
            break;
        if (flag + 6 <= n)
        {
            ans[r] = flag;
            visited[flag] = true;
            flag += 6;
            ans[r - 1] = flag;
            visited[flag] = true;
            flag += 6;
            r -= 3;
        }
        else
        {
            ans[r + 1] = flag;
            visited[flag] = true;
            break;
        }
    }

    flag = 1;
    for (auto &a : ans)
    {
        if (a)
            continue;
        while (visited[flag])
        {
            ++flag;
        }
        a = flag;
        visited[flag] = true;
        ++flag;
    }
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
