void solve()
{
    int a, b, k;
    std::cin >> a >> b >> k;
    std::vector<int> as(k);
    std::vector<int> bs(k);
    for (int i = 0; i < k; ++i)
    {
        std::cin >> as[i];
    }
    for (int i = 0; i < k; ++i)
    {
        std::cin >> bs[i];
    }

    std::map<int, int> counts_a;
    std::map<int, int> counts_b;
    for (auto aa : as)
    {
        ++counts_a[aa];
    }
    for (auto bb : bs)
    {
        ++counts_b[bb];
    }

    int64_t ans{ 0 };
    for (int i = 0; i < k; ++i)
    {
        ans += k - counts_a[as[i]] - counts_b[bs[i]] + 1;
    }
    ans /= 2;
    std::cout << ans << '\n';
}
