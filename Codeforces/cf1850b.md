void solve()
{
    int n;
    std::cin >> n;
    int ans;
    int max{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        int a, b;
        std::cin >> a >> b;
        if (a > 10)
            continue;
        if (b > max)
        {
            max = b;
            ans = i;
        }
    }
    std::cout << ans << '\n';
}
