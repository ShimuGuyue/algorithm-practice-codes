void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    for (int i{ 0 }; i < n; ++i)
    {
        int64_t ans{ 0 };
        std::vector<int> flags(n);
        for (int j{ (i - 1 + n) % n }; j != i; j = (j - 1 + n) % n)
        {
            flags[j] = std::max(as[j], flags[(j + 1) % n]);
        }
        for (int j{ (i + 1) % n }; j != i; j = (j + 1) % n)
        {
            if (flags[(j - 1 + n) % n] > as[(j - 1 + n) % n])
                flags[j] = flags[(j - 1 + n) % n];
            else
                flags[j] = std::min(flags[j], as[(j - 1 + n) % n]);
            ans += flags[j];
        }
        std::cout << ans << ' ';
    }
    std::cout << '\n';
}
