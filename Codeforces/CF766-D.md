class DisjointSet
{
private:
    struct Data
    {
        int father;
        int rank;
        int size;
    };
    std::vector<Data> nodes_;

    int count_;

public:
    DisjointSet(const int n)
    {
        build(n);
    }

public:
    int find(const int x)
    {
        return nodes_[x].father == x ? x : nodes_[x].father = find(nodes_[x].father);
    }

    bool merge(const int x, const int y)
    {
        const int set_x{ find(x) };
        const int set_y{ find(y) };
        if (set_x == set_y)
            return false;
        if (nodes_[set_x].rank < nodes_[set_y].rank)
        {
            nodes_[set_x].father = set_y;
            nodes_[set_y].size += nodes_[set_x].size;
        }
        else
        {
            nodes_[set_y].father = set_x;
            nodes_[set_x].size += nodes_[set_y].size;
            if (nodes_[set_x].rank == nodes_[set_y].rank)
                ++nodes_[set_x].rank;
        }
        --count_;
        return true;
    }

    int count() const
    {
        return count_;
    }

    int get_size(const int x)
    {
        return nodes_[find(x)].size;
    }

private:
    void build(const int n)
    {
        count_ = n;

        nodes_.assign(n, { });
        for (int i{ 0 }; i < n; ++i)
        {
            nodes_[i].father = i;
            nodes_[i].rank = 1;
            nodes_[i].size = 1;
        }
    }
};

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    std::vector<std::array<int, 2>> switchs(n + 1, {0, 0});
    for (int i{ 1 }; i <= m; ++i)
    {
        int x;
        std::cin >> x;
        while (x--)
        {
            int a;
            std::cin >> a;
            if (switchs[a][0] == 0)
                switchs[a][0] = i;
            else
                switchs[a][1] = i;
        }
    }

    DisjointSet ds(m * 2 + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] == 1)
        {
            ds.merge(switchs[i][0], switchs[i][1]);
            ds.merge(switchs[i][0] + m, switchs[i][1] + m);
        }
        else
        {
            ds.merge(switchs[i][0], switchs[i][1] + m);
            ds.merge(switchs[i][1], switchs[i][0] + m);
        }
    }

    for (int i{ 1 }; i <= m; ++i)
    {
        if (ds.find(i) == ds.find(i + m))
        {
            std::cout << "NO\n";
            return;
        }
    }
    std::cout << "YES\n";
}
