void solve()
{
    int64_t n;
    std::cin >> n;

    int64_t ans{ 1 };
    for (int i{ 2 }; i * i <= n; ++i)
    {
        if (n % i)
            continue;
        ans *= i;
        while (n % i == 0)
        {
            n /= i;
        }
    }
    if (n)
        ans *= n;
    std::cout << ans << '\n';
}
