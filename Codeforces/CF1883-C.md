void solve()
{
    int n, k;
    std::cin >> n >> k;

    std::vector<int> as(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }

    int ans{ k };
    if (k == 4)
    {
        int count4 = std::count_if(as.begin(), as.end(), [](int a){ return a % 4 == 0; });
        if (count4)
        {
            std::cout << 0 << '\n';
            return;
        }
        int count2 = std::count_if(as.begin(), as.end(), [](int a){ return a % 4 == 2; });
        if (count2 >= 2)
        {
            std::cout << 0 << '\n';
            return;
        }
        if (count2 == 1)
        {
            std::cout << 1 << '\n';
            return;
        }
        int count3 = std::count_if(as.begin(), as.end(), [](int a){ return a % 4 == 3; });
        if (count3)
        {
            std::cout << 1 << '\n';
            return;
        }
        ans = 2;
    }
    else
    {
        for (auto a : as)
        {
            if (a % k == 0)
                ans = 0;
            else
                ans = std::min(ans, k - a % k);
        }
    }
    std::cout << ans << '\n';
}
