void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int ans{ 1 };
    std::vector<int> dp(n + 1, 1);
    for (int i = 2; i <= n; ++i)
    {
        for (int j = 1; j * j <= i; ++j)
        {
            if (i % j != 0)
                continue;
            if (as[j] < as[i])
                dp[i] = std::max(dp[i], dp[j] + 1);
            if (as[i / j] < as[i])
                dp[i] = std::max(dp[i], dp[i / j] + 1);
        }
        ans = std::max(ans, dp[i]);
    }
    std::cout << ans << '\n';
}
