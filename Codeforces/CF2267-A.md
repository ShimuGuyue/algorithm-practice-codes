void solve()
{
    int n;
    char c;
    cin >> n >> c;
    string s;
    cin >> s;

    int ans = 0;
    for (int i = 0; i < n - 1 - i; ++i)
    {
        if (s[i] == s[n - i - 1])
            continue;
        if (s[i] == c || s[n - 1 - i] == c)
            ans += 1;
        else
            ans += 2;
    }
    cout << ans << '\n';
}
