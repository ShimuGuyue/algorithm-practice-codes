void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::pair<int, int>> ranges(n);
    for (auto& [a, b] : ranges)
    {
        std::cin >> a >> b;
    }

    auto check = [&ranges](int k) -> bool
    {
        int sum{ 0 };
        for (auto [a, b] : ranges)
        {
            if (k - sum - 1 > a)
                continue;
            if (sum > b)
                continue;
            ++sum;
            if (sum == k)
                return true;
        }
        return false;
    };
    int l{ 0 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    std::cout << l << '\n';
}
