void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::array<int, 20> counts{ };
    for (auto a : as)
    {
        for (int i{ 0 }; i < 20; ++i)
        {
            if (a >> i & 1)
                ++counts[i];
        }
    }

    int64_t ans{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        int64_t a{ 0 };
        for (int b{ 0 }; b < 20; ++b)
        {
            if (counts[b])
            {
                --counts[b];
                a |= 1 << b;
            }
        }
        ans += a * a;
    }
    std::cout << ans << '\n';
}
