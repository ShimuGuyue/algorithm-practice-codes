void solve()
{
    int n, k;
    std::cin >> n >> k;

    if (k > n)
    {
        std::cout << "NO\n";
        return;
    }

    int flag{ 1 };
    while (flag * 2 <= n)
    {
        flag *= 2;
    }

    using MaxHeap_Int = std::priority_queue<int, std::vector<int>, std::less<int>>;
    MaxHeap_Int h;
    while (n)
    {
        while (n >= flag)
        {
            n -= flag;
            h.push(flag);
            --k;
        }
        flag /= 2;
    }
    if (k < 0)
    {
        std::cout << "NO\n";
        return;
    }

    while (k--)
    {
        int t{ h.top() };
        h.pop();
        h.push(t / 2);
        h.push(t / 2);
    }

    std::cout << "YES\n";
    while (!h.empty())
    {
        std::cout << h.top() << ' ';
        h.pop();
    }
    std::cout << '\n';
}
