void solve()
{
    int64_t n;
    std::cin >> n;
    std::vector<std::pair<int, int>> abs(n);
    for (auto& [a, b] : abs)
    {
        std::cin >> a >> b;
    }

    std::vector<int64_t> counts_a(n + 1);
    std::vector<int64_t> counts_b(n + 1);
    for (auto [a, b] : abs)
    {
        ++counts_a[a];
        ++counts_b[b];
    }

    int64_t ans{ n * (n - 1) * (n - 2) / 6 };
    for (auto [a, b] : abs)
    {
        ans -= (counts_a[a] - 1) * (counts_b[b] - 1);
    }
    std::cout << ans << '\n';
}
