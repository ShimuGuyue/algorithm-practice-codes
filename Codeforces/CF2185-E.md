void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    std::vector<int> bs(m + 2);
    bs[0] = -1e9;
    bs[m + 1] = 2e9;
    for (int i = 1; i <= m; ++i)
    {
        std::cin >> bs[i];
    }
    std::string s;
    std::cin >> s;

    std::sort(bs.begin(), bs.end());

    std::map<int, std::set<int>> l_to_index;
    std::map<int, std::set<int>> r_to_index;
    std::vector<std::pair<int, int>> flags(n);
    for (int i = 0; i < n; ++i)
    {
        auto rt{ std::upper_bound(bs.begin(), bs.end(), as[i]) };
        auto lt{ std::prev(rt) };

        int l{ *lt - as[i] };
        int r{ *rt - as[i] };
        flags[i] = {l, r};
        l_to_index[l].insert(i);
        r_to_index[r].insert(i);
    }

    int ans{ n };

    int count{ 0 };
    for (char c : s)
    {
        if (c == 'L')
            --count;
        else
            ++count;
        if (count > 0)
        {
            if (r_to_index.count(count))
            {
                ans -= r_to_index[count].size();
                for (int i : r_to_index[count])
                {
                    int l{ flags[i].first };
                    l_to_index[l].erase(i);
                }
                r_to_index.erase(count);
            }
        }
        else
        {
            if (l_to_index.count(count))
            {
                ans -= l_to_index[count].size();
                for (int i : l_to_index[count])
                {
                    int r{ flags[i].second };
                    r_to_index[r].erase(i);
                }
                l_to_index.erase(count);
            }
        }

        std::cout << ans << ' ';
    }
    std::cout << '\n';
}
