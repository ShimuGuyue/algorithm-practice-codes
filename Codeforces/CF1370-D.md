void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    auto check = [&as, k](int m) -> bool
    {
        std::array<int, 2> flags{ };
        int index;
        int count;

        index = 0;
        count = 0;
        for (auto a : as)
        {
            if (index == 0)
            {
                if (a <= m)
                {
                    flags[index] = std::max(flags[index], a);
                    index = (index + 1) % 2;
                    ++count;
                }
            }
            else
            {
                index = (index + 1) % 2;
                ++count;
            }
            if (count == k)
                return true;
        }

        index = 0;
        count = 0;
        for (auto a : as)
        {
            if (index == 1)
            {
                if (a <= m)
                {
                    flags[index] = std::max(flags[index], a);
                    index = (index + 1) % 2;
                    ++count;
                }
            }
            else
            {
                index = (index + 1) % 2;
                ++count;
            }
            if (count == k)
                return true;
        }

        return false;
    };
    int l{ 1 }, r{ 1000000000 };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
