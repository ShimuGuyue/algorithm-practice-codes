```cpp
/**
 *  SIZE：问题中质数的值域
 */
template <const int SIZE>
class Eratosthenes
{
private:
    inline static bool is_init{ false };
    inline static std::vector<int> primes_{ };
    inline static std::vector<bool> is_primes_{ };

public:
    static bool judge(const int n)
    {
        if (!is_init)
            init();
        return (n & 1) ? is_primes_[n] : (n == 2);
    }

    static const std::vector<int>& primes()
    {
        if (!is_init)
            init();
        return primes_;
    }

private:
    static void init()
    {
        primes_.reserve(SIZE / std::log(SIZE));
        is_primes_.assign(SIZE + 1, true);

        if (SIZE >= 2)
            primes_.push_back(2);

        is_primes_[1] = false;
        for (int i{ 3 }; i <= SIZE; i += 2)
        {
            if (!is_primes_[i])
                continue;
            primes_.push_back(i);

            if (int64_t(i) * i > SIZE)
                continue;
            for (int j{ i * i }; j <= SIZE; j += i * 2)
            {
                is_primes_[j] = false;
            }
        }
    }
};
```

```cpp
void solve()
{
    auto primes{ Eratosthenes<15000000>::primes() };

    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int gcd{
        std::accumulate(as.begin(), as.end(), as[0],
            [](int pre, int cur)
            {
                return std::gcd(pre, cur);
            }
        )
    };

    std::vector<int> counts(15000000 + 1);
    for (auto& a : as)
    {
        a /= gcd;
        ++counts[a];
    }

    int ans{ n };
    for (int p : primes)
    {
        int sum{ 0 };
        for (int i{ p }; i <= 15000000; i += p)
        {
            sum += counts[i];
        }
        ans = std::min(ans, n - sum);
    }
    if (ans == n)
        ans = -1;
    std::cout << ans << '\n';
}
```
