void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> counts(n + 1);
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    int mex{ 0 };
    for (int i{ 0 }; i <= n; ++i)
    {
        if (counts[i])
            ++mex;
        else
            break;
    }

    int cost{ 0 };
    int count{ n - mex };
    std::vector<int> difs(n + 2);
    for (int i{ mex }; i >= 0; --i)
    {
        ++difs[cost];
        --difs[count + 1];
        ++count;
        if (i)
            cost = counts[i - 1];
    }
    std::vector<int> ans(n + 1);
    std::partial_sum(difs.begin(), difs.end() - 1, ans.begin());
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
