void solve()
{
    constexpr int mod{ 100000000 };

    int n, m, p, q;
    std::cin >> n >> m >> p >> q;
    std::vector<std::vector<std::vector<std::vector<int>>>> dp(n + 1,
                std::vector<std::vector<std::vector<int>>>    (m + 1,
                            std::vector<std::vector<int>>     (2,
                                        std::vector<int>      (10 + 1))));
    dp[1][0][0][1] = dp[0][1][1][1] = 1;
    for (int i = 0; i <= n; ++i)
    {
        for (int j = 0; j <= m; ++j)
        {
            for (int k = 1; k <= p; ++k)
            {
                if (i < n && k < p)
                {
                    dp[i + 1][j][0][k + 1] += dp[i][j][0][k];
                    dp[i + 1][j][0][k + 1] %= mod;
                }
                if (j < m)
                {
                    dp[i][j + 1][1][1] += dp[i][j][0][k];
                    dp[i][j + 1][1][1] %= mod;
                }
            }
            for (int k = 1; k <= q; ++k)
            {
                if (j < m && k < q)
                {
                    dp[i][j + 1][1][k + 1] += dp[i][j][1][k];
                    dp[i][j + 1][1][k + 1] %= mod;
                }
                if (i < n)
                {
                    dp[i + 1][j][0][1] += dp[i][j][1][k];
                    dp[i + 1][j][0][1] %= mod;
                }
            }
        }
    }
    int ans{ 0 };
    for (int i = 1; i <= p; ++i)
    {
        ans += dp[n][m][0][i];
        ans %= mod;
    }
    for (int i = 1; i <= q; ++i)
    {
        ans += dp[n][m][1][i];
        ans %= mod;
    }
    std::cout << ans << '\n';
}
