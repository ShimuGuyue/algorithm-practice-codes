void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    for (int i = 0; i < 1000; ++i)
    {
        for (auto& a : as)
        {
            a = turn(a);
        }
    }

    map<int, int> m;
    for (auto a : as)
    {
        ++m[a];
    }

    int64_t ans = 0;
    for (auto [k, v] : m)
    {
        ans += int64_t(v) * (v - 1) / 2;
    }
    cout << ans << '\n';
}

int turn(int a)
{
    int ans = 0;
    while (a)
    {
        ans += (a % 10) * (a % 10);
        a /= 10;
    }
    return ans;
}
