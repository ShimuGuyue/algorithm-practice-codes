```cpp
struct DisjointSet
{
    std::vector<int> fathers;
    std::vector<std::pair<int, int>> minmaxs;

    DisjointSet(int n) : fathers(n), minmaxs(n)
    {
        for (int i{ 0 }; i < n; ++i)
        {
            fathers[i] = i;
            minmaxs[i] = { i, i };
        }
    }

    int find(int x)
    {
        return fathers[x] == x ? x : fathers[x] = find(fathers[x]);
    }

    void merge(int x, int y)
    {
        x = find(x);
        y = find(y);
        int min{ std::min(minmaxs[x].first, minmaxs[y].first) };
        int max{ std::max(minmaxs[x].second, minmaxs[y].second) };
        fathers[x] = y;
        minmaxs[y] = { min, max };
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;

    DisjointSet ds(n + 1);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        ds.merge(u, v);
    }

    using Range = std::pair<int, int>;
    std::priority_queue<Range, std::vector<Range>, std::greater<Range>> h;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (ds.find(i) == i)
            h.push(ds.minmaxs[i]);
    }

    int ans{ 0 };
    while (h.size() > 1)
    {
        auto [l1, r1]{ h.top() };
        h.pop();
        auto [l2, r2]{ h.top() };
        h.pop();
        if (l2 > r1)
        {
            h.push({ l2, r2 });
        }
        else
        {
            ++ans;
            h.push({ l1, std::max(r1, r2) });
        }
    }
    std::cout << ans << '\n';
}
```
