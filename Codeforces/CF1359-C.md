void solve()
{
    int64_t h, c, t;
    std::cin >> h >> c >> t;

    if (t == h)
    {
        std::cout << 1 << '\n';
        return;
    }

    if (t * 2 <= h + c)
    {
        std::cout << 2 << '\n';
        return;
    }

    auto check = [h, c, t](int64_t k) -> bool
    {
        int64_t sum{ h * k + c * (k - 1) };
        return sum >= t * (k * 2 - 1);
    };
    int64_t l{ 1 }, r{ 1000000000 };
    while (l < r)
    {
        int64_t mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    /*
        sum1 / k1 - t <= t - sum2 / k2
        (sum1 - k1 * t) / k1 <= (k2 * t - sum2) / k2
        (sum1 - k1 * t) * k2 <= (k2 * t - sum2) * k1
    */
    int64_t sum1{ h * l + c * (l - 1) };
    int64_t sum2{ h * (l + 1) + c * l };
    int64_t k1{ l * 2 - 1 };
    int64_t k2{ l * 2 + 1 };
    if ((sum1 - k1 * t) * k2 <= (k2 * t - sum2) * k1)
        std::cout << k1 << '\n';
    else
        std::cout << k2 << '\n';
}
