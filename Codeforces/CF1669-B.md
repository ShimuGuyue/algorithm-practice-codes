void solve()
{
    int n;
    std::cin >> n;

    std::map<int, int> counts;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }
    for (auto [k, v] : counts)
    {
        if (v >= 3)
        {
            std::cout << k << '\n';
            return;
        }
    }
    std::cout << -1 << '\n';
}
