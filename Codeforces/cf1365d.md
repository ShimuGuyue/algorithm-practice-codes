static constexpr std::array<std::pair<int, int>, 4> dxy{ {{-1, 0}, {0, -1}, {0, 1}, {1, 0}} };

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> grid(n);
    for (auto& s : grid)
    {
        std::cin >> s;
    }

    for (int i{ 0 }; i < n; ++i)
    {
        for (int j{ 0 }; j < m; ++j)
        {
            if (grid[i][j] != 'B')
                continue;
            for (auto [dx, dy] : dxy)
            {
                int x{ i + dx };
                int y{ j + dy };
                if (x == -1 || x == n)
                    continue;
                if (y == -1 || y == m)
                    continue;
                if (grid[x][y] == 'G')
                {
                    std::cout << "No\n";
                    return;
                }
                if (grid[x][y] == '.')
                    grid[x][y] = '#';
            }
        }
    }

    int count{ 0 };
    for (auto& s : grid)
    {
        for (char c : s)
        {
            count += c == 'G';
        }
    }

    std::vector<std::vector<bool>> visited(n, std::vector<bool>(m));
    auto dfs = [&count, &grid, &visited, n, m](auto&& dfs, int x, int y) -> void
    {
        if (grid[x][y] == 'G')
            --count;
        for (auto [dx, dy] : dxy)
        {
            int xx{ x + dx };
            int yy{ y + dy };
            if (xx == -1 || xx == n)
                continue;
            if (yy == -1 || yy == m)
                continue;
            if (grid[xx][yy] == '#')
                continue;
            if (visited[xx][yy])
                continue;
            visited[xx][yy] = true;
            dfs(dfs, xx, yy);
        }
    };
    visited[n - 1][m - 1] = true;
    if (grid[n - 1][m - 1] != '#')
        dfs(dfs, n - 1, m - 1);
    std::cout << (count ? "No" : "Yes") << '\n';
}
