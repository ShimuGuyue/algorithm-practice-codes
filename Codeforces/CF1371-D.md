void solve()
{
    int n, k;
    std::cin >> n >> k;

    std::vector<int> count_rs(n);
    std::vector<int> count_cs(n);
    std::vector<std::vector<int>> grid(n, std::vector<int>(n));
    int i{ 0 }, j{ 0 };
    while (k--)
    {
        if (grid[i][j])
            j = (j + 1) % n;
        grid[i][j] = 1;
        ++count_rs[i];
        ++count_cs[j];
        i = (i + 1) % n;
        j = (j + 1) % n;
    }

    auto [min_r, max_r]{ std::minmax_element(count_rs.begin(), count_rs.end()) };
    auto [min_c, max_c]{ std::minmax_element(count_cs.begin(), count_cs.end()) };
    int ans{ (*max_r - *min_r) * (*max_r - *min_r) + (*max_c - *min_c) * (*max_c - *min_c) };
    std::cout << ans << '\n';
    for (auto& v : grid)
    {
        for (auto a : v)
        {
            std::cout << a;
        }
        std::cout << '\n';
    }
}
