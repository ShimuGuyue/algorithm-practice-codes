void solve()
{
    int64_t l, r;
    std::cin >> l >> r;

    static std::vector<std::vector<int64_t>> dp(20, std::vector<int64_t>(4, -1));

    auto calc = [](int64_t x) -> int64_t
    {
        std::vector<int> digits;
        while (x)
        {
            digits.push_back(x % 10);
            x /= 10;
        }

        auto dfs = [&digits](this auto& dfs, int index, int count, bool limit, bool front) -> int64_t
        {
            if (count > 3)
                return 0;
            if (index == -1)
                return front ? 0 : 1;

            if (!limit && !front && dp[index][count] != -1)
                return dp[index][count];

            int64_t ans{ 0 };
            int max{ limit ? digits[index] : 9 };
            for (int a{ 0 }; a <= max; ++a)
            {
                ans += dfs(
                    index - 1,
                    count + (a != 0),
                    limit && (a == max),
                    front && a == 0
                );
            }
            if (!limit && !front)
                dp[index][count] = ans;
            return ans;
        };
        return dfs(int(digits.size()) - 1, 0, true, true);
    };

    int64_t ans{ calc(r) - calc(l - 1) };
    std::cout << ans << '\n';
}
