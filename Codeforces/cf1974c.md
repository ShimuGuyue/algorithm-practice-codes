void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::map<std::array<int, 2>, int64_t> counts_12;
    std::map<std::array<int, 2>, int64_t> counts_13;
    std::map<std::array<int, 2>, int64_t> counts_23;
    std::map<std::array<int, 3>, int64_t> counts_123;
    for (int i = 0; i + 2 < n; ++i)
    {
        int a{ as[i] }, b{ as[i + 1] }, c{ as[i + 2] };
        ++counts_12[{ a, b }];
        ++counts_13[{ a, c }];
        ++counts_23[{ b, c }];
        ++counts_123[{ a, b, c }];
    }

    int64_t ans{ 0 };
    for (auto [k, v] : counts_12)
    {
        ans += (v * (v - 1) / 2);
    }
    for (auto [k, v] : counts_13 )
    {
        ans += (v * (v - 1) / 2);
    }
    for (auto [k, v] : counts_23)
    {
        ans += (v * (v - 1) / 2);
    }
    for (auto [k, v] : counts_123)
    {
        ans -= 3 * (v * (v - 1) / 2);
    }
    std::cout << ans << '\n';
}
