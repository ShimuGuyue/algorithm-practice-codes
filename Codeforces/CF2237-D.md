void solve()
{
    int64_t n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int64_t ans{ n * (n + 1) / 2 };

    std::array<int64_t, 3> counts{ { 1, 0, 0 } };
    int pre{ 0 };
    for (char c : s)
    {
        if (c == '0')
            pre = (pre + 2) % 3;
        else
            pre = (pre + 1) % 3;
        ++counts[pre];
    }
    for (int i{ 0 }; i < 3; ++i)
    {
        ans -= counts[i] * (counts[i] - 1) / 2;
    }

    std::vector<int64_t> flags(n + 1);
    for (int i{ 3 }; i <= n; ++i)
    {
        flags[i] = (i + 1) / 2 - 1;
    }
    std::vector<int64_t> presums(n + 1);
    std::partial_sum(flags.begin(), flags.end(), presums.begin());
    int len{ 1 };
    for (int i{ 1 }; i < n; ++i)
    {
        if (s[i] != s[i - 1])
        {
            ++len;
        }
        else
        {
            if (len >= 3)
                ans -= presums[len];
            len = 1;
        }
    }
    if (len >= 3)
        ans -= presums[len];

    std::cout << ans << '\n';
}
