void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n);
    std::vector<int> bs(m);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    std::sort(bs.rbegin(), bs.rend());

    int i{ 0 }, j{ 0 };
    while (i < n && j < m)
    {
        if (as[i] > bs[j])
        {
            std::cout << as[i] << ' ';
            ++i;
        }
        else
        {
            std::cout << bs[j] << ' ';
            ++j;
        }
    }
    while (i < n)
    {
        std::cout << as[i] << ' ';
        ++i;
    }
    while (j < m)
    {
        std::cout << bs[j] << ' ';
        ++j;
    }
    std::cout << '\n';
}
