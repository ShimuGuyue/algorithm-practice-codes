void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    std::vector<int> hs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& h : hs)
    {
        std::cin >> h;
    }

    if (*std::min_element(as.begin(), as.end()) > k)
    {
        std::cout << 0 << '\n';
        return;
    }

    int ans{ 1 };
    int sum{ as[0] };
    int l{ 0 }, r{ 1 };
    while (r < n)
    {
        if (hs[r - 1] % hs[r] == 0)
        {
            sum += as[r];
            while (sum > k)
            {
                sum -= as[l++];
            }
            ans = std::max(ans, r - l + 1);
        }
        else
        {
            sum = as[r];
            l = r;
        }
        ++r;
    }
    std::cout << ans << '\n';
}
