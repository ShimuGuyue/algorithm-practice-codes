void solve()
{
    int n;
    std::cin >> n;
    std::vector<double> xs(n);
    for (auto& x : xs)
    {
        std::cin >> x;
    }
    std::vector<double> ts(n);
    for (auto& t : ts)
    {
        std::cin >> t;
    }

    auto calc = [n, &xs, &ts](double k) -> double
    {
        double ans{ 0 };
        for (int i{ 0 }; i < n; ++i)
        {
            ans = std::max(fabs(xs[i] - k) + ts[i], ans);
        }
        return ans;
    };
    double l{ 0 }, r{ 1e8 };
    while (r - l > 1e-7)
    {
        double ll{ l + (r - l) / 3 };
        double rr{ r - (r - l) / 3 };
        double ans_l{ calc(ll) };
        double ans_r{ calc(rr) };
        if (ans_l >= ans_r)
            l = ll;
        if (ans_r >= ans_l)
            r = rr;
    }
    std::cout << std::fixed << std::setprecision(10) << l << '\n';
}
