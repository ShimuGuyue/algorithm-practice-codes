void solve()
{
    int64_t a, m;
    std::cin >> a >> m;

    int64_t g{ std::gcd(a, m) };
    int64_t flag{ m / g };

    std::vector<int64_t> facotrs;
    for (int64_t i{ 2 }; i * i <= flag; ++i)
    {
        if (flag % i)
            continue;
        facotrs.push_back(i);
        while (flag % i == 0)
        {
            flag /= i;
        }
    }
    if (flag > 1)
        facotrs.push_back(flag);

    int64_t ans{ 0 };

    int n{ int(facotrs.size()) };
    int64_t l{ (a + g - 1) / g }, r{ (a + m - 1) / g };
    for (int k{ 0 }; k < (1 << n); ++k)
    {
        int64_t cur{ 1 };
        int count{ 0 };
        bool ok{ true };
        for (int b{ 0 }; b < n; ++b)
        {
            if (k >> b & 1)
            {
                ++count;
                cur *= facotrs[b];
                if (cur > r)
                    ok = false;
            }
        }
        if (!ok)
            continue;

        int64_t sum{ r / cur - (l - 1) / cur };
        if (count & 1)
            ans -= sum;
        else
            ans += sum;
    }
    std::cout << ans << '\n';
}
