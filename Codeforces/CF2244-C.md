void solve()
{
    int n, x, y;
    std::cin >> n >> x >> y;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int g{ std::gcd(x, y) };

    for (int i{ 1 }; i <= n; ++i)
    {
        int dif{ std::abs(i - as[i]) };
        if (dif % g)
        {
            std::cout << "No\n";
            return;
        }
    }
    std::cout << "Yes\n";
}
