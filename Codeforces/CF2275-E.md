void solve()
{
    int n;
    cin >> n;
    vector<int> as(n + 1);
    vector<int> bs(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        cin >> as[i];
    }
    for (int i = 1; i <= n; ++i)
    {
        cin >> bs[i];
    }

    vector<int> pres(n + 2);
    vector<int> sufs(n + 2);
    for (int i = 2; i <= n; ++i)
    {
        pres[i] = pres[i - 1] + turn(as[i - 1], bs[i - 1]) + turn(bs[i - 1], as[i]);
    }
    for (int i = n - 1; i >= 1; --i)
    {
        sufs[i] = sufs[i + 1] + turn(as[i], bs[i + 1]) + turn(as[i + 1], bs[i]);
    }

    int ans = 0;
    for (int i = 1; i <= n; ++i)
    {
        ans = std::max(ans, pres[i] + sufs[i]);
    }
    ans += turn(as[n], bs[n]);
    cout << ans << '\n';
}

int turn(int x, int y)
{
    return x == y ? 2 : 1;
}
