void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::cout << quick_pow(n, k, 1000000007) << '\n';
}

int64_t quick_pow(int64_t base, int64_t power, int64_t mod)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}
