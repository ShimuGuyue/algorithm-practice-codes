void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    auto calc = [&as, n](int i) -> int
    {
        if (i == 0 || i == n - 1)
            return 0;
        if (as[i] > as[i - 1] && as[i] > as[i + 1])
            return 1;
        if (as[i] < as[i - 1] && as[i] < as[i + 1])
            return 1;
        return 0;
    };

    int sum{ 0 };
    for (int i{ 1 }; i < int(as.size()) - 1; ++i)
    {
        sum += calc(i);
    }

    int ans{ sum };
    for (int i{ 1 }; i < n - 1; ++i)
    {
        int temp{ as[i] };

        int cur{ sum };
        cur -= calc(i - 1) + calc(i) + calc(i + 1);
        as[i] = as[i - 1];
        cur += calc(i - 1) + calc(i) + calc(i + 1);
        ans = std::min(ans, cur);
        as[i] = temp;

        cur = sum;
        cur -= calc(i - 1) + calc(i) + calc(i + 1);
        as[i] = as[i + 1];
        cur += calc(i - 1) + calc(i) + calc(i + 1);
        ans = std::min(ans, cur);
        as[i] = temp;
    }
    std::cout << ans << '\n';
}
