void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> ans;
    int flag{ 1 };
    for (int i{ n }; i > 0; --i)
    {
        as[i] *= flag;
        if (as[i] > 0)
        {
            ans.push_back(i);
            flag *= -1;
        }
    }

    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
