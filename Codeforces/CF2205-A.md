void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int index{ std::max_element(as.begin(), as.end()) - as.begin() };
    std::swap(as[index], as[1]);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << as[i] << ' ';
    }
    std::cout << '\n';
}
