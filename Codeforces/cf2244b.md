void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t temp{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        as[i] += temp;
        temp = 0;
        if (as[i] <= i)
        {
            std::cout << "No\n";
            return;
        }
        temp = as[i] - (i + 1);
    }
    std::cout << "Yes\n";
}
