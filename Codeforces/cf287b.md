void solve()
{
    int64_t n, k;
    std::cin >> n >> k;

    auto calc = [k](int64_t n) -> int64_t
    {
        int64_t ans{ 1 };
        int64_t a{ k - 1 };
        int64_t b{ a - n + 1 };
        int64_t len{ a - b + 1 };
        ans += len * (a + b) / 2;
        return ans;
    };

    if (n == 1)
    {
        std::cout << 0 << '\n';
        return;
    }

    if (calc(k - 1) < n)
    {
        std::cout << -1 << '\n';
        return;
    }

    int64_t l{ 1 }, r{ k - 1 };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if (calc(mid) >= n)
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
