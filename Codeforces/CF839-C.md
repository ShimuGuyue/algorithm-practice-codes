void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i = 1; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    if (n == 1)
    {
        std::cout << 0 << '\n';
        return;
    }

    double ans{ 0 };
    auto dfs = [&tree, &ans](this auto& dfs, int u, int len, double p, int last)
    {
        if (tree[u].size() == 1)
        {
            ans += len * p;
            return;
        }
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, len + 1, p / (tree[u].size() - 1), u);
        }
    };
    for (int i : tree[1])
    {
        dfs(i, 1, 1.0 / tree[1].size(), 1);
    }

    std::cout << std::fixed << std::setprecision(10) << ans << '\n';
}
