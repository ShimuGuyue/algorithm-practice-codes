void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> premaxs(n);
    premaxs[0] = as[0];
    for (int i{ 1 }; i < n; ++i)
    {
        premaxs[i] = std::max(premaxs[i - 1], as[i]);
    }

    int ans{ 0 };
    for (int i{ n - 1 }; i >= 0; --i)
    {
        if (premaxs[i] == as[i])
            ++ans;
    }
    std::cout << ans << '\n';
}
