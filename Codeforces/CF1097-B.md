void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (auto &a : as)
    {
        std::cin >> a;
    }

    std::vector<std::vector<bool>> dp(n + 1, std::vector<bool>(360));
    dp[0][0] = true;
    for (int i = 0; i < n; ++i)
    {
        for (int j = 0; j < 360; ++j)
        {
            if (!dp[i][j])
                continue;
            dp[i + 1][(j + 360 - as[i]) % 360] = true;
            dp[i + 1][(j + as[i]) % 360] = true;
        }
    }
    std::cout << (dp[n][0] ? "yes" : "no") << '\n';
}
