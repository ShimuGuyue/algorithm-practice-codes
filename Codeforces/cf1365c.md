void solve()
{
    int n;
    std::cin >> n;

    std::vector<int> indexs(n + 1);
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        indexs[a] = i;
    }

    std::map<int, int> m;
    for (int i = 0; i < n; ++i)
    {
        int b;
        std::cin >> b;
        int dif{ (i - indexs[b] + n) % n };
        ++m[dif];
    }

    int ans{ 0 };
    for (auto [k, v] : m)
    {
        ans = std::max(ans, v);
    }
    std::cout << ans << '\n';
}
