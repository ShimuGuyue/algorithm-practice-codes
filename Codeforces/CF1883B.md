void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    std::map<char, int> m;
    for (char c : s)
    {
        ++m[c];
    }

    int need{ -1 };
    for (auto [c, v] : m)
    {
        need += v % 2;
    }
    std::cout << (need <= k ? "Yes" : "No") << '\n';
}
