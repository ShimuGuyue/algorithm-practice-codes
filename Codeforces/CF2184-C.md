void solve()
{
    int n, k;
    std::cin >> n >> k;

    if (k == n)
    {
        std::cout << 0 << '\n';
        return;
    }

    std::map<int, int> dp;
    dp[n] = 0;
    std::queue<int> q;
    q.push(n);
    while (!q.empty())
    {
        int t{ q.front() };
        q.pop();

        int x{ t / 2 };
        int y{ t - x };
        if (x >= k)
        {
            if (dp.count(x) == 0)
            {
                dp[x] = dp[t] + 1;
                q.push(x);
            }
        }
        if (y >= k)
        {
            if (dp.count(y) == 0)
            {
                dp[y] = dp[t] + 1;
                q.push(y);
            }
        }
        if (dp.count(k))
        {
            std::cout << dp[k] << '\n';
            return;
        }
    }
    std::cout << -1 << '\n';
}
