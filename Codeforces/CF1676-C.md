void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> vs(n);
    for (auto& s : vs)
    {
        std::cin >> s;
    }

    int ans{ 26 * m };
    for (int i = 0; i < n; ++i)
    {
        for (int j = i + 1; j < n; ++j)
        {
            int sum{ 0 };
            for (int k = 0; k < m; ++k)
            {
                sum += abs(vs[i][k] - vs[j][k]);
            }
            ans = std::min(ans, sum);
        }
    }
    std::cout << ans << '\n';
}
