void solve()
{
    int n, k;
    std::cin >> n >> k;

    std::string flag;
    for (int i{ 0 }; i < k; ++i)
    {
        char c1{ static_cast<char>('a' + i) };
        flag += c1;
        for (int j{ i + 1 }; j < k; ++j)
        {
            char c2{ static_cast<char>('a' + j) };
            flag += c1;
            flag += c2;
        }
    }

    for (int i{ 0 }; i < n; ++i)
    {
        std::cout << flag[i % flag.size()];
    }
    std::cout << '\n';
}
