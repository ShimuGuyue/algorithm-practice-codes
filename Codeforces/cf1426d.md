void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 0 };
    std::set<int64_t> t;
    int64_t sum{ as[0] };
    t.insert(sum);
    for (int i{ 1 }; i < n; ++i)
    {
        sum += as[i];
        if (t.find(sum) != t.end() || sum == 0)
        {
            ++ans;
            t.clear();
            sum = as[i];
        }
        t.insert(sum);
    }
    std::cout << ans << '\n';
}
