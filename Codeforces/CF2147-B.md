void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> ans(n * 2);

    int flag{ 0 };
    for (int i = n; i >= 1; --i)
    {
        while (ans[flag])
        {
            ++flag;
        }
        bool ok{ false };
        for (int j = flag + i; j < n * 2; j += i)
        {
            if (ans[j] == 0)
            {
                ok = true;
                ans[flag] = ans[j] = i;
            }
        }
        if (!ok)
        {
            ++flag;
            ++i;
        }
    }
    for (int a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
