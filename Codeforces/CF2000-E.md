void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    int w;
    std::cin >> w;
    std::vector<int64_t> as(w);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.rbegin(), as.rend());

    std::vector<std::vector<int>> counts_dif(n + 2, std::vector<int>(m + 2));
    for (int i = 1; i + k - 1 <= n; ++i)
    {
        for (int j = 1; j + k - 1 <= m; ++j)
        {
            ++counts_dif[i][j];
            --counts_dif[i][j + k];
            --counts_dif[i + k][j];
            ++counts_dif[i + k][j + k];
        }
    }
    std::vector<std::vector<int>> counts(n + 2, std::vector<int>(m + 2));
    for (int i = 1; i <= n; ++i)
    {
        for (int j = 1; j <= m; ++j)
        {
            counts[i][j] = counts[i - 1][j] + counts[i][j - 1] - counts[i - 1][j - 1] + counts_dif[i][j];
        }
    }

    std::vector<int> counts_each;
    counts_each.reserve(n * m);
    for (int i = 1; i <= n; ++i)
    {
        for (int j = 1; j <= m; ++j)
        {
            counts_each.push_back(counts[i][j]);
        }
    }
    std::sort(counts_each.rbegin(), counts_each.rend());

    int64_t ans{ 0 };
    for (int i = 0; i < w; ++i)
    {
        ans += as[i] * counts_each[i];
    }
    std::cout << ans << '\n';
}
