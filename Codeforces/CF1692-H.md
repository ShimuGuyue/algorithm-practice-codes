void solve()
{
    int n;
    std::cin >> n;
    std::map<int, std::vector<int>> indexs;
    for (int i = 1; i <= n; ++i)
    {
        int a;
        std::cin >> a;
        indexs[a].push_back(i);
    }

    int ans_a, ans_l, ans_r;
    int max{ 0 };
    for (auto& [a, v] : indexs)
    {
        int sum{ 1 };
        int l{ v.front() }, r{ v.front() };
        if (sum > max)
        {
            max = sum;
            ans_a = a;
            ans_l = l;
            ans_r = r;
        }
        for (int i = 1; i < v.size(); ++i)
        {
            sum += 1 - (v[i] - v[i - 1] - 1);
            if (sum <= 1)
            {
                sum = 1;
                l = r = v[i];
            }
            else
            {
                r = v[i];
            }
            if (sum > max)
            {
                max = sum;
                ans_a = a;
                ans_l = l;
                ans_r = r;
            }
        }
    }
    std::cout << ans_a << ' ' << ans_l << ' ' << ans_r << '\n';
}
