void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::pair<int, int>> seges(m);
    for (auto& [l, r] : seges)
    {
        std::cin >> l >> r;
    }
    int q;
    std::cin >> q;
    std::vector<int> xs(q + 1);
    for (int i{ 1 }; i <= q; ++i)
    {
        std::cin >> xs[i];
    }

    auto check = [&seges, &xs, n](int k) -> bool
    {
        std::vector<int> presums(n + 1);
        for (int i{ 1 }; i <= k; ++i)
        {
            presums[xs[i]] = 1;
        }
        for (int i{ 1 }; i <= n; ++i)
        {
            presums[i] = presums[i] + presums[i - 1];
        }
        for (auto [l, r] : seges)
        {
            int len{ r - l + 1 };
            int count{ presums[r] - presums[l - 1] };
            if (count > len - count)
                return true;
        }
        return false;
    };
    int l{ 1 }, r{ q };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    if (check(l))
        std::cout << l << '\n';
    else
        std::cout << -1 << '\n';
}
