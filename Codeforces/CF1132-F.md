void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    s = " " + s;

    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(n + 1, std::numeric_limits<int>::max()));
    for (int len{ 1 }; len <= n; ++len)
    {
        for (int i{ 1 }; i + len - 1 <= n; ++i)
        {
            if (len == 1)
            {
                dp[i][i] = 1;
                continue;
            }
            for (int j{ i }; j < i + len - 1; ++j)
            {
                dp[i][i + len - 1] = std::min(dp[i][i + len - 1], dp[i][j] + dp[j + 1][i + len - 1]);
            }
            if (s[i] == s[i + len - 1])
                --dp[i][i + len - 1];
        }
    }
    std::cout << dp[1][n] << '\n';
}
