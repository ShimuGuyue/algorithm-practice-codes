void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    s = "0" + s;

    if (s[1] == '0')
    {
        std::cout << -1 << '\n';
        return;
    }
    for (int i{ 1 }; i < n; ++i)
    {
        if (s[i] == '0' && s[i + 1] == '0')
        {
            std::cout << -1 << '\n';
            return;
        }
    }

    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(7, 9));
    dp[0][turn(0)] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ -3 }; j <= 3; ++j)
        {
            if (!match(j, s[i - 1]))
                continue;
            for (int k{ -3 }; k <= 3; ++k)
            {
                if (!match(k, s[i]))
                    continue;
                if (j == k)
                    continue;
                dp[i][turn(k)] = std::min(
                    dp[i][turn(k)],
                    std::max(dp[i - 1][turn(j)], abs(j - k))
                );
            }
        }
    }
    int ans{ *std::min_element(dp[n].begin(), dp[n].end()) };
    std::cout << ans << '\n';
}

int turn(int x)
{
    return x + 3;
}

bool match(int x, char c)
{
    if (x > 0)
        return c == '+';
    if (x < 0)
        return c == '-';
    return c == '0';
}
