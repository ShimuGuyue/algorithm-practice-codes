void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a: as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());
    for (int k = 1; k <= 1024; ++k)
    {
        std::vector<int> bs(as);
        for (auto &b : bs)
        {
            b ^= k;
        }
        std::sort(bs.begin(), bs.end());
        if (as == bs)
        {
            std::cout << k << '\n';
            return;
        }
    }
    std::cout << -1 << '\n';
}
