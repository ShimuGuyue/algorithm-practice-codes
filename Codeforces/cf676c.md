void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    auto calc = [&s, n, k](char c)
    {
        int ans{ 0 };
        int sum{ 0 };
        int count{ 0 };
        int l{ 0 }, r{ 0 };
        while (r < n)
        {
            ++sum;
            if (s[r] != c)
                ++count;
            while (count > k)
            {
                if (s[l] != c)
                    --count;
                ++l;
                --sum;
            }
            ++r;
            ans = std::max(ans, sum);
        }
        return ans;
    };
    int ans{ std::max(calc('a'), calc('b')) };
    std::cout << ans << '\n';
}
