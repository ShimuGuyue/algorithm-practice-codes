void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> vs(n * 2 - 1);
    for (auto &s : vs)
    {
        std::cin >> s;
    }

    for (int index = 0; index < m; ++index)
    {
        std::map<char, int> counts;
        for (int i = 0; i < n * 2 - 1; ++i)
        {
            ++counts[vs[i][index]];
        }
        for (auto [k, v] : counts)
        {
            if (v & 1)
                std::cout << k;
        }
    }
    std::cout << std::endl;
}
