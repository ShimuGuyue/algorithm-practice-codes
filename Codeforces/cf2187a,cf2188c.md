void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (std::is_sorted(as.begin(), as.end()))
    {
        std::cout << -1 << '\n';
        return;
    }

    std::vector<int> bs{ as };
    std::sort(bs.begin(), bs.end());

    int min{ bs.front() }, max{ bs.back() };

    int ans{ INT_MAX };
    for (int i = 0; i < n; ++i)
    {
        if (as[i] == bs[i])
            continue;
        ans = std::min(ans, std::max(std::abs(min - as[i]), std::abs(max - as[i])));
    }
    std::cout << ans << '\n';
}
