void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int ans{ 1 };
    int extra{ 0 };
    for (int i{ 1 }; i < n; ++i)
    {
        if (s[i] != s[i - 1])
            ++ans;
        else
            extra = 1;
    }
    if (s.front() != s.back())
        ans += extra;
    std::cout << ans << '\n';
}
