void solve()
{
    int n;
    std::cin >> n;

    std::vector<int64_t> odds;
    std::vector<int64_t> evens;
    odds.reserve(n);
    evens.reserve(n);
    for (int i = 0; i < n; ++i)
    {
        int64_t a;
        std::cin >> a;
        if (a & 1)
            odds.push_back(a);
        else
            evens.push_back(a);
    }
    std::sort(odds.rbegin(), odds.rend());
    std::sort(evens.rbegin(), evens.rend());

    std::vector<int64_t> presums_even(evens.size());
    std::partial_sum(evens.begin(), evens.end(), presums_even.begin());

    if (odds.empty())
    {
        while (n--)
        {
            std::cout << "0 ";
        }
        std::cout << '\n';
        return;
    }

    std::vector<int64_t> dp(n + 1);
    dp[1] = odds.front();
    std::cout << dp[1] << ' ';
    for (int i = 2; i <= n; ++i)
    {
        if (i == n && odds.size() % 2 == 0)
        {
            dp[i] = 0;
        }
        else if (i - 1 <= evens.size())
        {
            dp[i] = dp[1] + presums_even[i - 2];
        }
        else
        {
            if (odds.size() > 2)
                dp[i] = dp[i - 2];
            else
                dp[i] = 0;
        }
        std::cout << dp[i] << ' ';
    }
    std::cout << '\n';
}
