void solve()
{
    int n;
    std::cin >> n;

    std::vector<int> b_reverse;
    if (n == 0)
    b_reverse = {0};
    while (n)
    {
        b_reverse.push_back(n % 2);
        n /= 2;
    }

    int len = b_reverse.size();
    while (b_reverse.size() < len * 2)
    {
        if (b_reverse.size() & 1)
        {
            int mid = b_reverse.size() / 2;
            if (b_reverse[mid] == 1)
            {
                b_reverse.push_back(0);
                continue;
            }
            bool ok = true;
            for (int l = mid - 1, r = mid + 1; l >= 0; --l, ++r)
            {
                if (b_reverse[l] != b_reverse[r])
                    ok = false;
            }
            if (ok)
            {
                std::cout << "yes\n";
                return;
            }
        }
        else
        {
            bool ok = true;
            for (int r = b_reverse.size() / 2, l = r - 1; l >= 0; --l, ++r)
            {
                if (b_reverse[l] != b_reverse[r])
                    ok = false;
            }
            if (ok)
            {
                std::cout << "yes\n";
                return;
            }
        }
        b_reverse.push_back(0);
    }
    std::cout << "no\n";
}
