void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(n);
    std::vector<int64_t> bs(m);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    std::sort(bs.begin(), bs.end());

    auto check = [&as, &bs](int64_t k)
    {
        for (auto a : as)
        {
            auto it1{ std::lower_bound(bs.begin(), bs.end(), a) };
            bool b1{ it1 != bs.end() && *it1 <= a + k };
            if (b1)
                continue;
            if (it1 == bs.begin())
                return false;
            auto it2{ std::prev(it1) };
            bool b2{ *it2 >= a - k };
            if (!b2)
                return false;
        }
        return true;
    };

    int64_t l{ 0 }, r{ std::numeric_limits<int>::max() };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
