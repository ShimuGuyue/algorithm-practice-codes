void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> difs(k * 2 + 2);
    for (int i{ 0 }; i * 2 < n; ++i)
    {
        auto [min, max]{ std::minmax(as[i], as[n - i - 1]) };
        int sum{ min + max };
        difs[0] += 2;
        difs[min + 1] -= 2;
        difs[min + 1] += 1;
        difs[sum] -= 1;
        difs[sum + 1] += 1;
        difs[sum + 1 + k - min] -= 1;
        difs[sum + 1 + k - min] += 2;
    }

    std::vector<int> counts(k * 2 + 2);
    std::partial_sum(difs.begin(), difs.end(), counts.begin());

    int ans{ n };
    for (int i{ 1 }; i <= k * 2; ++i)
    {
        ans = std::min(ans, counts[i]);
    }
    std::cout << ans << '\n';
}
