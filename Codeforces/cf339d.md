void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as((1 << n) + 1);
    for (int i = 1; i <= (1 << n); ++i)
    {
        std::cin >> as[i];
    }

    std::map<std::pair<int, int>, int> flags;
    auto dfs1 = [&as, &flags](auto&& dfs, int l, int r, int level) -> int
    {
        if (l == r)
        {
            flags[{l, r}] = as[l];
            return flags[{l, r}];
        }
        int mid{ (l + r) / 2 };
        int a{ dfs(dfs, l, mid, level - 1) };
        int b{ dfs(dfs, mid + 1, r, level - 1) };
        if (level & 1)
            flags[{l, r}] = a | b;
        else
            flags[{l, r}] = a ^ b;
        return flags[{l, r}];
    };
    dfs1(dfs1, 1, 1 << n, n);

    while (m--)
    {
        int p, x;
        std::cin >> p >> x;
        as[p] = x;
        auto dfs2 = [&as, &flags, p](auto &&dfs, int l, int r, int level) -> void
        {
            if (l == r)
            {
                flags[{l, r}] = as[l];
                return;
            }
            int mid{ (l + r) / 2 };
            int a, b;
            if (p <= mid)
                dfs(dfs, l, mid, level - 1);
            else
                dfs(dfs, mid + 1, r, level - 1);
            a = flags[{l, mid}];
            b = flags[{mid + 1, r}];
            if (level & 1)
                flags[{l, r}] = a | b;
            else
                flags[{l, r}] = a ^ b;
        };
        dfs2(dfs2, 1, 1 << n, n);
        std::cout << flags[{1, 1 << n}] << '\n';
    }
}
