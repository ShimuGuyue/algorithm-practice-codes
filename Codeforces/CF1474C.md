void solve()
{
    int n;
    std::cin >> n;
    std::multiset<std::pair<int, int>> t;
    for (int i{ 1 }; i <= n * 2; ++i)
    {
        int a;
        std::cin >> a;
        t.insert({a, i});
    }

    for (auto it{ std::next(t.rbegin()) }; it != t.rend(); ++it)
    {
        int x, a, b;
        x = t.rbegin()->first;

        auto tt{ t };
        std::vector<std::pair<int, int>> ans;

        auto it1{ std::prev(tt.end()) };
        a = it1->first;
        tt.erase(it1);
        auto it2{ tt.find(*it) };
        b = it2->first;
        tt.erase(it2);
        ans.push_back({a, b});
        int flag{ a + b };
        while (!tt.empty())
        {
            it1 = std::prev(tt.end());
            a = it1->first;
            tt.erase(it1);
            b = x - a;
            it2 = tt.lower_bound({b, 0});
            if (it2 == tt.end() || it2->first != b)
                break;
            tt.erase(it2);
            ans.push_back({a, b});
            x = a;
        }
        if (tt.empty())
        {
            std::cout << "Yes\n";
            std::cout << flag << '\n';
            for (auto [a, b] : ans)
            {
                std::cout << a << ' ' << b << '\n';
            }
            return;
        }
    }
    std::cout << "No\n";
}
