void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    auto check = [k, &tree](int x) -> bool
    {
        int count{ 0 };
        auto dfs = [x, &tree, &count](this auto& dfs, int u, int last) -> int
        {
            int sum{ 1 };
            for (int v : tree[u])
            {
                if (v == last)
                    continue;
                sum += dfs(v, u);
            }
            if (sum >= x)
            {
                ++count;
                sum = 0;
            }
            return sum;
        };
        dfs(1, 0);
        return count > k;
    };
    int l{ 1 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    std::cout << l << '\n';
}
