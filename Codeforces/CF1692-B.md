void solve()
{
    int n;
    std::cin >> n;
    std::map<int, int> m;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++m[a];
    }

    int count_odd{ 0 };
    int count_even{ 0 };
    for (auto [k, v] : m)
    {
        if (v & 1)
            ++count_odd;
        else
            ++count_even;
    }

    int ans{ count_odd + count_even - (count_even & 1) };
    std::cout << ans << '\n';
}
