void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }

    std::vector<int64_t> difs(n);
    std::adjacent_difference(as.begin(), as.end(), difs.begin());

    std::vector<int64_t> ans(n);
    for (int i = 0; i < n; ++i)
    {
        if (difs[i] == i + 1)
            ans[i] = i + 1;
        else
            ans[i] = ans[i - difs[i]];
    }
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
