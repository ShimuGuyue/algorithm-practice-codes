void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::pair<int, int>> abs(n);
    for (auto& [a, b] : abs)
    {
        std::cin >> a;
    }
    for (auto& [a, b] : abs)
    {
        std::cin >> b;
    }

    std::vector<int64_t> flag1s(1, 0);
    std::vector<int64_t> flag2s(1, 0);
    for (auto [a, b] : abs)
    {
        if (b == 1)
            flag1s.push_back(a);
        else
            flag2s.push_back(a);
    }

    std::sort(flag1s.rbegin(), flag1s.rend() - 1);
    std::sort(flag2s.rbegin(), flag2s.rend() - 1);
    int x{ int(flag1s.size()) };
    int y{ int(flag2s.size()) };

    std::vector<int64_t> pre1s(x);
    std::vector<int64_t> pre2s(y);
    std::partial_sum(flag1s.begin(), flag1s.end(), pre1s.begin());
    std::partial_sum(flag2s.begin(), flag2s.end(), pre2s.begin());

    if (pre1s.back() + pre2s.back() < m)
    {
        std::cout << -1 << '\n';
        return;
    }

    int ans{ n * 2 };
    for (int i{ 0 }; i < x; ++i)
    {
        int l{ 0 }, r{ y - 1 };
        if (pre1s[i] + pre2s[r] < m)
            continue;
        while (l < r)
        {
            int mid{ (l + r) / 2 };
            if (pre1s[i] + pre2s[mid] >= m)
                r = mid;
            else
                l = mid + 1;
        }
        ans = std::min(ans, i + r * 2);
    }
    std::cout << ans << '\n';
}
