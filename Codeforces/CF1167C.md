```c++
class DisjointSet
{
private:
    struct Node
    {
        int parent;
        int rank;
        int size;
    };

    std::vector<Node> nodes;

public:
    DisjointSet(int n) : nodes(n)
    {
        for (int i = 0; i < n; ++i)
        {
            nodes[i] = {i, 1, 1};
        }
    }

public:
    int find(int x)
    {
        return (nodes[x].parent == x) ? x : (nodes[x].parent = find(nodes[x].parent));
    }

    void merge(int x, int y)
    {
        int set_x = find(x);
        int set_y = find(y);
        if (set_x == set_y)
            return;
        if (nodes[set_x].rank < nodes[set_y].rank)
        {
            nodes[set_x].parent = set_y;
            nodes[set_y].size += nodes[set_x].size;
        }
        else
        {
            nodes[set_y].parent = set_x;
            if (nodes[set_x].rank == nodes[set_y].rank)
                ++nodes[set_x].rank;
            nodes[set_x].size += nodes[set_y].size;
        }
    }

    int get_size(int x)
    {
        return nodes[find(x)].size;
    }
};
```

```c++
void solve()
{
    int n, m;
    std::cin >> n >> m;

    DisjointSet ds(n + 1);
    while (m--)
    {
        int k;
        std::cin >> k;
        if (k == 0)
            continue;
        int a;
        std::cin >> a;
        while (--k)
        {
            int b;
            std::cin >> b;
            ds.merge(a, b);
        }
    }
    for (int i = 1; i <= n; ++i)
    {
        std::cout << ds.get_size(i) << ' ';
    }
    std::cout << '\n';
}
```
