std::vector<std::vector<int>> factors;

void init()
{
    factors.resize(200001);
    for (int i = 1; i <= 200000; ++i)
    {
        for (int j = 1; j * j <= i ; ++j)
        {
            if (i % j)
                continue;
            if (j != 1)
                factors[i].push_back(j);
            if (i / j != 1 && j != i / j)
                factors[i].push_back(i / j);
        }
    }
}

void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }
    for (auto &b : bs)
    {
        std::cin >> b;
    }

    std::vector<int> counts(200001);
    for (int a : as)
    {
        for (int factor : factors[a])
        {
            ++counts[factor];
        }
    }
    // 0
    for (int c : counts)
    {
        if (c >= 2)
        {
            std::cout << 0 << '\n';
            return;
        }
    }
    // 1
    for (int a : as)
    {
        for (int factor : factors[a])
        {
            --counts[factor];
        }
        for (int factor : factors[a + 1])
        {
            if (counts[factor])
            {
                std::cout << 1 << '\n';
                return;
            }
        }
        for (int factor : factors[a])
        {
            ++counts[factor];
        }
    }
    // 2
    std::cout << 2 << '\n';
}
