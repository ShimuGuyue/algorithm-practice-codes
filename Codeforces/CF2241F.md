void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    int count1{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (s[i] == '1')
        {
            ++count1;
        }
        else
        {
            if (count1 & 1)
            {
                std::cout << "Alice\n";
                return;
            }
        }
    }
    int count0{ 0 };
    for (int i{ n - 1 }; i >= 0; --i)
    {
        if (s[i] == '0')
        {
            ++count0;
        }
        else
        {
            if (count0 & 1)
            {
                std::cout << "Alice\n";
                return;
            }
        }
    }

    std::cout << "Bob\n";
}
