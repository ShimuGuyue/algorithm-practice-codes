void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    std::map<std::pair<int, int>, int> m;
    int flag{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (as[i] == 0)
        {
            if (bs[i] == 0)
                ++flag;
            continue;
        }
        int g{ std::gcd(as[i], bs[i]) };
        as[i] /= g;
        bs[i] /= g;
        if (as[i] < 0)
        {
            as[i] *= -1;
            bs[i] *= -1;
        }
        ++m[{as[i], bs[i]}];
    }

    int ans{ 0 };
    for (auto [k, v] : m)
    {
        ans = std::max(ans, v);
    }
    ans += flag;
    std::cout << ans << '\n';
}
