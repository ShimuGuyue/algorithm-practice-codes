void solve()
{
    std::string s;
    std::cin >> s;
    int64_t x;
    std::cin >> x;

    int64_t n{ int64_t(s.length()) };

    int64_t l{ 1 }, r{ n };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if ((n + n - mid + 1) * mid / 2 >= x)
            r = mid;
        else
            l = mid + 1;
    }

    int64_t a{ r };
    int64_t b{ x - (n + n - r + 2) * (r - 1) / 2 };

    int64_t dif{ a - 1 };
    std::string str;
    for (char c : s)
    {
        while (!str.empty() && c < str.back())
        {
            if (dif == 0)
                break;
            --dif;
            str.pop_back();
        }
        str += c;
    }
    while (dif--)
    {
        str.pop_back();
    }
    std::cout << str[b - 1];
}
