

```cpp
class Eratosthenes
{
private:
    std::vector<int64_t> primes_;
    std::vector<bool> is_primes_;   // 仅保存奇数的映射

public:
    Eratosthenes(const int n)
    {
        build(n);
    }

public:
    bool judge(const int n) const
    {
        if (n % 2 == 0)
            return n == 2;
        int x{ n / 2 };
        return is_primes_[x];
    }

    const std::vector<int64_t>& primes() const
    {
        return primes_;
    }

private:
    void build(const int n)
    {
        primes_.reserve(n / std::log(n));
        is_primes_.assign((n + 1) / 2, true);

        if (n >= 2)
            primes_.push_back(2);

        is_primes_[0] = false;
        for (int i{ 1 }; i < is_primes_.size(); ++i)
        {
            int x{ i * 2 + 1 };

            if (!is_primes_[i])
                continue;

            primes_.push_back(x);
            if (int64_t(x) * x > n)
                continue;
            for (int y{ x * x }; y <= n; y += x * 2)
            {
                int j{ y / 2 };
                is_primes_[j] = false;
            }
        }
    }
};
```

```cpp
void solve()
{
    static Eratosthenes era{ 200000 };

    int n;
    std::cin >> n;
    for (int i{ 0 }; i < n; ++i)
    {
        std::cout << era.primes()[i] * era.primes()[i + 1] << ' ';
    }
    std::cout << '\n';
}
```
