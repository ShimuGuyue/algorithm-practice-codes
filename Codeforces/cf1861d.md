void solve()
{
    uint64_t n, x;
    std::cin >> n >> x;

    auto get_flags = [](uint64_t x)
    {
        std::array<bool, 10> flags{ };
        while (x)
        {
            flags[x % 10] = true;
            x /= 10;
        }
        return flags;
    };

    int ans{ -1 };
    std::unordered_set<uint64_t> t;
    std::queue<std::pair<uint64_t, int>> q;
    q.push({ x, 0 });
    while (!q.empty())
    {
        auto [a, sum]{ q.front() };
        q.pop();

        if (std::to_string(a).length() >= n)
        {
            ans = sum;
            break;
        }
        auto flags{ get_flags(a) };
        for (int i{ 9 }; i > 1; --i)
        {
            if (!flags[i])
                continue;
            if (!t.count(a * i))
            {
                q.push({a * i, sum + 1});
                t.insert(a * i);
            }
        }
    }
    std::cout << ans << '\n';
}
