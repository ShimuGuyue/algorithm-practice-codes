void solve()
{
    int a, b, c;
    std::cin >> a >> b >> c;

    std::string ans;
    if (a)
    {
        ans += std::string(a + 1, '0');
    }

    if (c)
    {
        if (a)
            --b;
        ans += std::string(c + 1, '1');
    }

    if (b)
    {
        if (ans.empty())
            ans += "1";
        for (int i = 0; i < b; ++i)
        {
            ans += ans.back() == '1' ? '0' : '1';
        }
    }

    std::cout << ans << '\n';
}
