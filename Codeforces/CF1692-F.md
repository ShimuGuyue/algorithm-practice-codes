void solve()
{
    int n;
    std::cin >> n;
    std::array<int, 10> counts{ };
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a % 10];
    }

    for (int i = 0; i < 10; ++i)
    {
        for (int j = 0; j < 10; ++j)
        {
            for (int k = 0; k < 10; ++k)
            {
                if ((i + j + k) % 10 != 3)
                    continue;
                if (i == j && j == k && counts[i] >= 3)
                {
                    std::cout <<"yes\n";
                    return;
                }
                if (i == j && j != k && counts[i] >= 2 && counts[k] >= 1)
                {
                    std::cout <<"yes\n";
                    return;
                }
                if (i == k && j != k && counts[i] >= 2 && counts[j] >= 1)
                {
                    std::cout <<"yes\n";
                    return;
                }
                if (j == k && i != j && counts[i] >= 1 && counts[j] >= 2)
                {
                    std::cout <<"yes\n";
                    return;
                }
                if (i != j && i != k && j != k && counts[i] && counts[j] && counts[k])
                {
                    std::cout << "yes\n";
                    return;
                }
            }
        }
    }
    std::cout << "no\n";
}
