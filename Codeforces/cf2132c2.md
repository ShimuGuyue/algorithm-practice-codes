template <typename T>
using MaxHeap = std::priority_queue<T, std::vector<T>, std::less<T>>;

void solve()
{
    static constexpr std::array<std::array<int64_t, 3>, 20> flags{{
        {1, 3, 0},  // 个数，花费，拆分节省
        {3, 10, 1},
        {9, 33, 3},
        {27, 108, 9},
        {81, 351, 27},
        {243, 1134, 81},
        {729, 3645, 243},
        {2187, 11664, 729},
        {6561, 37179, 2187},
        {19683, 118098, 6561},
        {59049, 373977, 19683},
        {177147, 1180980, 59049},
        {531441, 3720087, 177147},
        {1594323, 11691702, 531441},
        {4782969, 36669429, 1594323},
        {14348907, 114791256, 4782969},
        {43046721, 358722675, 14348907},
        {129140163, 1119214746, 43046721},
        {387420489, 3486784401, 129140163},
        {1162261467, 10847773692, 387420489}
    }};

    int64_t n, k;
    std::cin >> n >> k;

    MaxHeap<std::pair<int64_t, int64_t>> h; // save, count

    // judge
    int64_t ans{ 0 };
    int64_t min_count{ 0 };
    while (n)
    {
        int64_t sum{ 0 }, cost{ 0 }, save{ 0 };
        for (auto [a, b, c] : flags)
        {
            if (a <= n)
            {
                sum = a;
                cost = b;
                save = c;
            }
        }

        int64_t count{ n / sum };
        n %= sum;
        min_count += count;
        ans += count * cost;
        h.push({save, count});
    }
    if (min_count > k)
    {
        std::cout << -1 << '\n';
        return;
    }

    // solve
    k -= min_count;
    while (!h.empty() && k >= 2)
    {
        auto [save, count]{ h.top() };
        h.pop();

        // 不能继续拆分
        if (save == 0)
            break;

        // 拆分次数
        int64_t x{ std::min(k / 2, count) };
        k -= x * 2;

        ans -= save * x;
        h.push({save / 3, x * 3});
    }
    std::cout << ans << '\n';
}
