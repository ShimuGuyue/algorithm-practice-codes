void solve()
{
    int n, l, r;
    std::cin >> n >> l >> r;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    int64_t ans{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        auto ll{ std::lower_bound(as.begin() + i + 1, as.end(), l - as[i]) };
        auto rr{ std::upper_bound(as.begin() + i + 1, as.end(), r - as[i]) };
        ans += (rr - ll);
    }
    std::cout << ans << '\n';
}
