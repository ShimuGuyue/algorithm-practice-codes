void solve()
{
    constexpr int m{ 32768 };

    std::vector<int> dp(m, m);
    dp[0] = 0;
    std::queue<int> q;
    q.push(0);
    while (!q.empty())
    {
        int a{ q.front() };
        q.pop();

        int b{ a == 0 ? m - 1 : a - 1 };
        if (dp[b] == m)
        {
            dp[b] = dp[a] + 1;
            q.push(b);
        }

        int c{ a / 2 };
        if (dp[c] == m)
        {
            dp[c] = dp[a] + 1;
            q.push(c);
        }
        int d{ (a + m) / 2 };
        if (dp[d] == m)
        {
            dp[d] = dp[a] + 1;
            q.push(d);
        }
    }

    int n;
    std::cin >> n;
    while (n--)
    {
        int a;
        std::cin >> a;
        std::cout << dp[a] << ' ';
    }
    std::cout << '\n';
}
