void solve()
{
    int n, t;
    std::cin >> n >> t;
    int k;
    std::cin >> k;

    int l{ 1 }, r{ n };
    while (1)
    {
        if (l == r)
        {
            std::cout << "! " << l << std::endl;
            break;
        }
        int mid{ (l + r) / 2 };
        std::cout << "? " << l << ' ' << mid << std::endl;
        int x;
        std::cin >> x;
        int ll{ mid - l + 1 - x };
        int rr{ k - ll };
        if (k > ll)
        {
            k -= ll;
            l = mid + 1;
        }
        else
        {
            r = mid;
        }
    }
}
