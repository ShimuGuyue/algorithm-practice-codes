class SparseTable_MIN
{
private:
    std::vector<int> v_;
    std::vector<std::vector<int>> table_;

public:
    SparseTable_MIN(const std::vector<int>& v)
    {
        build(v);
    }

public:
    int query(const int l, const int r) const
    {
        const int power{ static_cast<int>(log2(r - l + 1)) };
        const int len{ 1 << power };
        return merge(l, r - len + 1, power);
    }

private:
    void build(const std::vector<int>& v)
    {
        v_ = v;

        const int n{ static_cast<int>(v.size()) };
        const int m{ static_cast<int>(std::log2(n)) + 1 };
        table_.assign(n, std::vector<int>(m));

            table_[i][0] = contribute(i);

        for (int j{ 1 }; j < m; ++j)
        {
            int len{ 1 << (j - 1) };
            for (int i{ 0 }; i + len * 2 - 1 < n; ++i)
            {
                table_[i][j] = merge(i, i + len, j - 1);
            }
        }
    }

void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    std::map<int, std::set<int>> indexs;
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        indexs[as[i]].insert(i);
    }

    SparseTable_MAX sp_max(as);
    SparseTable_MIN sp_min(as);

    int q;
    std::cin >> q;
    while (q--)
    {
        int l, r;
        std::cin >> l >> r;
        int max{ sp_max.query(l, r) };
        int min{ sp_min.query(l, r) };
        if (max == min)
            std::cout << -1 << ' ' << -1 << '\n';
        else
            std::cout << *indexs[max].lower_bound(l) << ' ' << *indexs[min].lower_bound(l) << '\n';
    }
}
