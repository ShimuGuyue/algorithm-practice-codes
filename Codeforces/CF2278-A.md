void solve()
{
    int n, m;
    cin >> n >> m;
    string a, b;
    cin >> a;
    cin >> b;

    map<char, int> count;
    for (char c : a)
    {
        ++count[c];
    }
    for (char c : b)
    {
        --count[c];
        if (count[c] == 0)
            count.erase(c);
    }

    if (count.size() == 3
        && count['s'] == 1
        && count['i'] == 1
        && count['x'] == 1)
        cout << "Yes\n";
    else
        cout << "No\n";
}
