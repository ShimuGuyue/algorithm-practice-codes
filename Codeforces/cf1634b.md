void solve()
{
    int64_t n, x, y;
    std::cin >> n >> x >> y;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t sum{ std::accumulate(as.begin(), as.end(), INT64_C(0)) };
    if ((x + sum - y) % 2 == 0)
        std::cout << "Alice\n";
    else
        std::cout << "Bob\n";
}
