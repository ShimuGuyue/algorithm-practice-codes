void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    int64_t ans{ 0 };
    int x{ 0 };
    int l{ 0 }, r{ n - 1 };
    while (l < r)
    {
        debug_(as[l], as[r], x);
        if (x + as[l] < as[r])
        {
            ans += as[l];
            x += as[l];
            as[l] = 0;
            ++l;
        }
        else if (x + as[l] == as[r])
        {
            ans += as[l] + 1;
            x = 0;
            as[l] = 0;
            as[r] = 0;
            ++l;
            --r;
        }
        else if (x + as[l] > as[r])
        {
            int dif{ as[r] - x };
            ans += dif + 1;
            as[l] -= dif;
            x = 0;
            as[r] = 0;
            --r;
        }
    }
    if (l == r)
    {
        debug_(as[l], as[r], x);
        if (as[l] == 1)
            ans += 1;
        else
        {
            int dif{ as[l] - x };
            ans += (dif + 1) / 2 + 1;
        }
        as[l] = 0;
    }
    std::cout << ans << '\n';
}
