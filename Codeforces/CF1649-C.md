void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::map<int, std::vector<int>> indexs_x;
    std::map<int, std::vector<int>> indexs_y;
    for (int i = 1; i <= n; ++i)
    {
        for (int j = 1; j <= m; ++j)
        {
            int a;
            std::cin >> a;
            indexs_x[a].push_back(i);
            indexs_y[a].push_back(j);
        }
    }

    int64_t ans{ 0 };
    for (auto& [a, v] : indexs_x)
    {
        std::sort(v.begin(), v.end());
        for (int i = 0; i < v.size(); ++i)
        {
            ans += int64_t(v[i]) * (i - (int(v.size()) - i - 1));
        }
    }
    for (auto& [a, v] : indexs_y)
    {
        std::sort(v.begin(), v.end());
        for (int i = 0; i < v.size(); ++i)
        {
            ans += int64_t(v[i]) * (i - (int(v.size()) - i - 1));
        }
    }
    std::cout << ans << '\n';
}
