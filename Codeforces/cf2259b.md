void solve()
{
    int n;
    cin >> n;
    int count1{ 0 }, count2{ 0 }, count4{ 0 };
    while (n--)
    {
        int a;
        cin >> a;
        if (a % 2 == 1)
            ++count1;
        else if (a % 4 == 2)
            ++count2;
        else
            ++count4;
    }
    int ans = std::max<int>({count1, count2, count4});
    cout << ans << '\n';
}
