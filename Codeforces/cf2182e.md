void solve()
{
    int64_t n, m, k;
    std::cin >> n >> m >> k;
    std::multiset<int> as;
    while (m--)
    {
        int a;
        std::cin >> a;
        as.insert(a);
    }
    std::vector<std::array<int, 3>> xyzs(n);
    for (auto& [x, y, z] : xyzs)
    {
        std::cin >> x >> y >> z;
    }

    std::sort(xyzs.begin(), xyzs.end(),
        [](std::array<int, 3>& a, std::array<int, 3>& b)
        {
            return a[2] - a[1] > b[2] - b[1];
        }
    );

    int ans{ 0 };
    std::vector<bool> visited(n);
    for (int i{ 0 }; i < n; ++i)
    {
        auto [x, y, z]{ xyzs[i] };
        k -= y;
        auto it{ as.lower_bound(x) };
        if (it != as.end())
        {
            ++ans;
            visited[i] = true;
            as.erase(it);
        }
    }
    for (int i{ n - 1 }; i >= 0; --i)
    {
        if (visited[i])
            continue;
        auto [x, y, z]{ xyzs[i] };
        int dif{ z - y };
        if (k >= dif)
        {
            ++ans;
            k -= dif;
        }
    }
    std::cout << ans << '\n';
}
