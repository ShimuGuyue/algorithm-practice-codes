void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::set<int> indexs;
    for (int i{ 1 }; i <= n; ++i)
    {
        indexs.insert(i);
    }

    while (q--)
    {
        int op;
        std::cin >> op;
        if (op == 1)
        {
            int l, r;
            std::cin >> l >> r;
            std::vector<int> temp;
            auto it{ indexs.lower_bound(l) };
            while (it != indexs.end() && *it <= r)
            {
                auto i{ *it };
                int a{ 0 };
                while (as[i])
                {
                    a += as[i] % 10;
                    as[i] /= 10;
                }
                as[i] = a;
                if (a < 10)
                    temp.push_back(i);
                ++it;
            }
            for (int i : temp)
            {
                indexs.erase(i);
            }
        }
        else
        {
            int x;
            std::cin >> x;
            std::cout << as[x] << '\n';
        }
    }
}
