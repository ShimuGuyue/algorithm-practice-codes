template<typename T, typename K = T>
class SparseTable
{
private:
    std::vector<std::vector<K>> table_;

    const std::vector<T>& v_;

public:
    SparseTable(const std::vector<T>& v) : v_(v)
    {
        const int n{ static_cast<int>(v.size()) };
        const int m{ static_cast<int>(std::log2(n)) + 1 };
        table_.assign(n, std::vector<K>(m));

        for (int i{ 0 }; i < n; ++i)
        {
            table_[i][0] = contribute(i);
        }
        for (int j{ 1 }; j < m; ++j)
        {
            const int len{ 1 << (j - 1) };
            for (int i{ 0 }; i + len * 2 - 1 < n; ++i)
            {
                table_[i][j] = merge(i, i + len, j - 1);
            }
        }
    }

public:
    K query(const int l, const int r) const
    {
        const int power{ static_cast<int>(log2(r - l + 1)) };
        const int len{ 1 << power };
        return merge(l, r - len + 1, power);
    }

private:
    K contribute(const int index) const
    {
        // TODO: 规定初始贡献值
        return v_[index];
    }

    K merge(const int i1, const int i2, const int j) const
    {
        // TODO: 规定贡献值合并结果
        return std::min(table_[i1][j], table_[i2][j]);
    }
};

void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 2);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    SparseTable<int> st(as);

    std::vector<int> ans(n + 2);
    for (int i{ 1 }; i <= n; ++i)
    {
        int ll, rr;
        int l, r;

        l = 1, r = i;
        while (l < r)
        {
            int mid{ (l + r) / 2 };
            if (st.query(mid, i) == as[i])
                r = mid;
            else
                l = mid + 1;
        }
        ll = r;

        l = i, r = n;
        while (l < r)
        {
            int mid{ (l + r + 1) / 2 };
            if (st.query(i, mid) == as[i])
                l = mid;
            else
                r = mid - 1;
        }
        rr = l;
        ans[rr - ll + 1] = std::max(ans[rr - ll + 1], as[i]);
    }

    for (int i{ n }; i > 0; --i)
    {
        ans[i] = std::max(ans[i], ans[i + 1]);
    }

    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
