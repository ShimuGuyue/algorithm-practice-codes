void solve()
{
    int64_t n, k;
    std::cin >> n >> k;
    std::multiset<int64_t> as;
    std::multiset<int64_t> bs;
    for (int i{ 0 }; i < n; ++i)
    {
        int64_t a;
        std::cin >> a;
        as.insert(a);
    }
    for (int i{ 0 }; i < n; ++i)
    {
        int64_t b;
        std::cin >> b;
        bs.insert(b);
    }

    int ans{ 0 };
    while (!as.empty() && !bs.empty())
    {
        auto lt{ as.begin() };
        auto rt{ std::prev(bs.end()) };
        if ((*lt + 1) * (*rt + 1) <= k + 1)
        {
            ++ans;
            as.erase(lt);
            bs.erase(rt);
        }
        else
        {
            bs.erase(rt);
        }
    }
    std::cout << ans << '\n';
}
