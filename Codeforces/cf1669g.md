void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> grid(n);
    for (auto& s : grid)
    {
        std::cin >> s;
    }
    grid.push_back(std::string(m, 'o'));

    std::vector<std::vector<char>> ans(n + 1, std::vector<char>(m, '.'));
    for (int i = 0; i < n; ++i)
    {
        for (int j = 0; j < m; ++j)
        {
            if (grid[i][j] == 'o')
                ans[i][j] = 'o';
        }
    }
    ans[n] = std::vector<char>(m, 'o');

    for (int j = 0; j < m; ++j)
    {
        int sum{ 0 };
        for (int i = 0; i <= n; ++i)
        {
            if (grid[i][j] == 'o')
            {
                for (int k = 1; k <= sum; ++k)
                {
                    ans[i - k][j] = '*';
                }
                sum = 0;
            }
            else if (grid[i][j] == '*')
            {
                ++sum;
            }
        }
    }

    for (int i = 0; i < n; ++i)
    {
        for (int j = 0; j < m; ++j)
        {
            std::cout << ans[i][j];
        }
        std::cout << '\n';
    }
}
