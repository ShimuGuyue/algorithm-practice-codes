void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> presums_1(n + 1);
    std::partial_sum(as.begin(), as.end(), presums_1.begin());

    std::vector<int> presums_adjsame(n + 1);
    for (int i = 1; i < n; ++i)
    {
        presums_adjsame[i] = presums_adjsame[i - 1] + (as[i] == as[i + 1]);
    }

    while (q--)
    {
        int l, r;
        std::cin >> l >> r;
        if ((r - l + 1) % 3 || (presums_1[r] - presums_1[l - 1]) % 3)
        {
            std::cout << -1 << '\n';
            continue;
        }

        int ans{ (r - l + 1) / 3 };
        if (presums_adjsame[r - 1] - presums_adjsame[l - 1])
            std::cout << ans << '\n';
        else
            std::cout << ans + 1 << '\n';
    }
}
