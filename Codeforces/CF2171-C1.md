void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }
    for (auto &b : bs)
    {
        std::cin >> b;
    }

    int x{ 0 };
    int last_dif_index{ 0 };
    for (int i = 0; i < n; ++i)
    {
        x ^= as[i] ^ bs[i];
        if (as[i] != bs[i])
            last_dif_index = i + 1;
    }
    if (x == 0)
    {
        std::cout << "Tie\n";
        return;
    }
    std::cout << (last_dif_index % 2 == 1 ? "Ajisai" : "Mai") << '\n';
}
