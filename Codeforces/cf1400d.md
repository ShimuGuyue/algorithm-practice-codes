void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<int>> presums(n + 1, std::vector<int>(n + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= n; ++j)
        {
            presums[i][j] = presums[i - 1][j];
        }
        ++presums[i][as[i]];
    }


    int64_t ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ i + 1 }; j <= n; ++j)
        {
            ans += presums[i - 1][as[j]] * (presums[n][as[i]] - presums[j][as[i]]);
        }
    }
    std::cout << ans << '\n';
}
