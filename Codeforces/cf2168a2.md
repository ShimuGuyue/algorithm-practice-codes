void solve()
{
    std::string op;

        int n;
        std::cin >> n;

        {
            std::cout << 'z';

        }

        std::string s;
        std::cin >> s;
        std::cout << std::count(s.begin(), s.end(), 'z') << '\n';
        s += 'z';
        std::string temp;
        for (char c : s)
        {
            if (c == 'z')
            {
                if (temp.empty())
                    continue;
                std::cout << decode(temp) << ' ';
                temp.clear();
            }
            else
            {
                temp += c;
            }
        }
    }
    std::cout << '\n';
}

std::string encode(int n)
{
    std::string ans_reverse;
    while (n)
    {
        int mod{ n % 25 };
        n /= 25;
        ans_reverse += 'a' + mod;
    }
    return std::string(ans_reverse.rbegin(), ans_reverse.rend());
}

int decode(std::string &s)
{
    int ans{};
    for (char c : s)
    {
        ans = ans * 25 + c - 'a';
    }
    return ans;
}
