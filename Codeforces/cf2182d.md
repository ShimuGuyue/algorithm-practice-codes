int64_t mod{ 998244353 };
std::vector<int64_t> factorials;
std::vector<int64_t> inv_factorials;

void init()
{
    int n{ 1000000 };
    factorials.assign(n + 1, 1);
    for (int i{ 2 }; i <= n; ++i)
    {
        factorials[i] = factorials[i - 1] * i % mod;
    }
    inv_factorials.assign(n + 1, 1);
    inv_factorials[n] = quickpow(factorials[n], mod - 2);
    for (int i{ n - 1 }; i >= 1; --i)
    {
        inv_factorials[i] = inv_factorials[i + 1] * (i + 1) % mod;
    }
}

int64_t quickpow(int64_t base, int64_t power)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}

void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int max{ *std::max_element(as.begin() + 1, as.end()) };
    int count_max{ std::count(as.begin() + 1, as.end(), max) };

    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] < max - 1)
        {
            as[0] -= max - 1 - as[i];
            as[i] = max - 1;
        }
    }
    if (as[0] < 0)
    {
        std::cout << 0 << '\n';
        return;
    }

    int64_t ans;
    if (count_max + as[0] <= n)
    {
        ans = combinate(n - count_max, as[0]);
        count_max += as[0];
        ans *= factorials[count_max] * factorials[n - count_max] % mod;
        ans %= mod;
    }
    else
    {
        as[0] -= n - count_max;
        count_max = as[0] % n;
        ans = combinate(n, count_max);
        ans *= factorials[count_max] * factorials[n - count_max] % mod;
        ans %= mod;
    }

    std::cout << ans << '\n';
}

int64_t combinate(int n, int k)
{
    if (k == 0 || k == n)
        return 1;
    return factorials[n] * inv_factorials[k] % mod * inv_factorials[n - k] % mod;
}
