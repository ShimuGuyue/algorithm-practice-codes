void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    s = " " + s;
    std::vector<int> ans;
    ans.reserve(n);

    for (int i = 1; i <= n; ++i)
    {
        if (s[i] == '0')
            ans.push_back(i);
    }
    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
