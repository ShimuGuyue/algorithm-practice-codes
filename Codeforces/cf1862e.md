void solve()
{
    int64_t n, m, d;
    std::cin >> n >> m >> d;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t ans{ 0 };
    int64_t sum{ 0 };
    std::priority_queue<int64_t, std::vector<int64_t>, std::greater<int64_t>> h;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] > 0)
        {
            h.push(as[i]);
            sum += as[i];
        }
        while (h.size() > m)
        {
            sum -= h.top();
            h.pop();
        }
        ans = std::max(ans, sum - d * i);
    }
    std::cout << ans << '\n';
}
