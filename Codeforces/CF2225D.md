void solve()
{
    static constexpr int64_t mod{ 998244353 };

    int64_t n, x;
    std::cin >> n >> x;

    auto calc = [](int64_t n, int64_t x) -> int64_t
    {
        if (n == 0)
            return 0;
        if (x == 0)
            return n / 4;
        return n / 4 + (n % 4 == 0 ? 0 : (n % 4 >= x));
    };

    int64_t l, r;

    int64_t ans{ 0 };
    // <= , % 4 == 0 || 1
    // >= , % 4 == 3
    l = (calc(x, 0) + 1) % mod;
    r = (calc(n, 3) - calc(x - 1, 3)) % mod;
    ans = (ans + l * r) % mod;
    // <= , % 4 == 2
    // >= , % 4 == 1
    l = (calc(x, 2)) % mod;
    r = (calc(n, 1) - calc(x - 1, 1)) % mod;
    ans = (ans + l * r) % mod;

    std::cout << ans << '\n';
}
