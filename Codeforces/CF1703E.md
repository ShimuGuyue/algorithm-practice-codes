void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::string> grid(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::string s;
        std::cin >> s;
        grid[i] = " " + s;
    }

    std::vector<std::vector<bool>> visited(n + 1, std::vector<bool>(n + 1));

    int ans{ 0 };
    for (int i = 1; i <= (n + 1) / 2; ++i)
    {
        for (int j = 1; j <= (n + 1) / 2; ++j)
        {
            if (visited[i][j])
                continue;
            int count0{ 0 };
            int count1{ 0 };
            if (grid[i][j] == '1')
                ++count1;
            else
                ++count0;
            if (grid[j][n - i + 1] == '1')
                ++count1;
            else
                ++count0;
            if (grid[n - i + 1][n - j + 1] == '1')
                ++count1;
            else
                ++count0;
            if (grid[n - j + 1][i] == '1')
                ++count1;
            else
                ++count0;
            if (count0 == 1 || count1 == 1)
                ans += 1;
            else if (count0 == 2 || count1 == 2)
                ans += 2;
            visited[i][j] = true;
            visited[j][n - i + 1] = true;
            visited[n - i + 1][n - j + 1] = true;
            visited[n - j + 1][i] = true;
        }
    }
    std::cout << ans << '\n';
}
