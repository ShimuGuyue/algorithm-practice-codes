void solve()
{
    int64_t n, x;
    std::cin >> n >> x;
    std::vector<std::array<int64_t, 3>> as(n);
    for (int i = 0; i < n; ++i)
    {
        std::cin >> as[i][0] >> as[i][1] >> as[i][2];
    }

    int64_t sum{ 0 };
    int64_t max{ 0 };
    for (auto [a, b, c] : as)
    {
        sum += a * (b - 1);
        max = std::max(max, a * b - c);
    }

    if (sum >= x)
    {
        std::cout << 0 << '\n';
        return;
    }
    if (max == 0)
    {
        std::cout << -1 << '\n';
        return;
    }
    int64_t ans{ (x - sum + max - 1) / max };
    std::cout << ans << '\n';
}
