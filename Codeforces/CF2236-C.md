void solve()
{
    int a, b, x;
    std::cin >> a >> b >> x;

    int ans{ 1 << 30 };
    int sum{ 0 };
    while (1)
    {
        if (a > b)
            std::swap(a, b);
        ans = std::min(ans, sum + b - a);
        if (a == b)
            break;
        ++sum;
        b /= x;
    }
    std::cout << ans << '\n';
}
