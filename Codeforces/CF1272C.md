void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;
    std::map<char, bool> m;
    for (int i = 0; i < k; ++i)
    {
        char c;
        std::cin >> c;
        m[c] = true;
    }
    s += '0';

    int64_t ans{ 0 };
    int64_t count{ 0 };
    for (char c : s)
    {
        if (!m[c])
        {
            ans += count * (count + 1) / 2;
            count = 0;
        }
        else
        {
            ++count;
        }
    }
    std::cout << ans << '\n';
}
