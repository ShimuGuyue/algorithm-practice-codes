void solve()
{
    int64_t h, n;
    std::cin >> h >> n;
    std::vector<int64_t> as(n);
    std::vector<int64_t> cs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& c : cs)
    {
        std::cin >> c;
    }

    auto check = [&as, &cs, h, n](int64_t k)
    {
        int64_t sum{ 0 };
        for (int i{ 0 }; i < n; ++i)
        {
            sum += as[i] * (k / cs[i] + 1);
            if (sum >= h)
                return true;
        }
        return false;
    };

    int64_t l{ 1 }, r{ 50000000000 };
    while (l < r)
    {
        int64_t mid{ (l + r) / 2 };
        if (check(mid - 1))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << r << '\n';
}
