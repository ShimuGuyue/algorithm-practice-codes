void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    std::vector<std::vector<std::string>> ss(n + 1, std::vector<std::string>(2));
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> ss[i][0];
        ss[i][1] = ss[i][0];
        std::reverse(ss[i][1].begin(), ss[i][1].end());
    }

    constexpr int64_t inf{ static_cast<int64_t>(1e15) };

    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(2, inf));
    dp[0][0] = dp[0][1] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (ss[i][0] >= ss[i - 1][0])
            dp[i][0] = std::min(dp[i][0], dp[i - 1][0]);
        if (ss[i][0] >= ss[i - 1][1])
            dp[i][0] = std::min(dp[i][0], dp[i - 1][1]);
        if (ss[i][1] >= ss[i - 1][0])
            dp[i][1] = std::min(dp[i][1], dp[i - 1][0] + as[i]);
        if (ss[i][1] >= ss[i - 1][1])
            dp[i][1] = std::min(dp[i][1], dp[i - 1][1] + as[i]);
    }
    int64_t ans{ std::min(dp[n][0], dp[n][1]) };
    if (ans == inf)
        ans = -1;
    std::cout << ans << '\n';
}
