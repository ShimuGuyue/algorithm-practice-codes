void solve()
{
    std::string s;
    std::cin >> s;
    int64_t a, b, c;
    int64_t x, y, z;
    std::cin >> a >> b >> c;
    std::cin >> x >> y >> z;
    int64_t n;
    std::cin >> n;

    int64_t per_a{ 0 }, per_b{ 0 }, per_c{ 0 };
    for (char c : s)
    {
        if (c == 'B')
            ++per_a;
        if (c == 'S')
            ++per_b;
        if (c == 'C')
            ++per_c;
    }

    auto check = [a, b, c, x, y, z, per_a, per_b, per_c, n](int64_t k)
    {
        int64_t cost{ 0 };
        if (a < per_a * k)
            cost += (per_a * k - a) * x;
        if (b < per_b * k)
            cost += (per_b * k - b) * y;
        if (c < per_c * k)
            cost += (per_c * k - c) * z;
        return cost <= n;
    };

    int64_t l{ 0 }, r{ 10000000000000 };
    while (l < r)
    {
        auto mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    std::cout << l << '\n';
}
