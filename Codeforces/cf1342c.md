void solve()
{
    int a, b, q;
    std::cin >> a >> b >> q;

    std::vector<int> presums(a * b + 1);
    for (int i{ 1 }; i <= a * b; ++i)
    {
        presums[i] = presums[i - 1];
        if (i % a % b != i % b % a)
            ++presums[i];
    }
    int64_t per{ presums[a * b] };
    int64_t len{ a * b };

    while (q--)
    {
        int64_t l, r;
        std::cin >> l >> r;
        int64_t ll{ l / len };
        int64_t rr{ r / len };
        l %= len;
        r %= len;
        int64_t ans{ 0 };
        if (rr - ll == 0)
        {
            ans += presums[r] - (l ? presums[l - 1] : 0);
        }
        else
        {
            ans = (rr - ll - 1) * per;
            ans += presums[len] - (l ? presums[l - 1] : 0);
            ans += presums[r];
        }
        std::cout << ans << ' ';
    }
    std::cout << '\n';
}
