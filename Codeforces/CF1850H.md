```cpp
struct DisjointSet
{
    std::vector<int> sets;
    std::vector<int64_t> dists;

    DisjointSet(int n)
    {
        sets.assign(n, 0);
        dists.assign(n, 0);
        for (int i{ 0 }; i < n; ++i)
        {
            sets[i] = i;
        }
    }

    int find(int x)
    {
        if (sets[x] == x)
            return x;
        int set{ find(sets[x]) };
        dists[x] += dists[sets[x]];
        sets[x] = set;
        return set;
    }

    bool merge(int x, int y, int dist)
    {
        int set_x{ find(x) };
        int set_y{ find(y) };
        if (set_x == set_y)
        {
            if (dists[x] - dists[y] != dist)
                return false;
        }
        else
        {
            sets[set_x] = set_y;
            dists[set_x] = (int64_t)dist + dists[y] - dists[x];
        }
        return true;
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;

    DisjointSet ds(n + 1);
    bool ok{ true };
    while (m--)
    {
        int64_t a, b, d;
        std::cin >> a >> b >> d;
        if (!ds.merge(a, b, d))
            ok = false;
    }
    std::cout << (ok ? "YES" : "NO") << '\n';
}
```
