int64_t mod{ 998244353 };

void solve()
{
    int64_t n, m, r, c;
    std::cin >> n >> m >> r >> c;

    int64_t ans{ quickpow(2, n * m - (n - r + 1) * (m - c + 1)) };
    std::cout << ans << '\n';
}

int64_t quickpow(int64_t base, int64_t k)
{
    int64_t ans{ 1 };
    while (k)
    {
        if (k & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        k /= 2;
    }
    return ans;
}
