void solve()
{
    int n;
    std::cin >> n;
    std::string as;
    std::string bs;
    std::cin >> as;
    std::cin >> bs;
    as = " " + as;
    bs = " " + bs;

    int count00{ 0 }, count01{ 0 }, count10{ 0 }, count11{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] == '0' && bs[i] == '0')
            ++count00;
        if (as[i] == '0' && bs[i] == '1')
            ++count01;
        if (as[i] == '1' && bs[i] == '0')
            ++count10;
        if (as[i] == '1' && bs[i] == '1')
            ++count11;
    }

    /*
        1. a + b + c + d = n / 2;
        2. c + d = 01 - b + 11 - d
            b + c + 2 * d = 01 + 11
    */
    for (int b{ 0 }; b <= count01; ++b)
    {
        for (int c{ 0 }; c <= count10; ++c)
        {
            int d{ (count01 + count11 - b - c) / 2 };
            if (d < 0 || d > count11)
                continue;
            if (b + c + 2 * d != count01 + count11)
                continue;
            int a{ n / 2 - b - c - d };
            if (a < 0 || a > count00)
                continue;

            for (int i{ 1 }; i <= n; ++i)
            {
                if (as[i] == '0' && bs[i] == '0' && a)
                {
                    --a;
                    std::cout << i << ' ';
                }
                if (as[i] == '0' && bs[i] == '1' && b)
                {
                    --b;
                    std::cout << i << ' ';
                }
                if (as[i] == '1' && bs[i] == '0' && c)
                {
                    --c;
                    std::cout << i << ' ';
                }
                if (as[i] == '1' && bs[i] == '1' && d)
                {
                    --d;
                    std::cout << i << ' ';
                }
            }
            std::cout << '\n';
            return;
        }
    }
    std::cout << -1 << '\n';
}
