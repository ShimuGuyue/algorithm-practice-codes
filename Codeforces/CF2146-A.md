void solve()
{
    int n;
    std::cin >> n;
    std::map<int, int> m;
    while (n--)
    {
        int a;
        std::cin >> a;
        ++m[a];
    }

    std::vector<int> counts;
    for (auto [k, v] : m)
    {
        counts.push_back(v);
    }
    std::sort(counts.rbegin(), counts.rend());

    int ans{ 0 };
    for (int i = 0; i < counts.size(); ++i)
    {
        ans = std::max(ans, (i + 1) * counts[i]);
    }
    std::cout << ans << '\n';
}
