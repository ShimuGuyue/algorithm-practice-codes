void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    as.push_back(0);
    std::vector<int64_t> pres(n + 1);
    std::partial_sum(as.begin(), as.end(), pres.begin());

    int ans{ 0 };
    int i{ 0 };
    while (i < n && as[i])
    {
        if (pres[i] == 0)
            ++ans;
        ++i;
    }
    while (i < n)
    {
        std::map<int64_t, int> m;
        do
        {
            ++m[pres[i]];
            ++i;
        } while (as[i]);

        int max{ 0 };
        for (auto [k, v] : m)
        {
            max = std::max(max, v);
        }
        ans += max;
    }
    std::cout << ans << '\n';
}
