void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    std::vector<int64_t> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    for (int i{ n - 1 }; i >= 0; --i)
    {
        if (as[i] < bs[i])
        {
            as[i] = bs[i];
        }
        else if (as[i] > bs[i])
        {
            if (i != 0)
            {
                int64_t dif{ as[i] - bs[i] };
                as[i] = bs[i];
                as[i - 1] += dif;
            }
        }
    }
    if (as[0] != bs[0])
        std::cout << "No\n";
    else
        std::cout << "Yes\n";
}
