void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> a1s;
    std::vector<int> a2s;
    std::vector<int> a3s;
    std::vector<int> a6s;
    for (auto a : as)
    {
        if (a % 6 == 0)
            a6s.push_back(a);
        else if (a % 2 == 0 || a % 4 == 0)
            a2s.push_back(a);
        else if (a % 3 == 0)
            a3s.push_back(a);
        else
            a1s.push_back(a);
    }

    for (auto a : a6s)
    {
        std::cout << a << ' ';
    }
    if (a2s.size() < a3s.size())
    {
        for (auto a : a2s)
        {
            std::cout << a << ' ';
        }
    }
    else
    {
        for (auto a : a3s)
        {
            std::cout << a << ' ';
        }
    }
    for (auto a : a1s)
    {
        std::cout << a << ' ';
    }
    if (a2s.size() < a3s.size())
    {
        for (auto a : a3s)
        {
            std::cout << a << ' ';
        }
    }
    else
    {
        for (auto a : a2s)
        {
            std::cout << a << ' ';
        }
    }
    std::cout << '\n';
}
