void solve()
{
    int64_t n, m;
    std::cin >> n >> m;

    int64_t zero{ n - m };
    int64_t count{ m + 1 };
    int64_t per{ zero / count };
    int64_t more{ zero % count };
    int64_t ans{ more * (per + 1) * (per + 2) / 2 + (count - more) * per * (per + 1) / 2 };
    std::cout << n * (n + 1) / 2 - ans << '\n';
}
