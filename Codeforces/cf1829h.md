void solve()
{
    static constexpr int64_t mod{ 1000000007 };

    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(64));
    for (int i{ 1 }; i <= n; ++i)
    {
        dp[i][as[i]] = 1;
        for (int j{ 0 }; j < 64; ++j)
        {
            dp[i][j] += dp[i - 1][j];
            dp[i][j] %= mod;
            dp[i][j & as[i]] += dp[i - 1][j];
            dp[i][j & as[i]] %= mod;
        }
    }

    int64_t ans{ 0 };
    for (uint32_t i{ 0 }; i < 64; ++i)
    {
        if (std::popcount(i) == k)
            ans += dp[n][i];
    }
    std::cout << ans % mod << '\n';
}
