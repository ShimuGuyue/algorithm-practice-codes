void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }
    for (auto &b : bs)
    {
        std::cin >> b;
    }

    int x{};
    for (int i = 0; i < n; ++i)
    {
        x ^= as[i] ^ bs[i];
    }
    if (x == 0)
    {
        std::cout << "Tie\n";
        return;
    }

    int highest_bit{};
    for (int b = 0; (1 << b) <= x; ++b)
    {
        if ((x >> b) & 1)
            highest_bit = b;
    }
    int last_dif_index{};
    for (int i = 0; i < n; ++i)
    {
        if ((as[i] ^ bs[i]) >> highest_bit & 1)
            last_dif_index = i + 1;
    }

    std::cout << (last_dif_index % 2 == 1 ? "Ajisai" : "Mai") << '\n';
}
