void solve()
{
    int k;
    cin >> k;

    vector<vector<vector<int>>> trees(k + 1);

    for (int i = 1; i <= k; ++i)
    {
        auto& tree = trees[i];

        int n;
        cin >> n;
        tree.resize(n + 1);
        for (int j = 1; j < n; ++j)
        {
            int u, v;
            cin >> u >> v;
            tree[u].push_back(v);
            tree[v].push_back(u);
        }
    }

    int flag = 1;
    for (int i = 1; i <= k; ++i)
    {
        if (trees[i].size() - 1 != 1)
        {
            flag = i;
            break;
        }
    }

    int index = 1;
    if (trees[flag].size() - 1 != 1)
    {
        auto& tree = trees[flag];
        int n = tree.size() - 1;
        for (int i = 1; i <= n; ++i)
        {
            if (tree[i].size() == 1)
            {
                index = tree[i][0];
                break;
            }
        }
    }

    for (int i = 1; i <= k; ++i)
    {
        if (i == flag)
            continue;
        auto& tree = trees[i];
        int n = tree.size() - 1;
        if (n == 1)
        {
            cout << std::format("{} {} {} {}\n", flag, index, i, 1);
            continue;
        }
        for (int j = 1; j <= n; ++j)
        {
            if (tree[j].size() == 1)
            {
                cout << std::format("{} {} {} {}\n", flag, index, i, j);
                break;
            }
        }
    }
