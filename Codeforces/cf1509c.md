void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::sort(as.begin() + 1, as.end());

    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(n + 1));
    for (int len{ 2 }; len <= n; ++len)
    {
        for (int l{ 1 }; l + len - 1 <= n; ++l)
        {
            int r{ l + len - 1 };
            dp[l][r] = std::min(dp[l + 1][r], dp[l][r - 1]) + as[r] - as[l];
        }
    }
    std::cout << dp[1][n] << '\n';
}
