void solve()
{
    int a, b, c;
    std::cin >> a >> b >> c;

    auto lower_bound = [c](int n) -> int
    {
        int l{ 1 }, r{ 100000 };
        while (l < r)
        {
            int mid{ (l + r + 1) / 2 };
            int sum{ mid * n };
            if (sum <= c)
                l = mid;
            else
                r = mid - 1;
        }
        return l * n;
    };
    std::vector<int> ans{ { 1 << 30, 0, 0, 0 } };
    for (int x{ 1 }; x <= 2 * a; ++x)
    {
        for (int y{ x }; y <= 2 * b; y += x)
        {
            int z1{ lower_bound(y) };
            int z2{ z1 + y };
            for (int z : { z1, z2 })
            {
                int sum{ std::abs(x - a) + std::abs(y - b) + std::abs(z - c) };
                if (sum < ans[0])
                    ans = { sum, x, y, z };
            }
        }
    }

    std::cout << ans[0] << '\n';
    std::cout << ans[1] << ' ' << ans[2] << ' ' << ans[3] << '\n';
}
