void solve()
{
    int n;
    std::cin >> n;
    std::array<int64_t, 30> counts{ };
    while (n--)
    {
        int a;
        std::cin >> a;
        ++counts[bit(a)];
    }

    int64_t ans{ 0 };
    for (auto a : counts)
    {
        ans += a * (a - 1) / 2;
    }
    std::cout << ans << '\n';
}

int bit(int n)
{
    int ans{ -1 };
    while (n)
    {
        n >>= 1;
        ++ans;
    }
    return ans;
}
