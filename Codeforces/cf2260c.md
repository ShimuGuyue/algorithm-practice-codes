void solve()
{
    int x, y;
    std::cin >> x >> y;
    int ini{ x };

    int ans1{ 0 }, ans2{ 0 };
    for (int b{ 28 }; b >= 0; --b)
    {
        if ((x >> b & 1) != (y >> b & 1))
            continue;
        if ((x >> b & 1) == 0)
            continue;
        int flag{ 1 << b };
        if (x < flag)
            continue;
        x -= flag;
        y += flag;

        for (int i{ b + 1 }; i <= 30; ++i)
        {
            if ((x >> i & 1) == 0)
                continue;
            if ((y >> i & 1) == 0)
                continue;
            x -= 1 << i;
            y += 1 << i;
        }
    }
    ans1 = x ^ y;

    x = 0;
    for (int b{ 28 }; b >= 0; --b)
    {
        if ((ans1 >> b & 1) == 0)
            continue;
        if (x + (1 << b) <= ini)
            x |= 1 << b;
    }
    ans2 = ini - x;

    std::cout << ans1 << ' ' << ans2 << '\n';
}
