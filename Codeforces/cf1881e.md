void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> dp(n + 2);
    for (int i = n; i > 0; --i)
    {
        dp[i] = dp[i + 1] + 1;
        if (i + as[i] <= n)
            dp[i] = std::min(dp[i], dp[i + as[i] + 1]);
    }
    std::cout << dp[1] << '\n';
}
