void solve()
{
    int n, k;
    std::cin >> n >> k;

    std::vector<bool> visited(n + 1);
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        visited[a] = true;
    }

    int mex{ 0 };
    for (int i = 0; i <= n; ++i)
    {
        if (visited[i])
            mex = i + 1;
        else
            break;
    }
    int ans { std::min(mex, k - 1) };
    std::cout << ans << '\n';
}
