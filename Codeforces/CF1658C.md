void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (std::count(as.begin(), as.end(), 1) != 1)
    {
        std::cout << "No\n";
        return;
    }

    int index{ std::find(as.begin(), as.end(), 1) - as.begin() };
    for (int i{ (index + 1) % n }; i != index; i = (i + 1) % n)
    {
        if (as[i] - as[(i - 1 + n) % n] > 1)
        {
            std::cout << "No\n";
            return;
        }
    }
    std::cout << "YES\n";
}
