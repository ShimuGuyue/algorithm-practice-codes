void solve()
{
    constexpr int64_t mod{ 1000000007 };

    int n;
    std::cin >> n;
    int64_t ans{ (quickpow(3, n * 3, mod) - quickpow(7, n, mod) + mod) % mod };
    std::cout << ans << '\n';
}

int64_t quickpow(int64_t base, int64_t power, int64_t mod)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}
