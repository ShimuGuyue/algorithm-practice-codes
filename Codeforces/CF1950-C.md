void solve()
{
    std::string time;
    std::cin >> time;

    std::string h(time.begin(), time.begin() + 2);
    std::string m(time.begin() + 3, time.end());
    int hh{ std::stoi(h) };

    if (hh == 0)
    {
        h = std::to_string(hh + 12);
        std::cout << h << ':' << m << " AM\n";
    }
    else if (hh == 12)
    {
        std::cout << h << ':' << m << " PM\n";
    }
    else if (hh > 12)
    {
        hh -= 12;
        h = std::to_string(hh);
        if (hh < 10)
            h = "0" + h;
        std::cout << h << ':' << m << " PM\n";
    }
    else
    {
        std::cout << h << ':' << m << " AM\n";
    }
}
