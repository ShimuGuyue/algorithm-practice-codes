static constexpr int64_t mod{ 998244353 };

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> edges(n + 1);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        edges[u].push_back(v);
        edges[v].push_back(u);
    }

    int64_t ans{ 1 };
    std::vector<int> colors(n + 1);
    std::queue<int> q;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (colors[i])
            continue;
        colors[i] = 1;
        int count1{ 1 }, count2{ 0 };
        q.push(i);
        while (!q.empty())
        {
            const int u{ q.front() };
            q.pop();
            for (const int v : edges[u])
            {
                if (colors[v] == colors[u])
                {
                    std::cout << 0 << '\n';
                    return;
                }
                if (colors[v] == 0)
                {
                    if (colors[u] == 1)
                    {
                        colors[v] = 2;
                        ++count2;
                    }
                    else
                    {
                        colors[v] = 1;
                        ++count1;
                    }
                    q.push(v);
                }
            }
        }
        ans = (ans * (quickpow(2, count1) + quickpow(2, count2))) % mod;
    }
    std::cout << ans << '\n';
}

int64_t quickpow(int64_t base, int64_t power)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}
