void solve()
{
    int n;
    std::cin >> n;

    int64_t ans{ std::max(calc(n), calc(n - 1)) };
    std::cout << ans << '\n';
}

int64_t calc(int64_t n)
{
    if (n < 3)
        return n;

    int64_t a{ n }, b{ 0 }, c{ 0 };
    while (n--)
    {
        if (!b)
        {
            if (std::gcd(a, n) == 1)
                b = n;
        }
        else if (!c)
        {
            if (std::gcd(a, n) == 1 && std::gcd(b, n) == 1)
                c = n;
        }
        else
        {
            break;
        }
    }
    return std::lcm(a, std::lcm(b, c));
}
