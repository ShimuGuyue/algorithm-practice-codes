void solve()
{
    int n;
    std::cin >> n;
    std::string s1;
    std::string s2;
    std::cin >> s1;
    std::cin >> s2;

    s1 = " " + s1;
    s2 = " " + s2;

    std::vector<int> dp1(n + 1, n);    // |
    std::vector<int> dp2(n + 1, n);    // -
    dp1[0] = dp2[0] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (i >= 1)
        {
            dp1[i] = std::min(dp1[i - 1], dp2[i - 1]);
            if (s1[i] != s2[i])
                ++dp1[i];
        }
        if (i >= 2)
        {
            dp2[i] = std::min(dp1[i - 2], dp2[i - 2]);
            if (s1[i] != s1[i - 1])
                ++dp2[i];
            if (s2[i] != s2[i - 1])
                ++dp2[i];
        }
    }
    int ans{ std::min(dp1[n], dp2[n]) };
    std::cout << ans << '\n';
}
