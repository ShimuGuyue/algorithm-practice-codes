```cpp
std::vector<std::vector<int>> factors;
void init()
{
    factors.resize(100000 + 1);
    for (int i{ 1 }; i <= 100000; ++i)
    {
        for (int j{ 1 }; j * j <= i; ++j)
        {
            if (i % j)
                continue;
            factors[i].push_back(j);
            if (j * j != i)
                factors[i].push_back(i / j);
        }
    }
}
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());
    as.erase(std::unique(as.begin(), as.end()), as.end());
    n = as.size();

    int ans{ 200000 };
    std::vector<int> counts(m + 1);
    int sum{ 0 };
    int l{ 0 }, r{ 0 };
    while (r < n)
    {
        int rr{ as[r] };
        for (int f : factors[rr])
        {
            if (f > m)
                continue;
            ++counts[f];
            if (counts[f] == 1)
                ++sum;
        }
        while (sum == m)
        {
            int ll{ as[l] };
            ans = std::min(ans, rr - ll);
            for (int f : factors[ll])
            {
                if (f > m)
                    continue;
                --counts[f];
                if (counts[f] == 0)
                    --sum;
            }
            ++l;
        }
        ++r;
    }
    if (ans == 200000)
        ans = -1;
    std::cout << ans << '\n';
}
```
