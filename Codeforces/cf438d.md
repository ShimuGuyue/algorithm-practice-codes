struct SegmentTree
{
    struct Data
    {
        int seg_l, seg_r;
        int64_t assign{ 0 };
        int64_t sum{ 0 };
        int64_t max{ 0 };
    };

    std::vector<int>& v;
    std::vector<Data> tree;

    SegmentTree(std::vector<int>& v) : v(v)
    {
        tree.assign(v.size() * 4, { });
        build(0, 0, v.size() - 1);
    }

    void build(int node, int l, int r)
    {
        tree[node].seg_l = l;
        tree[node].seg_r = r;
        if (l == r)
        {
            tree[node].sum = v[l];
            tree[node].max = v[l];
            return;
        }
        int mid{ (l + r) / 2 };
        build(node * 2 + 1, l, mid);
        build(node * 2 + 2, mid + 1, r);
        pushup(node);
    }

    void mod(int l, int r, int64_t data, int node = 0)
    {
        auto& [seg_l, seg_r, assign, sum, max]{ tree[node] };
        if (max < data)
            return;
        if (seg_l == seg_r)
        {
            sum %= data;
            max = sum;
            return;
        }
        pushdown(node);
        int mid{ (seg_l + seg_r) / 2 };
        if (r <= mid)
            mod(l, r, data, node * 2 + 1);
        else if (l > mid)
            mod(l, r, data, node * 2 + 2);
        else
            mod(l, mid, data, node * 2 + 1),
            mod(mid + 1, r, data, node * 2 + 2);
        pushup(node);
    }

    void update(int l, int r, int64_t data, int node = 0)
    {
        auto& [seg_l, seg_r, assign, sum, max]{ tree[node] };
        if (l == seg_l && r == seg_r)
        {
            assign = data;
            sum = data * (seg_r - seg_l + 1);
            max = sum;
            return;
        }
        pushdown(node);
        int mid{ (seg_l + seg_r) / 2 };
        if (r <= mid)
            update(l, r, data, node * 2 + 1);
        else if (l > mid)
            update(l, r, data, node * 2 + 2);
        else
            update(l, mid, data, node * 2 + 1),
            update(mid + 1, r, data, node * 2 + 2);
        pushup(node);
    }

    int64_t query(int l, int r, int node = 0)
    {
        auto& [seg_l, seg_r, assign, sum, max]{ tree[node] };
        if (l == seg_l && r == seg_r)
            return sum;
        pushdown(node);
        int mid{ (seg_l + seg_r) / 2 };
        if (r <= mid)
            return query(l, r, node * 2 + 1);
        else if (l > mid)
            return query(l, r, node * 2 + 2);
        else
            return query(l, mid, node * 2 + 1) +
                   query(mid + 1, r, node * 2 + 2);
    }

    void pushup(int node)
    {
        int node_l{ node * 2 + 1 };
        int node_r{ node * 2 + 2 };
        tree[node].sum = tree[node_l].sum + tree[node_r].sum;
        tree[node].max = std::max(tree[node_l].max, tree[node_r].max);
    }

    void pushdown(int node)
    {
        if (tree[node].assign == 0)
            return;
        for (int i : {1, 2})
        {
            int child{ node * 2 + i };
            auto& [seg_l, seg_r, assign, sum, max]{ tree[child] };
            assign = tree[node].assign;
            max = assign;
            sum = assign * (seg_r - seg_l + 1);
        }
        tree[node].assign = 0;
    }
};

void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    SegmentTree st(as);

    while (m--)
    {
        int op;
        std::cin >> op;
        switch (op)
        {
        case 1:
        {
            int l, r;
            std::cin >> l >> r;
            std::cout << st.query(l, r) << '\n';
            break;
        }
        case 2:
        {
            int l, r, x;
            std::cin >> l >> r >> x;
            st.mod(l, r, x);
            break;
        }
        case 3:
        {
            int k, x;
            std::cin >> k >> x;
            st.update(k, k, x);
            break;
        }
        }
    }
}
