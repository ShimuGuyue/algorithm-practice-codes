void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::map<int, std::vector<int>> m;
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        m[a % k].push_back(a);
    }

    int odd{ -1 };
    for (auto& [e, v] : m)
    {
        if (v.size() % 2 == 0)
            continue;
        if (odd != -1)
        {
            std::cout << -1 << '\n';
            return;
        }
        odd = e;
    }

    int64_t ans{ 0 };
    for (auto& [e, v] : m)
    {
        std::sort(v.begin(), v.end());
        if (odd == e)
        {
            std::vector<int64_t> difs_l(v.size());
            std::vector<int64_t> difs_r(v.size());
            for (int i{ 1 }; i < v.size(); i += 2)
            {
                difs_l[i] = v[i] - v[i - 1];
                if (i > 1)
                    difs_l[i] += difs_l[i - 2];
            }
            for (int i{ int(v.size()) - 2 }; i >= 0; i -= 2)
            {
                difs_r[i] = v[i + 1] - v[i];
                if (i < v.size() - 2)
                    difs_r[i] += difs_r[i + 2];
            }
            int64_t min{ std::numeric_limits<int64_t>::max() };
            for (int i{ 0 }; i < v.size(); i += 2)
            {
                min = std::min(min, (i - 1 >= 0 ? difs_l[i - 1] : 0) + (i + 1 < v.size() ? difs_r[i + 1] : 0));
            }
            ans += min;
        }
        else
        {
            for (int i{ 1 }; i < v.size(); i += 2)
            {
                ans += v[i] - v[i - 1];
            }
        }
    }
    ans /= k;
    std::cout << ans << '\n';
}
