void solve()
{
    constexpr int64_t mod{ 1000000007 };

    int n, m;
    std::cin >> n >> m;

    std::vector<std::vector<int64_t>> dp_a(m + 1, std::vector<int64_t>(n + 1));
    std::vector<std::vector<int64_t>> dp_b(m + 1, std::vector<int64_t>(n + 1));
    for (int i = 1; i <= n; ++i)
    {
        dp_a[1][i] = 1;
        dp_b[1][i] = 1;
    }
    for (int i = 2; i <= m; ++i)
    {
        for (int j = 1; j <= n; ++j)
        {
            for (int k = 1; k <= n; ++k)
            {
                if (k <= j)
                    dp_a[i][j] = (dp_a[i][j] + dp_a[i - 1][k]) % mod;
                if (k >= j)
                    dp_b[i][j] = (dp_b[i][j] + dp_b[i - 1][k]) % mod;
            }
        }
    }

    int64_t ans{ 0 };
    for (int i = 1; i <= n; ++i)
    {
        for (int j = i; j <= n; ++j)
        {
            ans = (ans + dp_a[m][i] * dp_b[m][j] % mod) % mod;
        }
    }
    std::cout << ans << '\n';
}
