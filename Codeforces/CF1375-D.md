void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> counts(n + 1);
    for (auto a : as)
    {
        ++counts[a];
    }

    std::vector<int> ans;
    while (1)
    {
        int mex{ 0 };
        while (counts[mex])
        {
            ++mex;
        }
        bool end{ true };
        if (mex == n)
        {
            for (int i{ 0 }; i < n; ++i)
            {
                if (as[i] == i)
                    continue;
                --counts[as[i]];
                ++counts[mex];
                as[i] = mex;
                ans.push_back(i + 1);
                end = false;
                break;
            }
        }
        else
        {
            --counts[as[mex]];
            ++counts[mex];
            as[mex] = mex;
            ans.push_back(mex + 1);
            end = false;
        }
        if (end)
            break;
    }

    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
