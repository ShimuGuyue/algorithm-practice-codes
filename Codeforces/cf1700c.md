void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int64_t> difs(n);
    std::adjacent_difference(as.begin(), as.end(), difs.begin());
    int64_t ans{ 0 };
    for (int i{ 1 }; i < n; ++i)
    {
        ans += std::abs(difs[i]);
        if (difs[i] < 0)
            difs[0] += difs[i];
    }
    ans += std::abs(difs[0]);
    std::cout << ans << '\n';
}
