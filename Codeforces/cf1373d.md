void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t sum{ 0 };
    for (int i{ 0 }; i < n; i += 2)
    {
        sum += as[i];
    }

    int64_t ans{ sum };

    std::vector<int64_t> bs;
    std::vector<int64_t> cs;
    for (int i{ 0 }; i + 1 < n; i += 2)
    {
        bs.push_back(as[i + 1] - as[i]);
    }
    for (int i{ 1 }; i +1 < n; i += 2)
    {
        cs.push_back(as[i] - as[i + 1]);
    }
    auto calc = [sum, &ans](std::vector<int64_t>& v)
    {
        int64_t max{ 0 };
        int64_t cur{ 0 };
        for (auto a : v)
        {
            if (cur + a < 0)
                cur = 0;
            else
                cur += a;
            max = std::max(max, cur);
        }
        ans = std::max(ans, sum + max);
    };
    calc(bs);
    calc(cs);
    std::cout << ans << '\n';
}
