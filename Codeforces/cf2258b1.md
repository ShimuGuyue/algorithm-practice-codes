void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> counts(m + 1);
    std::vector<int> precounts(m + 1);
    for (auto a : as)
    {
        ++counts[a];
    }
    std::partial_sum(counts.begin(), counts.end(), precounts.begin());

    int ans{ 0 };
    for (int i{ 1 }; i <= m; ++i)
    {
        int sum{ precounts[m] - precounts[i - 1] };
        if (i * 2 <= m)
            sum += counts[i * 2];
        ans = std::max(ans, sum);
    }
    std::cout << ans << '\n';
}
