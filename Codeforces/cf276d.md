void solve()
{
    int64_t l, r;
    std::cin >> l >> r;

    if (l == r)
    {
        std::cout << 0 << '\n';
        return;
    }

    for (int b = 59; b >= 0; --b)
    {
        if ((l >> b & 1) != (r >> b & 1))
        {
            std::cout << (INT64_C(1) << (b + 1)) - 1 << '\n';
            return;
        }
    }
}
