void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    std::map<int, int> sufsums;
    for (auto a : as)
    {
        ++sufsums[a];
    }
    for (auto it = std::next(sufsums.rbegin()); it != sufsums.rend(); ++it)
    {
        it->second += std::prev(it)->second;
    }

    std::vector<int> pres(n);
    std::partial_sum(bs.begin(), bs.end(), pres.begin());

    int64_t ans{ 0 };
    for (auto [x, count] : sufsums)
    {
        if (count < pres.front())
            continue;
        auto loc{ std::prev(std::upper_bound(pres.begin(), pres.end(), count)) };
        ans = std::max(ans, x * int64_t(loc - pres.begin() + 1));
    }
    std::cout << ans << '\n';
}
