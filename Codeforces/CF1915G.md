void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<std::pair<int, int64_t>>> graph(n + 1);
    while (m--)
    {
        int u, v, w;
        std::cin >> u >> v >> w;
        graph[u].push_back({ v, w });
        graph[v].push_back({ u, w });
    }
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<int64_t>> ans(n + 1, std::vector<int64_t>(1000 + 1, INT64_C(1) << 60));
    std::queue<std::array<int64_t, 3>> q;   // u, s. len
    q.push({ 1, as[1], 0 });
    ans[1][as[1]] = 0;
    while (!q.empty())
    {
        auto [u, s, len]{ q.front() };
        q.pop();

        s = std::min(s, as[u]);
        for (auto [v, w] : graph[u])
        {
            if (len + s * w < ans[v][s])
            {
                ans[v][s] = len + s * w;
                q.push({v, s, len + s * w});
            }
        }
    }
    std::cout << *std::min_element(ans[n].begin(), ans[n].end()) << '\n';
}
