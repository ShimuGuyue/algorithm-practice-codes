void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int sum{ std::accumulate(as.begin(), as.end(), 0) };
    int max{ *std::max_element(as.begin(), as.end()) };
    if (max > sum / 2)
    {
        std::cout << "T\n";
        return;
    }

    if (sum & 1)
        std::cout << "T\n";
    else
        std::cout << "HL\n";
}
