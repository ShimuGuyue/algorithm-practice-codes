void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    int max{ 1 };
    int count{ 1 };
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] - as[i - 1] <= k)
            ++count;
        else
            count = 1;
        max = std::max(max, count);
    }

    int ans{ n - max };
    std::cout << ans << '\n';
}
