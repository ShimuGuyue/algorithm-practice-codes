void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> grid(n + 1, std::vector<int>(m + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            std::cin >> grid[i][j];
        }
    }

    if ((n + m) % 2 == 0)
    {
        std::cout << "No\n";
        return;
    }

    std::vector<std::vector<int>> dp_max(n + 1, std::vector<int>(m + 1, -2000000));
    std::vector<std::vector<int>> dp_min(n + 1, std::vector<int>(m + 1, 2000000));
    dp_max[1][1] = grid[1][1];
    dp_min[1][1] = grid[1][1];
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            if (i != 1)
            {
                dp_max[i][j] = std::max(dp_max[i][j], dp_max[i - 1][j] + grid[i][j]);
                dp_min[i][j] = std::min(dp_min[i][j], dp_min[i - 1][j] + grid[i][j]);
            }
            if (j != 1)
            {
                dp_max[i][j] = std::max(dp_max[i][j], dp_max[i][j - 1] + grid[i][j]);
                dp_min[i][j] = std::min(dp_min[i][j], dp_min[i][j - 1] + grid[i][j]);
            }
        }
    }

    int max{ dp_max[n][m] };
    int min{ dp_min[n][m] };
    if (min > 0 || max < 0)
        std::cout << "No\n";
    else
        std::cout << "Yes\n";
}
