void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> pres(n + 1);
    std::partial_sum(as.begin(), as.end(), pres.begin(), std::bit_xor<int>());

    using Query = std::array<int, 3>;
    std::vector<Query> querys(m);
    for (int i{ 0 }; i < m; ++i)
    {
        std::cin >> querys[i][0] >> querys[i][1];
        --querys[i][0];
        querys[i][2] = i;
    }
    int block{ int(std::sqrt(n)) };
    std::sort(querys.begin(), querys.end(),
        [block](Query& a, Query& b)
        {
            int b1{ a[0] / block };
            int b2{ b[0] / block };
            if (b1 != b2)
                return b1 < b2;
            return (b1 & 1) ? (a[1] > b[1]) : (a[1] < b[1]);
        }
    );

    int64_t cur{ 0 };
    int l{ 0 }, r{ -1 };
    std::vector<int> counts(1 << 20);
    auto add = [&cur, &pres, &counts, k](int index) -> void
    {
        int sum{ pres[index] };
        cur += counts[sum ^ k];
        ++counts[sum];
    };
    auto del = [&cur, &pres, &counts, k](int index) -> void
    {
        int sum{ pres[index] };
        --counts[sum];
        cur -= counts[sum ^ k];
    };

    std::vector<int64_t> ans(m);
    for (auto [left, right, index] : querys)
    {
        while (l > left)
        {
            add(--l);
        }
        while (r < right)
        {
            add(++r);
        }
        while (l < left)
        {
            del(l++);
        }
        while (r > right)
        {
            del(r--);
        }
        ans[index] = cur;
    }

    for (auto a : ans)
    {
        std::cout << a << '\n';
    }
}
