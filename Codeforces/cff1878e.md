void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    SparseTable st(as);

    int q;
    std::cin >> q;
    while (q--)
    {
        int l, k;
        std::cin >> l >> k;
        int ll{ l }, rr{ n };
        while (ll < rr)
        {
            int mid{ (ll + rr + 1) / 2 };
            if (st.query(l, mid) >= k)
                ll = mid;
            else
                rr = mid - 1;
        }
        if (st.query(l, ll) < k)
            std::cout << -1 << ' ';
        else
            std::cout << ll << ' ';
    }
    std::cout << '\n';
}
