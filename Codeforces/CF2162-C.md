void solve()
{
    int a, b;
    std::cin >> a >> b;

    int flag{ 1 };
    while (flag <= b)
    {
        if (flag > a)
        {
            std::cout << "-1\n";
            return;
        }
        flag *= 2;
    }

    std::vector<int> ans;
    for (int i = 0; i < 30; ++i)
    {
        if ((a & (1 << i)) != (b & (1 << i)) && (1 << i) <= a)
            ans.push_back(1 << i);
    }
    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
