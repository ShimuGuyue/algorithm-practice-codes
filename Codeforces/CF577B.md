void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    if (n >= m)
    {
        std::cout << "YES\n";
        return;
    }

    std::set<int> t;
    for (int i = 1; i <= n; ++i)
    {
        std::set<int> temp(t);
        temp.insert(as[i] % m);
        for (int a : t)
        {
            temp.insert((a + as[i]) % m);
        }
        t = temp;
    }

    if (t.find(0) != t.end())
        std::cout << "YES\n";
    else
        std::cout << "NO\n";
}
