```cpp
static constexpr int64_t mod{ 1000000007 };
std::vector<int64_t> factorials;
std::vector<int64_t> inv_factorials;

void init()
{
    int m{ 1000000 };
    factorials.assign(m + 1, 1);
    inv_factorials.assign(m + 1, 1);
    for (int i{ 2 }; i <= m; ++i)
    {
        factorials[i] = factorials[i - 1] * i % mod;
    }
    inv_factorials[m] = power(factorials[m], mod - 2);
    for (int i{ m - 1 }; i >= 1; --i)
    {
        inv_factorials[i] = inv_factorials[i + 1] * (i + 1) % mod;
    }
}
```

```cpp
void solve()
{
    int a, b, n;
    std::cin >> a >> b >> n;

    int64_t ans{ 0 };
    for (int i{ 0 }; i <= n; ++i)
    {
        int flag{ a * i + b * (n - i) };
        bool ok{ true };
        while (flag)
        {
            if (flag % 10 != a && flag % 10 != b)
                ok = false;
            flag /= 10;
        }
        if (ok)
            ans = (ans + combinate(n, i)) % mod;
    }
    std::cout << ans << '\n';
}

int64_t combinate(int n, int k)
{
    if (k == n || k == 0)
        return 1;
    return factorials[n] * inv_factorials[k] % mod * inv_factorials[n - k] % mod;
}

int64_t power(int64_t base, int power)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
};
```
