void solve()
{
    int n;
    std::cin >> n;
    int count2{ 0 }, count_more{ 0 };
    while (n--)
    {
        int a;
        std::cin >> a;
        if (a == 2)
            ++count2;
        if (a > 2)
            ++count_more;
    }
    if (count2 >= 2 || count_more)
        std::cout << "Yes\n";
    else
        std::cout << "No\n";
}
