void solve()
{
    int n;
    cin >> n;
    vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        cin >> as[i];
    }

    int64_t ans = 0;
    map<int, int> m;
    ++m[-as[n    ] + as[n - 2] + as[n - 4]];
    ++m[-as[n - 1] + as[n - 3] + as[n - 5]];
    if (m.size() == 1)
        ++ans;

    for (int i = n - 2; i - 4 > 0; --i)
    {
        int cur = -as[i] + as[i - 2] + as[i - 4];
        ans += m[cur];
        if (i + 2 <= n && as[i - 2] + as[i] - as[i + 2] == cur)
            --ans;
        if (i + 4 <= n && as[i] + as[i + 2] - as[i + 4] == cur)
            --ans;

        ++m[cur];
    }
    cout << ans << '\n';
}
