void solve()
{
    std::string s;
    std::cin >> s;

    int n{ int(s.length()) };
    if (n < 3)
    {
        std::cout << "Just a legend\n";
        return;
    }

    std::vector<int> borders(n);
    for (int i = 1; i < n; ++i)
    {
        borders[i] = borders[i - 1];
        while (borders[i] > 0 && s[borders[i]] != s[i])
        {
            borders[i] = borders[borders[i] - 1];
        }
        if (s[borders[i]] == s[i])
            ++borders[i];
    }

    std::set<int> t(borders.begin(), borders.end() - 1);
    int x{ n - 1 };
    while (borders[x])
    {
        if (t.find(borders[x]) != t.end())
        {
            std::string str(s.begin(), s.begin() + borders[x]);
            std::cout << str << '\n';
            return;
        }
        x = borders[x] - 1;
    }
    std::cout << "Just a legend\n";
}
