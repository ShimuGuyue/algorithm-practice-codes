void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::sort(as.begin() + 1, as.end());

    std::vector<std::vector<int>> dp(n * 2 + 1, std::vector<int>(n + 1, 100000));
    for (int i{ 0 }; i <= n * 2; ++i)
    {
        dp[i][0] = 0;
    }
    for (int i{ 1 }; i <= n * 2; ++i)
    {
        for (int j{ 1 }; j <= std::min(i, n); ++j)
        {
            dp[i][j] = std::min(dp[i - 1][j], dp[i - 1][j - 1] + std::abs(i - as[j]));
        }
    }
    int ans{ 100000 };
    for (int i{ n }; i <= n * 2; ++i)
    {
        ans = std::min(ans, dp[i][n]);
    }
    std::cout << ans << '\n';
}
