void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::pair<int, int>> degrees(n + 1);
    std::set<std::pair<int, int>> edges;
    for (int i{ 1 }; i <= n; ++i)
    {
        degrees[i].second = i;
        if (i == n)
            break;
        int u, v;
        std::cin >> u >> v;
        ++degrees[u].first;
        ++degrees[v].first;
        auto [min, max]{ std::minmax(u, v) };
        edges.insert({min, max});
    }

    if (n == 2)
    {
        std::cout << 0 << '\n';
        return;
    }

    std::sort(degrees.begin() + 1, degrees.end(),
        [](std::pair<int, int> a, std::pair<int, int> b)
        {
            return a.first > b.first;
        }
    );
    int ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ i + 1 }; j <= n; ++j)
        {
            auto [min, max]{ std::minmax(degrees[i].second, degrees[j].second) };
            int cur;
            if (edges.count({min, max}))
                cur = degrees[i].first + degrees[j].first - 2;
            else
                cur = degrees[i].first + degrees[j].first - 1;
            if  (cur <= ans - 2)
                break;
            ans = std::max(ans, cur);
            if (!edges.count({min, max}))
                break;
        }
    }
    std::cout << ans << '\n';
}
