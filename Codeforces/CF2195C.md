void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    auto match = [](int a, int b)
    {
        return a + b != 7 && a != b;
    };

    std::vector<std::vector<int>> dp(n + 1, (std::vector<int>(7, n)));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= 6; ++j)
        {
            if (i == 1)
            {
                if (j != as[i])
                    dp[i][j] = 1;
                else
                    dp[i][j] = 0;
                continue;
            }

            for (int k{ 1 }; k <= 6; ++k)
            {
                if (!match(j, k))
                    continue;
                if (j == as[i])
                    dp[i][j] = std::min(dp[i][j], dp[i - 1][k]);
                else
                    dp[i][j] = std::min(dp[i][j], dp[i - 1][k] + 1);
            }
        }
    }
    int min{ *std::min_element(dp[n].begin() + 1, dp[n].end()) };
    std::cout << min << '\n';
}
