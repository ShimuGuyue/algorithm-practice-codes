void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::map<int64_t, std::vector<int64_t>> vs;
    for (int i{ 0 }; i < n; ++i)
    {
        vs[as[i]].push_back(i);
    }

    std::vector<int64_t> dp(n);
    for (int i{ 0 }; i < n; ++i)
    {
        auto& v{ vs[as[i] * k] };
        dp[i] = std::distance(std::upper_bound(v.begin(), v.end(), i), v.end());
    }
    debugln(dp);

    std::vector<int64_t> sufdps(n);
    for (auto& [a, v] : vs)
    {
        int64_t m{ int(v.size()) };
        for (int64_t i{ m - 1 }; i >= 0; --i)
        {
            int64_t index{ v[i] };
            sufdps[index] = dp[index];
            if (i == m - 1)
                continue;
            int64_t sufindex{ v[i + 1] };
            sufdps[index] += sufdps[sufindex];
        }
    }

    int64_t ans{ 0 };
    for (int64_t i{ 0 }; i < n; ++i)
    {
        auto& v{ vs[as[i] * k] };
        auto jt{ std::upper_bound(v.begin(), v.end(), i) };
        if (jt == v.end())
            continue;
        ans += sufdps[*jt];
    }
    std::cout << ans << '\n';
}
