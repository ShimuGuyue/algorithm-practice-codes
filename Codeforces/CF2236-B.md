void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    for (int i{ 0 }; i + k < n; ++i)
    {
        if (s[i] == '0')
            continue;
        s[i] = '0';
        s[i + k] = s[i + k] == '1' ? '0' : '1';
    }

    if (std::count(s.begin(), s.end(), '1') == 0)
        std::cout << "Yes\n";
    else
        std::cout << "No\n";
}
