void solve()
{
    int n, u, r, d, l;
    std::cin >> n >> u >> r >> d >> l;
    for (int bit = 0b0000; bit <= 0b1111; ++bit)
    {
        int uu{ u }, rr{ r }, dd{ d }, ll{ l }; // 去除四角的最大值
        if ((bit & 0b0001) == 0b0001)
            --uu, --ll;
        if ((bit & 0b0010) == 0b0010)
            --uu, --rr;
        if ((bit & 0b0100) == 0b0100)
            --dd, --rr;
        if ((bit & 0b1000) == 0b1000)
            --dd, --ll;
        if (uu < 0 || rr < 0 || dd < 0 || ll < 0)
            continue;
        if (uu <= n - 2 && rr <= n - 2 && dd <= n - 2 && ll <= n - 2)
        {
            std::cout << "yes\n";
            return;
        }
    }
    std::cout << "no\n";
}
