void solve()
{
    int64_t n, c, d;
    std::cin >> n >> c >> d;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.rbegin(), as.rend());

    if (as.front() * d < c)
    {
        std::cout << "Impossible\n";
        return;
    }
    if (std::accumulate(as.begin(), std::min(as.begin() + d, as.end()), INT64_C(0)) >= c)
    {
        std::cout << "Infinity\n";
        return;
    }

    auto check = [&as, c, d](int64_t k)
    {
        int64_t count{ d / (k + 1) };
        int64_t mod{ d % (k + 1) };
        int64_t sum{ 0 };
        for (int i{ 0 }; i <= std::min<int64_t>(k, as.size() - 1); ++i)
        {
            if (i < mod)
                sum += (count + 1) * as[i];
            else
                sum += count * as[i];
            if (sum >= c)
                return true;
        }
        return false;
    };
    int64_t l{ 0 }, r{ d };
    while (l < r)
    {
        int64_t mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    std::cout << l << '\n';
}
