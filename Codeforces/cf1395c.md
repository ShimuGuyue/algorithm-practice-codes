void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n);
    std::vector<int> bs(m);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    for (int ans = 0; ans < (1 << 9); ++ans)
    {
        bool find{ true };
        for (int a : as)
        {
            bool ok{ false };
            for (int b : bs)
            {
                if ((a & b | ans) == ans)
                    ok = true;
            }
            if (!ok)
            {
                find = false;
                break;
            }
        }
        if (find)
        {
            std::cout << ans << '\n';
            return;
        }
    }
}
