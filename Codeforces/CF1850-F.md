void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::map<int, int> m;
    for (auto a : as)
    {
        ++m[a];
    }

    std::vector<int> counts(n + 1);
    for (auto [k, v] : m)
    {
        for (int i{ k }; i <= n; i += k)
        {
            counts[i] += v;
        }
    }

    int ans{ *std::max_element(counts.begin(), counts.end()) };
    std::cout << ans << '\n';
}
