void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (n == 1)
    {
        std::cout << 0 << '\n';
        return;
    }

    int x{ std::accumulate(as.begin(), as.end(), 0, std::bit_xor<int>()) };

    int ans{ x ? 0 : 1 };
    for (auto a : as)
    {
        if ((x ^ a) < a)
            ++ans;
    }
    std::cout << ans << '\n';
}
