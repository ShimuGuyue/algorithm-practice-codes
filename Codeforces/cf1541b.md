void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int val{ as[i] }; val < n * 2; val += as[i])
        {
            int j{ val - i };
            if (j > i && j <= n && as[j] * as[i] == val)
                ++ans;
        }
    }
    std::cout << ans << '\n';
}
