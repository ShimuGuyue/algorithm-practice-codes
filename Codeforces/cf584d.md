void solve()
{
    int n;
    std::cin >> n;

    if (isprime(n))
    {
        std::cout << 1 << '\n';
        std::cout << n << '\n';
    }
    else if (isprime(n - 2))
    {
        std::cout << 2 << '\n';
        std::cout << n - 2 << ' ' << 2 << '\n';
    }
    else if (isprime(n -  4))
    {
        std::cout << 3 << '\n';
        std::cout << n - 4 << ' ' << 2 << ' ' << 2 << '\n';
    }
    else
    {
        std::cout << 3 << '\n';
        for (int x{ n }; x > 0; --x)
        {
            if (isprime(x))
            {
                std::cout << x << ' ';
                n -= x;
                break;
            }
        }
        for (int i{ 3 }; i < n; i += 2)
        {
            if (isprime(i) && isprime(n - i))
            {
                std::cout << i << ' ' << n - i << '\n';
                break;
            }
        }
    }
}

bool isprime(int n)
{
    if (n == 1)
        return false;
    for (int i{ 2 }; i * i <= n; ++i)
    {
        if (n % i == 0)
            return false;
    }
    return true;
}
