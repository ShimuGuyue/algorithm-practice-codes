void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n + 1);
    std::vector<int> bs(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> bs[i];
    }

    k %= 2;

    std::vector<int64_t> pre_maxsum(n + 2);
    std::vector<int64_t> suf_maxsum(n + 2);
    pre_maxsum[1] = as[1];
    for (int i = 2; i <= n; ++i)
    {
        pre_maxsum[i] = std::max(int64_t(as[i]), as[i] + pre_maxsum[i - 1]);
    }
    suf_maxsum[n] = as[n];
    for (int i = n - 1; i >= 1; --i)
    {
        suf_maxsum[i] = std::max(int64_t(as[i]), as[i] + suf_maxsum[i + 1]);
    }

    if (k == 0)
    {
        int64_t ans = *std::max_element(pre_maxsum.begin() + 1, pre_maxsum.begin() + n + 1);
        std::cout << ans << '\n';
        return;
    }

    int64_t ans{ INT64_MIN };
    for (int i = 1; i <= n; ++i)
    {
        int64_t temp = as[i] + llabs(bs[i]) + std::max(pre_maxsum[i - 1], INT64_C(0)) + std::max(suf_maxsum[i + 1], INT64_C(0));
        ans = std::max(ans, temp);
    }
    std::cout << ans << '\n';
}
