void solve()
{
    int n, a, b;
    std::cin >> n >> a >> b;

    std::vector<std::vector<int>> grid(n + 1);
    for (int i{ 0 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        grid[u].push_back(v);
        grid[v].push_back(u);
    }

    std::vector<int> degrees(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        degrees[i] = grid[i].size();
    }

    std::queue<int> q;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (degrees[i] == 1)
            q.push(i);
    }

    while (!q.empty())
    {
        int u{ q.front() };
        q.pop();
        for (int v : grid[u])
        {
            if (--degrees[v] == 1)
            {
                q.push(v);
            }
        }
    }

    auto [node, len1] = [&grid, &degrees, b] -> std::pair<int, int>
    {
        std::queue<std::array<int, 3>> q;
        q.push({b, 0, 0});
        while (!q.empty())
        {
            auto [u, len, last]{ q.front() };
            q.pop();
            if (degrees[u] > 1)
                return {u, len};
            for (int v : grid[u])
            {
                if (v == last)
                    continue;
                q.push({v, len + 1, u});
            }
        }
        return {0, 0};
    }();

    int len2 = [&grid, node, a]() -> int
    {
        std::queue<std::array<int, 3>> q;
        q.push({node, 0, 0});
        while (!q.empty())
        {
            auto [u, len, last]{ q.front() };
            q.pop();
            if (u == a)
                return len;
            for (int v : grid[u])
            {
                if (v == last)
                    continue;
                q.push({v, len + 1, u});
            }
        }
        return -1;
    }();

    if (len1 < len2)
        std::cout << "YES\n";
    else
        std::cout << "NO\n";
}
