void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t ans_nega{ 0 }, ans_posi{ 0 };
    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(2));
    for (int i = 1; i <= n; ++i)
    {
        if (as[i] > 0)
        {
            ++dp[i][1];
            dp[i][1] += dp[i - 1][1];
            dp[i][0] += dp[i - 1][0];
        }
        else
        {
            ++dp[i][0];
            dp[i][1] += dp[i - 1][0];
            dp[i][0] += dp[i - 1][1];
        }
        ans_nega += dp[i][0];
        ans_posi += dp[i][1];
    }
    std::cout << ans_nega << ' ' << ans_posi << '\n';
}
