void solve()
{
    int n, l, r, x;
    std::cin >> n >> l >> r >> x;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 0 };
    auto dfs = [n, l, r, x, &as, &ans](auto&& dfs, int index, int min, int max, int64_t sum, int count)
    {
        if (index == n)
        {
            if (sum >= l && sum <= r && max - min >= x && count >= 2)
                ++ans;
            return;
        }
        dfs(dfs, index + 1, min, max, sum, count);
        min = std::min(min, as[index]);
        max = std::max(max, as[index]);
        sum = sum + as[index];
        count = count + 1;
        if (sum > r)
            return;
        dfs(dfs, index + 1, min, max, sum, count);
    };
    dfs(dfs, 0, INT_MAX, INT_MIN, 0, 0);
    std::cout << ans << '\n';
}
