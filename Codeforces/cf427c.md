void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    int m;
    std::cin >> m;
    std::vector<std::vector<int>> graph(n + 1);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        graph[u].push_back(v);
    }

    int timer{ 1 };
    std::vector<std::vector<int>> sccs;
    std::stack<int> k;
    std::vector<bool> in_stacks(n + 1);
    std::vector<int> dfns(n + 1);
    std::vector<int> lows(n + 1);
    auto dfs = [&graph, &sccs, &k, &in_stacks, &dfns, &lows, &timer](this auto& dfs, int u) -> void
    {
        in_stacks[u] = true;
        dfns[u] = lows[u] = timer++;
        k.push(u);
        for (int v : graph[u])
        {
            if (dfns[v])
            {
                if (!in_stacks[v])
                    continue;
                lows[u] = std::min(lows[u], lows[v]);
            }
            else
            {
                dfs(v);
                lows[u] = std::min(lows[u], lows[v]);
            }
        }
        if (dfns[u] != lows[u])
            return;
        sccs.emplace_back();
        while (!k.empty())
        {
            int n{ k.top() };
            k.pop();
            in_stacks[n] = false;
            sccs.back().push_back(n);
            if (n == u)
                break;
        }
    };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (!dfns[i])
            dfs(i);
    }

    int64_t mod{ 1000000007 };
    int64_t sum{ 0 };
    int64_t count{ 1 };
    for (auto& v : sccs)
    {
        int min{ std::numeric_limits<int>::max() };
        for (auto i : v)
        {
            min = std::min(min, as[i]);
        }
        sum += min;
        int c{ 0 };
        for (auto i : v)
        {
            if (as[i] == min)
                ++c;
        }
        count = count * c % mod;
    }
    std::cout << sum << ' ' << count << '\n';
}
