void solve()
{
    constexpr int mod{ 1000000007 };

    constexpr int n{ 100000 };

    int t, k;
    std::cin >> t >> k;
    std::array<int64_t, n + 1> dp;
    dp[0] = 1;
    for (int i = 1; i <= n; ++i)
    {
        dp[i] = dp[i - 1];
        if (i >= k)
            dp[i] += dp[i - k];
        dp[i] %= mod;
    }

    std::array<int64_t, n + 1> presums;
    std::partial_sum(dp.begin(), dp.end(), presums.begin(),
        [](int64_t sum, int64_t next)
        {
            return (sum + next) % mod;
        }
    );

    while (t--)
    {
        int a, b;
        std::cin >> a >> b;
        int64_t ans{ (presums[b] - presums[a - 1] + mod) % mod };
        std::cout << ans << '\n';
    }
}
