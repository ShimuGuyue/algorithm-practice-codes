void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    bool b{ false };
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] < as[i - 1])
            b = true;
    }
    int ans{ b ? 1 : n };
    std::cout << ans << '\n';
}
