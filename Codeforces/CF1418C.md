void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(2, n));
    dp[0][0] = 0;
    for (int i = 1; i <= n; ++i)
    {
        dp[i][1] = dp[i - 1][0] + as[i];
        if (i > 1)
            dp[i][1] = std::min(dp[i][1], dp[i - 2][0] + as[i - 1] + as[i]);
        dp[i][0] = dp[i - 1][1];
        if (i > 1)
            dp[i][0] = std::min(dp[i - 1][1], dp[i - 2][1]);
    }
    int ans{ std::min(dp[n][0], dp[n][1]) };
    std::cout << ans << '\n';
}
