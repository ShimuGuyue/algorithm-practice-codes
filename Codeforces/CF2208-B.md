void solve()
{
    int n, k, p, m;
    std::cin >> n >> k >> p >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    {
        std::priority_queue<int, std::vector<int>, std::greater<int>> h;
        for (int i{ 1 }; i < p; ++i)
        {
            h.push(as[i]);
        }
        int dif{ std::max(p - k, 0) };
        while (dif--)
        {
            m -= h.top();
            h.pop();
        }
        m -= as[p];
        if (m < 0)
        {
            std::cout << 0 << '\n';
            return;
        }
    }

    int ans{ 1 };
    std::priority_queue<int, std::vector<int>, std::greater<int>> h;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (i == p)
            continue;
        h.push(as[i]);
    }
    int dif{ std::max(n - k, 0) };
    int sum{ 0 };
    while (dif--)
    {
        sum += h.top();
        h.pop();
    }
    sum += as[p];
    ans += m / sum;
    std::cout << ans << '\n';
}
