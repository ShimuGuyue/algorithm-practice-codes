        std::cout << "No\n";
        return;
    }
    if (n % 2 == 1 && k % 2 == 1)
    {
        if (k > n)
        {
            std::cout << "No\n";
            return;
        }
        std::cout << "Yes\n";
        for (int i = 1; i < k; ++i)
        {
            std::cout << 1 << ' ';
        }
        std::cout << n - (k - 1) << '\n';
    }
    if (n % 2 == 0 && k % 2 == 0)
    {
        if (k > n)
        {
            std::cout << "No\n";

        std::cout << "Yes\n";
        for (int i = 1; i < k; ++i)
        {
            std::cout << 1 << ' ';
        }
        std::cout << n - (k - 1) << '\n';
    }
    if (n % 2 == 0 && k % 2 == 1)
    {
        if (k * 2 > n)
        {
            std::cout << "No\n";
            return;

        for (int i = 1; i < k; ++i)
        {
            std::cout << 2 << ' ';
        }
        std::cout << n - 2 * (k - 1) << '\n';
    }
}
