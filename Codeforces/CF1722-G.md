void solve()
{
    int n;
    std::cin >> n;
    int x{ 0 };
    std::vector<int> as(n);
    for (int i{ 0 }; i < n - 2; ++i)
    {
        as[i] = i;
        x ^= i;
    }
    if (x == 0)
    {
        x ^= as[n - 3];
        as[n - 3] = 1 << 29;
        x ^= 1 << 29;
    }
    as[n - 2] = 1 << 30;
    x ^= 1 << 30;
    as[n - 1] = x;
    for (auto a : as)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
