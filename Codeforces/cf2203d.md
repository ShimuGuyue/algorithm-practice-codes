void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n);
    std::vector<int> bs(m);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    std::sort(as.begin(), as.end());
    as.erase(std::unique(as.begin(), as.end()), as.end());

    int max{ *std::max_element(bs.begin(), bs.end()) };

    int64_t lcm{ 1 };
    std::vector<bool> flags(max + 1);
    for (auto a : as)
    {
        for (int x{ a }; x <= max; x += a)
        {
            flags[x] = true;
        }
        if (lcm > max)
            continue;
        lcm = std::lcm(lcm, a);
    }

    int alice{ 0 }, bob{ 0 }, both{ 0 };
    for (auto b : bs)
    {
        if (b % lcm == 0)
        {
            ++alice;
            continue;
        }
        if (flags[b])
            ++both;
        else
            ++bob;
    }

    both %= 2;
    alice += both;
    if (alice > bob)
        std::cout << "Alice\n";
    else
        std::cout << "Bob\n";
}
