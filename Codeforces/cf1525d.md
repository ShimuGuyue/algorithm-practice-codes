void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> indexs(1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i])
            indexs.push_back(i);
    }
    int m{ int(indexs.size()) };

    std::vector<std::vector<int>> dp(m, std::vector<int>(n + 1, 1000000000));
    for (int i{ 0 }; i <= n; ++i)
    {
        dp[0][i] = 0;
    }
    for (int i{ 1 }; i < m; ++i)
    {
        for (int j{ 1 }; j <= n; ++j)
        {
            if (as[j] == 1)
                dp[i][j] = dp[i][j - 1];
            else
                dp[i][j] = std::min(dp[i][j - 1], dp[i - 1][j - 1] + std::abs(indexs[i] - j));
        }
    }
    std::cout << dp[m - 1][n] << '\n';
}
