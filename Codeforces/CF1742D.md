void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> last_indexs(1001);
    for (int i{ 1 }; i <= n; ++i)
    {
        int a;
        std::cin >> a;
        last_indexs[a] = i;
    }

    int ans{ -1 };
    for (int i{ 1000 }; i > 0; --i)
    {
        if (last_indexs[i] == 0)
            continue;
        for (int j{ i }; j > 0; --j)
        {
            if (last_indexs[j] == 0)
                continue;
            if (std::gcd(i, j) != 1)
                continue;
            ans = std::max(ans, last_indexs[i] + last_indexs[j]);
        }
    }
    std::cout << ans << '\n';
}
