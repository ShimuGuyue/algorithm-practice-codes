void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::string> vs(n);
    for (auto& s : vs)
    {
        std::cin >> s;
    }

    int m{ 26 };
    std::vector<int> counts(m);
    std::vector<std::vector<int>> nexts(m);
    for (int i = 1; i < n; ++i)
    {
        bool is_suf{ true };
        for (int j = 0; j < std::min(vs[i - 1].length(), vs[i].length()); ++j)
        {
            if (vs[i - 1][j] == vs[i][j])
                continue;
            nexts[vs[i - 1][j] - 'a'].push_back(vs[i][j] - 'a');
            ++counts[vs[i][j] - 'a'];
            is_suf = false;
            break;
        }
        if (is_suf && vs[i - 1].length() > vs[i].length())
        {
            std::cout << "Impossible\n";
            return;
        }
    }
    debug_(counts);

    std::vector<char> ans;
    std::queue<int> q;
    for (int i = 0; i < 26; ++i)
    {
        if (counts[i] == 0)
            q.push(i);
    }
    while (!q.empty())
    {
        int t{ q.front() };
        q.pop();
        ans.push_back(t + 'a');
        for (int i : nexts[t])
        {
            if (--counts[i] == 0)
                q.push(i);
        }
    }
    for (int i = 0; i < 26; ++i)
    {
        if (counts[i])
        {
            std::cout << "Impossible\n";
            return;
        }
    }
    for (auto a : ans)
    {
        std::cout << a;
    }
    std::cout << '\n';
}
