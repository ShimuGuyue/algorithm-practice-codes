```cpp
std::vector<int> turns;
int max;
void init()
{
    constexpr int n{ 1000 };
    turns.assign(n + 1, 1000);
    turns[1] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= i; ++j)
        {
            if (i + i / j > n)
                continue;
            turns[i + i / j] = std::min(turns[i + i / j], turns[i] + 1);
        }
    }
    max = *std::max_element(turns.begin() + 1, turns.end());
}
```

```cpp
void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
        as[i] = turns[as[i]];
    }
    std::vector<int> cs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> cs[i];
    }

    k = std::min(max * n, k);
    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(k + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 0 }; j <= k; ++j)
        {
            dp[i][j] = dp[i - 1][j];
            if (j >= as[i])
                dp[i][j] = std::max(dp[i][j], dp[i - 1][j - as[i]] + cs[i]);
        }
    }
    std::cout << dp[n][k] << '\n';
}
```
