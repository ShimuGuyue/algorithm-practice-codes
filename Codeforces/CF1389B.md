void solve()
{
    int n, k, z;
    std::cin >> n >> k >> z;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<std::vector<int>> dp(n, std::vector<int>(z + 1));
    dp[0][0] = as[0];
    for (int i = 0; i < n; ++i)
    {
        if (i != 0)
            dp[i][0] = dp[i - 1][0] + as[i];
        for (int j = 1; j <= z; ++j)
        {
            if (i + j * 2 > k)
                break;
            if (i != 0)
                dp[i][j] = std::max(dp[i][j], dp[i - 1][j] + as[i]);
            if (i != n - 1)
            {
                for (int x = 0; x <= j; ++x)
                {
                    if (dp[i][x])
                        dp[i][j] = std::max(dp[i][j], dp[i][x] + (as[i] + as[i + 1]) * (j - x));
                }
            }
        }
    }
    int ans{ 0 };
    for (int i = 0; i < n; ++i)
    {
        if (i > k)
            break;
        if ((k - i) & 1)
            continue;
        if ((k - i) / 2 > z)
            continue;
        ans = std::max(ans, dp[i][(k - i) / 2]);
    }
    std::cout << ans << '\n';
}
