void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    int64_t ans{ as.front() * as.back() };
    for (int l{ 0 }, r{ n - 1 }; l <= r; ++l, --r)
    {
        if (as[l] * as[r] != ans)
        {
            std::cout << -1 << '\n';
            return;
        }
    }

    int count{ 0 };
    for (int64_t i = 2; i * i <= ans; ++i)
    {
        if (ans % i)
            continue;
        ++count;
        if (i != ans / i)
            ++count;
    }
    if (count != n)
        ans = -1;

    std::cout << ans << '\n';
}
