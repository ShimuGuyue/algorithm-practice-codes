void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int count1{ 0 };
    for (int i{ 0 }; i < n - 2; ++i)
    {
        if (as[i] == 1)
            ++count1;
        if (count1 * 2 < i + 1)
            continue;
        int count3{ 0 };
        for (int j{ i + 1 }; j < n - 1; ++j)
        {
            if (as[j] == 3)
                ++count3;
            if (count3 * 2 > j - i)
                continue;
            std::cout << "Yes\n";
            return;
        }
    }
    std::cout << "No\n";
}
