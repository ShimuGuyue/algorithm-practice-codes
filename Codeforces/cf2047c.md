void solve()
{
    int n;
    std::cin >> n;

    for (int b = 0; b <= 30; ++b)
    {
        if ((1 << b) == n || (1 << b) - 1 == n)
        {
            std::cout << -1 << '\n';
            return;
        }
    }

    int ans;
    for (int b = 0; b <= 30; ++b)
    {
        if ((1 << b) > n)
            break;
        ans = 1 << b;
    }
    --ans;
    std::cout << ans << '\n';
}
