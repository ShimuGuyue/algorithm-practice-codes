void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] == as[i - 1])
        {
            std::cout << "NO\n";
            return;
        }
    }
    std::cout << "YES\n";
}
