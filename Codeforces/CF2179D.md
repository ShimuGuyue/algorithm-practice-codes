void solve()
{
    int n;
    std::cin >> n;

    std::vector<bool> visited(1 << n);
    for (int i = (1 << n) - 1; i > 0; i /= 2)
    {
        std::cout << i << ' ';
        visited[i] = true;
        for (int j = 0; j < (1 << n); ++j)
        {
            if (visited[j])
                continue;
            if ((j & i) == i)
            {
                std::cout << j << ' ';
                visited[j] = true;
            }
        }
    }
    for (int i = 0; i < (1 << n); ++i)
    {
        if (!visited[i])
            std::cout << i << ' ';
    }
    std::cout << '\n';
}
