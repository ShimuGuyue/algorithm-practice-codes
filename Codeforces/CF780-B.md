void solve()
{
    int n;
    std::cin >> n;
    std::vector<double> xs(n);
    std::vector<double> vs(n);
    for (auto& x : xs)
    {
        std::cin >> x;
    }
    for (auto& v : vs)
    {
        std::cin >> v;
    }

    auto check = [n, &xs, &vs](double t) -> bool
    {
        double l{ -2e9 }, r{ 2e9 };
        for (int i{ 0 }; i < n; ++i)
        {
            l = std::max(l, xs[i] - t * vs[i]);
            r = std::min(r, xs[i] + t * vs[i]);
        }
        return l <= r;
    };
    double l{ 0 }, r{ 1e9 };
    while (r - l > 1e-6)
    {
        double mid{ (l + r) / 2 };
        if (check(mid))
            r = mid;
        else
            l = mid;
    }
    std::cout << std::fixed << std::setprecision(10) << r << '\n';
}
