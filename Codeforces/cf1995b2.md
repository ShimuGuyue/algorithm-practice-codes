void solve()
{
    int64_t n, m;
    std::cin >> n >> m;
    std::vector<std::pair<int64_t, int64_t>> acs(n);
    for (auto& [a, c] : acs)
    {
        std::cin >> a;
    }
    for (auto& [a, c] : acs)
    {
        std::cin >> c;
    }

    std::sort(acs.begin(), acs.end());

    int64_t ans{ 0 };
    for (auto [a, c] : acs)
    {
        ans = std::max(ans, std::min(m / a, c) * a);
    }

    for (int i{ 1 }; i < n; ++i)
    {
        auto [a1, c1]{ acs[i - 1] };
        auto [a2, c2]{ acs[i] };
        if (a2 != a1 + 1)
            continue;
        int64_t count1{ std::min(m / a1, c1) };
        int64_t sum{ count1 * a1 };
        int64_t count2{ std::min((m - sum) / a2, c2) };
        sum += count2 * a2;
        int64_t remain{ c2 - count2 };
        sum = std::min(sum + std::min(remain, count1), m);
        ans = std::max(ans, sum);
    }
    std::cout << ans << '\n';
}
