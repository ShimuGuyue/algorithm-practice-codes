void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::priority_queue<int, std::vector<int>, std::less<int>> h;
    for (auto a : as)
    {
        h.push(a);
    }

    std::vector<int> ans(n);
    for (int i{ 0 }; i < n; i += 2)
    {
        ans[i] = h.top();
        h.pop();
    }
    if (n & 1)
    {
        for (int i{ 1 }; i < n; i += 2)
        {
            ans[i] = h.top();
            h.pop();
        }
    }
    else
    {
        ans[n - 1] = h.top();
        h.pop();
        for (int i{ 1 }; i < n - 1; i += 2)
        {
            ans[i] = h.top();
            h.pop();
        }
    }

    int count{ 0 };
    for (int i{ 1 }; i < n - 1; ++i)
    {
        count += (ans[i] < ans[i - 1] && ans[i] < ans[i + 1]);
    }
    std::cout << count << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
