void solve()
{
    int64_t n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    int64_t ans{ 0 };
    int64_t sum{ 0 };
    int l{ 0 }, r{ 0 };
    while (r < n)
    {
        sum += as[r];
        while (sum > m || (r < n && as[r] - as[l] > 1 && l < n))
        {
            sum -= as[l++];
        }
        ++r;
        ans = std::max(ans, sum);
    }
    std::cout << ans << '\n';
}
