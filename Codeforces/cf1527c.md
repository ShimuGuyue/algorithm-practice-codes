void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t ans{ 0 };
    std::map<int, int64_t> m;
    for (int i{ 1 }; i <= n; ++i)
    {
        ans += m[as[i]] * (n - i + 1);
        m[as[i]] += i;
    }
    std::cout << ans << '\n';
}
