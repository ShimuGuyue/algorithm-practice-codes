void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }

    std::vector<int> premins(n);
    std::vector<int> sufmaxs(n);
    std::partial_sum(as.begin(), as.end(), premins.begin(),
        [](int a, int b)
        {
            return std::min(a, b);
        });
    std::partial_sum(as.rbegin(), as.rend(), sufmaxs.rbegin(),
        [](int a, int b)
        {
            return std::max(a, b);
        });

    for (int i = 1; i < n; ++i)
    {
        if (premins[i - 1] > sufmaxs[i])
        {
            std::cout << "No\n";
            return;
        }
    }
    std::cout << "Yes\n";
}
