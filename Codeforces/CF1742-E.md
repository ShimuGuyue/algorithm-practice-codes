void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t sum{ 0 };
    int64_t max{ 0 };
    std::map<int, int64_t> m;
    for (auto a : as)
    {
        max = std::max(max, a);
        sum += a;
        m[max] = sum;
    }

    while (q--)
    {
        int k;
        std::cin >> k;
        auto it{ m.upper_bound(k) };
        if (it == m.begin())
            std::cout << 0 << ' ';
        else
            std::cout << std::prev(it)->second << ' ';
    }
    std::cout << '\n';
}
