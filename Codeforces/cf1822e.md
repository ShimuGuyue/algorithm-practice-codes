void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    if (n & 1)
    {
        std::cout << -1 << '\n';
        return;
    }

    std::vector<int> counts('z' + 1);
    for (char c : s)
    {
        ++counts[c];
        if (counts[c] > n / 2)
        {
            std::cout << -1 << '\n';
            return;
        }
    }
    counts.assign('z' + 1, 0);

    for (int i{ 0 }; i < n - i - 1; ++i)
    {
        if (s[i] == s[n - i - 1])
            ++counts[s[i]];
    }
    int sum{ std::accumulate(counts.begin(), counts.end(), 0) };
    int max{ *std::max_element(counts.begin(), counts.end()) };

    int ans{ max * 2 <= sum ? (sum + 1) / 2 : max };
    std::cout << ans << '\n';
}
