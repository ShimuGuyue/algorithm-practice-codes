void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int64_t> presums(n + 1);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    std::vector<std::vector<int64_t>> dp(k + 1, std::vector<int64_t>(n + 1));
    for (int i{ 1 }; i <= k; ++i)
    {
        for (int j{ i * m }; j <= n; ++j)
        {
            dp[i][j] = std::max(dp[i][j - 1], dp[i - 1][j - m] + presums[j] - presums[j - m]);
        }
    }
    std::cout << dp[k][n] << '\n';
}
