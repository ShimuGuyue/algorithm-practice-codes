void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> xs(n);
    std::vector<int> ys(n);
    for (int i = 0; i < n; ++i)
    {
        std::cin >> xs[i] >> ys[i];
    }

    std::sort(xs.begin(), xs.end());
    std::sort(ys.begin(), ys.end());

    int64_t ans{ 1 };
    if (n % 2 == 0)
    {
        int l{ n / 2 - 1 };
        int r{ n / 2 };
        ans *= xs[r] - xs[l] + 1;
        ans *= ys[r] - ys[l] + 1;
    }
    std::cout << ans << '\n';
}
