void solve()
{
    int64_t x, n;
    std::cin >> x >> n;

    int64_t len;
    if (n % 4 == 0)
        len = 0;
    else if (n % 4 == 1)
        len = -n;
    else if (n % 4 == 2)
        len = 1;
    else if (n % 4 == 3)
        len = 4 * (n / 4 + 1);

    if (x & 1)
        len = -len;

    int64_t ans{ x + len };
    std::cout << ans << '\n';
}
