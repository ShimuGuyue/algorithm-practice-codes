void solve()
{
    int rr, gg, bb;
    std::cin >> rr >> gg >> bb;
    std::vector<std::vector<int64_t>> colors(3);
    colors[0].resize(rr);
    colors[1].resize(gg);
    colors[2].resize(bb);
    for (auto& v : colors)
    {
        for (auto& a : v)
        {
            std::cin >> a;
        }
        std::sort(v.begin(), v.end());
        v.erase(std::unique(v.begin(), v.end()), v.end());
    }

    auto calc = [](int64_t a, int64_t b, int64_t c) -> int64_t
    {
        return (a - b) * (a - b)
            +  (a - c) * (a - c)
            +  (b - c) * (b - c);
    };
    int64_t ans{ std::numeric_limits<int64_t>::max() };
    std::array<int, 3> flags{ {0, 1, 2} };
    do
    {
        auto [i, j, k]{ flags };
        for (auto a : colors[i])
        {
            auto bt{ std::lower_bound(colors[j].begin(), colors[j].end(), a) };
            if (bt == colors[j].end())
                break;
            auto b{ *bt };
            auto ct{ std::upper_bound(colors[k].begin(), colors[k].end(), a) };
            if (ct == colors[k].begin())
                continue;
            auto c{ *std::prev(ct) };
            ans = std::min(ans, calc(a, b, c));
        }
    } while (std::next_permutation(flags.begin(), flags.end()));
    std::cout << ans << '\n';
}
