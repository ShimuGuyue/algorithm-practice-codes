void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> vs(n);
    for (auto& v : vs)
    {
        int nn;
        std::cin >> nn;
        v.resize(nn);
        for (auto& a : v)
        {
            std::cin >> a;
        }
        std::reverse(v.begin(), v.end());
    }

    std::vector<bool> visited_v(1000000 + 1);
    std::vector<bool> visited_a(1000000 + 1);
    for (int k{ 0 }; k < n; ++k)
    {
        std::vector<int> flag{ 1000000 + 1 };
        int index;
        for (int i{ 0 }; i < n; ++i)
        {
            if (visited_v[i])
                continue;
            std::vector<int> temp;
            std::set<int> t;
            for (auto a : vs[i])
            {
                if (visited_a[a])
                    continue;
                if (t.find(a) != t.end())
                    continue;
                temp.push_back(a);
                t.insert(a);
            }
            if (temp < flag)
            {
                flag = temp;
                index = i;
            }
        }
        for (auto a : flag)
        {
            std::cout << a << ' ';
            visited_a[a] = true;
        }
        visited_v[index] = true;
    }
    std::cout << '\n';
}
