void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (n > m)
    {
        std::cout << 0 << '\n';
        return;
    }

    int64_t ans{ 1 };
    for (int i = 0; i < n; ++i)
    {
        for (int j = i + 1; j < n; ++j)
        {
            ans *= abs(as[j] - as[i]);
            ans %= m;
        }
    }
    std::cout << ans << '\n';
}
