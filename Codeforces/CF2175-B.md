void solve()
{
    int n, l, r;
    std::cin >> n >> l >> r;
    std::vector<int> ans(n + 1);
    std::vector<int> prexors(n + 1);

    int flag{ 2 };
    for (int i = 1; i <= n; ++i)
    {
        if (i == r)
            ans[i] = prexors[l - 1] ^ prexors[i - 1];
        else
            ans[i] = flag++ ^ prexors[i - 1];
        prexors[i] = prexors[i - 1] ^ ans[i];
    }
    for (int i = 1; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
