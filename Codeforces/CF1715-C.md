void solve()
{
    int64_t n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int64_t ans{ n * (n + 1) / 2 };
    for (int64_t i{ 1 }; i < n; ++i)
    {
        int64_t j{ i + 1 };
        if (as[i] != as[j])
            ans += (i) * (n - j + 1);
    }

    while (m--)
    {
        int64_t i, x;
        std::cin >> i >> x;
        if (i != 1)
        {
            int64_t j{ i - 1 };
            if (as[i] != as[j])
                ans -= (j) * (n - i + 1);
            if (x != as[j])
                ans += (j) * (n - i + 1);
        }
        if (i != n)
        {
            int64_t j{ i + 1 };
            if (as[i] != as[j])
                ans -= (i) * (n - j + 1);
            if (x != as[j])
                ans += (i) * (n - j + 1);
        }
        as[i] = x;
        std::cout << ans << '\n';
    }
}
