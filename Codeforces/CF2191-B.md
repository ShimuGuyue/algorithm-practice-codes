void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (std::find(as.begin(), as.end(), 0) == as.end())
    {
        std::cout << "no\n";
        return;
    }

    if (std::count(as.begin(), as.end(), 0) == 1)
    {
        std::cout << "yes\n";
        return;
    }

    if (std::find(as.begin(), as.end(), 1) == as.end())
    {
        std::cout << "no\n";
        return;
    }

    std::cout << "yes\n";
}
