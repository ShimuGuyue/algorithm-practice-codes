void solve()
{
    int k;
    std::cin >> k;
    std::string s;
    std::cin >> s;

    std::vector<int> indexs;
    indexs.push_back(-1);
    for (int i{ 0 }; i < s.length(); ++i)
    {
        if (s[i] == '1')
            indexs.push_back(i);
    }
    indexs.push_back(s.length());

    int64_t ans{ 0 };
    if (k == 0)
    {
        for (int i{ 1 }; i < indexs.size(); ++i)
        {
            int64_t x{ indexs[i] - indexs[i - 1] - 1 };
            ans += x * (x + 1) / 2;
        }
    }
    else
    {
        for (int l{ 1 }, r{ k }; r < indexs.size() - 1; ++l, ++r)
        {
            int64_t x{ indexs[l] - indexs[l - 1] };
            int64_t y{ indexs[r + 1] - indexs[r] };
            ans += x * y;
        }
    }

    std::cout << ans << '\n';
}
