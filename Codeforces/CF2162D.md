void solve()
{
    int n;
    std::cin >> n;

    int sum1, sum2;
    std::cout << 1 << ' ' << 1 << ' ' << n << std::endl;
    std::cin >> sum1;
    std::cout << 2 << ' ' << 1 << ' ' << n << std::endl;
    std::cin >> sum2;
    int len{ sum2 - sum1 };

    int l{ 1 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        std::cout << 1 << ' ' << 1 << ' ' << mid << std::endl;
        std::cin >> sum1;
        std::cout << 2 << ' ' << 1 << ' ' << mid << std::endl;
        std::cin >> sum2;
        if (sum1 == sum2)
            l = mid + 1;
        else
            r = mid;
    }
    std::cout << "! " << r << ' ' << r + len - 1 << std::endl;
}
