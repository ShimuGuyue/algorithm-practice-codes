void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> flags(as);
    std::sort(flags.begin(), flags.end());
    int flag{ flags[n / 2] };

    std::vector<std::vector<int>> turns(n);
    for (int i{ 0 }; i < n; ++i)
    {
        int l{ 0 }, e{ 0 }, r{ 0 };
        for (int j{ i }; j < n; ++j)
        {
            if (as[j] < flag)
                ++l;
            if (as[j] == flag)
                ++e;
            if (as[j] > flag)
                ++r;
            if (e && (l + e + r) % 2 == 1 && std::abs(r - l) < e)
                turns[i].push_back(j + 1);
        }
    }

    std::vector<int> dp(n);
    for (int i{ n - 1 }; i >= 0; --i)
    {
        for (int next : turns[i])
        {
            if (next == n)
                dp[i] = std::max(dp[i], 1);
            if (next != n && dp[next])
                dp[i] = std::max(dp[i], dp[next] + 1);
        }
    }
    std::cout << dp[0] << '\n';
}
