void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n * 2);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());
    for (int i{ 0 }; i < n * 2; i += 2)
    {
        if (as[i] != as[i + 1])
        {
            std::cout << "No\n";
            return;
        }
    }

    std::vector<int64_t> flags(n + 2);
    std::vector<int64_t> sufs(n + 2);
    for (int i{ n }; i > 0; --i)
    {
        int index{ i * 2 - 1 };
        as[index] -= sufs[i + 1] * 2;
        if (as[index] % (i * 2))
        {
            std::cout << "No\n";
            return;
        }
        flags[i] = as[index] / (i * 2);
        if (flags[i] <= 0)
        {
            std::cout << "No\n";
            return;
        }
        if (i != n && flags[i] >= flags[i + 1])
        {
            std::cout << "No\n";
            return;
        }
        sufs[i] = sufs[i + 1] + flags[i];
    }
    std::cout << "Yes\n";
}
