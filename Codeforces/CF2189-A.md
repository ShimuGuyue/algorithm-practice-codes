void solve()
{
    int k, n, m;
    std::cin >> k >> n >> m;
    std::vector<int> as(k);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (n > m)
        std::swap(n, m);

    int count1{ 0 }, count2{ 0 };
    for (auto a : as)
    {
        if (a <= n)
            ++count1;
        else if (a <= m)
            ++count2;
    }

    int ans{ 0 };
    int min{ std::min(count1, count2) };
    ans += min;
    count1 -= min;
    count2 -= min;
    ans += count1 / 2;
    std::cout << ans << '\n';
}
