```c++
static constexpr int64_t mod{ 998244353 };
std::vector<int64_t> factorials;

void init()
{
    constexpr int n{ 200000 };
    factorials.assign(n + 1, 1);
    for (int i = 2; i <= n; ++i)
    {
        factorials[i] = factorials[i - 1] * i % mod;
    }
}
```

```c++
void solve()
{
    int n;
    std::cin >> n;
    std::map<int, int> counts;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    if (counts.size() == 1)
    {
        std::cout << factorials[n] << '\n';
        return;
    }

    auto [k1, v1]{ *counts.rbegin() };
    auto [k2, v2]{ *std::next(counts.rbegin()) };
    if (v1 >= 2)
    {
        std::cout << factorials[n] << '\n';
        return;
    }
    if (k1 >= k2 + 2)
    {
        std::cout << 0 << '\n';
        return;
    }

    int64_t ans{ factorials[n] * v2 % mod * quickpow(v2 + 1, mod - 2) % mod };
    std::cout << ans << '\n';
}

int64_t quickpow(int64_t base, int power)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}
```
