void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(m + 1);
    for (int i = 1; i <= m; ++i)
    {
        std::cin >> as[i];
    }

    std::sort(as.begin(), as.end());

    int64_t ans{ 0 };
    for (int i = 1; i < n; ++i)
    {
        auto it_l{ std::lower_bound(as.begin(), as.end(), i) };
        auto it_r{ std::lower_bound(as.begin(), as.end(), n - i) };
        int64_t count_l{ as.end() - it_l };
        int64_t count_r{ as.end() - it_r };

        auto it_same{ std::lower_bound(as.begin(), as.end(), std::max(i, n - i)) };
        int64_t count_same{ as.end() - it_same };

        ans += count_l * count_r - count_same;
    }
    std::cout << ans << '\n';
}
