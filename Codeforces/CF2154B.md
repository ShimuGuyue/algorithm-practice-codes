void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto &a : as)
    {
        std::cin >> a;
    }
    as.push_back(0);

    std::vector<int> premaxs(n + 1);
    std::partial_sum(as.begin(), as.end(), premaxs.begin(),
        [](int a, int b)
        {
            return std::max(a, b);
        });

    int64_t ans{ 0 };
    for (int i = 0; i < n; ++i)
    {
        if (i & 1)
        {
            as[i] = premaxs[i];
            if (as[i - 1] >= as[i])
            {
                ans += as[i - 1] - as[i] + 1;
                as[i - 1] = as[i] - 1;
            }
            if (as[i + 1] >= as[i])
            {
                ans += as[i + 1] - as[i] + 1;
                as[i + 1] = as[i] - 1;
            }
        }
    }
    std::cout << ans << '\n';
}
