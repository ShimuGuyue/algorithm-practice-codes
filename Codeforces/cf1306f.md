void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> ss(n);
    for (auto& s : ss)
    {
        std::cin >> s;
    }

    for (int i{ 0 }; i < m; ++i)
    {
        for (char c{ 'a' }; c <= 'z'; ++c)
        {
            std::string ans{ ss[0] };
            ans[i] = c;
            bool ok{ true };
            for (int j{ 0 }; j < n; ++j)
            {
                int count{ 0 };
                for (int k{ 0 }; k < m; ++k)
                {
                    count += ss[j][k] != ans[k];
                }
                if (count > 1)
                {
                    ok = false;
                    break;
                }
            }
            if (ok)
            {
                std::cout << ans << '\n';
                return;
            }
        }
    }
    std::cout << -1 << '\n';
}
