void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> indexs(n);
    for (int i{ 0 }; i < n; ++i)
    {
        std::cin >> as[i];
        indexs[as[i]] = i;
    }

    if (n <= 2)
    {
        std::cout << 1 << '\n';
        return;
    }

    int64_t mod{ 1000000007 };
    int64_t ans{ 1 };
    auto [min, max]{ std::minmax(indexs[0], indexs[1]) };
    int l{ min }, r{ max };
    for (int i{ 2 }; i < n; ++i)
    {
        if (indexs[i] < l)
            l = indexs[i];
        else if (indexs[i] > r)
            r = indexs[i];
        else
            ans = ans * (r - l + 1 - i) % mod;
    }
    std::cout << ans << '\n';
}
