void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    int count0{ 0 }, count1{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (as[i] == 0 && bs[i] == 1)
            ++count0;
        if (as[i] == 1 && bs[i] == 0)
            ++count1;
    }

    if (count1 & 1)
    {
        std::cout << 1 << '\n';
        return;
    }

    if (count1)
    {
        std::cout << 2 << '\n';
        return;
    }

    // count1 = 0

    if (count0 == 0)
    {
        std::cout << 0 << '\n';
        return;
    }

    if (std::count(bs.begin(), bs.end(), 0) && std::count(as.begin(), as.end(), 1))
    {
        std::cout << 2 << '\n';
        return;
    }

    std::cout << -1 << '\n';
}
