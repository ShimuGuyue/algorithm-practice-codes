void solve()
{
    int n, c;
    std::cin >> n >> c;
    std::vector<int> as(n + 1);
    std::vector<int> bs(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i{ 1 }; i < n; ++i)
    {
        std::cin >> bs[i];
    }

    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(2));
    dp[1][1] = c;
    for (int i{ 2 }; i <= n; ++i)
    {
        dp[i][0] = std::min(dp[i - 1][0] + as[i - 1], dp[i - 1][1] + as[i - 1]);
        dp[i][1] = std::min(dp[i - 1][0] + bs[i - 1] + c, dp[i - 1][1] + bs[i - 1]);
    }

    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << std::min(dp[i][0], dp[i][1]) << ' ';
    }
    std::cout << '\n';
}
