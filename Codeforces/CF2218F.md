void solve()
{
    int x, y;
    std::cin >> x >> y;
    int n{ x + y };

    if (y == 0)
    {
        std::cout << "No\n";
        return;
    }

    if (x == y)
    {
        std::cout << "Yes\n";
        for (int i{ 1 }; i < n; i += 2)
        {
            std::cout << i << ' ' << i + 1 << '\n';
        }
        for (int i{ 1 }; i + 2 < n; i += 2)
        {
            std::cout << i << ' ' << i + 2 << '\n';
        }
    }
    else if (x > y)
    {
        std::cout << "No\n";
        return;
    }
    else if (x < y)
    {
        int dif{ y - x };
        if (dif & 1)
        {
            std::cout << "Yes\n";
            for (int i{ 0 }; i < x; ++i)
            {
                std::cout << n - i << ' ' << n - i - x << '\n';
            }
            for (int i{ 2 }; i <= y; ++i)
            {
                std::cout << 1 << ' ' << i << '\n';
            }
        }
        else
        {
            if (x == 0)
            {
                std::cout << "No\n";
                return;
            }
            std::cout << "Yes\n";
            for (int i{ 0 }; i < x; ++i)
            {
                std::cout << i * 2 + 1 << ' ' << i * 2 + 2 << '\n';
            }
            for (int i{ 0 }; i < x - 1; ++i)
            {
                std::cout << i * 2 + 1 << ' ' << i * 2 + 3 << '\n';
            }
            for (int i{ x * 2 + 1 }; i <= n; ++i)
            {
                std::cout << 1 << ' ' << i << '\n';
            }
        }
    }
}
