void solve()
{
    std::string s;
    std::cin >> s;

    int n{ int(s.length()) };

    int ans{ 0 };
    ans += std::count(s.begin(), s.end(), '4');
    if (ans == n)
    {
        std::cout << ans << '\n';
        return;
    }

    std::vector<int> pre13s(n);
    std::vector<int> suf2s(n);
    for (int i{ 0 }; i < n; ++i)
    {
        if (i != 0)
            pre13s[i] = pre13s[i - 1];
        pre13s[i] += (s[i] == '1' || s[i] == '3');
    }
    for (int i{ n - 1 }; i >= 0; --i)
    {
        if (i != n - 1)
            suf2s[i] =suf2s[i + 1];
        suf2s[i] += (s[i] == '2');
    }

    int min{ std::min(pre13s.back(), suf2s.front()) };
    if (pre13s.back() == 0 || suf2s.front() == 0)
        min = 0;
    for (int i{ 1 }; i < n; ++i)
    {
        min = std::min(min, pre13s[i - 1] + suf2s[i]);
    }

    ans += min;
    std::cout << ans << '\n';
}
