void solve()
{
    int64_t x, y, k;
    std::cin >> x >> y >> k;
    int64_t ans{ 0 };
    for (int i{ 0 }; i < k; ++i)
    {
        ans += (y + i) % (x + i);
        if (y + i < (x + i * 2))
        {
            ans += (k - i - 1) * (y - x);
            break;
        }
    }
    std::cout << ans << '\n';
}
