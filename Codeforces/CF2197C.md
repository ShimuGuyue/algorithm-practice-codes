void solve()
{
    int64_t a, b;
    std::cin >> a >> b;

    if (a * 3 < b * 2 || a >= b)
    {
        std::cout << "Alice\n";
        return;
    }

    /*
        (a - x) / (b - x) = 2 / 3
        x = 3 * a - 2 * b;
    */
    int64_t x{ 3 * a - 2 * b };
    if (x <= a && x <= b)
        std::cout << "Bob\n";
    else
        std::cout << "Alice\n";
}
