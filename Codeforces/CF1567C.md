void solve()
{
    int n;
    std::cin >> n;

    std::vector<int> v;
    while (n)
    {
        v.push_back(n % 10);
        n /= 10;
    }
    std::reverse(v.begin(), v.end());

    int64_t a{ 0 }, b{ 0 };
    for (int i{ 0 }; i < v.size(); i += 2)
    {
        a = a * 10 + v[i];
    }
    for (int i{ 1 }; i < v.size(); i += 2)
    {
        b = b * 10 + v[i];
    }

    int64_t ans{ (a + 1) * (b + 1) - 2 };
    std::cout << ans << '\n';
}
