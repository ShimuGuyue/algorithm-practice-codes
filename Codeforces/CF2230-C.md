void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t ans{ 0 };
    int64_t count{ 0 };
    for (auto a : as)
    {
        if (a == 1)
            continue;
        ans += a;
        count += (a - 2) / 2;
    }
    if (ans == as.back())
        ++count;

    for (auto a : as)
    {
        if (a != 1)
            continue;
        if (count)
        {
            --count;
            ++ans;
        }
    }
    if (ans < 3)
        ans = 0;
    std::cout << ans << '\n';
}
