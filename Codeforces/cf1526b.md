void solve()
{
    int n;
    std::cin >> n;

    if (n > 111 * 11 - 111 - 11)
    {
        std::cout << "Yes\n";
        return;
    }
    for (int i = 0; i <= n; i += 111)
    {
        if ((n - i) % 11 == 0)
        {
            std::cout << "Yes\n";
            return;
        }
    }
    std::cout << "No\n";
}
