void solve()
{
    int n;
    cin >> n;
    vector<int64_t> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    if (n == 1)
    {
        cout << as[0] << '\n';
        return;
    }

    vector<int64_t> bs(as);
    for (int i = 0; i < n; i += 2)
    {
        bs[i] *= -1;
    }

    using max_heap = priority_queue<int64_t, vector<int64_t>, std::less<int64_t>>;
    max_heap h;
    for (auto b : bs)
    {
        h.push(b);
    }

    int64_t ans = 0;
    for (int i = 0; i < n / 2; ++i)
    {
        ans += h.top();
        h.pop();
    }
    while (!h.empty())
    {
        ans -= h.top();
        h.pop();
    }
    cout << ans << '\n';
}
