void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::set<int, std::greater<int>> t;
    for (int i{ 1 }; i <= n; ++i)
    {
        t.insert(i);
    }
    for (auto a : as)
    {
        t.erase(a);
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i])
            continue;
        as[i] = *t.begin();
        t.erase(t.begin());
    }

    int l{ 1 }, r{ n };
    while (l <= n && as[l] == l)
    {
        ++l;
    }
    while (r >= 1 && as[r] == r)
    {
        --r;
    }
    if (l > r)
        std::cout << 0 << '\n';
    else
        std::cout << r - l + 1 << '\n';
}
