void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int m{ 1000000 };

    int mod{ 1000000007 };

    std::vector<int64_t> dp(m + 1);
    dp[0] = 1;
    for (int i{ 1 }; i <= n; ++i)
    {
        std::vector<int> factors;
        for (int j{ 1 }; j * j <= as[i]; ++j)
        {
            if (as[i] % j)
                continue;
            factors.push_back(j);
            if (j * j != as[i])
                factors.push_back(as[i] / j);
        }
        std::sort(factors.rbegin(), factors.rend());
        for (int a : factors)
        {
            if (a > i)
                continue;
            dp[a] += dp[a - 1];
            dp[a] %= mod;
        }
    }
    int64_t ans{
        std::accumulate(dp.begin() + 1, dp.end(), INT64_C(0),
            [mod](int64_t a, int64_t b)
            {
                return (a + b) % mod;
            }
        )
    };
    std::cout << ans << '\n';
}
