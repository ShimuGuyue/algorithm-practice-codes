void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> ans(n + 1);
    ans[n] = 1;
    for (int i = n - 1; i > 1; --i)
    {
        ans[i] = 1 ^ i;
    }
    ans[1] = n & 1 ? n - 1 : n;
    for (int i = 1; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
