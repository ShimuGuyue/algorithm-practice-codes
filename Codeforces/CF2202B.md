void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    bool a{ true }, b{ false };
    for (int i{ 0 }; i < n; ++i)
    {
        bool aa{ false }, bb{ false };
        int len{ n - i };
        if (a)
        {
            char l{ 'a' };
            char r{ len % 2 ? 'a' : 'b' };
            if (s[i] == l || s[i] == '?')
                bb = true;
            if (s[i] == r || s[i] == '?')
                aa = true;
        }
        if (b)
        {
            char l{ 'b' };
            char r{ len % 2 ? 'b' : 'a' };
            if (s[i] == l || s[i] == '?')
                aa = true;
            if (s[i] == r || s[i] == '?')
                bb = true;
        }

        a = aa;
        b = bb;
    }
    std::cout << (a || b ? "YES\n" : "NO\n");
}
