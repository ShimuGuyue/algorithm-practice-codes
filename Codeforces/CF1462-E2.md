```cpp
template <const int N, const int64_t MOD = 0>
class Combination
{
private:
    inline static std::vector<int64_t> factorials_;
    inline static std::vector<int64_t> inv_factorials_;

public:
    Combination()
    {
        factorials_.assign(N + 1, 1);
        inv_factorials_.assign(N + 1, 1);
        for (int i{ 1 }; i <= N; ++i)
        {
            factorials_[i] = factorials_[i - 1] * i;
            if (MOD)
                factorials_[i] %= MOD;
        }
        if (MOD)
        {
            inv_factorials_[N] = power(factorials_[N], MOD - 2);
            for (int i{ N - 1 }; i > 0; --i)
            {
                inv_factorials_[i] = inv_factorials_[i + 1] * (i + 1) % MOD;
            }
        }
    }

public:
    int64_t calc(const int n, const int k)
    {
        if (k == 0 || k == n)
            return 1;
        if (k < 0 || k > n)
            return 0;
        return MOD
            ? factorials_[n] * inv_factorials_[k] % MOD * inv_factorials_[n - k] % MOD
            : factorials_[n] / factorials_[k] / factorials_[n - k];
    }

private:
    int64_t power(int64_t base, int64_t p)
    {
        int64_t ans{ 1 };
        while (p)
        {
            if (p & 1)
                ans = ans * base % MOD;
            base = base * base % MOD;
            p /= 2;
        }
        return ans;
    }
};
```

```cpp
void solve()
{
    static const int64_t mod{ 1000000007 };
    static Combination<200000, mod> combination;

    int n, m, k;
    std::cin >> n >> m >> k;
    std::map<int, int> counts;
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    std::map<int, int> presums{ counts };
    presums[0] = 0;
    for (auto it{ presums.begin() }; it != presums.end(); ++it)
    {
        it->second += std::prev(it)->second;
    }

    int64_t ans{ 0 };
    for (auto [a, c] : counts)
    {
        auto rt{ std::prev(presums.upper_bound(a + k)) };
        auto lt{ std::prev(presums.lower_bound(a)) };
        int sum{ rt->second - lt->second };
        ans = (ans + combination.calc(sum, m) - combination.calc(sum - c, m) + mod) % mod;
    }
    std::cout << ans << '\n';
}
```
