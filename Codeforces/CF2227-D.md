void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n * 2);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 1 };
    int index1{ -1 }, index2{ -1 };
    for (int i{ 0 }; i < n * 2; ++i)
    {
        if (as[i])
            continue;
        if (index1 == -1)
            index1 = i;
        else
            index2 = i;
    }

    for (int i : { index1, index2 })
    {
        std::vector<bool> visited(n + 1);
        for (int j{ 0 }; i + j < n * 2 && i - j >= 0; ++j)
        {
            if (as[i - j] != as[i + j])
                break;
            visited[as[i + j]] = true;
        }
        int mex{ 0 };
        while (mex <= n)
        {
            if (visited[mex])
                ++mex;
            else
                break;
        }
        ans = std::max(ans, mex);
    }

    bool p{ true };
    std::vector<bool> visited(n + 1);
    for (int i{ 0 }; index1 + i <= index2 - i; ++i)
    {
        if (as[index1 + i] != as[index2 - i])
        {
            p = false;
            break;
        }
        visited[as[index1 + i]] = true;
    }
    if (p)
    {
        for (int i{ 0 }; index1 - i >= 0 && index2 + i < n * 2; ++i)
        {
            if (as[index1 - i] != as[index2 + i])
                break;
            visited[as[index1 - i]] = true;
        }
        int mex{ 0 };
        while (mex <= n)
        {
            if (visited[mex])
                ++mex;
            else
                break;
        }
        ans = std::max(ans, mex);
    }
    std::cout << ans << '\n';
}
