void solve()
{
    int64_t n, k;
    std::cin >> n >> k;

    std::vector<int64_t> factors;
    factors.reserve(1e7);
    for (int64_t i = 1; i * i <= n; ++i)
    {
        if (n % i)
            continue;
        factors.push_back(i);
        if (i * i != n)
            factors.push_back(n / i);
    }

    std::sort(factors.begin(), factors.end());

    if (k > factors.size())
    {
        std::cout << -1 << '\n';
        return;
    }

    std::cout << factors[k - 1] << '\n';
}
