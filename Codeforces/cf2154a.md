void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    int ans{ 0 };
    bool first{ true };
    for (int i = 0; i < n; ++i)
    {
        if (s[i] == '0')
            continue;
        if (first)
        {
            first = false;
            ans = 1;
        }
        else
        {
            bool find1{ false };
            for (int j = 1; j < k; ++j)
            {
                if (i - j < 0)
                    break;
                if (s[i - j] == '1')
                    find1 = true;
            }
            if (!find1)
                ++ans;
        }
    }
    std::cout << ans << '\n';
}
