void solve()
{
    int64_t a, b;
    std::cin >> a >> b;

    int64_t ans{ std::numeric_limits<int>::max() };

    auto calc = [a, b](int64_t k) -> int64_t
    {
        return (a + k - 1) / k + (b + k - 1) / k + k - 1;
    };

    for (int i{ 1 }; i <= 100000; ++i)
    {
        ans = std::min(ans, calc(i));
    }
    std::cout << ans << '\n';
}
