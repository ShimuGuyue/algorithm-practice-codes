void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n + 1);
    std::vector<int> bs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> bs[i];
    }

    std::vector<int> indexs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (bs[i] != -1)
            indexs[bs[i]] = i;
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        if (bs[i] != -1)
            continue;
        if (indexs[as[i]])
            continue;
        indexs[as[i]] = i;
        bs[i] = as[i];
    }
    std::queue<int> q;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (indexs[as[i]] == 0)
            q.push(as[i]);
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        if (bs[i] == -1)
        {
            bs[i] = q.front();
            indexs[q.front()] = i;
            q.pop();
        }
    }

    if (!q.empty())
    {
        std::cout << "No\n";
        return;
    }
    std::vector<bool> va(n + 1);
    std::vector<bool> vb(n + 1);
    for (int i{ 1 }; i <= k; ++i)
    {
        va[as[i]] = true;
        vb[bs[i]] = true;
    }
    for (int i{ 1 }; i <= k; ++i)
    {
        if (!vb[as[i]])
        {
            std::cout << "No\n";
            return;
        }
    }
    for (int i{ k + 1 }; i <= n; ++i)
    {
        if (as[i - k] != bs[i - k] || as[i] != bs[i])
        {
            std::cout << "No\n";
            return;
        }
    }
    std::cout << "Yes\n";
}
