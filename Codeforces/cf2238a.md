void solve()
{
    int n, c;
    std::cin >> n >> c;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    int ans{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (as[i] < bs[i])
        {
            ans = c;
            break;
        }
    }

    std::priority_queue<std::pair<int, int>> ha;
    std::priority_queue<int> hb;
    for (int i{ 0 }; i < n; ++i)
    {
        ha.push({ as[i], i });
        hb.push(bs[i]);
    }

    while (!ha.empty())
    {
        auto [a, i]{ ha.top() };
        auto b{ hb.top() };
        ha.pop();
        hb.pop();
        if (a < b)
        {
            std::cout << -1 << '\n';
            return;
        }
        ans += a - b;
        as[i] = b;
    }
    std::cout << ans << '\n';
}
