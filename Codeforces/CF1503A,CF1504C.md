void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    if (s.front() == '0' || s.back() == '0')
    {
        std::cout << "NO\n";
        return;
    }

    int count0{ std::count(s.begin(), s.end(), '0') };
    if (count0 & 1)
    {
        std::cout << "NO\n";
        return;
    }
    int count1{ n - count0 };

    std::cout << "YES\n";
    std::string a(n, 0);
    std::string b(n, 0);
    int flag0{ 0 }, flag1{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (s[i] == '1')
            if (flag1++ < count1 / 2)
                a[i] = b[i] = '(';
            else
                a[i] = b[i] = ')';
        else
            if (flag0++ & 1)
                a[i] = ')', b[i] = '(';
            else
                a[i] = '(', b[i] = ')';
    }
    std::cout << a << '\n';
    std::cout << b << '\n';
}
