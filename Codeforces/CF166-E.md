void solve()
{
    constexpr int64_t mod{ 1000000007 };

    int n;
    std::cin >> n;
    std::vector<int64_t> dp(n + 1);
    dp[1] = 3;
    for (int i = 2; i <= n; ++i)
    {
        dp[i] = dp[i - 1] * 2 + dp[i - 2] * 3;
        dp[i] %= mod;
    }
    std::cout << dp[n - 1] << '\n';
}
