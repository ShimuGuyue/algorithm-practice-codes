void solve()
{
    int n;
    std::cin >> n;
    std::map<int, int> m;
    while (n--)
    {
        int a;
        std::cin >> a;
        ++m[a];
    }

    int ans{ 0 };
    for (auto [k ,v] : m)
    {
        if (v >= k)
            ans += v - k;
        else
            ans += v;
    }
    std::cout << ans << '\n';
}
