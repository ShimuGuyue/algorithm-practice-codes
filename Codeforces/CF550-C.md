void solve()
{
    std::string s;
    std::cin >> s;

    if (s.find('0') != std::string::npos)
    {
        std::cout << "YES\n";
        std::cout << "0\n";
        return;
    }

    int n{ int(s.length()) };

    for (int i = 0; i < n; ++i)
    {
        int a{ s[i] - '0' };
        if (a % 8 == 0)
        {
            std::cout << "YES\n";
            std::cout << a << '\n';
            return;
        }
        for (int j = i + 1; j < n; ++j)
        {
            int b{s[j] - '0'};
            if ((a * 10 + b) % 8 == 0)
            {
                std::cout << "YES\n";
                std::cout << a << b << '\n';
                return;
            }
            for (int k = j + 1; k < n; ++k)
            {
                int c{ s[k] - '0' };
                if ((a * 100 + b * 10 + c) % 8 == 0)
                {
                    std::cout << "YES\n";
                    std::cout << a << b << c << '\n';
                    return;
                }
            }
        }
    }
    std::cout << "NO\n";
}
