void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    struct Query
    {
        int l, r;
        int id;
        int block;
    };
    std::vector<Query> querys(m);
    for (int i{ 0 }; i < m; ++i)
    {
        auto& [l, r, id, block]{ querys[i] };
        std::cin >> l >> r;
        id = i;
        block = l / std::sqrt(n);
    }

    std::sort(querys.begin(), querys.end(),
        [](Query& a, Query& b)
        {
            if (a.block != b.block)
                return a.block < b.block;
            return a.r < b.r;
        }
    );

    std::vector<int> ans(m);
    std::vector<int> counts(n + 1);
    int cur{ 0 };
    auto add = [&ans, &counts, &cur, n](int x)
    {
        if (x > n)
            return;
        if (counts[x] == x)
            --cur;
        ++counts[x];
        if (counts[x] == x)
            ++cur;
    };
    auto remove = [&ans, &counts, &cur, n](int x)
    {
        if (x > n)
            return;
        if (counts[x] == x)
            --cur;
        --counts[x];
        if (counts[x] == x)
            ++cur;
    };
    int ll{ 1 }, rr{ 0 };
    for (auto [l, r, id, block] : querys)
    {
        while (ll > l)
        {
            add(as[--ll]);
        }
        while (rr < r)
        {
            add(as[++rr]);
        }
        while (ll < l)
        {
            remove(as[ll++]);
        }
        while (rr > r)
        {
            remove(as[rr--]);
        }
        ans[id] = cur;
    }

    for (auto a : ans)
    {
        std::cout << a << '\n';
    }
}
