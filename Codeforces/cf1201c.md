void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    std::map<int, int> m;
    for (int i = n / 2; i < n; ++i)
    {
        ++m[as[i]];
    }

    while (k)
    {
        auto it{ m.begin() };
        auto [a1, b1]{ *it };
        if (std::next(it) == m.end())
        {
            int dif{ k / b1 };
            std::cout << a1 + dif << '\n';
            return;
        }

        auto jt{ std::next(it) };
        auto [a2, b2]{ *jt };
        int dif{ a2 - a1 };
        if (k >= int64_t(dif) * b1)
        {
            k -= dif * b1;
            jt->second += b1;
            m.erase(it);
        }
        else
        {
            int dif{ k / b1 };
            std::cout << a1 + dif << '\n';
            return;
        }
    }

    int ans{ m.begin()->first };
    std::cout << ans << '\n';
}
