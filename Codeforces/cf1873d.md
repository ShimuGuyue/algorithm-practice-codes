void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    int ans{ 0 };
    int i{ 0 };
    while (i < n)
    {
        if (s[i] == 'B')
        {
            ++ans;
            i += k;
        }
        else
        {
            ++i;
        }
    }
    std::cout << ans << '\n';
}
