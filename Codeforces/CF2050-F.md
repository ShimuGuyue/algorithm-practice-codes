template<typename T, typename K = T>
class SparseTable
{
private:
    std::vector<std::vector<K>> table_;

    const std::vector<T>& v_;

public:
    SparseTable(const std::vector<T>& v) : v_(v)
    {
        const int n{ static_cast<int>(v.size()) };
        const int m{ static_cast<int>(std::log2(n)) + 1 };
        table_.assign(n, std::vector<K>(m));

        for (int i{ 0 }; i < n; ++i)
        {
            table_[i][0] = contribute(i);
        }
        for (int j{ 1 }; j < m; ++j)
        {
            const int len{ 1 << (j - 1) };
            for (int i{ 0 }; i + len * 2 - 1 < n; ++i)
            {
                table_[i][j] = merge(i, i + len, j - 1);
            }
        }
    }

public:
    K query(const int l, const int r) const
    {
        const int power{ static_cast<int>(log2(r - l + 1)) };
        const int len{ 1 << power };
        return merge(l, r - len + 1, power);
    }

private:
    K contribute(const int index) const
    {
        // TODO: 规定初始贡献值
        return v_[index];
    }

    K merge(const int i1, const int i2, const int j) const
    {
        // TODO: 规定贡献值合并结果
        return std::gcd(table_[i1][j], table_[i2][j]);
    }
};

void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> equals(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (i != n && as[i] == as[i + 1])
            continue;
        for (int j{ i }; j > 0; --j)
        {
            if (as[j] != as[i])
                break;
            equals[j] = i;
        }
    }

    std::vector<int> difs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        difs[i] = std::abs(as[i] - as[i - 1]);
    }

    SparseTable<int> st(difs);

    while (q--)
    {
        int l, r;
        std::cin >> l >> r;
        if (equals[l] >= r)
        {
            std::cout << 0 << ' ';
            continue;
        }
        std::cout << st.query(l + 1, r) << ' ';
    }
    std::cout << '\n';
}
