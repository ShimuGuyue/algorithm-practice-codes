void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::string s;
    std::cin >> s;

    s = " " + s;

    std::vector<std::vector<int>> presums(6, std::vector<int>(n + 1));
    std::array<char, 3> flags{ {'a', 'b', 'c'} };
    for (int i{ 0 }; i < 6; ++i)
    {
        for (int j{ 1 }; j <= n; ++j)
        {
            presums[i][j] = presums[i][j - 1];
            if (s[j] == flags[j % 3])
                ++presums[i][j];
        }
        std::next_permutation(flags.begin(), flags.end());
    }

    while (m--)
    {
        int l, r;
        std::cin >> l >> r;
        int ans{ n };
        for (int i{ 0 }; i < 6; ++i)
        {
            ans = std::min(ans, (r - l + 1) - (presums[i][r] - presums[i][l - 1]));
        }
        std::cout << ans << '\n';
    }
}
