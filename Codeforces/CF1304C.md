void solve()
{
    int n, m;
    std::cin >> n >> m;

    std::string ans{ "YES" };
    int l{ m }, r{ m };
    int time{ 0 };
    while (n--)
    {
        int t, a, b;
        std::cin >> t >> a >> b;
        int dist{ t - time };
        l = std::max(l - dist, a);
        r = std::min(r + dist, b);
        if (l > b || r < a)
            ans = "NO";
        time = t;
    }
    std::cout << ans << '\n';
}
