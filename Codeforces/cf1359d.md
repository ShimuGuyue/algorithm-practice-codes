void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::set<int> t;
    for (auto a : as)
    {
        t.insert(a);
    }

    int ans{ 0 };
    for (int max{ 1 }; max <= 30; ++max)
    {
        if (!t.count(max))
            continue;
        bool find{ false };
        int sum{ -max };
        for (int i{ 1 }; i <= n; ++i)
        {
            if (as[i] > max)
                sum = -max;
            else
                sum = std::max(-max + as[i], sum + as[i]);
            ans = std::max(ans, sum);
        }
    }
    std::cout << ans << '\n';
}
