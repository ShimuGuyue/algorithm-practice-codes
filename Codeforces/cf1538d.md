void solve()
{
    int a, b, k;
    std::cin >> a >> b >> k;

    if (k == 1)
    {
        if ((a % b == 0 || b % a == 0) && a != b)
            std::cout << "YES\n";
        else
            std::cout << "NO\n";
        return;
    }

    int count{ 0 };
    for (int i{ 2 }; i * i <= a; ++i)
    {
        while (a % i == 0)
        {
            ++count;
            a /= i;
        }
    }
    if (a != 1)
        ++count;
    for (int i{ 2 }; i * i <= b; ++i)
    {
        while (b % i == 0)
        {
            ++count;
            b /= i;
        }
    }
    if (b != 1)
        ++count;

    if (count < k)
        std::cout << "NO\n";
    else
        std::cout << "YES\n";
}
