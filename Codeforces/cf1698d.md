void solve()
{
    int n;
    std::cin >> n;

    auto check = [](int l, int r) -> bool
    {
        int count{ 0 };
        for (int i{ l }; i <= r; ++i)
        {
            int x;
            std::cin >> x;
            if (x >= l && x <= r)
                ++count;
        }
        return count & 1;
    };
    int l{ 1 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        std::cout << "? " << l << ' ' << mid << std::endl;
        if (check(l, mid))
            r = mid;
        else
            l = mid + 1;
    }
    std::cout << "! " << r << std::endl;
}
