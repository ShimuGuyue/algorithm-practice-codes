void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    auto check = [n, &as, &bs](int k) -> bool
    {
        std::vector<int> flags;
        for (int i{ 0 }; i < n; ++i)
        {
            int sum{ int(as[i] >= k) + int(bs[i] >= k) };
            if (sum == 0)
                if (flags.empty() || flags.back() != 0)
                    flags.push_back(0);
            if (sum == 2)
                flags.push_back(2);
        }
        int count0{ std::count(flags.begin(), flags.end(), 0) };
        int count2{ std::count(flags.begin(), flags.end(), 2) };
        return count2 > count0;
    };
    int l{ 1 }, r{ n * 2 };
    while (l < r)
    {
        int mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    std::cout << l << '\n';
}
