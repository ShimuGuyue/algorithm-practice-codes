void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t ans{ 0 };
    std::vector<int64_t> pres(n);
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] == 1 && as[i - 1] != 1)
        {
            std::cout << -1 << '\n';
            return;
        }
        pres[i] = pres[i - 1];
        int64_t a{ as[i] }, b{ as[i - 1] };
        while (a < b)
        {
            a = a * a;
            ++pres[i];
        }
        while (a >= b)
        {
            if (b == 1)
            {
                pres[i] = -1;
                break;
            }
            b = b * b;
            --pres[i];
        }
        ++pres[i];
        pres[i] = std::max(pres[i], INT64_C(0));
        ans += pres[i];
    }
    std::cout << ans << '\n';
}
