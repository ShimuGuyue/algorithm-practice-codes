```cpp
vector<vector<int>> factors;
void init()
{
    int n = 300000;
    factors.resize(n + 1);
    for (int i = 2; i <= n; ++i)
    {
        if (!factors[i].empty())
            continue;
        for (int j = i; j <= n; j += i)
        {
            factors[j].push_back(i);
        }
    }
}
```

```cpp
void solve()
{
    int n, x;
    cin >> n >> x;
    map<int, int> m;
    for (int i = 0; i < n; ++i)
    {
        int a;
        cin >> a;
        ++m[a];
    }

    if (x == 1)
    {
        cout << 0 << '\n';
        return;
    }

    int64_t ans = 0;
    for (int f : factors[x])
    {
        int64_t sum = 0;
        for (auto [k, v] : m)
        {
            if (k % f == 0)
                sum += int64_t(k) * v;
        }
        ans = std::max(ans, sum);
    }
    cout << ans << '\n';
}
```
