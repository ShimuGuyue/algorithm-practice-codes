void solve()
{
    int n;
    std::cin >> n;
    std::string a;
    std::string b;
    std::cin >> a;
    std::cin >> b;

    std::vector<int> ans;
    for (int i{ n - 1 }; i > 0; --i)
    {
        if (a[i] == b[i])
            continue;
        if (a[0] == b[i])
        {
            ans.push_back(1);
            a[0] = b[i] == '0' ? '1' : '0';
        }
        ans.push_back(i + 1);
        std::vector<char> flags;
        for (int j{ 0 }; j <= i; ++j)
        {
            flags.push_back(a[j] == '0' ? '1' : '0');
        }
        for (int j{ 0 }; j <= i; ++j)
        {
            a[i - j] = flags[j];
        }
    }
    if (a[0] != b[0])
        ans.push_back(1);

    std::cout << ans.size();
    for (auto a : ans)
    {
        std::cout << ' ' << a;
    }
    std::cout << '\n';
}
