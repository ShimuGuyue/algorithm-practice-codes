void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 2);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int64_t> pres_abs(n + 2);
    std::partial_sum(as.begin(), as.end(), pres_abs.begin(),
        [](int64_t pre, int64_t cur)
        {
            return pre + std::llabs(cur);
        }
    );
    std::vector<int64_t> sufs(n + 2);
    std::partial_sum(as.rbegin(), as.rend(), sufs.rbegin());

    int64_t max{ std::accumulate(as.begin(), as.end(), INT64_C(0)) };
    int index{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] <= 0)
            continue;
        if (sufs[i + 1] - as[i] + pres_abs[i - 1] > max)
        {
            index = i;
            max = sufs[i + 1] - as[i] + pres_abs[i - 1];
        }
    }
    if (index == 0)
    {
        std::cout << 0 << '\n';
        return;
    }

    std::vector<int> ans;
    int flag{ 1 };
    for (int i{ index - 1 }; i > 0; --i)
    {
        as[i] *= flag;
        if (as[i] > 0)
        {
            ans.push_back(i);
            flag *= -1;
        }
    }
    ans.push_back(index);

    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
