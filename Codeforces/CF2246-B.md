void solve()
{
    int n;
    std::cin >> n;

    if (n == 2)
    {
        std::cout << -1 << '\n';
        return;
    }

    std::vector<int64_t> as(50);
    as[0] = 1;
    as[1] = 2;
    as[2] = 3;
    for (int i{ 3 }; i < n; ++i)
    {
        as[i] = as[i - 1] * 2;
    }
    for (int i{ 0 }; i < n; ++i)
    {
        std::cout << as[i] << ' ';
    }
    std::cout << '\n';
}
