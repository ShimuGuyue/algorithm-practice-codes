void solve()
{
    constexpr int64_t mod{ 1000000007 };

    int64_t n;
    std::cin >> n;

    int64_t factorial{ 1 };
    for (int i = 1; i <= n; ++i)
    {
        factorial *= i;
        factorial %= mod;
    }

    int64_t dif{ 1 };
    for (int i = 1; i < n; ++i)
    {
        dif *= 2;
        dif %= mod;
    }

    int64_t ans { (factorial - dif + mod) % mod };
    std::cout << ans << '\n';
}
