void solve()
{
    int n, k, q;
    std::cin >> n >> k >> q;
    std::vector<int> as(k + 1);
    for (int i{ 1 }; i <= k; ++i)
    {
        std::cin >> as[i];
    }
    std::vector<int> bs(k + 1);
    for (int i{ 1 }; i <= k; ++i)
    {
        std::cin >> bs[i];
    }

    while (q--)
    {
        int d;
        std::cin >> d;
        auto it{ std::prev(std::upper_bound(as.begin(), as.end(), d)) };
        int index{ it - as.begin() };
        if (as[index] == d)
        {
            std::cout << bs[index] << ' ';
            continue;
        }
        int ans{ bs[index] };
        int dist{ d - as[index] };
        int len{ as[index + 1] - as[index] };
        int time{ bs[index + 1] - bs[index] };
        ans += int64_t(dist) * time / len;
        std::cout << ans << ' ';
    }
    std::cout << '\n';
}
