void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::string> vs(n);
    for (auto& s : vs)
    {
        std::cin >> s;
    }

    std::set<std::string> t(vs.begin(), vs.end());

    std::string ans;
    for (auto& s : vs)
    {
        bool ok{ false };
        for (int i = 1; i < s.length(); ++i)
        {
            auto s1{ s.substr(0, i) };
            auto s2{ s.substr(i) };
            if (t.find(s1) != t.end() && t.find(s2) != t.end())
                ok = true;
        }
        ans += (ok ? '1' : '0');
    }
    std::cout << ans << '\n';
}
