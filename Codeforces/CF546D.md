

```cpp
std::vector<int64_t> counts;
std::vector<int64_t> presums;

void init()
{
    counts.assign(5000000 + 1, 0);
    presums.assign(5000000 + 1, 0);
    for (int i{ 2 }; i <= 5000000; ++i)
    {
        if (counts[i] == 0)
            counts[i] = 1;
        for (int j{ 2 }; j <= i && j * i <= 5000000; ++j)
        {
            counts[i * j] = counts[i] + counts[j];
        }
    }
    std::partial_sum(counts.begin(), counts.end(), presums.begin());
}
```

```cpp
void solve()
{
    int a, b;
    std::cin >> a >> b;
    int64_t ans{ presums[a] - presums[b] };
    std::cout << ans << '\n';
}
```
