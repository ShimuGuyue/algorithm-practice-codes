void solve()
{
    int n, x;
    std::cin >> n >> x;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 0 };
    while (!std::is_sorted(as.begin(), as.end()))
    {
        bool turn{ false };
        for (auto& a : as)
        {
            if (a > x)
            {
                turn = true;
                std::swap(a, x);
                ++ans;
                break;
            }
        }
        if (!turn)
        {
            std::cout << -1 << '\n';
            return;
        }
    }
    std::cout << ans << '\n';
}
