void solve()
{
    int64_t n, c, k;
    std::cin >> n >> c >> k;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());
    for (auto a : as)
    {
        if (a > c)
            break;
        int64_t dif{ std::min(k, c - a) };
        c += a + dif;
        k -= dif;
    }
    std::cout << c << '\n';
}
