void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (int i{ 0 }; i < n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> dp(n + 1);
    for (auto a : as)
    {
        dp[a] = dp[a - 1] + 1;
    }
    int ans{ n };
    for (int i{ 1 }; i <= n; ++i)
    {
        int r{ i };
        int l{ r - dp[i] + 1 };
        ans = std::min(ans, std::max(l - 1, n - r));
    }
    std::cout << ans << '\n';
}
