void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    int k{ 30 };
    int l{ 1 }, r{ n };
    while (k--)
    {
        int mid{ (l + r) / 2 };

        int sum{};
        std::cout << "? " << mid - l + 1 << ' ';
        for (int i = l; i <= mid; ++i)
        {
            std::cout << i << ' ';
            sum += as[i];
        }
        std::cout << std::endl;

        int a;
        std::cin >> a;
        if (a == sum)
            l = mid + 1;
        else
            r = mid;

        if (l == r)
        {
            std::cout << "! " << l << std::endl;
            return;
        }
    }
}
