void solve()
{
    int64_t n, k;
    std::cin >> n >> k;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::sort(as.begin(), as.end());
    std::vector<int64_t> presums(n + 1);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    auto ans{ std::numeric_limits<int64_t>::max() };
    for (int i{ 0 }; i < n; ++i)
    {
        auto sum{ presums[n - i] - presums[1] };
        auto dif{ k - sum };
        auto flag{ dif >= 0 ? dif / (i + 1) : (dif - i) / (i + 1) };
        ans = std::min(ans, i + std::max(INT64_C(0), as[1] - flag));
    }
    std::cout << ans << '\n';
}
