void solve()
{
    static constexpr int64_t mod{ 1000000007 };

    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    std::priority_queue<int64_t> h;
    std::vector<int64_t> childs(n + 1);
    auto dfs = [n, &tree, &h, &childs](this auto&& dfs, int u, int last) -> void
    {
        childs[u] = 1;
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            h.push(childs[v] * (n - childs[v]));
            childs[u] += childs[v];
        }
    };
    dfs(1, 0);

    int m;
    std::cin >> m;
    std::vector<int64_t> ps(m);
    for (auto& p : ps)
    {
        std::cin >> p;
    }
    std::sort(ps.begin(), ps.end());
    while (ps.size() > n - 1)
    {
        ps[m - 2] = ps[m - 2] * ps[m - 1] % mod;
        --m;
        ps.pop_back();
    }
    if (ps.size() < n - 1)
    {
        int dif{ n - 1 - int(ps.size()) };
        std::vector<int64_t> temp(dif, 1);
        for (int p : ps)
        {
            temp.push_back(p);
        }
        ps = temp;
    }

    int64_t ans{ 0 };
    while (!h.empty())
    {
        int64_t a{ h.top() };
        h.pop();
        int64_t p{ ps.back() };
        ps.pop_back();
        ans = (ans + a % mod * p % mod) % mod;
    }
    std::cout << ans << '\n';
}
