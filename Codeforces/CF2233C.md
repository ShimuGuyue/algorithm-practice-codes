void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    int max{ n };
    std::vector<bool> ans;
    for (int i{ 0 }; i <= k; ++i)
    {
        std::vector<bool> flags(n);
        for (int j{ 0 }, count{ 0 }; j < n && count < i; ++j)
        {
            if (s[j] == '(')
            {
                flags[j] = true;
                ++count;
            }
        }
        for (int j{ n - 1 }, count{ 0 }; j >= 0 && count < k - i; --j)
        {
            if (s[j] == ')')
            {
                flags[j] = true;
                ++count;
            }
        }
        int sum{ 0 };
        int count{ 0 };
        for (int j{ 0 }; j < n; ++j)
        {
            if (flags[j])
                continue;
            if (s[j] == '(')
            {
                ++count;
            }
            else if (count)
            {
                --count;
                ++sum;
            }
        }
        if (sum < max)
        {
            max = sum;
            ans = flags;
        }
    }

    for (bool b : ans)
    {
        std::cout << int(b);
    }
    std::cout << '\n';
}
