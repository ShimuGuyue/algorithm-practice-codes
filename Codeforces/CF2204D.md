void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> graph(n + 1);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    int ans{ 0 };
    std::vector<int> colors(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (colors[i])
            continue;
        bool ok{ true };
        int count1{ 0 }, count2{ 0 };
        auto dfs = [&graph, &colors, &ok, &count1, &count2](this auto&& dfs, int u, int mode) -> void
        {
            colors[u] = mode;
            if (mode == 1)
                ++count1;
            else
                ++count2;
            for (int v : graph[u])
            {
                if (colors[v] == mode)
                    ok = false;
                if (colors[v] == 0)
                    dfs(v, mode ^ 3);
            }
        };
        dfs(i, 1);
        if (ok)
            ans += std::max(count1, count2);
    }
    std::cout << ans << '\n';
}
