void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<std::vector<int>> indexs(n + 1);
    for (int i{ 0 }; i < n; ++i)
    {
        indexs[as[i]].push_back(i);
    }

    std::vector<int> ans;
    int l{ 0 };
    while (l < n)
    {
        int r{ l };
        int mex{ 0 };
        while (1)
        {
            auto it{ std::lower_bound(indexs[mex].begin(), indexs[mex].end(), l) };
            if (it == indexs[mex].end())
                break;
            r = std::max(r, *it);
            ++mex;
        }
        l = r + 1;
        ans.push_back(mex);
    }
    std::cout << ans.size() << '\n';
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
