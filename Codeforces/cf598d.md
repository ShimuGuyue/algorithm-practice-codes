void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<std::vector<char>> grid(n + 1, std::vector<char>(m + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            std::cin >> grid[i][j];
        }
    }

    static constexpr std::array<std::pair<int, int>, 4> dxy{
        {{-1, 0}, {0, -1}, {0, 1}, {1, 0}}
    };

    std::vector<std::vector<bool>> visited(n + 1, std::vector<bool>(m + 1));
    std::vector<std::vector<int>> ans(n + 1, std::vector<int>(m + 1));
    auto dfs1 = [&grid, &visited](this auto& dfs, int x, int y) -> int
    {
        int ans{ 0 };
        for (auto [dx, dy] : dxy)
        {
            int xx{ x + dx };
            int yy{ y + dy };
            if (grid[xx][yy] == '*')
            {
                ++ans;
                continue;
            }
            if (visited[xx][yy])
                continue;
            visited[xx][yy] = true;
            ans += dfs(xx, yy);
        }
        return ans;
    };
    auto dfs2 = [&grid, &ans](this auto& dfs, int x, int y, int k) -> void
    {
        ans[x][y] = k;
        for (auto [dx, dy] : dxy)
        {
            int xx{ x + dx };
            int yy{ y + dy };
            if (grid[xx][yy] == '*')
                continue;
            if (ans[xx][yy])
                continue;
            dfs(xx, yy, k);
        }
    };
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 1 }; j <= m; ++j)
        {
            if (grid[i][j] == '*')
                continue;
            if (visited[i][j])
                continue;
            visited[i][j] = true;
            int flag{ dfs1(i, j) };
            dfs2(i, j, flag);
        }
    }

    while (k--)
    {
        int x, y;
        std::cin >> x >> y;
        std::cout << ans[x][y] << '\n';
    }
}
