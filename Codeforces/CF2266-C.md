void solve()
{
    int n;
    cin >> n;
    string s;
    cin >> s;

    if (s[0] == '1')
    {
        cout << std::count(s.begin(), s.end(), '0') << '\n';
        return;
    }

    s = " " + s + " ";
    vector<int> pre1s(n + 2);
    vector<int> suf0s(n + 2);
    for (int i = 1; i <= n; ++i)
    {
        pre1s[i] = pre1s[i - 1] + (s[i] == '1');
    }
    for (int i = n; i >= 1; --i)
    {
        suf0s[i] = suf0s[i + 1] + (s[i] == '0');
    }
    int ans = n;
    bool find1 = false;
    for (int i = 0; i <= n; ++i)
    {
        if (s[i + 1] == '1')
            find1 = true;
        if (find1)
            ans = std::min(ans, pre1s[i] + suf0s[i + 1]);
        else
            ans = std::min(ans, pre1s[n]);
    }
    cout << ans << '\n';
}
