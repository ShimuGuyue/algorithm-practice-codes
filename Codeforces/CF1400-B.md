    void solve()
    {
        int n, m;
        int counta, countb;
        int pera, perb;
        std::cin >> n >> m;
        std::cin >> counta >> countb;
        std::cin >> pera >> perb;

        if (pera > perb)
        {
            std::swap(counta, countb);
            std::swap(pera, perb);
        }

        int ans{ 0 };
        for (int i{ 0 }; i <= counta; ++i)
        {
            if (i * pera > n)
                break;
            int j{ std::min((n - i * pera) / perb, countb) };
            int aa{ counta - i };
            int bb{ countb - j };

            int x{ std::min(m / pera, aa) };
            int y{ std::min((m - x * pera) / perb, bb) };
            ans = std::max(ans, i + j + x + y);
        }
        std::cout << ans << '\n';
    }
