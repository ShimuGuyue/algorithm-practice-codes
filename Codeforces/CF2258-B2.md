void solve()
{
    int64_t n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int64_t sum{ std::accumulate(as.begin(), as.end(), INT64_C(0)) };

    std::vector<int64_t> counts(m + 1);
    std::vector<int64_t> pres(m + 1);
    for (auto a : as)
    {
        ++counts[a];
    }
    std::partial_sum(counts.begin(), counts.end(), pres.begin());

    std::vector<int64_t> ans(20);
    for (int64_t k{ 1 }; k < 20; ++k)
    {
        int64_t flag{ int64_t(1) << k };
        for (int64_t x{ 1 }; x <= m; ++x)
        {
            int64_t max{ std::min(m, x * flag) };
            int64_t sum{ 0 };
            for (int64_t step{ 1 }; step <= max / x; ++step)
            {
                int64_t l{ step * x };
                int64_t r{ std::min(max, l + x - 1) };
                sum += (pres[r] - pres[l - 1]) * step;
            }
            sum += (flag - 1) * (n - pres[max]);
            ans[k] = std::max(ans[k], sum);
        }
    }

    for (int64_t k{ 1 }; k <= m; ++k)
    {
        std::cout << (k < 20 ? ans[k] : sum) << ' ';
    }
    std::cout << '\n';
}
