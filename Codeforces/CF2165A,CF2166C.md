void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (int i = 0; i < n; ++i)
    {
        std::cin >> as[i];
    }
    int64_t ans{ 0 };
    for (int i = 0; i < n; ++i)
    {
        ans += std::max(as[i], as[(i + 1) % n]);
    }
    ans -= *std::max_element(as.begin(), as.end());
    std::cout << ans << '\n';
}
