void solve()
{
    std::string s;
    std::cin >> s;
    std::string t;
    std::cin >> t;

    std::map<char, int> m;
    for (char c : t)
    {
        ++m[c];
    }
    for (char c : s)
    {
        --m[c];
        if (m[c] < 0)
        {
            std::cout << "Impossible\n";
            return;
        }
    }

    std::string ans;
    int index{ 0 };
    for (char c = 'a'; c <= 'z' && index < s.length(); ++c)
    {
        if (c < s[index])
        {
            while (m[c])
            {
                ans += c;
                --m[c];
            }
        }
        else
        {
            ans += s[index++];
            --c;
        }
    }
    for (char c = 'a'; c <= 'z'; ++c)
    {
        while (m[c])
        {
            ans += c;
            --m[c];
        }
    }
    while (index < s.length())
    {
        ans += s[index++];
    }
    std::cout << ans << '\n';
}
