void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    std::vector<int64_t> sums(n + 1);
    std::vector<int64_t> dp(n + 1);

    auto dfs1 = [&as, &tree, &sums, &dp](this auto&& dfs, int u, int last, int64_t dist) -> void
    {
        sums[u] = as[u];
        dp[1] += dist * as[u];
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u, dist + 1);
            sums[u] += sums[v];
        }
    };
    dfs1(1, 0, 0);

    auto dfs2 = [&tree, &sums, &dp](this auto&& dfs, int u, int last) -> void
    {
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dp[v] = dp[u] + sums[1] - 2 * sums[v];
            dfs(v, u);
        }
    };
    dfs2(1, 0);

    int64_t ans{ *std::max_element(dp.begin() + 1, dp.end()) };
    std::cout << ans << '\n';
}
