    std::cin >> x5 >> y5 >> x6 >> y6;

    auto erase = [](std::set<std::pair<int, int>>& t, std::pair<int, int> seg)
    {
        auto [l, r]{ seg };
        std::vector<std::pair<std::pair<int, int>, std::pair<int, int>>> flags;

            if (r >= rr && l <= rr && l > ll)

                flags.push_back({ { ll, rr }, { ll, l } }),
                flags.push_back({ { ll, rr }, { r, rr } });
            if (l <= ll && r >= rr)
                flags.push_back({ { ll, rr }, { -1, -1 } });
        }
        for (auto [flag1, flag2] : flags)
        {
            t.erase(flag1);
            if (flag2 != std::make_pair(-1, -1))
                t.insert(flag2);
        }
    };

    {

            std::cout << "YES\n";
            return;
        }
    }

    std::cout << "NO\n";
}
