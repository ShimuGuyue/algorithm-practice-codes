void solve()
{
    int n, k;
    std::cin >> n >> k;

    int l{ 1 }, r{ n };
    while (l < r)
    {
        int mid{ (l + r) / 2 };
        if (int64_t(mid) * (mid + 1) / 2 >= k)
            r = mid;
        else
            l = mid + 1;
    }

    int floor{ l };
    int leave{ k - int64_t(floor - 1) * (floor) / 2 };

    std::string ans(n, 'a');
    ans[n - floor - 1] = 'b';
    ans[n - leave] = 'b';
    std::cout << ans << '\n';
}
