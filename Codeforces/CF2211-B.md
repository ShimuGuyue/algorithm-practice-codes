void solve()
{
    int x, y;
    std::cin >> x >> y;

    if (x == 0 || y == 0)
    {
        int a{ x ? x : y };
        int ans{ 0 };
        for (int i{ 1 }; i * i <= a; ++i)
        {
            if (a % i)
                continue;
            ++ans;
            if (i * i != a)
                ++ans;
        }
        std::cout << ans << '\n';
        for (int i{ 0 }; i < a; ++i)
        {
            std::cout << (x ? 1 : -1) << ' ';
        }
        std::cout << '\n';
        return;
    }

    int a{ std::abs(x - y) };
    int ans{ 0 };
    for (int i{ 1 }; i * i <= a; ++i)
    {
        if (a % i)
            continue;
        ++ans;
        if (i * i != a)
            ++ans;
    }
    if (ans == 0)
        ans = 1;
    std::cout << ans << '\n';
    while (x--)
    {
        std::cout << 1 << ' ';
    }
    while (y--)
    {
        std::cout << -1 << ' ';
    }
    std::cout << '\n';
}
