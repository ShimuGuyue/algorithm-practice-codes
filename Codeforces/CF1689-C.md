void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    std::vector<int> sizes(n + 1, 1);
    std::vector<int> dp(n + 1);
    auto dfs = [&tree, &sizes, &dp](this auto& dfs, int u, int last) -> void
    {
        int v1{ 0 }, v2{ 0 };
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            sizes[u] += sizes[v];
            if (v1 == 0)
                v1 = v;
            else
                v2 = v;
        }
        if (v2 == 0)
            dp[u] = sizes[v1] - 1;
        else
            dp[u] = std::max(dp[v1] + sizes[v2] - 1, dp[v2] + sizes[v1] - 1);
    };
    dfs(1, 0);
    std::cout << dp[1] << '\n';
}
