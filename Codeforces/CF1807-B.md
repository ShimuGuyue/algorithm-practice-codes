void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int sum_odd{ std::accumulate(as.begin(), as.end(), 0,
        [](int pre, int next)
        {
            return pre + (next % 2 == 0 ? 0 : next);
        }
    ) };
    int sum_even{ std::accumulate(as.begin(), as.end(), 0,
        [](int pre, int next)
        {
            return pre + (next % 2 == 0 ? next : 0);
        }
    ) };
    if (sum_even > sum_odd)
        std::cout << "yes\n";
    else
        std::cout << "no\n";
}
