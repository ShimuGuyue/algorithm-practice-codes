void solve()
{
    int n, x;
    cin >> n >> x;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    auto check = [&as, x, n](int k) -> bool
    {
        vector<int> flags;
        for (int i = 0; i < n; ++i)
        {
            if (i != 0 && as[i] - as[i - 1] <= k)
                continue;
            if (i != n - 1 && as[i + 1] - as[i] <= k)
                continue;
            flags.push_back(as[i]);
        }
        int m = flags.size();
        int count = 0;
        for (int i = 0; i < m; ++i)
        {
            if (i == m - 1)
            {
                if (++count > x)
                    return false;
            }
            else
            {
                if (flags[i + 1] - flags[i] <= k * 2)
                {
                    if (++count > x)
                        return false;
                    ++i;
                }
                else
                {
                    if (++count > x)
                        return false;
                }
            }
        }
        return true;
    };

    int l = 1, r = 1'000'000'000;
    while (l < r)
    {
        int mid = (l + r) / 2;
        if (check(mid))
            r = mid;
        else
            l = mid + 1;
    }
    cout << r << '\n';
}
