void solve()
{
    int n;
    cin >> n;
    string s;
    cin >> s;

    int count1 = std::count(s.begin(), s.end(), '1');
    if (count1 == 1)
    {
        cout << 0 << '\n';
        return;
    }
    if (count1 == 2)
    {
        cout << 1 << '\n';
        return;
    }
    int l = -1, r;
    for (int i = 0; i < n; ++i)
    {
        if (s[i] == '0')
            continue;
        if (l == -1)
            l = i;
        r = i;
    }
    cout << r - l << '\n';
}
