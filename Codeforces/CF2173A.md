void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    int ans{ 0 };
    int flag{ 0 };
    for (char c : s)
    {
        if (c == '1')
        {
            flag = k;
        }
        else
        {
            if (flag <= 0)
                ++ans;
            --flag;
        }
    }
    std::cout << ans << '\n';
}
