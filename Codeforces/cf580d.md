void solve()
{
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    std::vector<std::vector<int64_t>> exts(n + 1, std::vector<int64_t>(n + 1));
    while (k--)
    {
        int x, y, c;
        std::cin >> x >> y >> c;
        exts[x][y] = c;
    }

    std::vector<std::vector<int64_t>> dp(1 << (n + 1), std::vector<int64_t>(n + 1));
    for (int i{ 1 }; i <= n; ++i)
    {
        dp[1 << i][i] = as[i];
    }
    for (int i{ 1 }; i < (1 << (n + 1)); ++i)
    {
        for (int j{ 1 }; j <= n; ++j)
        {
            if ((i >> j & 1) == 0)
                continue;
            for (int k{ 1 }; k <= n; ++k)
            {
                if (i >> k & 1)
                    continue;
                dp[i | (1 << k)][k] = std::max(dp[i | (1 << k)][k], dp[i][j] + as[k] + exts[j][k]);
            }
        }
    }

    int64_t ans{ 0 };
    for (int i{ 1 }; i < (1 << (n + 1)); ++i)
    {
        if (count(i) != m)
            continue;
        for (int j{ 1 }; j <= n; ++j)
        {
            if ((i >> j & 1) == 0)
                continue;
            ans = std::max(ans, dp[i][j]);
        }
    }
    std::cout << ans << '\n';
}

int count(int n)
{
    return std::bitset<32>(n).count();
}
