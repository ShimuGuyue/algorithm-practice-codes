void solve()
{
    int n;
    std::cin >> n;

    std::vector<std::pair<int, int>> ans;
    int index{ n };
    while (index != 2)
    {
        int next{ int(std::ceil(std::sqrt(index))) };
        for (int i{ next + 1 }; i < index; ++i)
        {
            ans.push_back({ i, index });
        }
        ans.push_back({ index, next });
        ans.push_back({ index, next });
        index = next;
    }

    std::cout << ans.size() << '\n';
    for (auto [x, y] : ans)
    {
        std::cout << x << ' ' << y << '\n';
    }
}
