void solve()
{
    int n;
    std::cin >> n;
    std::array<int, 3> counts{ };
    while (n--)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    int ans{ counts[0] + std::min(counts[1], counts[2]) };
    int min{ std::min(counts[1], counts[2]) };
    counts[1] -= min;
    counts[2] -= min;
    ans += counts[1] / 3;
    ans += counts[2] / 3;
    std::cout << ans << '\n';
}
