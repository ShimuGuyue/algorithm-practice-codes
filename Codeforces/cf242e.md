struct SegmentTree
{
    struct Data
    {
        int seg_l, seg_r;
        std::array<int, 30> tag{ };
        std::array<int64_t, 30> bits{ };
    };

    std::vector<int64_t>& v;
    std::vector<Data> tree;

    SegmentTree(std::vector<int64_t>& v) : v(v)
    {
        tree.assign(v.size() * 4, { });
        build(0, 0, v.size() - 1);
    }

    void build(int node, int l, int r)
    {
        tree[node].seg_l = l;
        tree[node].seg_r = r;
        if (l == r)
        {
            for (int b{ 0 }; b < 30; ++b)
            {
                tree[node].bits[b] = v[l] >> b & 1;
            }
            return;
        }
        int mid{ (l + r) / 2 };
        build(node * 2 + 1, l, mid);
        build(node * 2 + 2, mid + 1, r);
        pushup(node);
    }

    void update(int l, int r, int64_t data, int node = 0)
    {
        auto& [seg_l, seg_r, tag, bits]{ tree[node] };
        if (l == seg_l && r == seg_r)
        {
            for (int b{ 0 }; b < 30; ++b)
            {
                tag[b] ^= data >> b & 1;
                if (data >> b & 1)
                    bits[b] = (seg_r - seg_l + 1) - bits[b];
            }
            return;
        }
        pushdown(node);
        int mid{ (seg_l + seg_r) / 2 };
        if (r <= mid)
            update(l, r, data, node * 2 + 1);
        else if (l > mid)
            update(l, r, data, node * 2 + 2);
        else
            update(l, mid, data, node * 2 + 1),
            update(mid + 1, r, data, node * 2 + 2);
        pushup(node);
    }

    int64_t query(int l, int r, int node = 0)
    {
        auto& [seg_l, seg_r, tag, bits]{ tree[node] };
        if (l == seg_l && r == seg_r)
        {
            int64_t ans{ 0 };
            for (int b{ 0 }; b < 30; ++b)
            {
                ans += bits[b] * (1 << b);
            }
            return ans;
        }
        pushdown(node);
        int mid{ (seg_l + seg_r) / 2 };
        if (r <= mid)
            return query(l, r, node * 2 + 1);
        else if (l > mid)
            return query(l, r, node * 2 + 2);
        else
            return query(l, mid, node * 2 + 1) +
                   query(mid + 1, r, node * 2 + 2);
        return -1;
    }

    void pushup(int node)
    {
        int node_l{ node * 2 + 1 };
        int node_r{ node * 2 + 2 };
        for (int b{ 0 }; b < 30; ++b)
        {
            tree[node].bits[b] = tree[node_l].bits[b] + tree[node_r].bits[b];
        }
    }

    void pushdown(int node)
    {
        for (int i : {1, 2})
        {
            int child{ node * 2 + i };
            auto& [seg_l, seg_r, tag, bits]{ tree[child] };
            for (int b{ 0 }; b < 30; ++b)
            {
                if (tree[node].tag[b] == 0)
                    continue;
                tag[b] ^= 1;
                bits[b] = seg_r - seg_l + 1 - bits[b];
            }
        }
        tree[node].tag = { };
    }
};

void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    SegmentTree st(as);

    int m;
    std::cin >> m;
    while (m--)
    {
        int op;
        std::cin >> op;
        if (op == 1)
        {
            int l, r;
            std::cin >> l >> r;
            std::cout << st.query(l, r) << '\n';
        }
        else
        {
            int l, r, x;
            std::cin >> l >> r >> x;
            st.update(l, r, x);
        }
    }
}
