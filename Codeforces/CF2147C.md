void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    std::vector<std::string> vs;
    int l{ 0 }, r{ 1 };
    while (r < n)
    {
        if (s[r] == s[r - 1])
        {
            std::string s_;
            for (int i = l; i < r; ++i)
            {
                s_ += s[i];
            }
            vs.push_back(s_);
            l = r;
        }
        ++r;
    }
    std::string s_;
    for (int i = l; i < r; ++i)
    {
        s_ += s[i];
    }
    vs.push_back(s_);

    for (auto &ss : vs)
    {
        int count0{ std::count(ss.begin(), ss.end(), '0') };
        bool ok{ false };
        if (count0 % 2 == 0 || ss.front() == '0' || ss.back() == '0')
            ok = true;
        if (!ok)
        {
            std::cout << "no\n";
            return;
        }
    }
    std::cout << "yes\n";
}
