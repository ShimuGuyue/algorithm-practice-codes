void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::map<int, int> m;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        if (a % k == 0)
            continue;
        ++m[k - a % k];
    }

    if (m.empty())
    {
        std::cout << 0 << '\n';
        return;
    }

    int key, value{ 0 };
    for (auto [a, b] : m)
    {
        if (b >= value)
        {
            key = a;
            value = b;
        }
    }

    int64_t ans{ k * int64_t(value - 1) + key + 1 };
    std::cout << ans << '\n';
}
