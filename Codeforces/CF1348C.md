void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    std::sort(s.begin(), s.end());

    int count{ 0 };
    std::string ans{ s[k - 1] };
    n -= k;
    if (s[0] != s[k - 1])
    {
        std::cout << ans << '\n';
        return;
    }
    if (s[k] != s.back())
        ans += std::string(s.begin() + k, s.end());
    else
        ans += std::string((n + k - 1) / k, s[k]);

    std::cout << ans << '\n';
}
