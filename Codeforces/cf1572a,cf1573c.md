void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> indegrees(n + 1);
    std::vector<std::vector<int>> graph(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        int k;
        std::cin >> k;
        indegrees[i] = k;
        while (k--)
        {
            int a;
            std::cin >> a;
            graph[a].push_back(i);
        }
    }

    std::vector<int> ans(n + 1);
    std::queue<int> q;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (indegrees[i] == 0)
        {
            q.push(i);
            ans[i] = 1;
        }
    }
    while (!q.empty())
    {
        int u{ q.front() };
        q.pop();
        for (int v : graph[u])
        {
            if (v > u)
                ans[v] = std::max(ans[v], ans[u]);
            else
                ans[v] = std::max(ans[v], ans[u] + 1);
            if (--indegrees[v] == 0)
                q.push(v);
        }
    }
    if (std::find_if(indegrees.begin(), indegrees.end(), [](int a){ return a > 0; }) != indegrees.end())
    {
        std::cout << -1 << '\n';
        return;
    }
    std::cout << *std::max_element(ans.begin(), ans.end()) << '\n';
}
