void solve()
{
    int64_t a, n;
    std::cin >> a >> n;
    std::vector<int> ds(n);
    for (auto& d : ds)
    {
        std::cin >> d;
    }

    int64_t ans{ std::numeric_limits<int64_t>::max() };

    std::string sa{ std::to_string(a) };
    int len{ int(sa.length()) };

    if (len == 1)
    {
        for (auto d : ds)
        {
            ans = std::min(ans, std::llabs(a - d));
        }
        int64_t b{ (ds[0] == 0 && n > 1 ? ds[1] : ds[0]) * 10 + ds[0] };
        if (b > a)
            ans = std::min(ans, b - a);
        std::cout << ans << '\n';
        return;
    }

    {
        std::string sb(len - 1, ds.back() + '0');
        ans = std::min(ans, a - std::stoll(sb));
    }
    {
        std::string sb(1, (ds.front() == 0 && ds.size() > 1 ? ds[1] : ds[0]) + '0');
        sb += std::string(len, ds[0] + '0');
        int64_t b{ std::stoll(sb) };
        if (b > a)
            ans = std::min(ans, std::stoll(sb) - a);
    }

    {
        std::string sb;
        bool ok{ false };
        bool find{ false };
        auto dfs = [&sa, len, ds, n, &sb, a, &ans, &ok, &find](this auto &&dfs, int index) -> void
        {
            if (index == len)
            {
                ans = std::min(ans, a - std::stoll(sb));
                ok = true;
                return;
            }
            if (!find)
            {
                for (int i{ int(n - 1) }; i >= 0; --i)
                {
                    if (ds[i] + '0' > sa[index])
                        continue;
                    if (ds[i] + '0' < sa[index])
                        find = true;
                    sb.push_back(ds[i] + '0');
                    dfs(index + 1);
                    sb.pop_back();
                    if (ok)
                        return;
                }
            }
            else
            {
                sb.push_back(ds.back() + '0');
                dfs(index + 1);
                if (ok)
                    return;
            }
        };
        dfs(0);
    }
    {
        std::string sb;
        bool ok{ false };
        bool find{ false };
        auto dfs = [&sa, len, ds, n, &sb, a, &ans, &ok, &find](this auto &&dfs, int index) -> void
        {
            if (index == len)
            {
                ans = std::min(ans, std::stoll(sb) - a);
                ok = true;
                return;
            }
            if (!find)
            {
                for (int i{ 0 }; i < n; ++i)
                {
                    if (ds[i] + '0' < sa[index])
                        continue;
                    if (ds[i] + '0' > sa[index])
                        find = true;
                    sb.push_back(ds[i] + '0');
                    dfs(index + 1);
                    sb.pop_back();
                    if (ok)
                        return;
                }
            }
            else
            {
                sb.push_back(ds.front() + '0');
                dfs(index + 1);
                if (ok)
                    return;
            }
        };
        dfs(0);
    }
    std::cout << ans << '\n';
}
