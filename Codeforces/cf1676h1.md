void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    auto merge = [&as](auto&& merge, int l, int r) -> int64_t
    {
        if (l == r)
            return 0;
        int64_t ans{ 0 };
        int mid{ (l + r) / 2 };
        ans += merge(merge, l, mid);
        ans += merge(merge, mid + 1, r);

        std::vector<int> temp;
        temp.reserve(r - l + 1);
        int i{ l }, j{ mid + 1};
        while (i <= mid && j <= r)
        {
            if (as[i] >= as[j])
            {
                ans += mid - i + 1;
                temp.push_back(as[j++]);
            }
            else
            {
                temp.push_back(as[i++]);
            }
        }
        while (i <= mid)
        {
            temp.push_back(as[i++]);
        }
        while (j <= r)
        {
            temp.push_back(as[j++]);
        }
        for (int x = 0; x < temp.size(); ++x)
        {
            as[l + x] = temp[x];
        }
        return ans;
    };

    int64_t ans{ merge(merge, 0, n - 1) };
    std::cout << ans << '\n';
}
