```c++
std::vector<int64_t> flags;
void init()
{
    std::set<int64_t> t;

    int64_t flag = 6;
    int k = 3;
    while (flag < 1000000000000)
    {
        t.insert(flag);
        ++k;
        flag *= k;
    }

    flags.assign(t.rbegin(), t.rend());
}
```

```c++
void solve()
{
    int64_t n;
    std::cin >> n;
    int ans{ INT_MAX };
    dfs(0, 0, 0, n, ans);
    std::cout << ans << '\n';
}

void dfs(int64_t sum, int index, int count, int64_t n, int& ans)
{
    if (sum > n)
        return ;
    ans = std::min(ans, count + std::popcount(uint64_t(n - sum)));
    if (index == flags.size())
        return ;
    dfs(sum + flags[index], index + 1, count + 1, n, ans);
    dfs(sum, index + 1, count, n, ans);
}
```
