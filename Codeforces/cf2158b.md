void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n * 2);
    std::vector<int> counts(n * 2 + 1);
    for (auto &a : as)
    {
        std::cin >> a;
        ++counts[a];
    }

    int count_odd{ 0 };
    int count_even{ 0 };
    for (auto &a : counts)
    {
        if (a == 0)
            continue;
        if (a & 1)
            ++count_odd;
        else
            ++count_even;
    }

    int ans{ count_odd + count_even * 2 };
    if (count_odd == 0 && count_even % 2 != n % 2)
        ans -= 2;
    std::cout << ans << '\n';
}
