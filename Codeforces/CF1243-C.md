void solve()
{
    int64_t n;
    std::cin >> n;
    if (is_prime(n))
    {
        std::cout << n << '\n';
    }
    else
    {
        int count{ 0 };
        for (int64_t i{ 2 }; i * i <= n; ++i)
        {
            if (n % i)
                continue;
            if (is_prime(i))
                ++count;
            if (i * i != n && is_prime(n / i))
                ++count;
        }
        if (count > 1)
        {
            std::cout << 1 << '\n';
        }
        else
        {
            for (int i{ 2 }; i <= n; ++i)
            {
                if (n % i == 0 && is_prime(i))
                {
                    std::cout << i << '\n';
                    break;
                }
            }
        }
    }
}

bool is_prime(int64_t n)
{
    for (int64_t i{ 2 }; i * i <= n; ++i)
    {
        if (n % i == 0)
            return false;
    }
    return true;
}
