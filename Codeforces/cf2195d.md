void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> fs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> fs[i];
    }

    std::vector<int64_t> as(n + 1);
    for (int i{ 2 }; i <= n - 1; ++i)
    {
        /*
            (f1 - f2) - (f2 - f3) = 2 * a2
        */
        as[i] = (fs[i - 1] - 2 * fs[i] + fs[i + 1]) / 2;
    }
    /*
        f1 = 0 * a1 + 1 * a2 + 2 * a3 + ... + (n - 1) * an
        fn = (n - 1) * a1 + (n - 2) * a2 + ... + 1 * an-1 + 0 * an
    */
    int64_t sum1{ 0 };
    for (int i{ 1 }; i <= n - 1; ++i)
    {
        sum1 += as[i] * (i - 1);
    }
    as[n] = (fs[1] - sum1) / (n - 1);

    int64_t sum2{ 0 };
    for (int i{ 2 }; i <= n; ++i)
    {
        sum2 += as[i] * (n - i);
    }
    as[1] = (fs[n] - sum2) / (n - 1);


    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << as[i] << ' ';
    }
    std::cout << '\n';
}
