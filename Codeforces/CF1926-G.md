void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 2 }; i <= n; ++i)
    {
        int a;
        std::cin >> a;
        tree[i].push_back(a);
        tree[a].push_back(i);
    }
    std::string s;
    std::cin >> s;
    s = " " + s;

    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(2, n));
    auto dfs = [&tree, &s, &dp](this auto& dfs, int u, int last) -> void
    {
        if (s[u] == 'S' || s[u] == 'C')
            dp[u][0] = 0;
        if (s[u] == 'P' || s[u] == 'C')
            dp[u][1] = 0;
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            dp[u][0] += std::min(dp[v][0], dp[v][1] + 1);
            dp[u][1] += std::min(dp[v][1], dp[v][0] + 1);
        }
    };
    dfs(1, 0);

    int ans{ std::min(dp[1][0], dp[1][1]) };
    std::cout << ans << '\n';
}
