void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    std::string ans(k, 'z');
    for (int i{ 1 }; i <= n; ++i)
    {
        std::string str;
        for (int j{ 0 }; j < k; ++j)
        {
            str += s[j % i];
        }
        ans = std::min(ans, str);
    }
    std::cout << ans << '\n';
}
