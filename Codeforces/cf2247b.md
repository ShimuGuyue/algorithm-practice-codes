void solve()
{
    int n, k, m;
    std::cin >> n >> k >> m;
    if (k > m)
    {
        std::cout << "No\n";
        return;
    }

    std::cout << "Yes\n";
    for (int i{ 1 }; i < k; ++i)
    {
        std::cout << 1 << ' ';
    }
    std::cout << m - k + 1 << ' ';
    for (int i{ k + 1 }; i <= n; ++i)
    {
        std::cout << 1 << ' ';
    }
    std::cout << '\n';
}
