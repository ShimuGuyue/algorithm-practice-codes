void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> digits(n);
    for (auto &v : digits)
    {
        int l;
        std::cin >> l;
        v.resize(l);
        for (auto &a : v)
        {
            std::cin >> a;
        }
    }

    std::vector<int> counts(m + 1);
    for (auto &v : digits)
    {
        for (auto a : v)
        {
            ++counts[a];
        }
    }

    for (int i = 1; i <= m; ++i)
    {
        if (counts[i] == 0)
        {
            std::cout << "no\n";
            return;
        }
    }

    int ans{ 1 };
    for (auto &v : digits)
    {
        bool ok{ true };
        for (auto a : v)
        {
            --counts[a];
            if (counts[a] == 0)
                ok = false;
        }
        for (auto a : v)
        {
            ++counts[a];
        }
        if (ok)
            ++ans;
        if (ans == 3)
        {
            std::cout << "yes\n";
            return;
        }
    }
    std::cout << "no\n";
}
