```cpp
class DisjointSet
{
private:
    struct Data
    {
        int father;
    };
    std::vector<Data> nodes_;

public:
    DisjointSet(const int n, const int count)
    {
        build(n, count);
    }

public:
    int find(const int x)
    {
        return nodes_[x].father == x ? x : nodes_[x].father = find(nodes_[x].father);
    }

    void merge(const int x, const int y)
    {
        const int set_x{ find(x) };
        const int set_y{ find(y) };
        if (set_x == set_y)
            return;
        nodes_[set_y].father = set_x;
    }

private:
    void build(const int n, const int count)
    {
        nodes_.assign(count * n, { });
        for (int i{ 0 }; i < count * n; ++i)
        {
            nodes_[i].father = i;
        }
    }
};
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;

    DisjointSet ds(n + 1, 2);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        if (ds.find(u) == ds.find(v))
        {
            std::cout << -1 << '\n';
            return;
        }
        ds.merge(u, v + n);
        ds.merge(v, u + n);
    }

    std::vector<int> flags(n * 2 + 1);
    std::vector<int> ans1;
    std::vector<int> ans2;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (flags[ds.find(i)] == 0)
        {
            flags[ds.find(i)] = 1;
            flags[ds.find(i + n)] = 2;
        }
        if (flags[ds.find(i)] == 1)
            ans1.push_back(i);
        else
            ans2.push_back(i);
    }

    std::cout << ans1.size() << '\n';
    for (auto a : ans1)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
    std::cout << ans2.size() << '\n';
    for (auto a : ans2)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
```
