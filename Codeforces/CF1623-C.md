void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    auto check = [&as, n](int k)
    {
        auto v{ as };
        for (int i{ n - 1 }; i >= 0; --i)
        {
            if (v[i] < k)
                return false;
            if (i < 2)
                continue;
            int per{ std::min(as[i] / 3, (v[i] - k) / 3) };
            v[i - 1] += per;
            v[i - 2] += per * 2;
        }
        return true;
    };
    int l{ 1 }, r{ 1000000000 };
    while (l < r)
    {
        int mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    std::cout << l << '\n';
}
