void solve()
{
    int n;
    std::cin >> n;

    std::vector<std::pair<int, int>> lens;
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << "? " << i << ' ' << n << ' ';
        for (int j{ 1 }; j <= n; ++j)
        {
            std::cout << j << ' ';
        }
        std::cout << std::endl;

        int x;
        std::cin >> x;
        lens.push_back({ i, x });
    }

    std::sort(lens.begin(), lens.end(),
        [](std::pair<int, int> a, std::pair<int, int> b)
        {
            return a.second > b.second;
        }
    );

    std::vector<int> ans;
    int last{ lens.begin()->first };
    ans.push_back(last);
    for (int i{ 1 }; i < n; ++i)
    {
        auto [x, len]{ lens[i] };
        std::cout << "? " << last << ' ' << 2 << ' ' << last << ' ' << x << std::endl;
        int k;
        std::cin >> k;
        if (k == 2)
        {
            last = x;
            ans.push_back(last);
        }
    }

    std::cout << "! " << ans.size() << ' ';
    for (int a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << std::endl;
}
