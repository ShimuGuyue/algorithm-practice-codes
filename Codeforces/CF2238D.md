

```cpp
std::vector<int> primes;

void init()
{
    int n{ 1000000 };
    std::vector<bool> is_primes(n + 1, true);
    primes.push_back(2);
    for (int i{ 3 }; i <= n; i += 2)
    {
        if (!is_primes[i])
            continue;
        primes.push_back(i);
        for (int j{ i }; j <= n; j += i)
        {
            is_primes[j] = false;
        }
    }
}
```

```cpp
void solve()
{
    int n;
    std::cin >> n;

    int ans{ -1 };
    for (int p : primes)
    {
        if (n % p)
            continue;
        ++ans;
        while (n % p == 0)
        {
            n /= p;
            ++ans;
        }
        if (n == 1)
            break;
    }
    std::cout << ans << '\n';
}
```
