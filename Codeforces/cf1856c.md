void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int l{ *std::max_element(as.begin(), as.end()) }, r{ 200000000 };
    while (l < r)
    {
        auto check = [&as, n, k](int max) -> bool
        {
            for (int i{ 0 }; i < n - 1; ++i)
            {
                int64_t kk{ k };
                int flag{ max };
                for (int j{ i }; j < n - 1; ++j)
                {
                    if (as[j] >= flag)
                    {
                        if (kk >= 0)
                            return true;
                        else
                            break;
                    }
                    int dif{ flag - as[j] };
                    kk -= dif;
                    flag = std::max(0, flag - 1);
                    if (j == n - 2 && kk >= 0 && flag <= as[n - 1])
                        return true;
                }
            }
            return false;
        };
        int mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }

    std::cout << l << '\n';
}
