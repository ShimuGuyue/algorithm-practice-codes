void solve()
{
    int n, k;
    std::cin >> n >> k;

    int bit{ 0 };
    while ((1 << bit) != n)
    {
        ++bit;
    }

    int64_t ans{ 0 };
    if (bit >= k)
        ++ans;
    for (int i = bit - 1; i > 0; --i)
    {
        for (int j = std::max(k - i, 0); j <= i; ++j)
        {
            ans += combinate(i, j);
        }
    }
    std::cout << ans << '\n';
}

int64_t combinate(int n, int k)
{
    if (k == n || k == 0)
        return 1;
    if (k > n)
        return 0;
    int64_t ans{ 1 };
    for (int i = 1; i <= k; ++i)
    {
        ans = ans * (n - i + 1) / i;
    }
    return ans;
}
