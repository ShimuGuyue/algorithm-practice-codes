```c++
static constexpr int64_t mod{ 1000000007 };

std::vector<int64_t> factorials;
std::vector<int64_t> inv_factorials;

void init()
{
    factorials.assign(1001, 1);
    for (int i = 2; i <= 1000; ++i)
    {
        factorials[i] = factorials[i - 1] * i % mod;
    }

    inv_factorials.assign(1001, 1);
    inv_factorials[1000] = quickpow(factorials[1000], mod - 2);
    for (int i = 999; i >= 0; --i)
        inv_factorials[i] = inv_factorials[i + 1] * (i + 1) % mod;
}
```

```c++
void solve()
{
    int n, k;
    std::cin >> n >> k;

    std::map<int, int> counts;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    std::vector<std::pair<int, int>> v(counts.rbegin(), counts.rend());

    int64_t ans{ 1 };
    for (auto& [a, count] : v)
    {
        if (k > count)
        {
            k -= count;
        }
        else
        {
            ans *= factorials[count] * inv_factorials[k] % mod * inv_factorials[count - k] % mod;
            break;
        }
    }
    std::cout << ans << '\n';
}

int64_t quickpow(int64_t base, int64_t n)
{
    int64_t ans{ 1 };
    base %= mod;
    while (n)
    {
        if (n & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        n /= 2;
    }
    return ans;
}
```
