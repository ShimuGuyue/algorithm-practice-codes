void solve()
{
    int n;
    std::cin >> n;
    if (n == 1)
    {
        std::cout << 1 << '\n';
    }
    else if (n == 2)
    {
        std::cout << 9 << '\n';
    }
    else
    {
        int ans1 = (n * (n - 1) - 1) + (n * n - 1) + (n * n - 2) + (n * n);
        int ans2 = (n * (n - 1) - 1) + (n * n - 1) + (n * (n - 1) - 2) + (n * (n - 1) + (n * (n - 2) - 1));
        std::cout << std::max(ans1, ans2) << '\n';
    }
}
