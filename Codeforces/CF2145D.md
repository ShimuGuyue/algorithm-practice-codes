void solve()
{
    int n, k;
    std::cin >> n >> k;

    int sum{ n * (n - 1) / 2 };
    k = sum - k;

    std::vector<std::vector<bool>> dp(n + 1, std::vector<bool>(sum + 1));
    std::vector<std::vector<int>> pres(n + 1, std::vector<int>(sum + 1));
    dp[0][0] = true;
    for (int i{ 1 }; i <= n; ++i)
    {
        for (int j{ 0 }; j <= sum; ++j)
        {
            for (int len{ 1 }; len <= i; ++len)
            {
                int cur{ len * (len - 1) / 2 };
                if (j >= cur && dp[i - len][j - cur])
                {
                    dp[i][j] = true;
                    pres[i][j] = len;
                    break;
                }
            }
        }
    }
    if (!dp[n][k])
    {
        std::cout << 0 << '\n';
        return;
    }

    std::vector<int> lens;
    int index{ n }, flag{ k };
    while (index)
    {
        int len{ pres[index][flag] };
        lens.push_back(len);
        index -= len;
        flag -= len * (len - 1) / 2;
    }
    std::reverse(lens.begin(), lens.end());

    int max{ n };
    for (int len : lens)
    {
        for (int i{ max - len + 1 }; i <= max; ++i)
        {
            std::cout << i << ' ';
        }
        max -= len;
    }
    std::cout << '\n';
}
