void solve()
{
    std::string s;
    std::cin >> s;
    std::string t;
    std::cin >> t;

    auto n{ int(s.length()) };
    auto m{ int(t.length()) };

    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    auto check = [&s, &t, &as, n, m](int k) -> bool
    {
        std::vector<bool> deleted(n + 1);
        for (int i{ 0 }; i < k; ++i)
        {
            deleted[as[i] - 1] = true;
        }
        std::string str;
        for (int i{ 0 }; i < n; ++i)
        {
            if (!deleted[i])
                str += s[i];
        }
        int i{ 0 }, j{ 0 };
        while (i < int(str.length()))
        {
            if (str[i] == t[j])
                ++j;
            if (j == m)
                return true;
            ++i;
        }
        return false;
    };
    int l{ 0 }, r{ n - m };
    while (l < r)
    {
        int mid{ (l + r + 1) / 2 };
        if (check(mid))
            l = mid;
        else
            r = mid - 1;
    }
    std::cout << l << '\n';
}
