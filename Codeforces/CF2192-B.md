void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    s = " " + s;

    int count0{ std::count(s.begin(), s.end(), '0') };
    int count1{ std::count(s.begin(), s.end(), '1') };

    if (count0 % 2 == 1)
    {
        std::cout << count0 << '\n';
        for (int i{ 1 }; i <= n; ++i)
        {
            if (s[i] == '0')
                std::cout << i << ' ';
        }
        std::cout << '\n';
        return;
    }
    if (count1 % 2 == 0)
    {
        std::cout << count1 << '\n';
        for (int i{ 1 }; i <= n; ++i)
        {
            if (s[i] == '1')
                std::cout << i << ' ';
        }
        std::cout << '\n';
        return;
    }
    std::cout << -1 << '\n';
}
