void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<std::pair<int, int>> flags;
    for (int i{ 0 }; i < n; ++i)
    {
        int a{ as[i] };
        int count{ 0 };
        bool flag1{ false };
        bool flag2{ false };
        while (1)
        {
            flags.push_back({ a, count });
            if (a == 1)
                flag1 = true;
            if (a == 2)
                flag2 = true;
            if (flag1 && flag2)
                break;
            ++count;
            a = (a % 2 == 1) ? a + 1 : a / 2;
        }
    }
    std::sort(flags.begin(), flags.end());

    int ans{ std::numeric_limits<int>::max() };
    int l{ 0 }, r{ 0 };
    int sum{ 0 };
    while (r < flags.size())
    {
        sum += flags[r].second;
        if (flags[r].first == flags[l].first)
        {
            if (r - l + 1 == n)
                ans = std::min(ans, sum);
            ++r;
        }
        else
        {
            l = r;
            sum = 0;
        }
    }
    std::cout << ans << '\n';
}
