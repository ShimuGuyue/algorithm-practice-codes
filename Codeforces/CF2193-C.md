void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n + 1);
    std::vector<int> bs(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i = 1; i <= n; ++i)
    {
        std::cin >> bs[i];
    }

    as[n] = std::max(as[n], bs[n]);
    for (int i = n - 1; i >= 1; --i)
    {
        as[i] = std::max({as[i], as[i + 1], bs[i]});
    }

    std::vector<int> presums(n + 1);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    while (q--)
    {
        int l, r;
        std::cin >> l >> r;
        std::cout << presums[r] - presums[l - 1] << ' ';
    }
    std::cout << '\n';
}
