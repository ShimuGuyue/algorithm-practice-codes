void solve()
{
    int n, m;
    std::cin >> n >> m;
    std::vector<int64_t> as(n);
    std::vector<int64_t> bs(m);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }

    if (n == 1)
    {
        for (auto b : bs)
        {
            std::cout << as[0] + b << ' ';
        }
        std::cout << '\n';
        return;
    }

    int64_t g{ std::llabs(as[1] - as[0]) };
    for (int i = 2; i < n; ++i)
    {
        g = std::gcd(g, std::llabs(as[i] - as[0]));
    }

    for (auto b : bs)
    {
        std::cout << std::gcd(as[0] + b, g) << ' ';
    }
    std::cout << '\n';
}
