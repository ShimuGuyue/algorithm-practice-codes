void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::string s;
    std::cin >> s;

    s = " " + s;

    auto judge1 = [&s](int i) -> bool
    {
        return (i % 2 == 1 && s[i] == '1') || (i % 2 == 0 && s[i] == '0');
    };
    auto judge2 = [&s](int i) -> bool
    {
        return (i % 2 == 1 && s[i] == '0') || (i % 2 == 0 && s[i] == '1');
    };

    std::vector<int> ls1;
    std::vector<int> rs1;
    std::vector<int> ls2;
    std::vector<int> rs2;
    for (int i{ 1 }, l{ 0 }, r{ 0 }; i <= n; ++i)
    {
        if (judge1(i))
        {
            if (l != 0)
            {
                ls1.push_back(l);
                rs1.push_back(r);
            }
            l = 0;
            r = 0;
        }
        else
        {
            if (l == 0)
                l = i;
            r = i;
        }
        if (l != 0 && i == n)
        {
            ls1.push_back(l);
            rs1.push_back(r);
        }
    }
    for (int i{ 1 }, l{ 0 }, r{ 0 }; i <= n; ++i)
    {
        if (judge2(i))
        {
            if (l != 0)
            {
                ls2.push_back(l);
                rs2.push_back(r);
            }
            l = 0;
            r = 0;
        }
        else
        {
            if (l == 0)
                l = i;
            r = i;
        }
        if (l != 0 && i == n)
        {
            ls2.push_back(l);
            rs2.push_back(r);
        }
    }

    while (q--)
    {
        int l, r, k;
        std::cin >> l >> r >> k;

        int sum1, sum2;

        int r1 = std::lower_bound(rs1.begin(), rs1.end(), l) - rs1.begin();
        int l1 = std::upper_bound(ls1.begin(), ls1.end(), r) - ls1.begin();
        sum1 = (r1 < l1) ? (l1 - r1) : 0;

        int r2 = std::lower_bound(rs2.begin(), rs2.end(), l) - rs2.begin();
        int l2 = std::upper_bound(ls2.begin(), ls2.end(), r) - ls2.begin();
        sum2 = (r2 < l2) ? (l2 - r2) : 0;

        int ans{ std::min(sum1, sum2) };
        std::cout << (ans <= k ? "Yes" : "No") << '\n';
    }
}
