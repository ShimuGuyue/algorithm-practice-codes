void solve()
{
    int n;
    std::cin >> n;
    std::map<int64_t, int64_t> m;
    for (int i{ 0 }; i < n; ++i)
    {
        int64_t a;
        std::cin >> a;
        ++m[a];
    }

    int q;
    std::cin >> q;
    while (q--)
    {
        int64_t x, y;
        std::cin >> x >> y;
        /*
            a + b = x   b = x - a
            a * b = y   a * x - a * a = y   a^2 - x * a + y = 0
        */
        int64_t dert{ x * x - 4 * 1 * y };
        if (dert < 0)
        {
            std::cout << 0 << ' ';
            continue;
        }
        int64_t x1{ (x + int64_t(std::sqrt(dert))) / 2 };
        int64_t x2{ (x - int64_t(std::sqrt(dert))) / 2 };
        if (x1 + x2 != x || x1 * x2 != y)
        {
            std::cout << 0 << ' ';
            continue;
        }
        if (x1 == x2)
            std::cout << m[x1] * (m[x1] - 1) / 2 << ' ';
        else
            std::cout << m[x1] * m[x2] << ' ';
    }
    std::cout << '\n';
}
