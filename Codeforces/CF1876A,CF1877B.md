void solve()
{
    int64_t n, p;
    std::cin >> n >> p;
    std::vector<std::pair<int64_t, int64_t>> abs(n);
    for (auto& [a, b] : abs)
    {
        std::cin >> a;
    }
    for (auto& [a, b] : abs)
    {
        std::cin >> b;
    }

    std::sort(abs.begin(), abs.end(),
        [](std::pair<int64_t, int64_t> a, std::pair<int64_t, int64_t> b)
        {
            return a.second < b.second;
        }
    );

    int64_t ans{ 0 };
    std::queue<std::pair<int64_t, int64_t>> q;
    for (auto [a, b] : abs)
    {
        if (q.empty())
        {
            ans += p;
        }
        else
        {
            auto& [aa, bb]{ q.front() };
            ans += bb;
            --aa;
            if (aa == 0)
                q.pop();
        }
        if (b < p)
            q.push({ a, b });
    }
    std::cout << ans << '\n';
}
