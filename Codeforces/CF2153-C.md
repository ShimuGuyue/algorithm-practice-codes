void solve()
{
    int n;
    std::cin >> n;
    std::map<int64_t, int> m;
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++m[a];
    }

    int count{ 0 };
    for (auto [k, v] : m)
    {
        count += v / 2;
    }

    if (count == 0)
    {
        std::cout << 0 << '\n';
        return;
    }
    if (count == 1)
    {
        bool ok{ false };
        int64_t ans{ 0 };
        int64_t len2;
        for (auto &[k, v] : m)
        {
            if (v == 1)
                continue;
            len2 = k;
            v %= 2;
        }
        std::vector<int64_t> flags;
        flags.reserve(n);
        for (auto [k, v]: m)
        {
            if (v)
                flags.push_back(k);
        }
        std::reverse(flags.begin(), flags.end());
        for (int i = 0; i < flags.size(); ++i)
        {
            if (len2 * 2 > flags[i])
            {
                ok = true;
                ans = std::max(ans, len2 * 2 + flags[i]);
            }
            for (int j = i + 1; j < flags.size(); ++j)
            {
                if (len2 * 2 + flags[j] > flags[i])
                {
                    ok = true;
                    ans = std::max(ans, len2 * 2 + flags[j] + flags[i]);
                    break;
                }
            }
        }
        std::cout << (ok ? ans : 0) << '\n';
        return;
    }

    int64_t ans{ 0 };
    for (auto &[k, v] : m)
    {
        ans += k * (v / 2 * 2);
        v %= 2;
    }
    int64_t sum2{ ans };
    std::vector<int64_t> flags;
    flags.reserve(n);
    for (auto [k, v] : m)
    {
        if (v)
            flags.push_back(k);
    }
    std::reverse(flags.begin(), flags.end());
    for (int i = 0; i < flags.size(); ++i)
    {
        if (sum2 > flags[i])
        {
            ans = std::max(ans, sum2 + flags[i]);
        }
        for (int j = i + 1; j < flags.size(); ++j)
        {
            if (sum2 + flags[j] > flags[i])
            {
                ans = std::max(ans, sum2 + flags[j] + flags[i]);
                break;
            }
        }
    }
    std::cout << ans << '\n';
}
