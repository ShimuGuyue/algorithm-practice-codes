void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }
    int max = *max_element(as.begin(), as.end());
    if (max == 0)
    {
        cout << 0 << '\n';
        return;
    }

    int ans = max;
    array<int, 31> visited = { };
    for (int i = 30; i >= 0; --i)
    {
        if (max >> i & 1)
            visited[i] = true;
    }
    for (int& a : as)
    {
        for (int i = 30; i >= 0; --i)
        {
            if (visited[i])
                continue;
            if ((a | (1 << i)) > max)
                continue;
            a |= 1 << i;
            visited[i] = true;
            ans |= 1 << i;
        }
    }
    cout << ans << '\n';
}
