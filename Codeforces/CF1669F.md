void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> presums(n);
    std::vector<int> sufsums(n);
    std::partial_sum(as.begin(), as.end(), presums.begin());
    std::partial_sum(as.rbegin(), as.rend(), sufsums.rbegin());

    std::map<int, int> pres;
    std::map<int, int> sufs;
    for (int i = 0; i < n; ++i)
    {
        pres[presums[i]] = i;
        sufs[sufsums[i]] = i;
    }

    int sum(presums.back());

    int ans{ 0 };
    for (auto [k, v] : pres)
    {
        if (sufs.find(k) == sufs.end())
            continue;
        int index1{ v };
        int index2{ sufs[k] };
        if (index1 >= index2)
            continue;;
        ans = n - (index2 - index1 - 1);
    }
    std::cout << ans << '\n';
}
