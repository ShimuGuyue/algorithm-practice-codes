void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int count0{ std::count(as.begin(), as.end(), 0) };
    if (count0 == n)
    {
        std::cout << "No\n";
        return;
    }

    std::cout << "Yes\n";

    std::vector<int64_t> posis;
    std::vector<int64_t> negas;
    for (auto a : as)
    {
        if (a >= 0)
            posis.push_back(a);
        else
            negas.push_back(a);
    }

    int64_t sum{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (sum <= 0)
        {
            std::cout << posis.back() << ' ';
            sum += posis.back();
            posis.pop_back();
        }
        else
        {
            std::cout << negas.back() << ' ';
            sum += negas.back();
            negas.pop_back();
        }
    }
    std::cout << '\n';
}
