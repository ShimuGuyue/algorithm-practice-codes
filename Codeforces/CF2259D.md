void solve()
{
    int n;
    cin >> n;
    vector<int> as(n);
    for (auto& a : as)
    {
        cin >> a;
    }

    int count0{ std::count(as.begin(), as.end(), 0) };
    if (count0 == 0)
    {
        cout << "Yes\n";
        cout << string(n, 'A') << '\n';
        return;
    }
    if (count0 == 1)
    {
        cout << "No\n";
        return;
    }
    if (count0 >= 2)
    {
        cout << "Yes\n";
        string ans(n, ' ');
        bool flag{ false };
        for (int i{ 0 }; i < n; ++i)
        {
            if (as[i] == 0)
            {
                if (flag)
                {
                    ans[i] = 'B';
                }
                else
                {
                    ans[i] = 'A';
                    flag = true;
                }
            }
            else
            {
                ans[i] = 'C';
            }
        }
        cout << ans << '\n';
        return;
    }
}
