void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::set<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].insert(v);
        tree[v].insert(u);
    }
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (as.front() != 1)
    {
        std::cout << "No\n";
        return;
    }

    std::vector<bool> visited(n + 1);
    std::queue<int> q;
    q.push(1);
    visited[1] = true;
    int index{ 1 };
    while (!q.empty())
    {
        int u{ q.front() };
        q.pop();

        auto t{ tree[u] };
        for (auto a : tree[u])
        {
            if (visited[a])
                t.erase(a);
        }

        while (!t.empty())
        {
            if (t.find(as[index]) == t.end())
            {
                std::cout << "No\n";
                return;
            }
            visited[as[index]] = true;
            t.erase(as[index]);
            q.push(as[index]);
            ++index;
        }
    }
    std::cout << "Yes\n";
}
