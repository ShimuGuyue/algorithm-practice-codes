```cpp
inline static constexpr int64_t mod{ 1000000007 };
inline static std::array<std::array<int64_t, 200001>, 10> dp{ };

void init()
{
    for (int i{ 0 }; i < 10; ++i)
    {
        dp[i][0] = 1;
        std::array<int64_t, 10> flags{ };
        flags[i] = 1;
        for (int j{ 1 }; j <= 200000; ++j)
        {
            std::array<int64_t, 10> cur{ };
            for (int k{ 0 }; k < 10; ++k)
            {
                if (k == 9)
                {
                    cur[1] = (cur[1] + flags[k]) % mod;
                    cur[0] = (cur[0] + flags[k]) % mod;
                }
                else
                {
                    cur[k + 1] = (cur[k + 1] + flags[k]) % mod;
                }
            }
            flags = cur;
            dp[i][j] = std::accumulate(flags.begin(), flags.end(), INT64_C(0)) % mod;
        }
    }
}
```

```cpp
void solve()
{
    int n, m;
    std::cin >> n >> m;
    int64_t ans{ 0 };
    while (n)
    {
        int i{ n % 10 };
        n /= 10;
        ans = (ans + dp[i][m]) % mod;
    }
    std::cout << ans << '\n';
}
```
