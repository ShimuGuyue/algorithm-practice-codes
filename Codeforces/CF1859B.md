void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> as(n);
    for (auto& v : as)
    {
        int m;
        std::cin >> m;
        while (m--)
        {
            int a;
            std::cin >> a;
            v.push_back(a);
        }
        std::sort(v.begin(), v.end());
    }

    int min{ std::numeric_limits<int>::max() };
    int index;
    for (int i{ 0 }; i < n; ++i)
    {
        if (as[i][1] < min)
        {
            min = as[i][1];
            index = i;
        }
    }

    std::for_each(as.begin(), as.end(),
        [&min](std::vector<int>& v)
        {
            min = std::min(min, v[0]);
        }
    );

    int64_t ans{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (i == index)
            ans += min;
        else
            ans += as[i][1];
    }
    std::cout << ans << '\n';
}
