void solve()
{
    int n, k;
    std::cin >> n >> k;

    std::array<int, 31> counts{ };
    for (int i = 0; i < n; ++i)
    {
        int a;
        std::cin >> a;
        for (int b = 0; b < 31; ++b)
        {
            if (a >> b & 1)
                ++counts[b];
        }
    }

    int ans{ 0 };
    for (int b = 30; b >= 0; --b)
    {
        int dif{ n - counts[b] };
        if (k >= dif)
        {
            k -= dif;
            ans |= 1 << b;
        }
    }
    std::cout << ans << '\n';
}
