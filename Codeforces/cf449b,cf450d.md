void solve()
{
    using Edge = std::pair<int64_t, int>;

    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<std::vector<Edge>> graph(n + 1);
    std::vector<Edge> edges(k);
    while (m--)
    {
        int u, v, w;
        std::cin >> u >> v >> w;
        graph[u].push_back({ w, v });
        graph[v].push_back({ w, u });
    }
    for (auto& [w, v] : edges)
    {
        std::cin >> v >> w;
        graph[1].push_back({ w, v });
    }

    std::vector<int64_t> minlens(n + 1, std::numeric_limits<int64_t>::max());
    std::vector<int> counts(n + 1);
    std::priority_queue<Edge, std::vector<Edge>, std::greater<Edge>> h;
    minlens[1] = 0;
    h.push({ 0, 1 });
    while (!h.empty())
    {
        auto [len, u]{ h.top() };
        h.pop();
        if (len > minlens[u])
            continue;
        for (auto [w, v] : graph[u])
        {
            if (len + w > minlens[v])
                continue;
            if (len + w == minlens[v])
                ++counts[v];
            if (len + w < minlens[v])
            {
                minlens[v] = len + w;
                counts[v] = 1;
                h.push({ minlens[v], v });
            }
        }
    }

    int ans{ 0 };
    for (auto [w, v] : edges)
    {
        if (w > minlens[v])
            ++ans;
        if (w == minlens[v] && counts[v] > 1)
        {
            --counts[v];
            ++ans;
        }
    }
    std::cout << ans << '\n';
}
