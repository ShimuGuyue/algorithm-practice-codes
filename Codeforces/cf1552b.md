void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> grades(n + 1, std::vector<int>(5));
    for (int i = 1; i <= n; ++i)
    {
        for (int j = 0; j < 5; ++j)
        {
            std::cin >> grades[i][j];
        }
    }

    int index{ 1 };
    for (int i = 2; i <= n; ++i)
    {
        int count{ 0 };
        for (int j = 0; j < 5; ++j)
        {
            count += grades[index][j] < grades[i][j];
        }
        if (count < 3)
            index = i;
    }
    for (int i = 1; i <= n; ++i)
    {
        if (i == index)
            continue;
        int count{ 0 };
        for (int j = 0; j < 5; ++j)
        {
            count += grades[index][j] < grades[i][j];
        }
        if (count < 3)
        {
            index = -1;
            break;
        }
    }
    std::cout << index << '\n';
}
