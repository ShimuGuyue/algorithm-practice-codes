void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int mex{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (as[i] == mex)
            ++mex;
        else
            break;
    }

    int x{ mex };
    while (1)
    {
        std::cout << x << std::endl;

        int y;
        std::cin >> y;
        if (y == -1)
            break;

        x = y;
    }
}
