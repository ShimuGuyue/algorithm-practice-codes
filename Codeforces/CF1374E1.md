void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as;
    std::vector<int> bs;
    std::vector<int> cs;
    while (n--)
    {
        int t, a, b;
        std::cin >> t >> a >> b;
        if (a && !b)
            as.push_back(t);
        if (!a && b)
            bs.push_back(t);
        if (a && b)
            cs.push_back(t);
    }

    if (cs.size() + std::min(as.size(), bs.size()) < k)
    {
        std::cout << -1 << '\n';
        return;
    }

    std::sort(as.begin(), as.end());
    std::sort(bs.begin(), bs.end());
    for (int i{ 0 }; i < std::min<int>(as.size(), bs.size()); ++i)
    {
        cs.push_back(as[i] + bs[i]);
    }
    std::sort(cs.begin(), cs.end());
    int ans{ std::accumulate(cs.begin(), cs.begin() + k, 0) };
    std::cout << ans << '\n';
}
