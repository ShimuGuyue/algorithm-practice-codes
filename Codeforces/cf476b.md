void solve()
{
    int n{ 10 };
    std::string s1;
    std::cin >> s1;
    for (char c : s1)
    {
        if (c == '+')
            ++n;
        else
            --n;
    }

    std::string s2;
    std::cin >> s2;
    std::vector<std::vector<double>> dp(s2.length() + 1, std::vector<double>(20 + 1));
    dp[0][10] = 1;
    s2 = " " + s2;
    for (int i = 1; i < s2.length(); ++i)
    {
        for (int j = 0; j <= 20; ++j)
        {
            if (dp[i - 1][j] == 0)
                continue;
            if (s2[i] == '+')
                dp[i][j + 1] += dp[i - 1][j];
            if (s2[i] == '-')
                dp[i][j - 1] += dp[i - 1][j];
            if (s2[i] == '?')
            {
                dp[i][j + 1] += dp[i - 1][j] / 2;
                dp[i][j - 1] += dp[i - 1][j] / 2;
            }
        }
    }
    double ans{ dp[s2.length() - 1][n] };
    std::cout << std::fixed << std::setprecision(10) << ans << '\n';
}
