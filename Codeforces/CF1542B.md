void solve()
{
    int n, a, b;
    std::cin >> n >> a >> b;

    if (a == 1)
    {
        std::cout << ((n - 1) % b == 0 ? "yes" : "no") << '\n';
        return;
    }

    int64_t x{ 1 };
    while (x <= n)
    {
        if ((n - x) % b == 0)
        {
            std::cout << "yes\n";
            return;
        }
        x *= a;
    }
    std::cout << "no\n";
}
