void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    std::vector<int> bs(n);
    std::vector<int> cs(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }
    for (auto& b : bs)
    {
        std::cin >> b;
    }
    for (auto& c : cs)
    {
        std::cin >> c;
    }

    std::vector<bool> ok_ab(n, true);
    for (int i = 0; i < n; ++i)
    {
        for (int j = 0; j < n; ++j)
        {
            if (bs[(j + i) % n] <= as[j])
                ok_ab[i] = false;
        }
    }
    std::vector<bool> ok_bc(n, true);
    for (int i = 0; i < n; ++i)
    {
        for (int j = 0; j < n; ++j)
        {
            if (cs[(j + i) % n] <= bs[j])
                ok_bc[i] = false;
        }
    }
    int count_ab{ std::count(ok_ab.begin(), ok_ab.end(), true) };
    int count_bc{ std::count(ok_bc.begin(), ok_bc.end(), true) };
    int64_t ans{ count_ab * count_bc * int64_t(n) };
    std::cout << ans << '\n';
}
