void solve()
{
    int n, k;
    std::cin >> n >> k;
    while (k > 3)
    {
        --k;
        --n;
        std::cout << 1 << ' ';
    }

    if (n & 1)
        std::cout << 1 << ' ' << n / 2 << ' ' << n / 2 << '\n';
    else
        if ((n / 2) & 1)
            std::cout << 2 << ' ' << (n - 2) / 2 << ' ' << (n - 2) / 2 << '\n';
        else
            std::cout << n / 2 << ' ' << n / 4 << ' ' << n / 4 << '\n';
}
