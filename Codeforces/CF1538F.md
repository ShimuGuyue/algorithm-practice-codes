void solve()
{
    int l, r;
    std::cin >> l >> r;

    int count1{ 0 }, count2{ 0 };
    while (l)
    {
        count1 += l % 10 + l / 10 * 10;
        l /= 10;
    }
    while (r)
    {
        count2 += r % 10 + r / 10 * 10;
        r /= 10;
    }
    int ans{ count2 - count1 };
    std::cout << ans << '\n';
}
