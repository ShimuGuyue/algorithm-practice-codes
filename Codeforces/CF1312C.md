void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int64_t> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int64_t> flags{{ 1 }};
    int64_t flag{ 1 };
    while (flag <= 1e16)
    {
        flag *= k;
        flags.push_back(flag);
    }

    std::map<int64_t, int> counts;
    for (auto a : as)
    {
        while (a > 0)
        {
            int64_t x = *std::prev(std::upper_bound(flags.begin(), flags.end(), a));
            ++counts[x];
            a -= x;
        }
    }

    for (auto [k, v] : counts)
    {
        if (v > 1)
        {
            std::cout << "no\n";
            return;
        }
    }
    std::cout << "yes\n";
}
