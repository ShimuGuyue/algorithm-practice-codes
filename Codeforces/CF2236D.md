void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::map<int, int> m;
    for (auto a : as)
    {
        ++m[a];
    }

    std::vector<std::vector<std::pair<int, int>>> segs;
    int last;
    for (auto [a, c] : m)
    {
        if (segs.empty() || a - last > k)
            segs.emplace_back();
        segs.back().push_back({ a, c });
        last = a;
    }

    for (auto& v : segs)
    {
        if (v.back().second % 2 == 0)
        {
            std::cout << "Yes\n";
            return;
        }
        if (v.size() > 1)
        {
            std::cout << "Yes\n";
            return;
        }
    }
    std::cout << "No\n";
}
