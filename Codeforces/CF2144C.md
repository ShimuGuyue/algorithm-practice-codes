int64_t mod{ 998244353 };

void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    std::vector<int> bs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> bs[i];
    }

    std::vector<std::vector<int64_t>> dp(n + 1, std::vector<int64_t>(2));
    dp[0][0] = 1;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (as[i] >= as[i - 1] && bs[i] >= bs[i - 1])
        {
            dp[i][0] = (dp[i][0] + dp[i - 1][0]) % mod;
            dp[i][1] = (dp[i][1] + dp[i - 1][1]) % mod;
        }
        if (as[i] >= bs[i - 1] && bs[i] >= as[i - 1])
        {
            dp[i][0] = (dp[i][0] + dp[i - 1][1]) % mod;
            dp[i][1] = (dp[i][1] + dp[i - 1][0]) % mod;
        }
    }
    std::cout << (dp[n][0] + dp[n][1]) % mod << '\n';
}
