void solve()
{
    int n, x, y;
    std::cin >> n >> x >> y;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> v_out;
    std::vector<int> v_in;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (i <= x || i > y)
            v_out.push_back(as[i]);
        else
            v_in.push_back(as[i]);
    }

    if (v_in.empty())
    {
        for (auto a : as)
        {
            std::cout << a << ' ';
        }
        std::cout << '\n';
    }

    [&v_in]
    {
        int index{ std::min_element(v_in.begin(), v_in.end()) - v_in.begin() };
        std::vector<int> temp;
        for (int i{ 0 }; i < v_in.size(); ++i)
        {
            temp.push_back(v_in[(index + i) % v_in.size()]);
        }
        v_in = temp;
    }();

    bool find{ false };
    for (auto a : v_out)
    {
        if (!find && v_in[0] < a)
        {
            find = true;
            for (auto b : v_in)
            {
                std::cout << b << ' ';
            }
        }
        std::cout << a << ' ';
    }
    if (!find)
    {
        for (auto b : v_in)
        {
            std::cout << b << ' ';
        }
    }
    std::cout << '\n';
}
