void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (n == 1)
    {
        std::cout << 0 << '\n';
        return;
    }

    int ans{ as.back() - as.front() };
    for (int i{ 1 }; i < n; ++i)
    {
        ans = std::max(ans, as[i] - as[0]);
    }
    for (int i{ 0 }; i < n - 1; ++i)
    {
        ans = std::max(ans, as[n - 1] - as[i]);
    }
    for (int i{ 1 }; i < n; ++i)
    {
        ans = std::max(ans, as[i - 1] - as[i]);
    }
    std::cout << ans << '\n';
}
