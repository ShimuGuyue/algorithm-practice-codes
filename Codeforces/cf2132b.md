void solve()
{
    int64_t n;
    std::cin >> n;

    std::vector<int> ans;
    for (int64_t flag = 11; flag <= n; flag = (flag - 1) * 10 + 1)
    {
        if (n % flag == 0)
            ans.push_back(n / flag);
    }
    std::reverse(ans.begin(), ans.end());
    std::cout << ans.size() << '\n';
    if (ans.empty())
        return;
    for (auto a : ans)
    {
        std::cout << a <<' ';
    }
    std::cout << '\n';
}
