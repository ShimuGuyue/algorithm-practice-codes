void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    std::vector<int> bs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> bs[i];
    }

    std::vector<std::vector<int>> ias(n + 2);
    std::vector<std::vector<int>> ibs(n + 2);
    for (int i{ 1 }; i <= n; ++i)
    {
        ias[as[i]].push_back(i);
        ibs[bs[i]].push_back(i);
    }

    int64_t ans{ 0 };
    std::vector<int> dp(n + 2);
    for (int i{ n }; i > 0; --i)
    {
        if (as[i] == bs[i])
        {
            int flag{ as[i] + 1 };

            auto& va{ ias[flag] };
            auto& vb{ ibs[flag] };
            auto at{ std::lower_bound(va.begin(), va.end(), i) };
            auto bt{ std::lower_bound(vb.begin(), vb.end(), i) };
            int ia{ at == va.end() ? n + 1 : *at };
            int ib{ bt == vb.end() ? n + 1 : *bt };

            if (ia == ib)
                dp[i] = (ia == n + 1) ? n : dp[ia];
            else
                dp[i] = std::min(ia, ib) - 1;
        }

        auto& va1{ ias[1] };
        auto& vb1{ ibs[1] };
        auto at1{ std::lower_bound(va1.begin(), va1.end(), i) };
        auto bt1{ std::lower_bound(vb1.begin(), vb1.end(), i) };
        int ia1{ at1 == va1.end() ? n + 1 : *at1 };
        int ib1{ bt1 == vb1.end() ? n + 1 : *bt1 };

        int cur_ans;
        if (ia1 == ib1)
            cur_ans = (ia1 == n + 1) ? n : dp[ia1];
        else
            cur_ans = std::min(ia1, ib1) - 1;

        ans += cur_ans - i + 1;
    }
    std::cout << ans << '\n';
}
