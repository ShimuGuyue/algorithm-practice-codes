void solve()
{
    int n;
    std::cin >> n;

    if (std::popcount(uint32_t(n)) == 1)
    {
        std::cout << -1 << '\n';
        return;
    }

    std::vector<int> ans(n + 1);
    ans[n] = 1;
    for (int i = n - 1; i > 1; --i)
    {
        ans[i] = 1 ^ i;
    }
    ans[1] = n & 1 ? n - 1 : n;

    if (n % 2 == 0)
    {
        int lowbit{ n & (-n) };
        std::swap(ans[1], ans[lowbit]);
    }

    for (int i = 1; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
