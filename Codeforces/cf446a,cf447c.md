void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (n == 1)
    {
        std::cout << 1 << '\n';
        return;
    }

    std::vector<int> dpl(n, 1);
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] > as[i - 1])
            dpl[i] += dpl[i - 1];
    }
    std::vector<int> dpr(n, 1);
    for (int i{ n - 2 }; i >= 0; --i)
    {
        if (as[i] < as[i + 1])
            dpr[i] += dpr[i + 1];
    }

    int ans{ *std::max_element(dpl.begin(), dpl.end()) };
    for (int i{ 0 }; i < n; ++i)
    {
        int cur;
        if (i == 0)
        {
            cur = dpr[i + 1] + 1;
        }
        else if (i == n - 1)
        {
            cur = dpl[i - 1] + 1;
        }
        else
        {
            if (as[i + 1] - as[i - 1] <= 1)
                cur = std::max(dpl[i - 1], dpr[i + 1]) + 1;
            else
                cur = dpl[i - 1] + dpr[i + 1] + 1;
        }
        ans = std::max(ans, cur);
    }
    std::cout << ans << '\n';
}
