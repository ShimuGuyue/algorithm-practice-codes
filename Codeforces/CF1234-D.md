void solve()
{
    std::string s;
    std::cin >> s;

    s = " " + s;

    std::array<std::set<int>, 26> ts;
    for (int i{ 1 }; i < s.length(); ++i)
    {
        int index{ s[i] - 'a' };
        ts[index].insert(i);
    }

    int q;
    std::cin >> q;
    while (q--)
    {
        int op;
        std::cin >> op;
        if (op == 1)
        {
            int p;
            char c;
            std::cin >> p >> c;
            ts[s[p] - 'a'].erase(p);
            s[p] = c;
            ts[s[p] - 'a'].insert(p);
        }
        else
        {
            int l, r;
            std::cin >> l >> r;
            int ans{ 0 };
            for (int i{ 0 }; i < 26; ++i)
            {
                auto it{ ts[i].lower_bound(l) };
                if (it != ts[i].end() && *it <= r)
                    ++ans;
            }
            std::cout << ans << '\n';
        }
    }
}
