void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::map<int, int> m;
    for (auto a : as)
    {
        ++m[a];
    }

    for (auto a : as)
    {
        if (a == m.rbegin()->first)
            if (m.rbegin()->second != 1)
                std::cout << 0 << ' ';
            else
                std::cout << a - std::next(m.rbegin())->first << ' ';
        else
            std::cout << a - m.rbegin()->first << ' ';
    }
    std::cout << '\n';
}
