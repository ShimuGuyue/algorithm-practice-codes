void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int ans{ 0 };
    int sum{ 0 };
    for (auto a : as)
    {
        if (a == 0)
            ++sum;
        else
            sum = 0;
        ans = std::max(ans, sum);
    }
    std::cout << ans << '\n';
}
