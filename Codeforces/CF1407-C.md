void solve()
{
    using HeapInt = std::priority_queue<int>;

    int n;
    std::cin >> n;

    int sum{ n * (n + 1) / 2 };

    HeapInt h;
    for (int i = 1; i <= n; ++i)
    {
        h.push(i);
    }
    std::vector<int> ans(n + 1);
    int k{ n * 2 };
    while (k > 0)
    {
        if (h.size() == 1)
            break;

        int index1{ h.top() };
        h.pop();
        int index2{ h.top() };
        h.pop();
        std::cout << "? " << index1 << " " << index2 << std::endl;
        int mod1;
        std::cin >> mod1;
        std::cout << "? " << index2 << " " << index1 << std::endl;
        int mod2;
        std::cin >> mod2;

        if (mod1 > mod2)
        {
            ans[index1] = mod1;
            sum -= mod1;
            h.push(index2);
        }
        else
        {
            ans[index2] = mod2;
            sum -= mod2;
            h.push(index1);
        }
    }
    ans[h.top()] = sum;

    std::cout << "! ";
    for (int i = 1; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << std::endl;
}
