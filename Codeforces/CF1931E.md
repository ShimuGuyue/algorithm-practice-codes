void solve()
{
    int n, m;
    std::cin >> n >> m;

    int sum{ 0 };
    std::priority_queue<int, std::vector<int> , std::less<int>> h;
    for (int i = 0; i < n; ++i)
    {
        std::string s;
        std::cin >> s;
        sum += s.length();
        int suf0{ 0 };
        for (int i = s.length() - 1; i >= 0; --i)
        {
            if (s[i] == '0')
                ++suf0;
            else
                break;
        }
        h.push(suf0);
    }

    while (!h.empty())
    {
        sum -= h.top();
        h.pop();

        if (!h.empty())
            h.pop();
    }
    std::cout << (sum > m ? "Sasha" : "Anna") << '\n';
}
