void solve()
{
    int64_t n, m, k;
    std::cin >> n >> m >> k;
    std::string s;
    std::cin >> s;

    int ans{ 0 };

    int used{ 0 };
    int a{ 0 };
    for (char c : s)
    {
        if (c == 'I')
        {
            if (used < m)
            {
                ++used;
                ++ans;
            }
        }
        else if (c == 'E')
        {
            if (ans < used * k)
            {
                ++ans;
            }
            else
            {
                if (used < m && a)
                {
                    --a;
                    ++used;
                    ++ans;
                }
            }
        }
        else if (c == 'A')
        {
            if (ans < used * k)
            {
                ++ans;
                ++a;
            }
            else if (used < m)
            {
                ++ans;
                ++used;
            }
        }
    }
    std::cout << ans << '\n';
}
