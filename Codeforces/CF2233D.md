void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> digits{ as };
    std::sort(digits.begin(), digits.end());
    digits.erase(std::unique(digits.begin(), digits.end()), digits.end());
    std::map<int, int> ids;
    for (int i{ 0 }; i < digits.size(); ++i)
    {
        ids[digits[i]] = i;
    }
    std::vector<int> turns(n);
    for (int i{ 0 }; i < n; ++i)
    {
        turns[i] = ids[as[i]];
    }

    std::map<int, std::vector<std::pair<int, int>>> segs;
    for (int i{ 0 }; i < n; ++i)
    {
        auto& v{ segs[as[i]] };
        if (v.empty() || v.back().second != i - 1)
            v.push_back({ i, i });
        else
            v.back().second = i;
    }
    int count{ 0 };
    for (auto& [a, v] : segs)
    {
        if (v.size() > 3)
        {
            std::cout << "No\n";
            return;
        }
        if (v.size() > 1)
            ++count;
    }
    if (count == 0)
    {
        std::cout << "Yes\n";
        return;
    }
    if (count > 2)
    {
        std::cout << "No\n";
        return;
    }

    std::vector<int> indexs;
    for (auto& [a, v] : segs)
    {
        if (v.size() == 1)
            continue;
        for (auto [l, r] : v)
        {
            indexs.push_back(l);
            indexs.push_back(r);
            if (l != 0)
                indexs.push_back(l - 1);
            if (r != n - 1)
                indexs.push_back(r + 1);
        }
    }
    std::sort(indexs.begin(), indexs.end());
    indexs.erase(std::unique(indexs.begin(), indexs.end()), indexs.end());

    for (int i{ 0 }; i < indexs.size(); ++i)
    {
        for (int j{ i + 1 }; j < indexs.size(); ++j)
        {
            if (as[indexs[i]] == as[indexs[j]])
                continue;
            std::swap(as[indexs[i]], as[indexs[j]]);
            std::swap(turns[indexs[i]], turns[indexs[j]]);
            bool ok{ true };
            std::vector<int> lasts(n, -1);
            for (int k{ 0 }; k < n; ++k)
            {
                int cur{ turns[k] };
                if (lasts[cur] != -1 && lasts[cur] != k - 1)
                {
                    ok = false;
                    break;
                }
                lasts[cur] = k;
            }
            if (ok)
            {
                std::cout << "Yes\n";
                return;
            }
            std::swap(as[indexs[i]], as[indexs[j]]);
            std::swap(turns[indexs[i]], turns[indexs[j]]);
        }
    }
    std::cout << "No\n";
}
