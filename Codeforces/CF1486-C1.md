void solve()
{
    int n;
    std::cin >> n;

    std::cout << "? " << 1 << ' ' << n << std::endl;
    int loc;
    std::cin >> loc;

    bool left{ false };
    if (loc != 1)
    {
        std::cout << "? " << 1 << ' ' << loc << std::endl;
        int x;
        std::cin >> x;
        if (x == loc)
            left = true;
    }

    if (left)
    {
        int l{ 1 }, r{ loc - 1 };
        while (l < r)
        {
            int mid{ (l + r + 1) / 2 };
            std::cout << "? " << mid << ' ' << loc << std::endl;
            int x;
            std::cin >> x;
            if (loc == x)
                l = mid;
            else
                r = mid - 1;
        }
        std::cout << "! " << l << std::endl;
    }
    else
    {
        int l{ loc + 1 }, r{ n };
        while (l < r)
        {
            int mid{ (l + r) / 2 };
            std::cout << "? " << loc << ' ' << mid << std::endl;
            int x;
            std::cin >> x;
            if (loc == x)
                r = mid;
            else
                l = mid + 1;
        }
        std::cout << "! " << r << std::endl;
    }
}
