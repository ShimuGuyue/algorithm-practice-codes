static constexpr int64_t mod{ 1000000007 };

void solve()
{

    int n, k;
    std::cin >> n >> k;

    int64_t ans{ 0 };
    if (n & 1)
    {
        ans = (ans + quickpow((quickpow(2, n - 1)) + 1, k)) % mod;
    }
    else
    {
        ans = (ans + quickpow((quickpow(2, n - 1) - 1), k)) % mod;
        for (int i{ 1 }; i <= k; ++i)
        {
            int64_t before{ quickpow((quickpow(2, n - 1) - 1), i - 1) };
            int64_t cur{ 1 };
            int64_t after{ quickpow(quickpow(2, n), k - i) };
            ans = (ans + before * cur % mod * after % mod) % mod;
        }
    }
    std::cout << ans << '\n';
}

int64_t quickpow(int64_t base, int64_t power)
{
    int64_t ans{ 1 };
    while (power)
    {
        if (power & 1)
            ans = ans * base % mod;
        base = base * base % mod;
        power /= 2;
    }
    return ans;
}
