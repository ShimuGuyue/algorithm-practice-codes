void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::string s;
    std::cin >> s;

    s = " " + s;

    std::vector<int> pre0s(n + 1);
    std::vector<int> pre1s(n + 1);
    std::vector<int> predifs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        pre0s[i] = pre0s[i - 1] + (s[i] == '0');
        pre1s[i] = pre1s[i - 1] + (s[i] == '1');
        predifs[i] = predifs[i - 1] + (s[i] != s[i - 1]);
    }

    while (q--)
    {
        int l, r;
        std::cin >> l >> r;
        if (l == r)
        {
            std::cout << 3 << '\n';
            continue;
        }

        int len{ r - l + 1 };

        int count0{ pre0s[r] - pre0s[l - 1] };
        int count1{ pre1s[r] - pre1s[l - 1] };
        int seg{ predifs[r] - predifs[l] + (s[l] != s[r]) };

        int k = std::max<int>({
            (count0 + 1) / 2,
            (count1 + 1) / 2,
            seg / 2,
            (len - (seg / 2) + 2) / 3
        });

        int ans{ k * 4 - len };
        std::cout << ans << '\n';
    }
}
