void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    int min{ n };
    int c1{ 0 }, c2{ 0 };
    auto dfs1 = [&tree, &min, &c1, &c2, n](this auto&& dfs, int u, int last) -> int
    {
        int sum{ 1 };
        int max{ 0 };
        int f{ n - 1 };
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            int child{ dfs(v, u) };
            sum += child;
            f -= child;
            max = std::max(max, child);
        }
        max = std::max(max, f);
        if (max < min)
        {
            min = max;
            c1 = u;
            c2 = 0;
        }
        else if (max == min)
        {
            c2 = u;
        }
        return sum;
    };
    dfs1(1, 0);
    if (c2 == 0)
    {
        std::cout << 1 << ' ' << tree[1][0] << '\n';
        std::cout << 1 << ' ' << tree[1][0] << '\n';
        return;
    }

    bool find{ false };
    auto dfs2 = [&tree, &c2, &find](this auto&& dfs, int u, int last) -> void
    {
        if (find)
            return;
        if (tree[u].size() == 1)
        {
            std::cout << last << ' ' << u << '\n';
            std::cout << c2 << ' ' << u << '\n';
            find = true;
            return;
        }
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
        }
    };
    dfs2(c1, c2);
}
