void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::string t;
    std::cin >> s;
    std::cin >> t;

    for (int i = 0; i < n; ++i)
    {
        if (s[i] == 'G')
            s[i] = 'B';
        if (t[i] == 'G')
            t[i] = 'B';
        if (s[i] != t[i])
        {
            std::cout << "no\n";
            return;
        }
    }
    std::cout << "yes\n";
}
