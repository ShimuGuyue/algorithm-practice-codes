void solve()
{
    std::string s;
    std::cin >> s;

    int n{ int(s.length()) };

    int ans{ n };
    for (int i = 0; i < n; ++i)
    {
        ans = std::min({ans, calc(s, i, '0'), calc(s, i, '1')});
    }
    std::cout << ans << '\n';
}

int calc(std::string& s, int index, char c)
{
    int ans{ 0 };
    for (int i = 0; i < index; ++i)
    {
        if (s[i] == c)
            ++ans;
    }
    for (int i = index; i < s.length(); ++i)
    {
        if (s[i] != c)
            ++ans;
    }
    return ans;
}
