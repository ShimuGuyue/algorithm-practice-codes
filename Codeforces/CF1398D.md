void solve()
{
    int r, g, b;
    std::cin >> r >> g >> b;
    std::vector<int> rs(r + 1);
    std::vector<int> gs(g + 1);
    std::vector<int> bs(b + 1);
    for (int i{ 1 }; i <= r; ++i)
    {
        std::cin >> rs[i];
    }
    for (int i{ 1 }; i <= g; ++i)
    {
        std::cin >> gs[i];
    }
    for (int i{ 1 }; i <= b; ++i)
    {
        std::cin >> bs[i];
    }

    std::sort(rs.begin() + 1, rs.end(), std::greater<int>());
    std::sort(gs.begin() + 1, gs.end(), std::greater<int>());
    std::sort(bs.begin() + 1, bs.end(), std::greater<int>());

    int64_t ans{ 0 };
    std::vector<std::vector<std::vector<int64_t>>> dp(r + 1,
                std::vector<std::vector<int64_t>>    (g + 1,
                            std::vector<int64_t>     (b + 1)));
    for (int i{ 0 }; i <= r; ++i)
    {
        for (int j{ 0 }; j <= g; ++j)
        {
            for (int k{ 0 }; k <= b; ++k)
            {
                if (i && j)
                    dp[i][j][k] = std::max(dp[i][j][k], dp[i - 1][j - 1][k] + rs[i] * gs[j]);
                if (i && k)
                    dp[i][j][k] = std::max(dp[i][j][k], dp[i - 1][j][k - 1] + rs[i] * bs[k]);
                if (j && k)
                    dp[i][j][k] = std::max(dp[i][j][k], dp[i][j - 1][k - 1] + gs[j] * bs[k]);

                ans = std::max(ans, dp[i][j][k]);
            }
        }
    }
    std::cout << ans << '\n';
}
