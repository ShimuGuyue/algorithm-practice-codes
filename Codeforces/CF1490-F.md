void solve()
{
    int n;
    std::cin >> n;
    std::map<int, int> m;
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++m[a];
    }

    std::map<int, int> counts;
    for (auto [k, v] : m)
    {
        ++counts[v];
    }

    int ans{ n };
    for (auto [k, v] : counts)
    {
        int sum{ 0 };
        for (auto [kk, vv] : counts)
        {
            if (kk == k)
                continue;
            sum += kk < k ? kk * vv : (kk - k) * vv;
        }
        ans = std::min(ans, sum);
    }
    std::cout << ans << '\n';
}
