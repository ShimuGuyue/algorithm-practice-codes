```cpp
struct BinaryIndexedTree
{
    std::vector<int> tree;

    BinaryIndexedTree(int n) : tree(n + 1)
    { }

    void update(int index, int data)
    {
        while (index < tree.size())
        {
            tree[index] += data;
            index += lowbit(index);
        }
    }

    int query(int index)
    {
        int ans{ 0 };
        while (index)
        {
            ans += tree[index];
            index -= lowbit(index);
        }
        return ans;
    }

    int lowbit(int x)
    {
        return x & -x;
    }
};
```

```cpp
void solve()
{
    int n, q;
    std::cin >> n >> q;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    BinaryIndexedTree bit(n);
    for (int i{ 1 }; i <= n; ++i)
    {
        bit.update(as[i], 1);
    }

    while (q--)
    {
        int k;
        std::cin >> k;
        if (k > 0)
        {
            bit.update(k, 1);
        }
        else
        {
            k = -k;
            int l{ 1 }, r{ n };
            while (l < r)
            {
                int mid{ (l + r) / 2 };
                if (bit.query(mid) >= k)
                    r = mid;
                else
                    l = mid + 1;
            }
            bit.update(r, -1);
        }
    }
    for (int i{ 1 }; i <= n; ++i)
    {
        if (bit.query(i))
        {
            std::cout << i << '\n';
            return;
        }
    }
    std::cout << 0 << '\n';
}
```
