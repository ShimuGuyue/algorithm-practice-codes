void solve()
{
    int n, k, q;
    std::cin >> n >> k >> q;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> flags;
    int sum{ 0 };
    for (auto a : as)
    {
        if (a <= q)
        {
            ++sum;
        }
        else
        {
            if (sum)
                flags.push_back(sum);
            sum = 0;
        }
    }
    if (sum)
        flags.push_back(sum);

    int64_t ans{ 0 };
    for (auto a : flags)
    {
        int64_t x{ a - k + 1 };
        if (x <= 0)
            continue;
        ans += x * (x + 1) / 2;
    }
    std::cout << ans << '\n';
}
