void solve()
{
    int n, m;
    std::cin >> n >> m;

    std::vector<int> degrees(n + 1);
    while (m--)
    {
        int u, v;
        std::cin >> u >> v;
        ++degrees[u];
        ++degrees[v];
    }

    int count{ std::count(degrees.begin(), degrees.end(), 1) };
    int x{ n - count - 1 };
    int y{ count / x };
    std::cout << x << ' ' << y << '\n';
}
