void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());

    int count{ (n - 1) / 2 };
    std::cout << count << '\n';

    std::vector<int> ans(n);
    for (int i{ 0 }, r{ n - 1 }; i < n; i += 2, --r)
    {
        ans[i] = as[r];
    }
    for (int i{ 1 }, l{ 0 }; i < n; i += 2, ++l)
    {
        ans[i] = as[l];
    }
    for (auto a : ans)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
