void solve()
{
    int n, x;
    std::cin >> n >> x;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int64_t> presums(n + 1);
    std::partial_sum(as.begin(), as.end(), presums.begin());

    std::vector<int64_t> ans(n + 1);
    for (int i{ n }; i > 0; --i)
    {
        int64_t flag{ x + presums[i - 1] };
        auto it{ std::upper_bound(presums.begin() + i, presums.end(), flag) };
        int j{ std::distance(presums.begin(), it) };
        ans[i] += j - i;
        if (j + 1 <= n)
            ans[i] += ans[j + 1];
    }
    int64_t sum{ std::accumulate(ans.begin(), ans.end(), INT64_C(0)) };
    std::cout << sum << '\n';
}
