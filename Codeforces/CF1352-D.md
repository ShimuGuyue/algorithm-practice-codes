void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int count{ 0 };
    int sum_a{ 0 }, sum_b{ 0 };

    bool is_a{ true };
    int i{ 0 }, j{ n - 1 };
    int sum_last{ 0 };
    while (i <= j)
    {
        ++count;
        int sum{ 0 };
        if (is_a)
        {
            while (sum <= sum_last && i <= j)
            {
                sum += as[i++];
            }
            sum_a += sum;
        }
        else
        {
            while (sum <= sum_last && i <= j)
            {
                sum += as[j--];
            }
            sum_b += sum;
        }
        is_a = !is_a;
        sum_last = sum;
    }
    std::cout << count << ' ' << sum_a << ' ' << sum_b << '\n';
}
