void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> dp1(n, 1);
    for (int i{ 1 }; i < n; ++i)
    {
        if (as[i] > as[i - 1])
            dp1[i] = dp1[i - 1] + 1;
    }
    std::vector<int> dp2(n, 1);
    for (int i{ 1 }; i < n - 1; ++i)
    {
        if (as[i + 1] > as[i])
        dp2[i + 1] = dp2[i] + 1;
        if (as[i + 1] > as[i - 1])
            dp2[i + 1] = std::max(dp1[i - 1] + 1, dp2[i + 1]);
    }

    int ans{ std::max(
        *std::max_element(dp1.begin(), dp1.end()),
        *std::max_element(dp2.begin(), dp2.end())
    ) };
    std::cout << ans << '\n';
}
