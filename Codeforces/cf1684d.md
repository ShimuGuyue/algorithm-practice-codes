void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::priority_queue<std::pair<int, int>, std::vector<std::pair<int, int>>> h;
    for (int i{ 0 }; i < n; ++i)
    {
        h.push({ as[i] - n + i, i });
    }

    std::vector<bool> selects(n);
    while (k--)
    {
        selects[h.top().second] = true;
        h.pop();
    }

    int64_t ans{ 0 };
    int ext{ 0 };
    for (int i{ 0 }; i < n; ++i)
    {
        if (selects[i])
            ++ext;
        else
            ans += as[i] + ext;
    }
    std::cout << ans << '\n';
}
