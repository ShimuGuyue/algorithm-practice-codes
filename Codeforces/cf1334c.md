void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n);
    std::vector<int64_t> bs(n);
    for (int i{ 0 }; i < n; ++i)
    {
        std::cin >> as[i] >> bs[i];
    }

    int64_t min_value{ std::numeric_limits<int64_t>::max() };
    int index;
    for (int i{ 0 }; i < n; ++i)
    {
        int64_t value{ std::min(bs[i], as[(i + 1) % n]) };
        if (value < min_value)
        {
            min_value = value;
            index = i;
        }
    }

    for (int i{ 0 }; i < n; ++i)
    {
        if (i == index)
            continue;
        int64_t value{ std::min(bs[i], as[(i + 1) % n]) };
        as[(i + 1) % n] -= value;
    }

    int64_t ans{ std::accumulate(as.begin(), as.end(), INT64_C(0)) };
    std::cout << ans << '\n';
}
