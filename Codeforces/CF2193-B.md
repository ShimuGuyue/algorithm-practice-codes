void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int flag{ n };
    for (int i = 0; i < n; ++i)
    {
        if (as[i] == flag)
        {
            --flag;
            continue;
        }
        int j{ i };
        while (as[j] != flag)
        {
            ++j;
        }
        for (int l = i, r = j; l < r; ++l, --r)
        {
            std::swap(as[l], as[r]);
        }
        break;;
    }
    for (auto a : as)
    {
        std::cout << a << ' ';
    }
    std::cout << '\n';
}
