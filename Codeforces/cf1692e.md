void solve()
{
    int n, s;
    std::cin >> n >> s;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::vector<int> indexs;
    indexs.push_back(-1);
    for (int i = 0; i < n; ++i)
    {
        if (as[i] == 1)
            indexs.push_back(i);
    }
    indexs.push_back(n);

    int ans{ n };
    for (int i = 1; i + s < indexs.size(); ++i)
    {
        int l{ indexs[i - 1] + 1 };
        int r{ indexs[i + s] - 1 };
        int range{ r - l + 1 };
        ans = std::min(ans, n - range);
    }
    std::cout << (ans == n ? -1 : ans) << '\n';
}
