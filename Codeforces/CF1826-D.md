void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }

    std::vector<int> premaxs(n + 1);
    std::vector<int> sufmaxs(n + 1);
    auto cmp = [](int pre, int cur)
    {
        return std::max(pre - 1, cur);
    };
    std::partial_sum(as.begin() + 1, as.end(), premaxs.begin() + 1, cmp);
    std::partial_sum(as.rbegin(), as.rend() - 1, sufmaxs.rbegin(), cmp);

    int ans{ 0 };
    for (int i{ 2 }; i < n; ++i)
    {
        ans = std::max(ans, premaxs[i - 1] + sufmaxs[i + 1] + as[i] - 2);
    }
    std::cout << ans << '\n';
}
