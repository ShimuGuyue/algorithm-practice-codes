void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> counts(n + 1);
    for (int i{ 0 }; i < n; ++i)
    {
        int a;
        std::cin >> a;
        ++counts[a];
    }

    bool ok{ true };
    int64_t pre{ 0 };
    std::stack<int> k;
    for (int i{ 0 }; i <= n; ++i)
    {
        int64_t ans{ counts[i] };
        if (i > 0 && ok)
        {
            if (!k.empty())
            {
                pre += i - 1 - k.top();
                k.pop();
            }
            else
            {
                ok = false;
            }
            ans += pre;
        }        
        for (int j{ 0 }; j < counts[i]; ++j)
        {
            k.push(i);
        }
        if (!ok)
            ans = -1;
        std::cout << ans << ' ';
    }
    std::cout << '\n';
}
