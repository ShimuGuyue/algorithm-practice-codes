void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    if (std::is_sorted(as.begin(), as.end()))
    {
        std::cout << "Bob\n";
        return;
    }

    std::vector<int> flags;
    flags.reserve(n);
    for (auto a : as)
    {
        if (a == 1)
        {
            flags.push_back(1);
            continue;
        }
        std::set<int> t;
        for (int i{ 2 }; i * i <= a; ++i)
        {
            if (a % i)
                continue;
            t.insert(i);
            while (a % i == 0)
            {
                a /= i;
            }
        }
        if (a != 1)
            t.insert(a);
        if (t.size() > 1)
        {
            std::cout << "Alice\n";
            return;
        }
        for (auto b : t)
        {
            flags.push_back(b);
        }
    }
    if (std::is_sorted(flags.begin(), flags.end()))
        std::cout << "Bob\n";
    else
        std::cout << "Alice\n";
}
