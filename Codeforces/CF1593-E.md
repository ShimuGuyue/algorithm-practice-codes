void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::vector<int> degrees(n + 1);
    std::vector<std::vector<int>> graph(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        ++degrees[u];
        ++degrees[v];
        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    std::vector<int> flags(n + 1);
    std::queue<int> q;
    for (int i{ 1 }; i <= n; ++i)
    {
        if (degrees[i] == 1)
            q.push(i);
    }
    while (!q.empty())
    {
        int u{ q.front() };
        q.pop();
        for (int v : graph[u])
        {
            if (--degrees[v] == 1)
            {
                q.push(v);
                flags[v] = flags[u] + 1;
            }
        }
    }

    int ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        if (flags[i] >= k)
            ++ans;
    }
    std::cout << ans << '\n';
}
