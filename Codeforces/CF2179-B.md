void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (int i = 0; i < n; ++i)
    {
        std::cin >> as[i];
    }

    int flag{ 0 };
    for (int i = 1; i < n; ++i)
    {
        flag += abs(as[i] - as[i - 1]);
    }
    int ans{ flag };
    for (int i = 1; i < n - 1; ++i)
    {
        ans = std::min(
            ans,
            flag - std::abs(as[i] - as[i - 1]) - std::abs(as[i] - as[i + 1]) + std::abs(as[i + 1] - as[i - 1])
        );
    }
    ans = std::min(ans, flag - abs(as[1] - as[0]));
    ans = std::min(ans, flag - abs(as[n - 1] - as[n - 2]));
    std::cout << ans << '\n';
}
