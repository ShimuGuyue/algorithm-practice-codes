void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    int min{ *std::min_element(as.begin(), as.end()) };

    int64_t ans{ 0 };
    std::for_each(as.begin(), as.end(),
        [&ans, min](int x)
        {
            ans += x - min;
        }
    );
    std::cout << ans << '\n';
}
