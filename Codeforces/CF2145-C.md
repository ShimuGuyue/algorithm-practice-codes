void solve()
{
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;

    std::vector<int> v(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        v[i] = s[i - 1] == 'a' ? 1 : -1;
    }

    std::vector<int> pres(n + 1);
    std::partial_sum(v.begin(), v.end(), pres.begin());

    if (pres[n] == 0)
    {
        std::cout << 0 << '\n';
        return;
    }

    int ans{ n };
    int sum{ pres[n] };
    std::map<int, int> m;
    m[0] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        int flag{ pres[i] - sum };
        if (m.count(flag))
        {
            ans = std::min(ans, i - m[flag]);
        }
        m[pres[i]] = i;
    }
    if (ans == n)
        ans = -1;
    std::cout << ans << '\n';
}
