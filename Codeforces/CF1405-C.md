void solve()
{
    int n, k;
    std::cin >> n >> k;
    std::string s;
    std::cin >> s;

    for (int i{ 0 }; i < k; ++i)
    {
        bool b0{ false }, b1{ false };
        for (int j{ i }; j < n; j += k)
        {
            if (s[j] == '0')
                b0 = true;
            if (s[j] == '1')
                b1 = true;
            if (b0 && b1)
            {
                std::cout << "NO\n";
                return;
            }
            if (s[j] == '?')
                s[j] = b0 ? '0' : b1 ? '1' : '?';
        }
    }
    int count0{ 0 }, count1{ 0 };
    for (int i{ 1 }; i <= k; ++i)
    {
        if (s[n - i] == '0')
            ++count0;
        if (s[n - i] == '1')
            ++count1;
    }
    if (count0 > k / 2 || count1 > k / 2)
        std::cout << "NO\n";
    else
        std::cout << "YES\n";
}
