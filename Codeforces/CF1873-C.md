void solve()
{
    int ans{ 0 };
    for (int i{ 1 }; i <= 10; ++i)
    {
        for (int j{ 1 }; j <= 10; ++j)
        {
            char c;
            std::cin >> c;
            if (c == '.')
                continue;
            int x{ std::max(i, 10 - i + 1) - 5 };
            int y{ std::max(j, 10 - j + 1) - 5 };
            ans += 5 - std::max(x, y) + 1;
        }
    }
    std::cout << ans << '\n';
}
