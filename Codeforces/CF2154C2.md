std::vector<std::vector<int>> prime_factors;

void init()
{
    prime_factors.resize(200001);
    for (int i = 2; i <= 200000; ++i)
    {
        int temp{ i };
        for (int j = 2; j * j <= temp; ++j)
        {
            if (temp % j)
                continue;
            prime_factors[i].push_back(j);
            while (temp % j == 0)
            {
                temp /= j;
            }
        }
        if (temp > 1)
            prime_factors[i].push_back(temp);
    }
}

void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::pair<int, int64_t>> datas(n);
    for (auto &[a, b] : datas)
    {
        std::cin >> a;
    }
    for (auto &[a, b] : datas)
    {
        std::cin >> b;
    }

    std::sort(datas.begin(), datas.end(),
        [](std::pair<int, int64_t> &a, std::pair<int, int64_t> &b)
        {
            return a.second < b.second;
        });

    // 0
    std::vector<int> counts(200001);
    for (auto &[a, b] : datas)
    {
        for (int factor : prime_factors[a])
        {
            ++counts[factor];
            if (counts[factor] >= 2)
            {
                std::cout << 0 << '\n';
                return;
            }
        }
    }

    // not 0
    // 如果代价最小两个数均为奇数，则最坏情况为最小两代价相加
    // 否则应为其中之一，该答案偏大，在后续讨论中被替换为正确答案
    int64_t ans{ datas[0].second + datas[1].second };

    // if 1
    // 选择一个数 +1
    for (auto &[a, b] : datas)
    {
        // a 和 a + 1 的质因子集合完全不相交，不需要先去除计数
        for (int factor : prime_factors[a + 1])
        {
            if (counts[factor])
                ans = std::min(ans, b);
        }
    }
    //代价最小的数 +1 若干次
    std::set<int> factors;
    for (int i = 1; i < n; ++i)
    {
        for (int factor : prime_factors[datas[i].first])
        {
            factors.insert(factor);
        }
    }
    for (int f : factors)
    {
        int mod{ datas[0].first % f };
        int dif{ f - mod };
        ans = std::min(ans, datas[0].second * dif);
    }

    std::cout << ans << '\n';
}
