void solve()
{
    int n, x;
    std::cin >> n >> x;
    std::vector<std::pair<int, int>> chs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> chs[i].first >> chs[i].second;
    }

    int m{ std::accumulate(chs.begin(), chs.end(), 0,
        [](int pre, std::pair<int, int> cur)
        {
            return pre + cur.second;
        }
    ) };

    std::vector<int64_t> dp(m + 1, -1);
    dp[0] = 0;
    for (int i{ 1 }; i <= n; ++i)
    {
        auto [c, h]{ chs[i] };
        for (int j{ m }; j >= 0; --j)
        {
            if (dp[j] >= c && j + h <= m)
                dp[j + h] = std::max(dp[j + h], dp[j] - c);
        }
        for (int j{ m }; j >= 0; --j)
        {
            if (dp[j] != -1)
                dp[j] += x;
        }
    }

    int ans{ 0 };
    for (int i{ 0 }; i <= m; ++i)
    {
        if (dp[i] != -1)
            ans = i;
    }
    std::cout << ans << '\n';
}
