    void solve()
    {
        std::string s;
        std::string t;
        std::cin >> s;
        std::cin >> t;

        std::vector<std::vector<int>> indexs(26);
        for (int i{ 0 }; i < s.length(); ++i)
        {
            indexs[s[i] - 'a'].push_back(i);
        }

        int ans{ 1 };
        int index{ -1 };
        for (char c : t)
        {
            auto& v{ indexs[c - 'a'] };
            auto it{ std::upper_bound(v.begin(), v.end(), index) };
            if (it != v.end())
            {
                index = *it;
                continue;
            }
            ++ans;
            index = -1;
            it = std::upper_bound(v.begin(), v.end(), index);
            if (it == v.end())
            {
                std::cout << -1 << '\n';
                return;
            }
            index = *it;
        }
        std::cout << ans << '\n';
    }
