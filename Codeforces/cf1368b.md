void solve()
{
    int64_t k;
    std::cin >> k;

    std::array<int, 10> counts{ };
    std::fill(counts.begin(), counts.end(), 1);

    int64_t sum{ 1 };
    int index{ 0 };
    while (sum < k)
    {
        sum = sum / counts[index] * (counts[index] + 1);
        ++counts[index];
        index = (index + 1) % 10;
    }

    std::string s{ "codeforces" };
    for (int i = 0; i < 10; ++i)
    {
        for (int j = 0; j < counts[i]; ++j)
        {
            std::cout << s[i];
        }
    }
    std::cout << '\n';
}
