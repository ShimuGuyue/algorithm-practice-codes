void solve()
{
    int n;
    std::cin >> n;
    std::vector<int64_t> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    std::vector<int64_t> bs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> bs[i];
    }

    std::vector<int64_t> presums(n + 2);
    std::partial_sum(bs.begin(), bs.end(), presums.begin());

    std::vector<int64_t> ans(n + 1);
    std::vector<int> difs(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        if (presums[n] - presums[i - 1] <= as[i])
        {
            ++difs[i];
            continue;
        }
        int l{ i }, r{ n };
        while (l < r)
        {
            int mid{ (l + r) / 2 };
            if (presums[mid] - presums[i - 1] > as[i])
                r = mid;
            else
                l = mid + 1;
        }
        ans[r] += as[i] - (presums[r - 1] - presums[i - 1]);
        ++difs[i];
        --difs[r];
    }
    std::vector<int64_t> counts(n + 1);
    std::partial_sum(difs.begin(), difs.end(), counts.begin());
    for (int i{ 1 }; i <= n; ++i)
    {
        ans[i] += counts[i] * bs[i];
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
