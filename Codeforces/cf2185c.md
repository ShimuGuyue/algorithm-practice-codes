void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n);
    for (auto& a : as)
    {
        std::cin >> a;
    }

    std::sort(as.begin(), as.end());
    as.erase(std::unique(as.begin(), as.end()), as.end());

    int ans{ 1 };
    int count{ 1 };
    for (int i = 1; i < as.size(); ++i)
    {
        if (as[i] == as[i - 1] + 1)
            ++count;
        else
            count = 1;
        ans = std::max(ans, count);
    }
    std::cout << ans << '\n';
}
