void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> as(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        std::cin >> as[i];
    }
    std::vector<std::vector<int>> tree(n + 1);
    for (int i{ 1 }; i < n; ++i)
    {
        int u, v;
        std::cin >> u >> v;
        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    std::vector<int> sizes(n + 1);
    auto dfs = [&sizes, &tree](this auto&& dfs, int u, int last) -> void
    {
        sizes[u] = 1;
        for (int v : tree[u])
        {
            if (v == last)
                continue;
            dfs(v, u);
            sizes[u] += sizes[v];
        }
    };
    dfs(1, 0);

    int64_t ans{ 0 };
    for (int i{ 1 }; i <= n; ++i)
    {
        int s{ static_cast<int>(std::sqrt(as[i])) };
        if (s * s != as[i])
            continue;

        std::vector<int> counts;
        for (int v : tree[i])
        {
            if (sizes[v] < sizes[i])
                counts.push_back(sizes[v]);
            else
                counts.push_back(n - sizes[i]);
        }

        int64_t c2{ 0 }, c3{ 0 };
        int64_t sum1{ 0 }, sum2{ 0 };
        for (int c : counts)
        {
            c2 += c * sum1;
            c3 += c * sum2;
            sum2 += c * sum1;
            sum1 += c;
        }
        ans += c2 + c3;
    }
    std::cout << ans << '\n';
}
