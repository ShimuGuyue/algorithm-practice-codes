void solve()
{
    std::string s;
    std::cin >> s;

    s += " ";

    std::vector<int> as;
    std::vector<int> bs;

    int count{ 1 };
    for (int i{ 1 }; i < s.length(); ++i)
    {
        if (s[i] != s[i - 1])
        {
            if (s[i - 1] == 'A')
            {
                as.push_back(count);
            }
            else
            {
                bs.push_back(1);
                --count;
                if (count)
                    bs.push_back(count);
            }
            count = 1;
        }
        else
        {
            ++count;
        }
    }

    int min{ *std::min_element(as.begin(), as.end()) };
    int ans{ std::accumulate(as.begin(), as.end(), 0) };
    if (as.size() > bs.size())
        ans -= min;
    std::cout << ans << '\n';
}
