void solve()
{
    int n;
    std::cin >> n;
    std::vector<int> us(n);
    std::vector<int> ss(n);
    for (auto& u : us)
    {
        std::cin >> u;
    }
    for (auto& s : ss)
    {
        std::cin >> s;
    }

    std::vector<std::vector<int64_t>> as(n + 1);
    for (int i{ 0 }; i < n; ++i)
    {
        as[us[i]].push_back(ss[i]);
    }

    std::vector<int64_t> ans(n + 1);
    for (int i{ 1 }; i <= n; ++i)
    {
        auto& v{ as[i] };
        std::sort(v.rbegin(), v.rend());
        std::partial_sum(v.begin(), v.end(), v.begin());
        for (int j{ 1 }; j <= v.size(); ++j)
        {
            ans[j] += v[int(v.size()) / j * j - 1];
        }
    }

    for (int i{ 1 }; i <= n; ++i)
    {
        std::cout << ans[i] << ' ';
    }
    std::cout << '\n';
}
