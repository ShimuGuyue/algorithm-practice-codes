void solve()
{
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> as(2, std::vector<int>(n + 1));
    for (int i = 0; i < 2; ++i)
    {
        for (int j = 1; j <= n; ++j)
        {
            std::cin >> as[i][j];
        }
    }

    std::vector<std::vector<int64_t>> dp(2, std::vector<int64_t>(n + 1));
    dp[0][1] = as[0][1];
    dp[1][1] = as[1][1];
    for (int i = 2; i <= n; ++i)
    {
        dp[0][i] = std::max({dp[1][i - 1], dp[0][i - 2], dp[1][i - 2]}) + as[0][i];
        dp[1][i] = std::max({dp[0][i - 1], dp[0][i - 2], dp[1][i - 2]}) + as[1][i];
    }
    int64_t ans = std::max(
        *std::max_element(dp[0].begin(), dp[0].end()),
        *std::max_element(dp[1].begin(), dp[1].end())
    );
    std::cout << ans << '\n';
}
